"""Deterministic scene sampler and renderer for a self-echo rail search."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 260
BACKGROUND = (250, 249, 245)
PANEL = (240, 239, 234)
FRAME = (212, 211, 205)
CYAN = (31, 139, 170)
INK = (35, 39, 43)
TEXT = (91, 95, 99)
COLORS = {
    "red": (211, 63, 63),
    "gold": (218, 159, 35),
    "violet": (132, 79, 185),
    "green": (45, 145, 91),
}
COLOR_NAMES = tuple(COLORS)
HALVES = tuple(
    tuple(word)
    for word in itertools.product((-1, 1), repeat=4)
    if sum(word) == 0
)
START_XS = (58, 198, 338, 478)
MOVE_DX = 10
MOVE_DY = 9


def _hamming(left: list[int] | tuple[int, ...], right: list[int] | tuple[int, ...]) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    first_halves = list(HALVES)
    rng.shuffle(first_halves)

    target_half = list(first_halves[target_index])
    candidates: list[list[int]] = []
    desired_mismatches = [2, 2, 4]
    distractor_number = 0
    for index in range(4):
        first = list(first_halves[index])
        if index == target_index:
            second = first[:]
        else:
            desired = desired_mismatches[distractor_number]
            options = [list(word) for word in HALVES if _hamming(first, word) == desired]
            second = options[rng.randrange(len(options))]
            distractor_number += 1
        candidates.append(first + second)

    # Over every 16 consecutive seeds, target position and answer color form
    # the full 4x4 grid. Remaining colors are independently permuted.
    answer_color = COLOR_NAMES[(int(seed) // 4) % 4]
    remaining_colors = [name for name in COLOR_NAMES if name != answer_color]
    rng.shuffle(remaining_colors)
    gate_colors = []
    for index in range(4):
        gate_colors.append(answer_color if index == target_index else remaining_colors.pop())
    return {
        "candidate_moves": candidates,
        "gate_colors": gate_colors,
        "base_level": 144 + (int(seed) % 3 - 1) * 7,
    }


def _points(start_x: int, start_y: int, moves: list[int]) -> list[tuple[int, int]]:
    points = [(start_x, start_y)]
    x, y = start_x, start_y
    for move in moves:
        x += MOVE_DX
        y += int(move) * MOVE_DY
        points.append((x, y))
    return points


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    base = int(scene["base_level"])
    moves = [[int(value) for value in word] for word in scene["candidate_moves"]]
    colors = [str(name) for name in scene["gate_colors"]]

    draw.rounded_rectangle((20, 78, 620, 202), radius=18, fill=PANEL, outline=FRAME, width=2)
    draw.text((27, 218), "ONE CONTINUOUS RAIL  •  FIND THE SELF-ECHO STRETCH", fill=TEXT, font=font)

    sections = [_points(x, base, word) for x, word in zip(START_XS, moves, strict=True)]
    draw.line(((30, base), sections[0][0]), fill=CYAN, width=5)
    for index, points in enumerate(sections):
        draw.line(points, fill=CYAN, width=5, joint="curve")
        if index + 1 < len(sections):
            draw.line((points[-1], sections[index + 1][0]), fill=CYAN, width=5)
    draw.line((sections[-1][-1], (610, base)), fill=CYAN, width=5)

    # Equal-size address gates sit just above the still-unbroken cyan rail.
    for name, x in zip(colors, START_XS, strict=True):
        draw.line((x, base - 5, x, base - 2), fill=INK, width=2)
        draw.ellipse((x - 6, base - 18, x + 6, base - 6), fill=COLORS[name], outline=INK, width=1)
    return image


def _mismatches(scene: dict) -> list[int]:
    values = []
    for word in scene["candidate_moves"]:
        values.append(_hamming(word[:4], word[4:]))
    return values


def analytic_gold(scene: dict) -> str:
    scores = _mismatches(scene)
    winners = [index for index, value in enumerate(scores) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene has no unique self-echo stretch")
    return str(scene["gate_colors"][winners[0]])


def margin(scene: dict) -> float:
    ordered = sorted(_mismatches(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    scores = _mismatches(scene)
    return scores.count(0) != 1 or sorted(scores)[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
