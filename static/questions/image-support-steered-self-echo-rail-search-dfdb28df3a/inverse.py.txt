"""Independent final-PNG inverse arm for the self-echo rail world."""

from __future__ import annotations

import numpy as np
from PIL import Image

CYAN = np.array((31, 139, 170), dtype=np.uint8)
COLORS = {
    "red": np.array((211, 63, 63), dtype=np.uint8),
    "gold": np.array((218, 159, 35), dtype=np.uint8),
    "violet": np.array((132, 79, 185), dtype=np.uint8),
    "green": np.array((45, 145, 91), dtype=np.uint8),
}
MOVE_COUNT = 8
MOVE_WIDTH = 10.0


def _component_sizes(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for every 8-connected component."""
    remaining = set(map(tuple, np.argwhere(mask).tolist()))
    components = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        ys = [start[0]]
        xs = [start[1]]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
                        ys.append(neighbor[0])
                        xs.append(neighbor[1])
        components.append((len(xs), min(xs), min(ys), max(xs), max(ys)))
    return sorted(components, reverse=True)


def _marker(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if not 80 <= len(xs) <= 130:
        raise ValueError("a colored address gate is missing or malformed")
    if not 10 <= int(xs.max() - xs.min()) <= 13 or not 10 <= int(ys.max() - ys.min()) <= 13:
        raise ValueError("a colored address gate has invalid shape")
    return float(np.median(xs)), float(np.median(ys))


def _section_word(rail: np.ndarray, start_x: float) -> tuple[int, ...]:
    word = []
    for index in range(MOVE_COUNT):
        left = int(round(start_x + index * MOVE_WIDTH + 2))
        right = int(round(start_x + (index + 1) * MOVE_WIDTH - 2))
        samples_x = []
        samples_y = []
        for x in range(left, right + 1):
            ys = np.flatnonzero(rail[:, x])
            if len(ys):
                samples_x.append(float(x))
                samples_y.append(float(np.median(ys)))
        if len(samples_x) < 4:
            raise ValueError("rail move evidence is interrupted")
        slope = float(np.polyfit(np.asarray(samples_x), np.asarray(samples_y), 1)[0])
        if abs(slope) < 0.55:
            raise ValueError("rail move direction is ambiguous")
        word.append(1 if slope > 0 else -1)
    return tuple(word)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    rail = np.all(array == CYAN, axis=2)
    components = _component_sizes(rail)
    if len(components) != 1:
        raise ValueError("the decisive cyan rail is not one connected trace")
    area, x0, _y0, x1, _y1 = components[0]
    if area < 2700 or x1 - x0 < 570:
        raise ValueError("the cyan rail is missing or materially erased")

    markers = []
    for name, rgb in COLORS.items():
        x, y = _marker(np.all(array == rgb, axis=2))
        markers.append((x, y, name))
    markers.sort()
    gaps = [markers[index + 1][0] - markers[index][0] for index in range(3)]
    if any(not 135.0 <= gap <= 145.0 for gap in gaps):
        raise ValueError("candidate gates are not distinctly and uniformly ordered")

    scores = []
    for x, _y, name in markers:
        word = _section_word(rail, x)
        score = sum(a != b for a, b in zip(word[:4], word[4:], strict=True))
        scores.append((score, name))
    scores.sort()
    if scores[0][0] != 0 or scores[1][0] - scores[0][0] < 2:
        raise ValueError("no unique self-echo stretch with runner-up margin")
    return scores[0][1]
