"""Independent image-only inverse; imports no candidate implementation module."""
from __future__ import annotations

import numpy as np
from PIL import Image

N, CELL = 5, 66
GOLDS = ((240, 174, 38), (235, 164, 45))
ONS = ((37, 62, 92), (63, 54, 91))
OFFS = ((237, 232, 216), (229, 235, 220))

def _dist(a, color):
    return np.linalg.norm(a.astype(float) - np.asarray(color, dtype=float), axis=-1)

def _step(cells):
    out = [[0]*N for _ in range(N)]
    for r in range(N):
        for c in range(N):
            count = 0
            for dr, dc in ((-1,0),(1,0),(0,-1),(0,1)):
                rr, cc = r+dr, c+dc
                if 0 <= rr < N and 0 <= cc < N:
                    count += cells[rr][cc]
            out[r][c] = cells[r][c] ^ int(count == 2)
    return out

def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape != (650, 640, 3):
        raise ValueError("unexpected raster size")
    # Recover the ring's connected rectangular footprint from gold pixels.
    gold_d = np.minimum(_dist(arr, GOLDS[0]), _dist(arr, GOLDS[1]))
    ys, xs = np.where(gold_d < 20)
    if len(xs) < 900:
        raise ValueError("missing gold tracking ring")
    gx0, gx1, gy0, gy1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (38 <= gx1-gx0 <= 52 and 38 <= gy1-gy0 <= 52):
        raise ValueError("ambiguous tracking ring")
    tracked_cx, tracked_cy = (gx0+gx1)//2, (gy0+gy1)//2
    x0, y0 = tracked_cx - 2*CELL - CELL//2, tracked_cy - 2*CELL - CELL//2
    if not (135 <= x0 <= 180 and 180 <= y0 <= 235):
        raise ValueError("board location invalid")
    cells = []
    for r in range(N):
        row = []
        for c in range(N):
            cx, cy = x0+c*CELL+CELL//2, y0+r*CELL+CELL//2
            patch = arr[cy-8:cy+9, cx-8:cx+9]
            on_fraction = min(float(np.mean(_dist(patch, col) < 20)) for col in ONS)
            # min above is intentionally not used for classification; palette is
            # determined jointly by the closest mean flat color below.
            mean = patch.reshape(-1,3).mean(axis=0)
            on_d = min(float(np.linalg.norm(mean-np.asarray(col))) for col in ONS)
            off_d = min(float(np.linalg.norm(mean-np.asarray(col))) for col in OFFS)
            if on_d < 15 and off_d > 60:
                row.append(1)
            elif off_d < 15 and on_d > 60:
                row.append(0)
            else:
                raise ValueError("missing or ambiguous cell interior")
        cells.append(row)
    if not (8 <= sum(map(sum, cells)) <= 17):
        raise ValueError("degenerate or corrupted field")
    cells = _step(_step(cells))
    return "yes" if cells[2][2] else "no"
