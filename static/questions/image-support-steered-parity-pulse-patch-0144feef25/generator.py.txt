"""Deterministic scene and rasterizer for Parity Pulse Patch."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

W, H = 640, 650
N, CELL = 5, 66
BASE_X, BASE_Y = 155, 204
PALETTES = (
    {"on": (37, 62, 92), "off": (237, 232, 216), "gold": (240, 174, 38), "ink": (25, 32, 42), "paper": (250, 248, 241)},
    {"on": (63, 54, 91), "off": (229, 235, 220), "gold": (235, 164, 45), "ink": (29, 34, 43), "paper": (248, 247, 240)},
)

def _step(cells: list[list[int]]) -> list[list[int]]:
    out = [[0] * N for _ in range(N)]
    for r in range(N):
        for c in range(N):
            neighbors = 0
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                rr, cc = r + dr, c + dc
                if 0 <= rr < N and 0 <= cc < N:
                    neighbors += int(cells[rr][cc])
            out[r][c] = int(cells[r][c]) ^ int(neighbors == 2)
    return out

def _replay(scene: dict) -> int:
    cells = [[int(v) for v in row] for row in scene["cells"]]
    cells = _step(_step(cells))
    return cells[int(scene["tracked_row"])][int(scene["tracked_col"])]

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed & 1
    for _ in range(5000):
        cells = [[rng.randrange(2) for _ in range(N)] for _ in range(N)]
        # Prevent degenerate nearly blank/full fields and require mixed evidence in
        # the radius-two tracked neighborhood.
        total = sum(map(sum, cells))
        local = sum(cells[r][c] for r in range(N) for c in range(N) if abs(r-2)+abs(c-2) <= 2)
        scene = {
            "cells": cells, "tracked_row": 2, "tracked_col": 2,
            "palette_variant": rng.randrange(2),
            "offset_x": rng.choice((-12, -4, 4, 12)),
            "offset_y": rng.choice((-10, -2, 6, 14)),
        }
        if 8 <= total <= 17 and 4 <= local <= 9 and _replay(scene) == target:
            return scene
    raise RuntimeError("balanced scene sampling failed")

def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene) else "no"

def margin(scene: dict) -> float:
    p = PALETTES[int(scene["palette_variant"])]
    colors = [p[k] for k in ("on", "off", "gold", "ink", "paper")]
    return float(min(sum((a-b)**2 for a, b in zip(x, y)) ** 0.5 for i, x in enumerate(colors) for y in colors[i+1:]))

def is_quarantined(scene: dict) -> bool:
    return False

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []

def _font(size: int, bold: bool = False):
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()

def render(scene: dict) -> Image.Image:
    p = PALETTES[int(scene["palette_variant"])]
    im = Image.new("RGB", (W, H), p["paper"])
    d = ImageDraw.Draw(im)
    ink, gold = p["ink"], p["gold"]
    d.text((W//2, 25), "PARITY PULSE PATCH", anchor="ma", font=_font(27, True), fill=ink)
    d.rounded_rectangle((64, 62, 576, 154), radius=12, fill=(242, 239, 229), outline=ink, width=3)
    d.text((W//2, 78), "TWO SYNCHRONOUS PULSES", anchor="ma", font=_font(18, True), fill=ink)
    d.text((W//2, 109), "EXACTLY 2 ORTHOGONAL ON NEIGHBORS = FLIP", anchor="ma", font=_font(15, True), fill=ink)
    d.text((W//2, 133), "ANY OTHER COUNT = KEEP", anchor="ma", font=_font(15), fill=ink)
    x0 = BASE_X + int(scene["offset_x"])
    y0 = BASE_Y + int(scene["offset_y"])
    cells = scene["cells"]
    for r in range(N):
        for c in range(N):
            xa, ya = x0 + c*CELL, y0 + r*CELL
            fill = p["on"] if int(cells[r][c]) else p["off"]
            d.rectangle((xa, ya, xa+CELL, ya+CELL), fill=fill, outline=ink, width=4)
    tr, tc = int(scene["tracked_row"]), int(scene["tracked_col"])
    xa, ya = x0 + tc*CELL, y0 + tr*CELL
    d.rounded_rectangle((xa+9, ya+9, xa+CELL-9, ya+CELL-9), radius=10, outline=gold, width=8)
    d.text((W//2, 606), "GOLD RING = TRACKED CELL", anchor="mm", font=_font(17, True), fill=ink)
    return im
