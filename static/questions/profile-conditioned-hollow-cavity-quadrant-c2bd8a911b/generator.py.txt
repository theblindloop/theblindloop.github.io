"""Hollow-cavity oddball search world.

The image is a 2x2 grid of quadrants split by a light divider. Each quadrant
holds exactly one dark closed shape. Three of the four shapes are SOLID filled
disks (fully dark with no empty interior). Exactly one shape is a HOLLOW ring
(an annulus): a filled disk of the same outer size with a circular cavity of
background ink removed from its middle, so a completely enclosed hole is
visible at the disk's centre.

The decision is the quadrant whose shape is hollow (contains the enclosed
interior cavity). Every attribute that decides the answer (which shape has the
background cavity) is drawn in the raster; nothing else in the scene is the
answer. The three solid disks share the same outer radius, so the answer can
not be found by size, colour, position, or ink salience alone: only the hidden
interior cavity identifies the winner.
"""

from __future__ import annotations

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
POLY = (25, 25, 25)
DIV = (205, 205, 205)
MG = SIZE // 2  # 240, split line

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, SIZE // 4),
    "TR": (3 * SIZE // 4, SIZE // 4),
    "BL": (SIZE // 4, 3 * SIZE // 4),
    "BR": (3 * SIZE // 4, 3 * SIZE // 4),
}

MIN_WALL = 13.0   # ring wall thickness in pixels
MAX_WALL = 22.0
_QUARANTINE_THRESH = 15.0


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    hollow = SECTORS[int(rng.integers(0, len(SECTORS)))]
    radius = float(rng.uniform(54.0, 62.0))
    wall = float(rng.uniform(MIN_WALL, MAX_WALL))
    return {"radius": radius, "wall": wall, "hollow_quadrant": hollow}


def _draw_shape(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    r: float,
    wall: float,
    hollow: bool,
) -> None:
    box = [cx - r, cy - r, cx + r, cy + r]
    draw.ellipse(box, fill=POLY)
    if hollow:
        inner = r - wall
        inner_box = [cx - inner, cy - inner, cx + inner, cy + inner]
        draw.ellipse(inner_box, fill=BG)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    draw.line([MG, 0, MG, SIZE], fill=DIV, width=2)
    draw.line([0, MG, SIZE, MG], fill=DIV, width=2)
    r = scene["radius"]
    wall = scene["wall"]
    hollow_quadrant = scene["hollow_quadrant"]
    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        _draw_shape(draw, cx, cy, r, wall, hollow=sector == hollow_quadrant)
    return img


def analytic_gold(scene: dict) -> str:
    return scene["hollow_quadrant"]


def margin(scene: dict) -> float:
    return float(scene["wall"])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < _QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
