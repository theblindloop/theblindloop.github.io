"""Independent pixel-only inverse arm for hollow_cavity_quadrant.

Receives only the final RGB image. It separates each of the four quadrant
cells, thresholds the dark shape ink, and decides for each cell whether the
dark shape is HOLLOW: a hollow shape fully encloses a pocket of background
ink (its interior cavity) that is not connected to the cell border through
background, whereas a solid filled disk leaves no such enclosed background
pocket.

Detects the enclosed background pocket by flood-filling all background
reachable from the cell border (4-connectivity) and checking whether any
background pixel remains unreached. The single cell with an enclosed pocket
is the hollow quadrant. Imports nothing from renderer/prompts/generate/verify.
"""

from __future__ import annotations

import numpy as np

SIZE = 480
MG = SIZE // 2  # 240
DARK_SUM = 300  # poly ink (25,25,25) sums to 75; divider (205,205,205) to 615

SECTORS = ("TL", "TR", "BL", "BR")
_CELL_X = {"TL": (0, MG), "TR": (MG, SIZE), "BL": (0, MG), "BR": (MG, SIZE)}
_CELL_Y = {"TL": (0, MG), "TR": (0, MG), "BL": (MG, SIZE), "BR": (MG, SIZE)}


def _cell_has_cavity(dark: np.ndarray, y0: int, y1: int, x0: int, x1: int) -> bool:
    cell = dark[y0:y1, x0:x1]
    bg = ~cell
    h, w = cell.shape
    reach = np.zeros((h, w), dtype=bool)
    reach[0, :] |= bg[0, :]
    reach[h - 1, :] |= bg[h - 1, :]
    reach[:, 0] |= bg[:, 0]
    reach[:, w - 1] |= bg[:, w - 1]
    while True:
        prop = np.zeros_like(reach)
        prop[1:, :] |= reach[:-1, :]
        prop[:-1, :] |= reach[1:, :]
        prop[:, 1:] |= reach[:, :-1]
        prop[:, :-1] |= reach[:, 1:]
        prop &= bg
        if not (prop & ~reach).any():
            break
        reach |= prop
    return bool((bg & ~reach).any())


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < DARK_SUM
    for sector in SECTORS:
        x0, x1 = _CELL_X[sector]
        y0, y1 = _CELL_Y[sector]
        if _cell_has_cavity(dark, y0, y1, x0, x1):
            return sector
    return SECTORS[0]
