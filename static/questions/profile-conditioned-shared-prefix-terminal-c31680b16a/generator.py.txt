"""Deterministic latent sampler, raster renderer, and analytic graph arm."""

from __future__ import annotations

import random
from collections import deque
from itertools import combinations

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
BACKGROUND = (250, 250, 247)
INK = (28, 31, 36)
PALETTE = {
    "green": (35, 166, 91),
    "purple": (137, 92, 235),
    "red": (226, 67, 75),
    "blue": (43, 111, 218),
    "orange": (240, 137, 35),
}
COLOR_ORDER = {"red": 0, "blue": 1, "orange": 2}
ANSWER_PAIRS = (("red", "blue"), ("red", "orange"), ("blue", "orange"))


def _pair_label(left: str, right: str) -> str:
    ordered = sorted((left, right), key=COLOR_ORDER.__getitem__)
    return f"{ordered[0]} and {ordered[1]}"


def _transform_point(
    point: tuple[float, float], rotation: int, reflect: bool, scale: float, dx: float, dy: float
) -> tuple[int, int]:
    x, y = point[0] - WIDTH / 2, point[1] - HEIGHT / 2
    if reflect:
        x = -x
    for _ in range(rotation):
        x, y = -y, x
    return round(WIDTH / 2 + scale * x + dx), round(HEIGHT / 2 + scale * y + dy)


def sample_scene(seed: int) -> dict:
    """Create an equal-hop rooted tree with a 1--3 link common-prefix gap."""

    rng = random.Random(int(seed))
    prefix_gap = 1 + (int(seed) // 3) % 3
    selected_pair = list(ANSWER_PAIRS[int(seed) % len(ANSWER_PAIRS)])
    remaining_color = next(color for color in COLOR_ORDER if color not in selected_pair)
    rng.shuffle(selected_pair)

    xs = (70, 132, 194, 256, 318, 380, 442, 504, 566)
    points: dict[str, tuple[float, float]] = {}
    kinds: dict[str, tuple[str, str]] = {}
    edges: list[list[str]] = []

    def jitter() -> int:
        return rng.randint(-4, 4)

    def add(node_id: str, level: int, y: float, kind: str, color: str) -> None:
        points[node_id] = (xs[level] + jitter(), y + jitter())
        kinds[node_id] = (kind, color)

    def connect(left: str, right: str) -> None:
        edges.append([left, right])

    add("start", 0, 320, "start", "green")
    add("trunk_1", 1, 318, "junction", "purple")
    add("trunk_2", 2, 322, "junction", "purple")
    add("fork_0", 3, 320, "junction", "purple")
    connect("start", "trunk_1")
    connect("trunk_1", "trunk_2")
    connect("trunk_2", "fork_0")

    # The singleton arm always has eight links from START, matching both paired arms.
    previous = "fork_0"
    singleton_y = (238, 185, 142, 108, 82)
    for level, y in zip(range(4, 9), singleton_y):
        node_id = "terminal_single" if level == 8 else f"single_{level}"
        kind = "terminal" if level == 8 else "junction"
        color = remaining_color if level == 8 else "purple"
        add(node_id, level, y, kind, color)
        connect(previous, node_id)
        previous = node_id

    # The paired arms share prefix_gap additional links after fork_0.
    previous = "fork_0"
    for offset in range(1, prefix_gap + 1):
        level = 3 + offset
        node_id = "fork_1" if offset == prefix_gap else f"pair_shared_{level}"
        y = 320 + 110 * offset / prefix_gap
        add(node_id, level, y, "junction", "purple")
        connect(previous, node_id)
        previous = node_id

    fork_id = previous
    for arm, target_y, terminal_color in (
        ("upper", 300, selected_pair[0]),
        ("lower", 558, selected_pair[1]),
    ):
        previous = fork_id
        first_level = 4 + prefix_gap
        steps = 8 - (3 + prefix_gap)
        for level in range(first_level, 9):
            progress = (level - (3 + prefix_gap)) / steps
            y = 430 + (target_y - 430) * progress
            node_id = f"terminal_{arm}" if level == 8 else f"{arm}_{level}"
            kind = "terminal" if level == 8 else "junction"
            color = terminal_color if level == 8 else "purple"
            add(node_id, level, y, kind, color)
            connect(previous, node_id)
            previous = node_id

    rotation = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    scale = rng.uniform(0.94, 1.0)
    dx = rng.uniform(-8, 8)
    dy = rng.uniform(-8, 8)

    nodes = []
    for node_id, point in points.items():
        x, y = _transform_point(point, rotation, reflect, scale, dx, dy)
        kind, color = kinds[node_id]
        nodes.append({"id": node_id, "x": x, "y": y, "kind": kind, "color": color})
    rng.shuffle(nodes)
    rng.shuffle(edges)
    return {"nodes": nodes, "edges": edges}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    node_map = {node["id"]: node for node in scene["nodes"]}
    for left, right in scene["edges"]:
        a, b = node_map[left], node_map[right]
        draw.line((a["x"], a["y"], b["x"], b["y"]), fill=INK, width=7)
    for node in scene["nodes"]:
        radius = 15 if node["kind"] in {"start", "terminal"} else 12
        x, y = node["x"], node["y"]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE[node["color"]])
    return image


def _path(adjacency: dict[str, set[str]], start: str, goal: str) -> list[str]:
    parents: dict[str, str | None] = {start: None}
    queue = deque([start])
    while queue:
        here = queue.popleft()
        if here == goal:
            break
        for neighbor in adjacency[here]:
            if neighbor not in parents:
                parents[neighbor] = here
                queue.append(neighbor)
    if goal not in parents:
        raise ValueError("terminal is disconnected from START")
    result = []
    here: str | None = goal
    while here is not None:
        result.append(here)
        here = parents[here]
    return list(reversed(result))


def _shared_links(left: list[str], right: list[str]) -> int:
    shared_nodes = 0
    for a, b in zip(left, right):
        if a != b:
            break
        shared_nodes += 1
    return max(0, shared_nodes - 1)


def _pair_scores(nodes: list[dict], edges: list[list[str]]) -> dict[str, int]:
    adjacency = {node["id"]: set() for node in nodes}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    start = next(node["id"] for node in nodes if node["color"] == "green")
    terminals = {
        node["color"]: node["id"]
        for node in nodes
        if node["color"] in {"red", "blue", "orange"}
    }
    paths = {color: _path(adjacency, start, node_id) for color, node_id in terminals.items()}
    return {
        _pair_label(left, right): _shared_links(paths[left], paths[right])
        for left, right in combinations(("red", "blue", "orange"), 2)
    }


def analytic_gold(scene: dict) -> str:
    scores = _pair_scores(scene["nodes"], scene["edges"])
    ordered = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ordered[0][1] == ordered[1][1]:
        raise ValueError("shared-prefix winner is tied")
    return ordered[0][0]


def margin(scene: dict) -> float:
    scores = sorted(_pair_scores(scene["nodes"], scene["edges"]).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {"nodes": list(reversed(scene["nodes"])), "edges": list(reversed(scene["edges"]))}
    return [("reverse_storage_order", twin)]
