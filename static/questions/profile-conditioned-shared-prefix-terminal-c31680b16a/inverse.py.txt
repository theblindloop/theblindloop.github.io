"""Independent PNG-only inverse arm for rooted shared-prefix comparison.

The implementation has no access to latent node IDs or coordinates. It segments
colored circle components and disconnected black link strokes, binds each stroke
to its two nearest visible circles, reconstructs the tree, and compares paths.
"""

from __future__ import annotations

from collections import deque
from itertools import combinations

import numpy as np
from PIL import Image

INK = np.array((28, 31, 36), dtype=np.int16)
COLORS = {
    "green": np.array((35, 166, 91), dtype=np.int16),
    "purple": np.array((137, 92, 235), dtype=np.int16),
    "red": np.array((226, 67, 75), dtype=np.int16),
    "blue": np.array((43, 111, 218), dtype=np.int16),
    "orange": np.array((240, 137, 35), dtype=np.int16),
}
COLOR_ORDER = {"red": 0, "blue": 1, "orange": 2}


def _near_color(array: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    """Return 8-connected foreground components as arrays of (y, x) pixels."""

    remaining = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    found: list[np.ndarray] = []
    while remaining:
        first = remaining.pop()
        stack = [first]
        points = [first]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
                        points.append(neighbor)
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _detect_nodes(array: np.ndarray) -> list[dict]:
    nodes: list[dict] = []
    for color_name, rgb in COLORS.items():
        components = _components(_near_color(array, rgb), minimum=180)
        expected = None if color_name == "purple" else 1
        if expected is not None and len(components) != expected:
            raise ValueError(f"expected one {color_name} circle, found {len(components)}")
        if color_name == "purple" and not components:
            raise ValueError("no purple junction circles found")
        for component in components:
            cy, cx = component.mean(axis=0)
            nodes.append({"x": float(cx), "y": float(cy), "color": color_name})
    return nodes


def _detect_edges(array: np.ndarray, nodes: list[dict]) -> set[tuple[int, int]]:
    components = _components(_near_color(array, INK, tolerance=14), minimum=55)
    edges: set[tuple[int, int]] = set()
    for component in components:
        ys, xs = component[:, 0], component[:, 1]
        distances = []
        for index, node in enumerate(nodes):
            squared = (xs - node["x"]) ** 2 + (ys - node["y"]) ** 2
            distances.append((float(squared.min()), index))
        distances.sort()
        if len(distances) < 2 or distances[1][0] > 25.0**2:
            raise ValueError("a black link cannot be bound to two circle endpoints")
        edge = tuple(sorted((distances[0][1], distances[1][1])))
        if edge[0] == edge[1] or edge in edges:
            raise ValueError("duplicate or self-loop link recovered")
        edges.add(edge)
    if len(edges) != len(nodes) - 1:
        raise ValueError(f"recovered {len(nodes)} nodes but {len(edges)} links; not a tree")
    return edges


def _path(adjacency: dict[int, set[int]], start: int, goal: int) -> list[int]:
    parent: dict[int, int | None] = {start: None}
    queue = deque([start])
    while queue:
        here = queue.popleft()
        if here == goal:
            break
        for neighbor in adjacency[here]:
            if neighbor not in parent:
                parent[neighbor] = here
                queue.append(neighbor)
    if goal not in parent:
        raise ValueError("colored terminal is disconnected from green START")
    result = []
    here: int | None = goal
    while here is not None:
        result.append(here)
        here = parent[here]
    return list(reversed(result))


def _shared_links(left: list[int], right: list[int]) -> int:
    shared_nodes = 0
    for a, b in zip(left, right):
        if a != b:
            break
        shared_nodes += 1
    return max(0, shared_nodes - 1)


def _pair_label(left: str, right: str) -> str:
    ordered = sorted((left, right), key=COLOR_ORDER.__getitem__)
    return f"{ordered[0]} and {ordered[1]}"


def measure_graph(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"))
    nodes = _detect_nodes(array)
    edges = _detect_edges(array, nodes)
    adjacency = {index: set() for index in range(len(nodes))}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    start = next(index for index, node in enumerate(nodes) if node["color"] == "green")
    terminals = {
        node["color"]: index
        for index, node in enumerate(nodes)
        if node["color"] in {"red", "blue", "orange"}
    }
    if set(terminals) != {"red", "blue", "orange"}:
        raise ValueError("terminal palette is incomplete")
    paths = {color: _path(adjacency, start, target) for color, target in terminals.items()}
    scores = {
        _pair_label(left, right): _shared_links(paths[left], paths[right])
        for left, right in combinations(("red", "blue", "orange"), 2)
    }
    return {"node_count": len(nodes), "edge_count": len(edges), "scores": scores}


def decision_from_image(image: Image.Image) -> str:
    """Return the terminal pair using only visible PNG pixels."""

    scores = measure_graph(image)["scores"]
    ordered = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ordered[0][1] == ordered[1][1]:
        raise ValueError("top two pixel-derived shared-prefix counts are tied")
    return ordered[0][0]
