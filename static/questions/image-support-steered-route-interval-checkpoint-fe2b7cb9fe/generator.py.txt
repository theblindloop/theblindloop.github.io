"""Deterministic renderer and analytic arm for a gated route interval question."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE_RGB = (30, 55, 78)
GREEN_RGB = (36, 174, 103)
RED_RGB = (218, 64, 62)
AMBER_RGB = (234, 150, 31)
VIOLET_RGB = (145, 77, 190)
GATE_RGB = (128, 151, 168)
RING_RGB = (18, 25, 34)
QUARANTINE_MARGIN = 25.0

_SHAPES: tuple[tuple[tuple[float, float], ...], ...] = (
    (
        (0.12, 0.86),
        (0.12, 0.20),
        (0.19, 0.12),
        (0.81, 0.12),
        (0.88, 0.20),
        (0.88, 0.86),
    ),
    (
        (0.12, 0.14),
        (0.12, 0.80),
        (0.19, 0.88),
        (0.81, 0.88),
        (0.88, 0.80),
        (0.88, 0.14),
    ),
    (
        (0.12, 0.14),
        (0.86, 0.14),
        (0.88, 0.22),
        (0.88, 0.47),
        (0.16, 0.53),
        (0.12, 0.61),
        (0.12, 0.86),
        (0.86, 0.86),
    ),
)


def _distance(a: Iterable[float], b: Iterable[float]) -> float:
    ax, ay = (float(value) for value in a)
    bx, by = (float(value) for value in b)
    return math.hypot(ax - bx, ay - by)


def _points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in scene["route_points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(cumulative[-1] + _distance(first, second))
    return cumulative, float(cumulative[-1])


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span <= 0.0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0)
    return points[-1]


def _transform_point(
    point: tuple[float, float],
    *,
    orientation: int,
    mirror_x: bool,
    mirror_y: bool,
    scale: float,
    dx: float,
    dy: float,
) -> tuple[float, float]:
    x, y = point
    if mirror_x:
        x = WIDTH - x
    if mirror_y:
        y = HEIGHT - y
    x = 256.0 + (x - 256.0) * scale
    y = 256.0 + (y - 256.0) * scale
    if orientation == 1:
        x, y = HEIGHT - y, x
    elif orientation == 2:
        x, y = WIDTH - x, HEIGHT - y
    elif orientation == 3:
        x, y = y, WIDTH - x
    return x + dx, y + dy


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced long route and one checkpoint inside a visible interval."""

    rng = random.Random(int(seed))
    shape = _SHAPES[rng.randrange(len(_SHAPES))]
    # Keep the path well inside the canvas, then apply only mild transforms so
    # all markers remain separated after rasterization.
    base = [(52.0 + 408.0 * x, 52.0 + 408.0 * y) for x, y in shape]
    scale = rng.uniform(0.96, 1.01)
    orientation = rng.randrange(4)
    mirror_x = bool(rng.randrange(2))
    mirror_y = bool(rng.randrange(2))
    dx = rng.uniform(-6.0, 6.0)
    dy = rng.uniform(-6.0, 6.0)
    transformed = [
        _transform_point(
            point,
            orientation=orientation,
            mirror_x=mirror_x,
            mirror_y=mirror_y,
            scale=scale,
            dx=dx,
            dy=dy,
        )
        for point in base
    ]
    # The transform is shared by all points; add a tiny deterministic bend
    # only to interior vertices, retaining a simple non-self-intersecting route.
    route_points: list[list[float]] = []
    for index, (x, y) in enumerate(transformed):
        if 0 < index < len(transformed) - 1:
            x += rng.uniform(-2.5, 2.5)
            y += rng.uniform(-2.5, 2.5)
        route_points.append([round(x, 4), round(y, 4)])
    points = [(float(x), float(y)) for x, y in route_points]
    _, total = _route_lengths(points)
    if total < 600.0:
        raise ValueError("sampled route is unexpectedly short")

    gate_a = rng.uniform(0.18, 0.25)
    gate_b = rng.uniform(0.75, 0.82)
    inside = rng.uniform(0.43, 0.57)
    if rng.randrange(2) == 0:
        outside = rng.uniform(0.055, 0.095)
    else:
        outside = rng.uniform(0.905, 0.945)
    inside_label = "amber" if rng.randrange(2) == 0 else "violet"
    outside_label = "violet" if inside_label == "amber" else "amber"
    return {
        "route_points": route_points,
        "gate_fractions": [round(gate_a, 5), round(gate_b, 5)],
        "checkpoint_fractions": {
            inside_label: round(inside, 5),
            outside_label: round(outside, 5),
        },
        "route_width": round(rng.uniform(7.2, 9.0), 3),
        "marker_radius": round(rng.uniform(9.0, 11.0), 3),
    }


def _scaled_points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _draw_route(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], width: int) -> None:
    scaled = _scaled_points(points)
    radius = max(1, width // 2)
    draw.line(scaled, fill=ROUTE_RGB, width=width, joint="curve")
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ROUTE_RGB)


def _draw_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    inner = max(1, round(radius * SCALE))
    outer = inner + 3 * SCALE
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=RING_RGB)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize only visible route, marker, and color fields."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    route_width = max(1, round(float(scene["route_width"]) * SCALE))
    marker_radius = float(scene["marker_radius"])
    _draw_route(draw, points, route_width)

    for fraction in scene["gate_fractions"]:
        _draw_marker(draw, _point_at_fraction(points, float(fraction)), GATE_RGB, marker_radius)
    colors = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
    for label, fraction in scene["checkpoint_fractions"].items():
        _draw_marker(draw, _point_at_fraction(points, float(fraction)), colors[label], marker_radius)
    _draw_marker(draw, points[0], GREEN_RGB, marker_radius)
    _draw_marker(draw, points[-1], RED_RGB, marker_radius)

    # Preserve a dark, connected centerline through every marker. The colored
    # annuli remain visible around the thin centerline for pixel segmentation.
    _draw_route(draw, points, route_width)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    gates = sorted(float(value) for value in scene["gate_fractions"])
    if len(gates) != 2 or not gates[0] < gates[1]:
        raise ValueError("scene needs two ordered gates")
    inside = [
        label
        for label, fraction in scene["checkpoint_fractions"].items()
        if gates[0] < float(fraction) < gates[1]
    ]
    if len(inside) != 1:
        raise ValueError("scene needs exactly one checkpoint inside the gate interval")
    return str(inside[0])


def margin(scene: dict[str, Any]) -> float:
    points = _points(scene)
    _, total = _route_lengths(points)
    gates = sorted(float(value) for value in scene["gate_fractions"])
    distances = [
        abs(float(fraction) - gate)
        for fraction in scene["checkpoint_fractions"].values()
        for gate in gates
    ]
    return float(min(distances) * total)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All label-bearing geometry is visible in the PNG; no hidden alias is declared.
    return []
