"""Deterministic switchboard-strip sampler and raster renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

ROWS = 10
COLS = 8
LABELS = ("rising_staircase", "falling_staircase", "valley", "peak")

# A base strip is a two-cell vertical dash. Every label uses every base strip
# exactly once, so the unordered pieces and total ink are exactly class-balanced.
ORDERS: dict[str, tuple[int, ...]] = {
    "rising_staircase": (7, 6, 5, 4, 3, 2, 1, 0),
    "falling_staircase": (0, 1, 2, 3, 4, 5, 6, 7),
    "valley": (0, 2, 4, 6, 7, 5, 3, 1),
    "peak": (7, 5, 3, 1, 0, 2, 4, 6),
}

ROUTE_COLORS = (
    (224, 161, 24),
    (229, 76, 38),
    (190, 45, 128),
    (121, 69, 190),
    (38, 100, 205),
    (24, 160, 178),
    (39, 151, 82),
    (145, 88, 39),
)
TILE_PALETTES = (
    {"dark": (27, 48, 66), "light": (216, 227, 229)},
    {"dark": (68, 42, 91), "light": (231, 221, 232)},
    {"dark": (43, 73, 51), "light": (224, 231, 211)},
)
BACKGROUNDS = ((247, 244, 236), (238, 244, 246), (245, 239, 244))
PANEL = (226, 229, 226)
FRAME = (55, 61, 67)
CANVAS = (760, 560)


def _strip_cells(strip_id: int) -> tuple[int, ...]:
    strip_id = int(strip_id)
    if not 0 <= strip_id < COLS:
        raise ValueError("strip identifier is outside 0..7")
    return tuple(int(row in (strip_id, strip_id + 1)) for row in range(ROWS))


def _canonical_order(scene: dict) -> tuple[int, ...]:
    return ORDERS[str(scene["label"])]


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random((seed ^ 0xCAB1_E5A7_91D3) & 0xFFFFFFFFFFFFFFFF)
    block = seed // len(LABELS)
    label = LABELS[seed % len(LABELS)]
    source_order = list(range(COLS))
    rng.shuffle(source_order)
    route_colors = list(range(COLS))
    rng.shuffle(route_colors)
    return {
        "label": label,
        "source_order": source_order,
        "route_colors": route_colors,
        "cell_width": (38, 42, 46)[block % 3],
        "cell_height": (20, 22, 24)[(block // 3) % 3],
        "column_gap": (22, 25, 28)[(block // 5) % 3],
        "origin_x": (62, 74, 86)[(block // 7) % 3],
        "vertical_shift": (0, 8, 16)[(block // 11) % 3],
        "tile_palette_index": block % len(TILE_PALETTES),
        "background_index": (block // 2) % len(BACKGROUNDS),
    }


def _geometry(scene: dict) -> tuple[list[float], float, float, float]:
    width = int(scene["cell_width"])
    gap = int(scene["column_gap"])
    origin = int(scene["origin_x"])
    centers = [origin + slot * (width + gap) + (width - 1) / 2.0 for slot in range(COLS)]
    shift = int(scene["vertical_shift"])
    return centers, float(42 + shift), float(220 + shift), float(252 + shift)


def render(scene: dict) -> Image.Image:
    label_order = _canonical_order(scene)
    source_order = [int(value) for value in scene["source_order"]]
    route_colors = [int(value) for value in scene["route_colors"]]
    if sorted(source_order) != list(range(COLS)):
        raise ValueError("source_order must be a permutation of eight strips")
    if sorted(value % COLS for value in route_colors) != list(range(COLS)):
        raise ValueError("route_colors must be a permutation modulo eight")

    background = BACKGROUNDS[int(scene["background_index"]) % len(BACKGROUNDS)]
    palette = TILE_PALETTES[int(scene["tile_palette_index"]) % len(TILE_PALETTES)]
    image = Image.new("RGB", CANVAS, background)
    draw = ImageDraw.Draw(image)
    centers, top_y, bottom_y, tile_top = _geometry(scene)
    left = int(min(centers) - int(scene["cell_width"]) / 2 - 20)
    right = int(max(centers) + int(scene["cell_width"]) / 2 + 20)
    draw.rounded_rectangle((left, int(top_y - 22), right, int(bottom_y + 18)), radius=12,
                           fill=PANEL, outline=FRAME, width=2)
    draw.line((centers[0] - 13, top_y, centers[-1] + 13, top_y), fill=FRAME, width=3)

    routes: list[tuple[list[tuple[float, float]], tuple[int, int, int]]] = []
    for destination, strip_id in enumerate(label_order):
        source = source_order.index(strip_id)
        color = ROUTE_COLORS[route_colors[destination] % COLS]
        routes.append(([(centers[destination], top_y), (centers[source], bottom_y)], color))
    for points, _ in routes:
        draw.line(points, fill=(250, 249, 245), width=9)
    for points, color in routes:
        draw.line(points, fill=color, width=5)
    radius = 10
    for points, color in routes:
        for x_value, y_value in (points[0], points[-1]):
            draw.ellipse((x_value - radius, y_value - radius, x_value + radius,
                          y_value + radius), fill=color, outline=(250, 249, 245), width=2)

    cell_width = int(scene["cell_width"])
    cell_height = int(scene["cell_height"])
    row_gap = 3
    for source, strip_id in enumerate(source_order):
        x0 = int(round(centers[source] - (cell_width - 1) / 2.0))
        for row, value in enumerate(_strip_cells(strip_id)):
            y0 = int(round(tile_top + row * (cell_height + row_gap)))
            draw.rectangle((x0, y0, x0 + cell_width - 1, y0 + cell_height - 1),
                           fill=palette["dark" if value else "light"])
        bottom = int(round(tile_top + ROWS * cell_height + (ROWS - 1) * row_gap - 1))
        draw.rectangle((x0 - 3, int(tile_top) - 3, x0 + cell_width + 2, bottom + 3),
                       outline=FRAME, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["label"])


def _template(label: str) -> tuple[tuple[int, ...], ...]:
    columns = [_strip_cells(strip_id) for strip_id in ORDERS[label]]
    return tuple(tuple(columns[column][row] for column in range(COLS)) for row in range(ROWS))


def margin(scene: dict) -> float:
    target = _template(str(scene["label"]))
    gaps = []
    for name in LABELS:
        if name == str(scene["label"]):
            continue
        rival = _template(name)
        gaps.append(sum(target[r][c] != rival[r][c] for r in range(ROWS) for c in range(COLS)))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0 or int(scene["cell_width"]) < 34 or int(scene["cell_height"]) < 18


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["route_colors"] = [int(value) + COLS for value in scene["route_colors"]]
    twin["tile_palette_index"] = int(scene["tile_palette_index"]) + len(TILE_PALETTES)
    twin["background_index"] = int(scene["background_index"]) + len(BACKGROUNDS)
    return [("add_full_visible_palette_periods", twin)]
