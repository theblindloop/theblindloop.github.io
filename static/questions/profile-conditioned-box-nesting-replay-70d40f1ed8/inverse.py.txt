"""Scene-free pixel inverse arm for box_nesting_replay.

Receives only the rendered PNG.  It re-finds the separator band, reads the drawn
containment forest in the tray from the nesting of the coloured borders, reads
the ordered row of cards and the source and target colour drawn on each, replays
the moves in drawn left-to-right order, and reports which box then directly
holds the yellow box and how deep it sits.

It imports no renderer, prompt, generator, verifier or gold code and never sees
a scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "box_nesting_replay-oracle-0.1.0"

# reference palette, matched with tolerance; these are the colour names the
# public question already uses to talk about the picture
_REF = {
    "yellow": (245, 196, 0),
    "red": (214, 45, 45),
    "green": (34, 150, 72),
    "blue": (36, 86, 200),
    "purple": (140, 58, 170),
}
_SEP = (90, 90, 90)
_CREAM = (243, 233, 210)
_TOL = 40
TRAY_NAME = "tray"


def _near(arr: np.ndarray, ref) -> np.ndarray:
    return np.all(np.abs(arr.astype(np.int16) - np.array(ref, dtype=np.int16)) <= 12, axis=-1)


def _bbox(mask: np.ndarray):
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return None
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _inside(a, b) -> bool:
    return a[0] > b[0] and a[1] > b[1] and a[2] < b[2] and a[3] < b[3]


def _split(arr: np.ndarray) -> tuple[int, int]:
    """Row index of the full-width separator band."""
    sep = _near(arr, _SEP)
    frac = sep.mean(axis=1)
    rows = np.nonzero(frac > 0.5)[0]
    if rows.size == 0:
        raise ValueError("no separator band found")
    return int(rows.min()), int(rows.max())


def _boxes_in(region: np.ndarray) -> dict[str, tuple]:
    found = {}
    for name, ref in _REF.items():
        bb = _bbox(_near(region, ref))
        if bb is not None:
            found[name] = bb
    return found


def _read_cards(strip: np.ndarray) -> list[tuple[str, str]]:
    cream = _near(strip, _CREAM)
    cols = np.nonzero(cream.any(axis=0))[0]
    if cols.size == 0:
        raise ValueError("no cards found")
    runs, start, prev = [], int(cols[0]), int(cols[0])
    for c in cols[1:]:
        c = int(c)
        if c != prev + 1:
            runs.append((start, prev))
            start = c
        prev = c
    runs.append((start, prev))

    cards = []
    for x0, x1 in runs:
        sub = strip[:, x0 : x1 + 1]
        swatches = []
        for name, ref in _REF.items():
            bb = _bbox(_near(sub, ref))
            if bb is not None:
                swatches.append(((bb[0] + bb[2]) / 2.0, name))
        if len(swatches) != 2:
            raise ValueError(f"card does not carry exactly two colours: {swatches}")
        swatches.sort()
        cards.append((swatches[0][1], swatches[1][1]))
    return cards


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    lo, hi = _split(arr)
    tray_region = arr[:lo]
    strip = arr[hi + 1 :]

    boxes = _boxes_in(tray_region)
    if len(boxes) != len(_REF):
        raise ValueError(f"expected {len(_REF)} boxes, found {sorted(boxes)}")

    # direct container = the smallest drawn box that strictly encloses it
    parents: dict[str, str] = {}
    for name, bb in boxes.items():
        holders = [
            ((ob[2] - ob[0]) * (ob[3] - ob[1]), other)
            for other, ob in boxes.items()
            if other != name and _inside(bb, ob)
        ]
        parents[name] = min(holders)[1] if holders else TRAY_NAME

    for src, dst in _read_cards(strip):
        parents[src] = dst

    depth, cur, seen = 0, parents["yellow"], set()
    while cur != TRAY_NAME:
        if cur in seen:
            raise ValueError("cyclic containment recovered from the image")
        seen.add(cur)
        depth += 1
        cur = parents[cur]
    return f"{parents['yellow']}|{depth}"
