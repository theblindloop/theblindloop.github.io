"""box_nesting_replay: containment-tree replay.

A tray holds five coloured boxes drawn nested inside one another, so the picture
shows a forest of containment.  Below the tray an ordered row of cream cards
each names a source colour and a target colour: "take that box out, with
everything still inside it, and put it directly inside the target box".  The
tray after the moves is never drawn.  The answer is which box directly contains
the yellow box once every card has been applied, and how many boxes it is then
inside.

The scene is stored as unordered bags: a bag of boxes each carrying its own
drawn parent and its own sibling rank, and a bag of cards each carrying its own
drawn slot.  Nothing about the drawing or the answer depends on Python list
order, which ``latent_symmetries`` exercises directly.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "box_nesting_replay-renderer-0.1.0"

CANVAS = (900, 560)
TRAY = (40, 28, 858, 348)          # outer edge of the drawn tray
SEP_Y = (362, 369)                 # full-width separator band
CARD_TOP = 392
CARD_W, CARD_H = 134, 88
SWATCH = 34
STROKE = 4
LEAF_W, LEAF_H = 112, 72
N_CARDS = 4

BG = (255, 255, 255)
INK = (20, 20, 20)
SEP = (90, 90, 90)
CREAM = (243, 233, 210)
ARROW = (120, 120, 120)

COLORS = {
    "yellow": (245, 196, 0),
    "red": (214, 45, 45),
    "green": (34, 150, 72),
    "blue": (36, 86, 200),
    "purple": (140, 58, 170),
}
TRACKED = "yellow"
BOXES = (TRACKED, "red", "green", "blue", "purple")
TRAY_NAME = "tray"
# a card can only put a box *into* a box, so the queried yellow box always ends
# up inside one of the four other boxes; the tray is a possible starting place
# and a possible container for other boxes, never a possible answer
PARENT_CLASSES = ("red", "green", "blue", "purple")

QUARANTINE_THRESHOLD = 6.0


# --------------------------------------------------------------------------
# scene semantics
# --------------------------------------------------------------------------
def _parent_map(scene: dict) -> dict[str, str]:
    return {b["color"]: b["parent"] for b in scene["boxes"]}


def _ranks(scene: dict) -> dict[str, int]:
    return {b["color"]: b["rank"] for b in scene["boxes"]}


def _cards_in_order(scene: dict) -> list[dict]:
    return sorted(scene["cards"], key=lambda c: c["slot"])


def _ancestors(parents: dict[str, str], node: str) -> list[str]:
    chain, cur = [], parents.get(node, TRAY_NAME)
    while cur != TRAY_NAME:
        chain.append(cur)
        cur = parents[cur]
    return chain


def _legal(parents: dict[str, str], src: str, dst: str) -> bool:
    """A move is legal when the target is not inside the box being moved."""
    if src == dst:
        return False
    return src not in _ancestors(parents, dst)


def _replay(scene: dict, upto: int | None = None) -> dict[str, str]:
    parents = _parent_map(scene)
    cards = _cards_in_order(scene)
    if upto is not None:
        cards = cards[:upto]
    for card in cards:
        parents[card["src"]] = card["dst"]
    return parents


def _consequences(parents: dict[str, str]) -> tuple[str, int]:
    return parents[TRACKED], len(_ancestors(parents, TRACKED))


def analytic_gold(scene: dict) -> str:
    parent, depth = _consequences(_replay(scene))
    return f"{parent}|{depth}"


# --------------------------------------------------------------------------
# layout
# --------------------------------------------------------------------------
def _layout(scene: dict) -> dict:
    """Deterministic geometry.  Sibling order comes from each box's own rank,
    never from bag order, so re-ordering the bags cannot move a pixel."""
    parents = _parent_map(scene)
    ranks = _ranks(scene)
    pad = scene["pad"]
    gap = scene["gap"]

    kids: dict[str, list[str]] = {p: [] for p in (TRAY_NAME,) + BOXES}
    for color in BOXES:
        kids[parents[color]].append(color)
    for key in kids:
        kids[key].sort(key=lambda c: ranks[c])

    def size(node: str) -> tuple[int, int]:
        ch = kids[node]
        if not ch:
            return LEAF_W, LEAF_H
        sizes = [size(c) for c in ch]
        w = 2 * pad + sum(s[0] for s in sizes) + gap * (len(ch) - 1)
        h = 2 * pad + max(s[1] for s in sizes)
        return max(w, LEAF_W), max(h, LEAF_H)

    rects: dict[str, tuple[int, int, int, int]] = {}

    def place(node: str, x: int, y: int) -> None:
        w, h = size(node)
        rects[node] = (x, y, x + w - 1, y + h - 1)
        cx = x + pad
        for child in kids[node]:
            place(child, cx, y + pad)
            cx += size(child)[0] + gap

    roots = kids[TRAY_NAME]
    sizes = [size(r) for r in roots]
    total_w = sum(s[0] for s in sizes) + gap * (len(roots) - 1)
    total_h = max(s[1] for s in sizes)
    tx0, ty0, tx1, ty1 = TRAY
    x = (tx0 + tx1) // 2 - total_w // 2
    y = (ty0 + ty1) // 2 - total_h // 2
    fits = total_w <= (tx1 - tx0) - 2 * pad and total_h <= (ty1 - ty0) - 2 * pad
    for root, s in zip(roots, sizes):
        place(root, x, y)
        x += s[0] + gap

    cards = _cards_in_order(scene)
    card_gap = scene["card_gap"]
    strip_w = len(cards) * CARD_W + card_gap * (len(cards) - 1)
    cx = CANVAS[0] // 2 - strip_w // 2
    card_rects = []
    for _ in cards:
        card_rects.append((cx, CARD_TOP, cx + CARD_W - 1, CARD_TOP + CARD_H - 1))
        cx += CARD_W + card_gap

    return {"rects": rects, "card_rects": card_rects, "fits": fits, "kids": kids}


def _swatch_boxes(card_rect) -> tuple[tuple, tuple]:
    x0, y0, x1, y1 = card_rect
    cy = (y0 + y1) // 2
    left = (x0 + 14, cy - SWATCH // 2, x0 + 14 + SWATCH - 1, cy - SWATCH // 2 + SWATCH - 1)
    right = (x1 - 13 - SWATCH + 1, left[1], x1 - 13, left[3])
    return left, right


# --------------------------------------------------------------------------
# margin / quarantine
# --------------------------------------------------------------------------
def margin(scene: dict) -> float:
    """Smallest drawn clearance the reader must resolve: the white gap between a
    box border and the border of the box that holds it, the gap between boxes
    that hold each other in neither direction, and the gaps that separate the
    cards and the two swatches inside a card."""
    lay = _layout(scene)
    rects, kids = lay["rects"], lay["kids"]
    clearances = []
    for node in (TRAY_NAME,) + BOXES:
        px0, py0, px1, py1 = rects[node] if node != TRAY_NAME else TRAY
        prev_x1 = None
        for child in kids[node]:
            cx0, cy0, cx1, cy1 = rects[child]
            clearances.append(min(cx0 - px0, cy0 - py0, px1 - cx1, py1 - cy1) - STROKE)
            if prev_x1 is not None:
                clearances.append(cx0 - prev_x1 - 1)
            prev_x1 = cx1
    prev = None
    for rect in lay["card_rects"]:
        if prev is not None:
            clearances.append(rect[0] - prev - 1)
        prev = rect[2]
        left, right = _swatch_boxes(rect)
        clearances.append(right[0] - left[2] - 1)
        clearances.append(left[0] - rect[0] - 1)
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _try_scene(rng: random.Random, target_parent: str) -> dict | None:
    order = list(BOXES)
    rng.shuffle(order)
    parents: dict[str, str] = {}
    placed: list[str] = []
    for color in order:
        pool = [TRAY_NAME] + [p for p in placed if len(_ancestors(parents, p)) < 3]
        parents[color] = rng.choice(pool)
        placed.append(color)

    initial = dict(parents)
    cards = []
    work = dict(parents)
    for slot in range(N_CARDS):
        options = [
            (s, d)
            for s in BOXES
            for d in BOXES
            if _legal(work, s, d) and work[s] != d and len(_ancestors(work, d)) < 3
        ]
        if not options:
            return None
        src, dst = rng.choice(options)
        cards.append({"slot": slot, "src": src, "dst": dst})
        work[src] = dst

    tight = rng.random() < 0.22
    scene = {
        "canvas": list(CANVAS),
        "boxes": [
            {"color": c, "parent": initial[c], "rank": r}
            for c, r in zip(BOXES, rng.sample(range(100), len(BOXES)))
        ],
        "cards": cards,
        "pad": rng.randint(7, 9) if tight else rng.randint(11, 20),
        "gap": rng.randint(14, 26),
        "card_gap": rng.randint(24, 38),
    }
    rng.shuffle(scene["boxes"])
    rng.shuffle(scene["cards"])

    lay = _layout(scene)
    if not lay["fits"]:
        return None
    parent, _ = _consequences(_replay(scene))
    if parent != target_parent:
        return None
    # the drawn program must be load bearing: erasing the last card must move
    # the answer, so the picture cannot be answered by ignoring the cards
    if analytic_gold(scene) == f"{_consequences(_replay(scene, N_CARDS - 1))[0]}|" \
            f"{_consequences(_replay(scene, N_CARDS - 1))[1]}":
        return None
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = PARENT_CLASSES[seed % len(PARENT_CLASSES)]
    for _ in range(4000):
        scene = _try_scene(rng, target)
        if scene is not None:
            return scene
    raise RuntimeError(f"no scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    lay = _layout(scene)
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)

    d.rectangle(TRAY, outline=INK, width=STROKE)
    for color in BOXES:
        d.rectangle(lay["rects"][color], outline=COLORS[color], width=STROKE)

    d.rectangle((0, SEP_Y[0], scene["canvas"][0] - 1, SEP_Y[1]), fill=SEP)

    for card, rect in zip(_cards_in_order(scene), lay["card_rects"]):
        d.rectangle(rect, fill=CREAM)
        left, right = _swatch_boxes(rect)
        d.rectangle(left, fill=COLORS[card["src"]])
        d.rectangle(right, fill=COLORS[card["dst"]])
        ay = (rect[1] + rect[3]) // 2
        d.rectangle((left[2] + 5, ay - 2, right[0] - 10, ay + 2), fill=INK)
        d.polygon(
            [(right[0] - 4, ay), (right[0] - 14, ay - 8), (right[0] - 14, ay + 8)],
            fill=INK,
        )

    first, last = lay["card_rects"][0], lay["card_rects"][-1]
    ay = last[3] + 26
    d.rectangle((first[0], ay - 3, last[2] - 18, ay + 3), fill=ARROW)
    d.polygon([(last[2], ay), (last[2] - 22, ay - 14), (last[2] - 22, ay + 14)], fill=ARROW)
    return img


# --------------------------------------------------------------------------
# declared latent aliases
# --------------------------------------------------------------------------
def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Both scene bags are unordered.  Re-ordering them must redraw a
    byte-identical raster and leave the answer fixed, which catches any gold or
    layout keyed to Python list position instead of to the drawn parent, the
    drawn sibling rank, or the drawn card slot."""
    def rebag(boxes_perm, cards_perm) -> dict:
        alt = dict(scene)
        alt["boxes"] = boxes_perm
        alt["cards"] = cards_perm
        return alt

    b, c = list(scene["boxes"]), list(scene["cards"])
    return [
        ("boxes_reversed", rebag(b[::-1], c)),
        ("cards_reversed", rebag(b, c[::-1])),
        ("boxes_rotated", rebag(b[2:] + b[:2], c)),
        ("both_reordered", rebag(b[::-1], c[1:] + c[:1])),
    ]
