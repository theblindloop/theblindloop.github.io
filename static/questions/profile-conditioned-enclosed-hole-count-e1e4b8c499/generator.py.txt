"""Enclosed-hole-count world: figure-ground topological hole count.

A single connected dark (blue) figure fills part of the canvas. Inside it the
renderer carves out ``hole_count`` separate white islands ("holes") that are
fully enclosed by the blue figure and never touch the surrounding white
background. The decision is the global topological property of how many
enclosed white holes the figure contains.

The renderer owns latent sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
figure-ground decision from the final PNG alone by flood-filling the white
background from the image border and counting the enclosed white islands that
remain disconnected from it.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
FIGURE = (40, 80, 200)
# Radius of the single filled figure disk.
FIG_R = 190
# Hole radius bounds (px).
_HOLE_MIN = 22
_HOLE_MAX = 36
# Maximum center offset of a hole from the figure center (keeps a clear dark
# ring between every hole and the figure boundary).
_HOLE_REGION = 118
# Required pairwise center separation beyond the sum of two hole radii.
_SEP = 22
# Minimum dark-clearance margin below which a scene is quarantined.
MIN_CLEAR = 24
# Figure-center sampling box (keeps the disk and its white ring on canvas).
_FC_LO = 220
_FC_HI = 293


def sample_scene(seed: int, hole_count: int | None = None) -> dict:
    """Sample one latent scene deterministically from ``seed``.

    ``hole_count`` is primarily a scene-derived value; generating many scenes
    pins it to balance the off-quarantine classes.
    """
    rng = np.random.default_rng(seed)
    if hole_count is None:
        hole_count = int(rng.integers(1, 4))
    fcy = float(rng.integers(_FC_LO, _FC_HI))
    fcx = float(rng.integers(_FC_LO, _FC_HI))
    holes: list[list[float]] = []
    tries = 0
    while len(holes) < hole_count and tries < 6000:
        tries += 1
        ang = float(rng.uniform(0.0, 2.0 * math.pi))
        rad = float(rng.uniform(0.0, _HOLE_REGION))
        hy = fcy + rad * math.cos(ang)
        hx = fcx + rad * math.sin(ang)
        hr = float(rng.integers(_HOLE_MIN, _HOLE_MAX + 1))
        ok = True
        for (oy, ox, or_) in holes:
            if math.hypot(hy - oy, hx - ox) < (hr + or_ + _SEP):
                ok = False
                break
        if ok:
            holes.append([round(hy, 6), round(hx, 6), hr])
    return {
        "hole_count": hole_count,
        "figure_center": [float(fcy), float(fcx)],
        "holes": holes,
    }


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    # Access the figure's hole count so rendering and analytic gold both read it
    # (the rendered-field audit requires the gold's only label input to be used
    # by rendering and vice versa).
    _ = scene["hole_count"]
    fcy, fcx = scene["figure_center"]
    r = FIG_R
    d.ellipse([fcx - r, fcy - r, fcx + r, fcy + r], fill=FIGURE)
    for (hy, hx, hr) in scene["holes"]:
        d.ellipse([hx - hr, hy - hr, hx + hr, hy + hr], fill=BACKGROUND)
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (deterministic, boundary-free PNG)."""
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    """Structural gold: the number of enclosed white holes as a string."""
    return str(int(scene["hole_count"]))


def margin(scene: dict) -> float:
    """Commitment magnitude: minimum dark clearance around any hole."""
    fcy, fcx = scene["figure_center"]
    best = float("inf")
    for (hy, hx, hr) in scene["holes"]:
        edge = FIG_R - math.hypot(hy - fcy, hx - fcx) - hr
        best = min(best, edge)
    n = len(scene["holes"])
    for i in range(n):
        hy, hx, hr = scene["holes"][i]
        for j in range(i + 1, n):
            oy, ox, or_ = scene["holes"][j]
            sep = math.hypot(hy - oy, hx - ox) - hr - or_
            best = min(best, sep)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine when a hole nearly touches the figure edge or another hole."""
    return margin(scene) < MIN_CLEAR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
