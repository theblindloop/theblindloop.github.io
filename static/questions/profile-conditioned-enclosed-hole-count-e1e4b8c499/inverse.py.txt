"""Independent pixel-only inverse arm for the enclosed-hole-count world.

Recovers the number of enclosed white holes from the final rasterized PNG
alone: it thresholds the white background, flood-fills the white region that is
connected to the image border, and counts each white island that remains
disconnected from the border (an enclosed hole). It shares no geometry, palette
or thresholds with the renderer beyond the physical facts that the figure is
clearly darker than the white background and that the canvas border is part of
the outside background.

This module never imports renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from functools import lru_cache
from collections import deque

import numpy as np
from PIL import Image

_WHITE = 235


@lru_cache(maxsize=None)
def _decide(arr_bytes: bytes, shape: tuple[int, int]) -> str:
    a = np.frombuffer(arr_bytes, dtype=np.uint8).reshape(shape)
    white = a >= _WHITE
    return str(_count_holes(white))


def _count_holes(white: np.ndarray) -> int:
    h, w = white.shape
    visited = np.zeros((h, w), dtype=bool)
    outside = 0
    fringe = deque()
    for y in range(h):
        for x in (0, w - 1):
            if white[y, x] and not visited[y, x]:
                visited[y, x] = True
                fringe.append((y, x))
    for x in range(w):
        for y in (0, h - 1):
            if white[y, x] and not visited[y, x]:
                visited[y, x] = True
                fringe.append((y, x))
    outside = 1
    while fringe:
        y, x = fringe.popleft()
        if y > 0 and white[y - 1, x] and not visited[y - 1, x]:
            visited[y - 1, x] = True
            fringe.append((y - 1, x))
        if y < h - 1 and white[y + 1, x] and not visited[y + 1, x]:
            visited[y + 1, x] = True
            fringe.append((y + 1, x))
        if x > 0 and white[y, x - 1] and not visited[y, x - 1]:
            visited[y, x - 1] = True
            fringe.append((y, x - 1))
        if x < w - 1 and white[y, x + 1] and not visited[y, x + 1]:
            visited[y, x + 1] = True
            fringe.append((y, x + 1))
    holes = 0
    remaining = np.argwhere(white & ~visited)
    idx = 0
    total = remaining.shape[0]
    while idx < total and holes <= 3:
        sy, sx = int(remaining[idx, 0]), int(remaining[idx, 1])
        if visited[sy, sx]:
            idx += 1
            continue
        fringe.clear()
        fringe.append((sy, sx))
        visited[sy, sx] = True
        holes += 1
        idx += 1
        while fringe:
            y, x = fringe.popleft()
            if y > 0 and white[y - 1, x] and not visited[y - 1, x]:
                visited[y - 1, x] = True
                fringe.append((y - 1, x))
            if y < h - 1 and white[y + 1, x] and not visited[y + 1, x]:
                visited[y + 1, x] = True
                fringe.append((y + 1, x))
            if x > 0 and white[y, x - 1] and not visited[y, x - 1]:
                visited[y, x - 1] = True
                fringe.append((y, x - 1))
            if x < w - 1 and white[y, x + 1] and not visited[y, x + 1]:
                visited[y, x + 1] = True
                fringe.append((y, x + 1))
    return holes


def decision_from_image(image: Image.Image) -> str:
    """Return the enclosed-white-hole count ('1','2','3',...) from the PNG."""
    arr = np.asarray(image.convert("RGB"))
    h, w, _ = arr.shape
    gray = np.asarray(arr, dtype=np.uint8).min(axis=2)
    return _decide(gray.tobytes(), (h, w))
