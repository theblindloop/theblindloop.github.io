"""Renderer and analytic contract for the ringed trace checkpoint world.

The answer is the first labeled checkpoint encountered while walking the solid
navy trace from its green ringed port.  The route is a single, winding,
non-self-crossing polyline.  All geometric work in this module is latent-scene
work; the independent inverse in ``oracle.py`` only receives the final PNG.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "ringed_trace_checkpoint_order-0.1.0"
WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 247, 242)
ROUTE_COLOR = (39, 54, 73)
PORT_COLOR = (44, 137, 91)
MARKER_FILL = (249, 247, 239)
GLYPH_COLOR = (236, 171, 45)
MARKER_COLORS = (
    (218, 75, 83),
    (36, 150, 163),
    (126, 89, 181),
    (224, 143, 45),
)
QUARANTINE_GAP_PX = 50.0
GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
}


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _cross(a: tuple[float, float], b: tuple[float, float]) -> float:
    return a[0] * b[1] - a[1] * b[0]


def _sub(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    return (a[0] - b[0], a[1] - b[1])


def _point_segment_distance(
    p: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    ab = _sub(b, a)
    denom = ab[0] * ab[0] + ab[1] * ab[1]
    if denom <= 1e-9:
        return _distance(p, a)
    t = max(0.0, min(1.0, ((p[0] - a[0]) * ab[0] + (p[1] - a[1]) * ab[1]) / denom))
    q = (a[0] + t * ab[0], a[1] + t * ab[1])
    return _distance(p, q)


def _segments_intersect(
    a: tuple[float, float],
    b: tuple[float, float],
    c: tuple[float, float],
    d: tuple[float, float],
) -> bool:
    ab = _sub(b, a)
    ac = _sub(c, a)
    ad = _sub(d, a)
    cd = _sub(d, c)
    ca = _sub(a, c)
    cb = _sub(b, c)
    values = (_cross(ab, ac), _cross(ab, ad), _cross(cd, ca), _cross(cd, cb))
    eps = 1e-7
    return (values[0] * values[1] < -eps) and (values[2] * values[3] < -eps)


def _segment_distance(
    a: tuple[float, float],
    b: tuple[float, float],
    c: tuple[float, float],
    d: tuple[float, float],
) -> float:
    if _segments_intersect(a, b, c, d):
        return 0.0
    return min(
        _point_segment_distance(a, c, d),
        _point_segment_distance(b, c, d),
        _point_segment_distance(c, a, b),
        _point_segment_distance(d, a, b),
    )


def _polyline_length(points: Iterable[Iterable[float]]) -> float:
    pts = [(float(p[0]), float(p[1])) for p in points]
    return sum(_distance(a, b) for a, b in zip(pts, pts[1:]))


def _point_at_distance(points: list[tuple[float, float]], distance: float) -> tuple[float, float]:
    remaining = max(0.0, min(distance, _polyline_length(points)))
    for a, b in zip(points, points[1:]):
        length = _distance(a, b)
        if remaining <= length:
            ratio = remaining / max(length, 1e-9)
            return (a[0] + ratio * (b[0] - a[0]), a[1] + ratio * (b[1] - a[1]))
        remaining -= length
    return points[-1]


def _point_at_t(points: list[tuple[float, float]], t: float) -> tuple[float, float]:
    return _point_at_distance(points, float(t) * _polyline_length(points))


def _tangent_at(points: list[tuple[float, float]], t: float) -> tuple[float, float]:
    d = _polyline_length(points)
    eps = min(7.0 / max(d, 1.0), 0.03)
    p0 = _point_at_t(points, max(0.0, t - eps))
    p1 = _point_at_t(points, min(1.0, t + eps))
    dx, dy = p1[0] - p0[0], p1[1] - p0[1]
    length = math.hypot(dx, dy) or 1.0
    return (dx / length, dy / length)


def _path_is_legal(points: list[tuple[float, float]]) -> bool:
    if not (430.0 <= _polyline_length(points) <= 760.0):
        return False
    for i in range(len(points) - 1):
        for j in range(i + 2, len(points) - 1):
            if _segment_distance(points[i], points[i + 1], points[j], points[j + 1]) < 34.0:
                return False
    return True


def _sample_path(rng: random.Random) -> list[list[float]]:
    """Make a winding route with a clearance margin from non-neighboring edges."""
    for _ in range(2000):
        x = rng.uniform(72.0, 440.0)
        y = rng.uniform(72.0, 440.0)
        heading = rng.uniform(0.0, 2.0 * math.pi)
        points: list[tuple[float, float]] = [(x, y)]
        for _step in range(7):
            length = rng.uniform(55.0, 92.0)
            turn = rng.uniform(-1.12, 1.12)
            heading += turn
            nx = points[-1][0] + length * math.cos(heading)
            ny = points[-1][1] + length * math.sin(heading)
            if not (48.0 <= nx <= 464.0 and 48.0 <= ny <= 464.0):
                break
            points.append((nx, ny))
        if len(points) == 8 and _path_is_legal(points):
            return [[round(px, 4), round(py, 4)] for px, py in points]
    raise RuntimeError("could not sample a legal winding route")


def _sample_distractor(rng: random.Random) -> list[list[float]]:
    x = rng.uniform(70.0, 410.0)
    y = rng.uniform(70.0, 410.0)
    heading = rng.uniform(0.0, 2.0 * math.pi)
    points = [(x, y)]
    for _ in range(2):
        heading += rng.uniform(-0.9, 0.9)
        length = rng.uniform(42.0, 86.0)
        points.append((points[-1][0] + length * math.cos(heading), points[-1][1] + length * math.sin(heading)))
    if any(not (34.0 <= x0 <= 478.0 and 34.0 <= y0 <= 478.0) for x0, y0 in points):
        return _sample_distractor(rng)
    return [[round(x0, 4), round(y0, 4)] for x0, y0 in points]


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene dictionary for ``seed``."""
    rng = random.Random(int(seed))
    bucket = int(seed) % 8
    order = "A_first" if int(seed) % 2 == 0 else "B_first"
    if bucket == 0:
        delta_range = (0.060, 0.085)
    elif bucket == 1:
        delta_range = (0.105, 0.140)
    else:
        delta_range = (0.17, 0.34)

    # Marker centres must remain visibly separate even in the quarantined band.
    # Rejection here also prevents a winding route from bringing two distant
    # along-path positions into a touching spatial configuration.
    for _ in range(500):
        path = _sample_path(rng)
        points = [(float(p[0]), float(p[1])) for p in path]
        delta = rng.uniform(*delta_range)
        t0 = rng.uniform(0.16, min(0.47, 0.84 - delta))
        if order == "A_first":
            a_t, b_t = t0, t0 + delta
        else:
            a_t, b_t = t0 + delta, t0
        a_center = _point_at_t(points, a_t)
        b_center = _point_at_t(points, b_t)
        a_radius = rng.uniform(13.0, 16.0)
        b_radius = rng.uniform(13.0, 16.0)
        start_center = points[0]
        if _distance(a_center, b_center) < a_radius + b_radius + 9.0:
            continue
        if _distance(start_center, a_center) < a_radius + 24.0:
            continue
        if _distance(start_center, b_center) < b_radius + 24.0:
            continue
        break
    else:
        raise RuntimeError("could not place separated checkpoints")

    selected = rng.sample(list(MARKER_COLORS), 2)
    distractors = [_sample_distractor(rng), _sample_distractor(rng)]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "path": path,
        "stroke_width": round(rng.uniform(6.0, 9.0), 3),
        "order": order,
        "a_t": round(a_t, 6),
        "b_t": round(b_t, 6),
        "a_color": list(selected[0]),
        "b_color": list(selected[1]),
        "a_radius": round(a_radius, 3),
        "b_radius": round(b_radius, 3),
        "distractors": distractors,
        "port_radius": round(rng.uniform(17.0, 21.0), 3),
        "route_color": list(ROUTE_COLOR),
        "port_color": list(PORT_COLOR),
        "marker_fill": list(MARKER_FILL),
        "glyph_color": list(GLYPH_COLOR),
    }


def _scaled_points(points: Iterable[Iterable[float]], scale: int) -> list[tuple[float, float]]:
    return [(float(p[0]) * scale, float(p[1]) * scale) for p in points]


def _circle(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: float, *, fill, outline, width: int, scale: int) -> None:
    cx, cy = center[0] * scale, center[1] * scale
    box = ((cx - radius * scale, cy - radius * scale), (cx + radius * scale, cy + radius * scale))
    draw.ellipse(box, fill=fill, outline=outline, width=max(1, int(round(width * scale))))


def _draw_glyph(draw: ImageDraw.ImageDraw, center: tuple[float, float], label: str, color, scale: int) -> None:
    pattern = GLYPHS[label]
    cell = 3.0 * scale
    x0 = center[0] * scale - 2.5 * cell
    y0 = center[1] * scale - 3.5 * cell
    for row, line in enumerate(pattern):
        for col, bit in enumerate(line):
            if bit == "1":
                left = round(x0 + col * cell)
                top = round(y0 + row * cell)
                right = round(left + cell - 0.35 * scale)
                bottom = round(top + cell - 0.35 * scale)
                draw.rectangle((left, top, right, bottom), fill=color)


def render(scene: dict) -> Image.Image:
    """Render only from the scene dictionary, with deterministic antialiasing."""
    if scene["order"] not in ("A_first", "B_first"):
        raise ValueError("invalid checkpoint order")
    width, height = int(scene["width"]), int(scene["height"])
    scale = 4
    background = tuple(int(v) for v in scene["background"])
    route_color = tuple(int(v) for v in scene["route_color"])
    port_color = tuple(int(v) for v in scene["port_color"])
    marker_fill = tuple(int(v) for v in scene["marker_fill"])
    glyph_color = tuple(int(v) for v in scene["glyph_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    for distractor in scene["distractors"]:
        draw.line(
            _scaled_points(distractor, scale),
            fill=(208, 207, 201),
            width=2 * scale,
            joint="curve",
        )

    path = [(float(p[0]), float(p[1])) for p in scene["path"]]
    route_width = float(scene["stroke_width"])
    draw.line(_scaled_points(path, scale), fill=route_color, width=max(1, round(route_width * scale)), joint="curve")
    for point in path[1:-1]:
        _circle(draw, point, route_width / 2.0, fill=route_color, outline=route_color, width=1, scale=scale)

    _circle(
        draw,
        path[0],
        float(scene["port_radius"]),
        fill=marker_fill,
        outline=port_color,
        width=4,
        scale=scale,
    )

    for label, t_key, color_key, radius_key in (
        ("A", "a_t", "a_color", "a_radius"),
        ("B", "b_t", "b_color", "b_radius"),
    ):
        t = float(scene[t_key])
        center = _point_at_t(path, t)
        radius = float(scene[radius_key])
        color = tuple(int(v) for v in scene[color_key])
        _circle(draw, center, radius, fill=marker_fill, outline=color, width=4, scale=scale)
        tangent = _tangent_at(path, t)
        bridge = radius * 1.05
        p0 = ((center[0] - tangent[0] * bridge) * scale, (center[1] - tangent[1] * bridge) * scale)
        p1 = ((center[0] + tangent[0] * bridge) * scale, (center[1] + tangent[1] * bridge) * scale)
        draw.line((p0, p1), fill=route_color, width=max(1, round(route_width * scale)))
        _draw_glyph(draw, center, label, glyph_color, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the latent answer; only the explicit label input is read here."""
    return "A" if scene["order"] == "A_first" else "B"


def margin(scene: dict) -> float:
    """Runner-up gap: absolute along-trace distance between A and B."""
    points = [(float(p[0]), float(p[1])) for p in scene["path"]]
    return abs(float(scene["a_t"]) - float(scene["b_t"])) * _polyline_length(points)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This parametrization has no nontrivial pixel-identical latent aliases."""
    return []
