"""Notch-keyed tile de-rotation world.

A square sheet is ruled into a 4x4 grid of tiles.  Every tile carries exactly
one solid block - a two-by-two square of cells parked in one of the tile's four
corners - and exactly one small coloured notch sitting on the middle of one of
the tile's four edges.  The printed block corner is *not* the answer: each tile
has been turned in place by a random quarter turn, and the notch is the visible
record of that turn.

The question declares one raster operation: turn every tile about its own
centre until its notch is back at the tile's top edge.  Only then do the sixteen
blocks settle into one of five named arrangements.

Because the printed corner is (true corner + turn) mod 4 with the turn drawn
uniformly and independently of the answer, the printed arrangement of blocks is
marginally uniform: it carries no information about the symbol at all.  Every
scene also prints exactly sixteen blocks and sixteen notches of fixed size, so
ink, count, colour, size and coarse layout are constant across answers.  The
answer lives only in the binding of each block's corner to its own notch edge.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

TILES = 4  # tiles per sheet side
NTILES = TILES * TILES
CELLS = 3  # cells per tile side; the block covers a 2x2 corner of them

# Directions are corner indices, numbered clockwise from the top left, so a
# clockwise quarter turn of a tile is exactly "+1 mod 4" on both the block
# corner and the notch edge (edges are numbered clockwise from the top too).
DIR_TL, DIR_TR, DIR_BR, DIR_BL = 0, 1, 2, 3
BLOCK_ORIGIN = {DIR_TL: (0, 0), DIR_TR: (0, 1), DIR_BR: (1, 1), DIR_BL: (1, 0)}


def _template(name: str) -> tuple[int, ...]:
    dirs = []
    for r in range(TILES):
        for c in range(TILES):
            if name == "quads":  # every block hugs the corner nearest the sheet centre
                dirs.append((DIR_BR if c < 2 else DIR_BL) if r < 2 else (DIR_TR if c < 2 else DIR_TL))
            elif name == "window":  # every block hugs the corner furthest from the centre
                dirs.append((DIR_TL if c < 2 else DIR_TR) if r < 2 else (DIR_BL if c < 2 else DIR_BR))
            elif name == "grid":  # all sixteen at the top-left of their own tile
                dirs.append(DIR_TL)
            elif name == "bars":  # vertical pairs, all against the left edge
                dirs.append(DIR_BL if r % 2 == 0 else DIR_TL)
            elif name == "dashes":  # horizontal pairs, all against the top edge
                dirs.append(DIR_TR if c % 2 == 0 else DIR_TL)
            else:
                raise KeyError(name)
    return tuple(dirs)


SYMBOLS: tuple[str, ...] = ("bars", "dashes", "grid", "quads", "window")
TEMPLATES: dict[str, tuple[int, ...]] = {name: _template(name) for name in SYMBOLS}


def hamming(a, b) -> int:
    return sum(1 for x, y in zip(a, b) if x != y)


MIN_TEMPLATE_GAP = min(
    hamming(TEMPLATES[a], TEMPLATES[b])
    for i, a in enumerate(SYMBOLS)
    for b in SYMBOLS[i + 1 :]
)
assert MIN_TEMPLATE_GAP >= 8, MIN_TEMPLATE_GAP

# ------------------------------------------------------------------ palette --
# Paper, rule and ink stay in widely separated luminance bands; the notches and
# corner fiducials are the only strongly saturated colour.  The pixel arm uses
# that ordering only, never a specific hue.
PALETTES = [
    {"bg": (245, 244, 241), "line": (176, 178, 180), "ink": (38, 40, 44)},
    {"bg": (239, 241, 244), "line": (170, 174, 178), "ink": (44, 42, 38)},
    {"bg": (244, 241, 244), "line": (180, 176, 180), "ink": (32, 38, 42)},
    {"bg": (241, 244, 239), "line": (174, 178, 172), "ink": (46, 42, 36)},
]
ACCENTS = [(226, 96, 24), (28, 150, 196), (206, 42, 128), (46, 158, 72)]

PAD = 8  # gutter inside each tile that carries the notch, never any cell
NOTCH = 3  # half-side of a notch square
FID = 9  # half-side of a corner fiducial square
MARGIN_PX = 26


def _luma(rgb) -> float:
    return 0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]


# -------------------------------------------------------------------- scene --
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 6151 + 29)
    # Answer identity is cycled against the seed, so it is independent of every
    # visual draw below: palette, hue, pitch, jitter, noise and the turn keys.
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    dirs = list(TEMPLATES[symbol])

    # Near-boundary scenes: one or two tiles are parked on a wrong corner, which
    # moves the arrangement off its template and shrinks the runner-up gap.
    flips = rng.choice([0, 0, 0, 1, 1, 2])
    for tile in rng.sample(range(NTILES), flips):
        dirs[tile] = (dirs[tile] + rng.choice([1, 2, 3])) % 4

    # The turn applied to each tile.  Uniform and independent of the answer, so
    # the printed block corners are uniform and answer-free on their own.
    keys = [rng.randrange(4) for _ in range(NTILES)]

    return {
        "dirs": dirs,
        "keys": keys,
        "palette": rng.randrange(len(PALETTES)),
        "accent": rng.randrange(len(ACCENTS)),
        "cell": rng.choice([16, 18, 20]),
        "jitter_x": rng.randrange(0, 14),
        "jitter_y": rng.randrange(0, 14),
        "noise": rng.choice([0, 1, 2, 3]),
        "noise_seed": rng.randrange(1, 10_000),
    }


def true_dirs(scene: dict) -> tuple[int, ...]:
    return tuple(int(d) % 4 for d in scene["dirs"])


def turn_keys(scene: dict) -> tuple[int, ...]:
    return tuple(int(k) % 4 for k in scene["keys"])


def printed_dirs(scene: dict) -> tuple[int, ...]:
    """What the raster actually shows: the true corner turned by the key."""

    return tuple((d + k) % 4 for d, k in zip(true_dirs(scene), turn_keys(scene)))


def _ranked(scene: dict) -> list[tuple[int, str]]:
    dirs = true_dirs(scene)
    return sorted((hamming(dirs, TEMPLATES[s]), s) for s in SYMBOLS)


# ------------------------------------------------------------------- layout --
def _layout(scene: dict) -> dict:
    cell = int(scene["cell"])
    pitch = CELLS * cell + 2 * PAD
    side = TILES * pitch
    x0 = MARGIN_PX + int(scene["jitter_x"])
    y0 = MARGIN_PX + int(scene["jitter_y"])
    size = (x0 + side + MARGIN_PX + 14, y0 + side + MARGIN_PX + 14)
    return {"cell": cell, "pitch": pitch, "side": side, "x0": x0, "y0": y0, "size": size}


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    accent = ACCENTS[int(scene["accent"]) % len(ACCENTS)]
    lay = _layout(scene)
    cell, pitch, side, x0, y0 = lay["cell"], lay["pitch"], lay["side"], lay["x0"], lay["y0"]
    shown = printed_dirs(scene)
    keys = turn_keys(scene)

    image = Image.new("RGB", lay["size"], palette["bg"])
    draw = ImageDraw.Draw(image)
    for r in range(TILES):
        for c in range(TILES):
            tile = r * TILES + c
            bx = x0 + c * pitch
            by = y0 + r * pitch
            draw.rectangle([bx, by, bx + pitch, by + pitch], outline=palette["line"], width=2)
            ix = bx + PAD
            iy = by + PAD
            for cr in range(CELLS):  # the faint cell ruling of the tile
                for cc in range(CELLS):
                    draw.rectangle(
                        [ix + cc * cell, iy + cr * cell, ix + (cc + 1) * cell, iy + (cr + 1) * cell],
                        outline=palette["line"],
                        width=1,
                    )
            br, bc = BLOCK_ORIGIN[shown[tile]]
            draw.rectangle(
                [ix + bc * cell, iy + br * cell, ix + (bc + 2) * cell, iy + (br + 2) * cell],
                fill=palette["ink"],
            )
            # The notch: an edge-midpoint mark inside the gutter, at edge index
            # equal to the applied turn, so "notch back to the top" undoes it.
            mid = pitch // 2
            offs = PAD // 2
            nx, ny = {
                0: (bx + mid, by + offs),
                1: (bx + pitch - offs, by + mid),
                2: (bx + mid, by + pitch - offs),
                3: (bx + offs, by + mid),
            }[keys[tile]]
            draw.rectangle([nx - NOTCH, ny - NOTCH, nx + NOTCH, ny + NOTCH], fill=accent)

    for px, py in ((x0, y0), (x0 + side, y0), (x0, y0 + side), (x0 + side, y0 + side)):
        draw.rectangle([px - FID, py - FID, px + FID, py + FID], fill=accent)

    noise = int(scene["noise"])
    if noise:
        rng = np.random.default_rng(int(scene["noise_seed"]))
        arr = np.asarray(image, dtype=np.int16)
        grain = rng.integers(-noise, noise + 1, size=arr.shape[:2], dtype=np.int16)
        arr = np.clip(arr + grain[:, :, None], 0, 255).astype(np.uint8)
        image = Image.fromarray(arr, "RGB")
    return image


# --------------------------------------------------------------------- gold --
def analytic_gold(scene: dict) -> str:
    return _ranked(scene)[0][1]


def _legibility(scene: dict) -> float:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    bg = _luma(palette["bg"])
    ink = _luma(palette["ink"])
    line = _luma(palette["line"])
    thresh = (bg + ink) / 2.0
    sep = min(bg - line, line - thresh, thresh - ink)
    return sep / (1.0 + 0.5 * int(scene["noise"]))


def margin(scene: dict) -> float:
    ranked = _ranked(scene)
    gap = float(ranked[1][0] - ranked[0][0])
    factor = max(0.0, min(1.0, _legibility(scene) / 24.0))
    return round(gap * factor, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical re-encodings of one and the same scene."""

    dirs_mod = dict(scene)
    dirs_mod["dirs"] = [int(d) + 4 for d in scene["dirs"]]
    keys_mod = dict(scene)
    keys_mod["keys"] = [int(k) + 4 for k in scene["keys"]]
    palette_mod = dict(scene)
    palette_mod["palette"] = int(scene["palette"]) + len(PALETTES)
    accent_mod = dict(scene)
    accent_mod["accent"] = int(scene["accent"]) + len(ACCENTS)
    return [
        ("direction_index_modulus", dirs_mod),
        ("turn_key_index_modulus", keys_mod),
        ("palette_index_modulus", palette_mod),
        ("accent_index_modulus", accent_mod),
    ]
