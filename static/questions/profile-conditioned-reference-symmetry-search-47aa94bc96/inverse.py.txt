"""Independent pixel-only inverse arm for reference_symmetry_search.

Receives only a PIL image and the public question contract. It segments the
dark filled triangles, isolates the reference triangle (top panel) and the
four quadrant triangles, and for each measures the overlap of its pixels with
its mirror image reflected about its own vertical bounding-box axis. A shape
whose reflection self-overlap is high is symmetric; a shape with a low value
is asymmetric. The oracle selects the single quadrant whose class matches the
reference triangle's class.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

FEATURE_THRESHOLD = 0.78


def _reflection_iou(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if xs.size == 0:
        return 1.0
    cmin, xmax = int(xs.min()), int(xs.max())
    axis = (cmin + xmax) / 2.0
    height, width = mask.shape
    reflected = np.zeros_like(mask)
    for x in range(width):
        mir = int(round(2.0 * axis - x))
        if 0 <= mir < width:
            reflected[:, mir] |= mask[:, x]
    inter = int(np.logical_and(mask, reflected).sum())
    union = int(np.logical_or(mask, reflected).sum())
    return inter / union if union else 1.0


def _classify(mask: np.ndarray) -> bool:
    return _reflection_iou(mask) >= FEATURE_THRESHOLD


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    x_split = w * 0.5
    y_ref = h * 0.31
    y_upper = h * 0.416
    y_split = h * 0.6875

    rows = np.arange(h)[:, None]
    ref_sel = dark.copy()
    ref_sel[int(y_ref):, :] = False
    ref_sym = _classify(ref_sel)

    def quadrant_mask(sector: str) -> np.ndarray:
        cols = np.arange(w)[None, :]
        if sector in ("TL", "BL"):
            left = cols < x_split
        else:
            left = cols >= x_split
        if sector in ("TL", "TR"):
            band = (rows >= y_upper) & (rows < y_split)
        else:
            band = rows >= y_split
        return dark & band & left

    for sector in ("TL", "TR", "BL", "BR"):
        if _classify(quadrant_mask(sector)) == ref_sym:
            return sector
    return "TL"
