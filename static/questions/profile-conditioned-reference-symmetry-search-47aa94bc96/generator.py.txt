"""Reference reflection-symmetry visual search world.

A dark filled reference triangle is shown in the top box. Below it the image
is split into four quadrants, each holding one dark filled triangle. Each
triangle is either reflection-symmetric about its own vertical axis (apex
centered) or asymmetric (apex shifted sideways). Exactly one quadrant's
triangle has the SAME reflective-symmetry status as the reference triangle;
the decision is the quadrant whose triangle shares the reference's symmetry
status. All rendering is deterministic and every decision-relevant surface is
visible in the raster.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
INK = (20, 20, 20)
GRAY = (190, 190, 190)
BOX = (170, 170, 170)

REF_CX = SIZE // 2
REF_BASE_Y = 126
BASE_HALF = 55
APEX_H = 90

# Classification boundary expressed as a "symmetry feature" g = 1 - offset_frac.
# Symmetric triangles have g = 1.0 (exact reflection self-overlap); asymmetric
# triangles have g <= 0.65. The boundary at FEATURE_BOUNDARY cleanly separates
# the two classes with a deterministic margin.
FEATURE_BOUNDARY = 0.78

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, 265),
    "TR": (3 * SIZE // 4, 265),
    "BL": (SIZE // 4, 395),
    "BR": (3 * SIZE // 4, 395),
}

# Asymmetric apex lateral displacement as a fraction of base half-width.
OFF_MIN = 0.40
OFF_MAX = 0.58

QUARANTINE_THRESH = 0.10


def _sample_off(rng: np.random.Generator) -> float:
    frac = float(rng.uniform(OFF_MIN, OFF_MAX))
    return frac


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_sym = bool(rng.integers(0, 2))
    if ref_sym:
        ref_off = 0.0
        ref_side = 0
    else:
        ref_off = _sample_off(rng)
        ref_side = int(rng.choice([-1, 1]))
    reference_shape = {"sym": ref_sym, "off": ref_off, "side": ref_side}

    # Exactly one quadrant matches the reference's symmetry status; the other
    # three are the opposite class. The winner quadrant and, for asymmetric
    # shapes, the apex side are randomized to counterbalance position.
    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    candidates: dict[str, dict] = {}
    for sector in SECTORS:
        if sector == winner:
            sym = ref_sym
        else:
            sym = not ref_sym
        if sym:
            off = 0.0
            side = 0
        else:
            off = _sample_off(rng)
            side = int(rng.choice([-1, 1]))
        candidates[sector] = {"sym": sym, "off": off, "side": side}

    return {"reference_shape": reference_shape, "candidates": candidates}


def _draw_triangle(draw: ImageDraw.ImageDraw, cx: int, base_y: int,
                   base_half: int, apex_h: int, off_frac: float,
                   side: int) -> None:
    apex_x = cx + side * off_frac * base_half
    pts = [(cx - base_half, base_y), (cx + base_half, base_y),
           (apex_x, base_y - apex_h)]
    draw.polygon(pts, fill=INK)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    draw.rectangle([SIZE // 2 - 170, 20, SIZE // 2 + 170, 140],
                   outline=BOX, width=3)
    draw.line([0, 330, SIZE, 330], fill=GRAY, width=2)
    draw.line([SIZE // 2, 200, SIZE // 2, SIZE], fill=GRAY, width=2)

    ref = scene["reference_shape"]
    _draw_triangle(draw, REF_CX, REF_BASE_Y, BASE_HALF, APEX_H,
                   ref["off"], ref["side"])

    for sector in SECTORS:
        c = scene["candidates"][sector]
        cx, cy = SECTOR_CENTERS[sector]
        _draw_triangle(draw, cx, cy + 45, BASE_HALF, APEX_H,
                       c["off"], c["side"])
    return img


def _decision(scene: dict) -> tuple[str, float]:
    ref_sym = scene["reference_shape"]["sym"]
    winner = None
    for sector in SECTORS:
        c = scene["candidates"][sector]
        if c["sym"] == ref_sym:
            winner = sector
            break
    clearances = []
    for shape in [scene["reference_shape"]] + [
        scene["candidates"][s] for s in SECTORS
    ]:
        feature = 1.0 - shape["off"]
        clearances.append(abs(feature - FEATURE_BOUNDARY))
    return winner, float(min(clearances))


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
