"""Deterministic latent sampler and renderer for the yin-yang area world."""

from __future__ import annotations

import copy
import math
import random

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 2
BACKGROUND = (248, 246, 240)
INK = (35, 39, 51)
PROOF_GRAY = (151, 154, 162)
PALETTES = (
    ((20, 151, 157), (239, 166, 48)),
    ((18, 137, 148), (226, 145, 37)),
    ((27, 164, 160), (244, 176, 58)),
)


def sample_scene(seed: int) -> dict:
    """Sample matched triples: amber-larger, equal control, teal-larger."""

    group_seed = int(seed) // 3
    rng = random.Random(group_seed ^ 0x59A17EA)
    magnitude = rng.choice((14, 18, 22))
    phase = int(seed) % 3
    divider_shift = (-magnitude, 0, magnitude)[phase]
    center_x = 256 + rng.randint(-5, 5)
    center_y = 256 + rng.randint(-5, 5)
    radius = rng.randint(198, 204)
    wave_amplitude = rng.randint(56, 64)
    dot_radius = rng.randint(18, 22)
    palette_index = rng.randrange(len(PALETTES))

    slots = ((42, 64), (470, 64), (42, 448), (470, 448))
    proof_marks = []
    for index, (base_x, base_y) in enumerate(slots):
        proof_marks.append(
            {
                "x": base_x + rng.randint(-4, 4),
                "y": base_y + rng.randint(-4, 4),
                "size": rng.randint(5, 8),
                "kind": "ring" if (index + group_seed) % 2 else "cross",
            }
        )
    rng.shuffle(proof_marks)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": center_x,
        "center_y": center_y,
        "radius": radius,
        "divider_shift": divider_shift,
        "wave_amplitude": wave_amplitude,
        "dot_radius": dot_radius,
        "palette_index": palette_index,
        "proof_marks": proof_marks,
    }


def _base_coordinates(width: int, height: int) -> tuple[np.ndarray, np.ndarray]:
    scale = SUPERSAMPLE
    xs = (np.arange(width * scale, dtype=np.float32) + 0.5) / scale
    ys = (np.arange(height * scale, dtype=np.float32) + 0.5) / scale
    return np.meshgrid(xs, ys)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    shift = float(scene["divider_shift"])
    wave = float(scene["wave_amplitude"])
    dot_radius = float(scene["dot_radius"])
    teal, amber = PALETTES[int(scene["palette_index"])]

    xx, yy = _base_coordinates(width, height)
    distance2 = (xx - cx) ** 2 + (yy - cy) ** 2
    disk = distance2 <= radius**2
    boundary_x = cx + shift + wave * np.sin(np.pi * (yy - cy) / radius)
    teal_field = disk & (xx <= boundary_x)
    amber_field = disk & ~teal_field

    array = np.empty((height * SUPERSAMPLE, width * SUPERSAMPLE, 3), dtype=np.uint8)
    array[:] = BACKGROUND
    array[teal_field] = teal
    array[amber_field] = amber

    # Equal-sized opposite-color dots cancel in the area comparison while making
    # the object unmistakably yin-yang-like. Their geometry is 180-degree paired.
    upper_y = cy - 0.42 * radius
    lower_y = cy + 0.42 * radius
    upper_d2 = (xx - cx) ** 2 + (yy - upper_y) ** 2
    lower_d2 = (xx - cx) ** 2 + (yy - lower_y) ** 2
    dot_inner = max(1.0, dot_radius - 2.5)
    array[upper_d2 <= dot_radius**2] = INK
    array[upper_d2 <= dot_inner**2] = teal
    array[lower_d2 <= dot_radius**2] = INK
    array[lower_d2 <= dot_inner**2] = amber

    divider = disk & (np.abs(xx - boundary_x) <= 2.25)
    outline = disk & (distance2 >= (radius - 6.0) ** 2)
    array[divider | outline] = INK

    # Four small exterior proof marks are visible nuisance objects. They are
    # deliberately non-overlapping, so their list order is a true latent alias.
    for mark in scene["proof_marks"]:
        mx = float(mark["x"])
        my = float(mark["y"])
        size = float(mark["size"])
        if mark["kind"] == "ring":
            radial = np.hypot(xx - mx, yy - my)
            mask = np.abs(radial - size) <= 1.35
        else:
            vertical = (np.abs(xx - mx) <= 1.2) & (np.abs(yy - my) <= size)
            horizontal = (np.abs(yy - my) <= 1.2) & (np.abs(xx - mx) <= size)
            mask = vertical | horizontal
        array[mask] = PROOF_GRAY

    high_res = Image.fromarray(array, mode="RGB")
    return high_res.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    shift = float(scene["divider_shift"])
    if shift > 0:
        return "teal"
    if shift < 0:
        return "amber"
    return "same"


def margin(scene: dict) -> float:
    return abs(float(scene["divider_shift"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["proof_marks"] = list(reversed(twin["proof_marks"]))
    return [("reverse_nonoverlapping_proof_mark_order", twin)]
