"""Independent final-PNG inverse arm for visible color-area balance.

This module intentionally duplicates only the public visual vocabulary. It has
no access to latent parameters, renderer code, filenames, or analytic labels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_TEAL_PROTOTYPES = np.asarray(
    ((20, 151, 157), (18, 137, 148), (27, 164, 160)), dtype=np.int16
)
_AMBER_PROTOTYPES = np.asarray(
    ((239, 166, 48), (226, 145, 37), (244, 176, 58)), dtype=np.int16
)
_COLOR_DISTANCE2_MAX = 62 * 62
# The two palettes have different antialiased rim brightness, so their nearest-
# prototype masks can differ by a few hundred boundary pixels on a mathematically
# symmetric control. Counterfactual shifts differ by more than 12,000 pixels.
_SAME_PIXEL_TOLERANCE = 512


class OracleAbstention(ValueError):
    """The raster no longer contains one intact, measurable medallion."""


def _minimum_distance2(array: np.ndarray, prototypes: np.ndarray) -> np.ndarray:
    delta = array.astype(np.int32)[:, :, None, :] - prototypes.astype(np.int32)[None, None, :, :]
    return np.min(np.sum(delta * delta, axis=3), axis=2)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 128:
        raise OracleAbstention("image is too small or malformed")

    teal_distance = _minimum_distance2(array, _TEAL_PROTOTYPES)
    amber_distance = _minimum_distance2(array, _AMBER_PROTOTYPES)
    teal_mask = (teal_distance < amber_distance) & (teal_distance <= _COLOR_DISTANCE2_MAX)
    amber_mask = (amber_distance < teal_distance) & (amber_distance <= _COLOR_DISTANCE2_MAX)
    colored = teal_mask | amber_mask
    ys, xs = np.nonzero(colored)
    if len(xs) < 60000:
        raise OracleAbstention("too little chromatic medallion area remains")

    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    span_x = x1 - x0 + 1
    span_y = y1 - y0 + 1
    if min(span_x, span_y) < 300 or not 0.93 <= span_x / span_y <= 1.07:
        raise OracleAbstention("chromatic support is not a large circular field")

    center_x = (x0 + x1) / 2.0
    center_y = (y0 + y1) / 2.0
    radius_estimate = (span_x + span_y) / 4.0
    yy, xx = np.indices(colored.shape, dtype=np.float32)
    protected_interior = (
        (xx - center_x) ** 2 + (yy - center_y) ** 2
        <= (0.96 * radius_estimate) ** 2
    )

    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0)
    ink = np.linalg.norm(array.astype(np.float32) - background, axis=2) > 24.0
    interior_coverage = float(np.mean(ink[protected_interior]))
    if interior_coverage < 0.999:
        raise OracleAbstention("the broad medallion interior has been erased or corrupted")

    teal_pixels = int(teal_mask.sum())
    amber_pixels = int(amber_mask.sum())
    if min(teal_pixels, amber_pixels) < 26000:
        raise OracleAbstention("both interlocking color fields must remain visible")
    difference = teal_pixels - amber_pixels
    if abs(difference) <= _SAME_PIXEL_TOLERANCE:
        return "same"
    return "teal" if difference > 0 else "amber"
