"""Deterministic scientific-plot world for ordered target crossings."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
PLOT_LEFT = 74
PLOT_RIGHT = 594
PLOT_TOP = 52
PLOT_BOTTOM = 400
QUARANTINE_GAP_PX = 18.0

COLOR_ORDER = ("red", "blue", "green", "orange")
PALETTE_THEMES: tuple[dict[str, tuple[int, int, int]], ...] = (
    {
        "red": (211, 54, 48),
        "blue": (39, 105, 205),
        "green": (34, 151, 91),
        "orange": (238, 132, 28),
    },
    {
        "red": (190, 43, 62),
        "blue": (38, 92, 181),
        "green": (24, 132, 76),
        "orange": (222, 112, 18),
    },
    {
        "red": (226, 65, 47),
        "blue": (48, 121, 222),
        "green": (42, 166, 99),
        "orange": (246, 146, 35),
    },
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, bounded scene from one integer seed.

    Consecutive seeds cycle the winning color. Slopes, shade theme, target
    height, and all non-winning crossings are independently randomized.
    """

    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    winner = COLOR_ORDER[int(seed) % len(COLOR_ORDER)]

    start = rng.randint(120, 150)
    gap_1 = rng.randint(72, 112)
    gap_2 = rng.randint(72, 112)
    # Deterministic near-boundary scenes provide quarantined gallery coverage.
    final_gap = rng.randint(9, 14) if int(seed) % 11 == 0 else rng.randint(38, 82)
    ordered_crossings = [
        float(start),
        float(start + gap_1),
        float(start + gap_1 + gap_2),
        float(start + gap_1 + gap_2 + final_gap),
    ]
    other_labels = [label for label in COLOR_ORDER if label != winner]
    rng.shuffle(other_labels)
    crossing_for = dict(zip(other_labels, ordered_crossings[:3]))
    crossing_for[winner] = ordered_crossings[3]

    slope_pool = [-0.30, -0.24, -0.17, -0.11, 0.12, 0.18, 0.25, 0.31]
    chosen_slopes = rng.sample(slope_pool, 4)
    theme = PALETTE_THEMES[rng.randrange(len(PALETTE_THEMES))]
    series = []
    for label, slope in zip(COLOR_ORDER, chosen_slopes):
        series.append(
            {
                "label": label,
                "rgb": list(theme[label]),
                "cross_x": crossing_for[label],
                "slope": float(slope),
            }
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "plot_left": PLOT_LEFT,
        "plot_right": PLOT_RIGHT,
        "plot_top": PLOT_TOP,
        "plot_bottom": PLOT_BOTTOM,
        "target_y": float(rng.randint(218, 248)),
        "series": series,
    }


def _series_crossings(scene: dict) -> list[tuple[float, str]]:
    return [(float(item["cross_x"]), str(item["label"])) for item in scene["series"]]


def analytic_gold(scene: dict) -> str:
    """Return the label at the greatest latent crossing x-coordinate."""

    return max(_series_crossings(scene))[1]


def margin(scene: dict) -> float:
    """Top-two crossing separation in rendered-coordinate pixels."""

    xs = sorted(value for value, _ in _series_crossings(scene))
    return float(xs[-1] - xs[-2])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the pixel-identical alias induced by irrelevant list storage order."""

    twin = copy.deepcopy(scene)
    twin["series"] = list(reversed(twin["series"]))
    return [("reverse_series_storage_order", twin)]


def _scaled_point(x: float, y: float, scale: int) -> tuple[int, int]:
    return (round(x * scale), round(y * scale))


def _line_y(target_y: float, cross_x: float, slope: float, x: float) -> float:
    return target_y + slope * (x - cross_x)


def render(scene: dict) -> Image.Image:
    """Render a clean RGB plot; every answer-bearing value affects visible ink."""

    width = int(scene["width"])
    height = int(scene["height"])
    left = float(scene["plot_left"])
    right = float(scene["plot_right"])
    top = float(scene["plot_top"])
    bottom = float(scene["plot_bottom"])
    target_y = float(scene["target_y"])
    scale = 3

    image = Image.new("RGB", (width * scale, height * scale), (250, 250, 248))
    draw = ImageDraw.Draw(image)

    # Plot paper and subtle measurement grid.
    draw.rectangle(
        [_scaled_point(left, top, scale), _scaled_point(right, bottom, scale)],
        fill=(255, 255, 255),
    )
    grid = (224, 226, 226)
    for index in range(1, 6):
        x = left + (right - left) * index / 6.0
        draw.line(
            [_scaled_point(x, top, scale), _scaled_point(x, bottom, scale)],
            fill=grid,
            width=scale,
        )
    for index in range(1, 5):
        y = top + (bottom - top) * index / 5.0
        draw.line(
            [_scaled_point(left, y, scale), _scaled_point(right, y, scale)],
            fill=grid,
            width=scale,
        )

    # The observable target boundary is dashed and shared by all four trends.
    target_color = (82, 82, 82)
    dash = 12.0
    gap = 7.0
    x = left + 2.0
    while x < right - 2.0:
        x2 = min(x + dash, right - 2.0)
        draw.line(
            [_scaled_point(x, target_y, scale), _scaled_point(x2, target_y, scale)],
            fill=target_color,
            width=3 * scale,
        )
        x += dash + gap

    # Stable label sorting makes storage-order permutations exactly pixel-identical.
    series_by_label = sorted(
        scene["series"], key=lambda item: COLOR_ORDER.index(str(item["label"]))
    )
    x0, x1 = left + 8.0, right - 8.0
    marker_xs = [left + (right - left) * part for part in (0.16, 0.37, 0.63, 0.84)]
    for item in series_by_label:
        cross_x = float(item["cross_x"])
        slope = float(item["slope"])
        color = tuple(int(channel) for channel in item["rgb"])
        y0 = _line_y(target_y, cross_x, slope, x0)
        y1 = _line_y(target_y, cross_x, slope, x1)
        draw.line(
            [_scaled_point(x0, y0, scale), _scaled_point(x1, y1, scale)],
            fill=color,
            width=5 * scale,
        )
        radius = 4.5 * scale
        for marker_x in marker_xs:
            marker_y = _line_y(target_y, cross_x, slope, marker_x)
            mx, my = marker_x * scale, marker_y * scale
            draw.ellipse(
                [round(mx - radius), round(my - radius), round(mx + radius), round(my + radius)],
                fill=color,
            )

    # A crisp plot frame is non-answer-bearing and visually exposes the coordinate area.
    frame = (38, 43, 46)
    draw.rectangle(
        [_scaled_point(left, top, scale), _scaled_point(right, bottom, scale)],
        outline=frame,
        width=2 * scale,
    )
    image = image.resize((width, height), Image.Resampling.LANCZOS)

    # Text is deliberately black, not target gray or any answer color.
    final_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    final_draw.text((left + 4, top - 18), "RESPONSE TRENDS", fill=(25, 28, 30), font=font)
    final_draw.text((right - 47, target_y - 15), "target", fill=(25, 28, 30), font=font)
    final_draw.text(((left + right) / 2 - 30, bottom + 18), "measurement", fill=(25, 28, 30), font=font)
    return image.convert("RGB")
