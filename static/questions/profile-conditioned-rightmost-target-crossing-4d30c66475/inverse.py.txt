"""Independent image-only inverse arm for rightmost target crossing."""

from __future__ import annotations

import numpy as np
from PIL import Image

# These are public rendering conventions, not latent scene values. Multiple shades
# per semantic color make label-preserving palette perturbations recoverable.
COLOR_PROTOTYPES: dict[str, tuple[tuple[int, int, int], ...]] = {
    "red": ((211, 54, 48), (190, 43, 62), (226, 65, 47)),
    "blue": ((39, 105, 205), (38, 92, 181), (48, 121, 222)),
    "green": ((34, 151, 91), (24, 132, 76), (42, 166, 99)),
    "orange": ((238, 132, 28), (222, 112, 18), (246, 146, 35)),
}


def _semantic_color_masks(array: np.ndarray) -> dict[str, np.ndarray]:
    pixels = array.astype(np.float32)
    labels = tuple(COLOR_PROTOTYPES)
    distances = []
    for label in labels:
        variants = np.asarray(COLOR_PROTOTYPES[label], dtype=np.float32)
        delta = pixels[:, :, None, :] - variants[None, None, :, :]
        distances.append(np.min(np.sum(delta * delta, axis=3), axis=2))
    stack = np.stack(distances, axis=2)
    best = np.argmin(stack, axis=2)
    best_distance = np.min(stack, axis=2)
    return {
        label: (best == index) & (best_distance <= 68.0**2)
        for index, label in enumerate(labels)
    }


def _target_height(array: np.ndarray) -> float:
    # The target is the only neutral mid-dark dashed structure; the frame/text are
    # darker and grid/background are lighter. Long horizontal support rejects noise.
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    intensity = array.mean(axis=2)
    neutral = (spread <= 5) & (intensity >= 52) & (intensity <= 125)
    ys, xs = np.nonzero(neutral)
    if len(xs) < 250:
        raise ValueError("dashed target is absent or too damaged")
    counts = np.bincount(ys, minlength=array.shape[0])
    peak = int(np.argmax(counts))
    support = ys[np.abs(ys - peak) <= 2]
    if len(support) < 200:
        raise ValueError("dashed target lacks horizontal support")
    # Pixel indices name upper-left corners; averaging the two antialiased core
    # rows recovers the half-pixel center more accurately than choosing one row.
    return float(np.mean(support))


def _fit_colored_line(mask: np.ndarray, label: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 350:
        raise ValueError(f"{label} trend is absent or too damaged")
    design = np.column_stack((xs.astype(np.float64), np.ones(len(xs))))
    slope, intercept = np.linalg.lstsq(design, ys.astype(np.float64), rcond=None)[0]
    if abs(float(slope)) < 0.075:
        raise ValueError(f"{label} trend is too flat for a stable crossing")
    residual = np.abs(ys - (slope * xs + intercept))
    if float(np.median(residual)) > 5.0:
        raise ValueError(f"{label} pixels do not form a stable trend")
    return float(slope), float(intercept)


def decision_from_image(image: Image.Image) -> str:
    """Recover and rank all four target intersections from final PNG pixels only."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    target_y = _target_height(array)
    crossings: list[tuple[float, str]] = []
    for label, mask in _semantic_color_masks(array).items():
        slope, intercept = _fit_colored_line(mask, label)
        cross_x = (target_y - intercept) / slope
        if not (-5.0 <= cross_x <= array.shape[1] + 5.0):
            raise ValueError(f"{label} crossing lies outside the visible image")
        crossings.append((float(cross_x), label))
    crossings.sort()
    if crossings[-1][0] - crossings[-2][0] < 2.0:
        raise ValueError("rightmost crossings are unresolved at raster precision")
    return crossings[-1][1]
