"""Independent image-only inverse arm for the slope-code world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

NAVY = np.array((28, 48, 72), dtype=np.int16)
CYAN = np.array((38, 190, 198), dtype=np.int16)
CREAM = np.array((245, 224, 170), dtype=np.int16)
COLORS = {
    "red": np.array((219, 67, 66), dtype=np.int16),
    "green": np.array((52, 158, 94), dtype=np.int16),
    "blue": np.array((68, 105, 205), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 40) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        count = 0
        minx = maxx = int(sx)
        miny = maxy = int(sy)
        while queue:
            y, x = queue.popleft()
            count += 1
            minx, maxx = min(minx, x), max(maxx, x)
            miny, maxy = min(miny, y), max(maxy, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= minimum:
            output.append((minx, miny, maxx, maxy, count))
    return output


def _center(component: tuple[int, int, int, int, int]) -> tuple[float, float]:
    x0, y0, x1, y1, _ = component
    return ((x0 + x1) / 2.0, (y0 + y1) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy = sorted(_components(_mask(array, NAVY), minimum=500), key=lambda c: _center(c)[0])
    navy = [component for component in navy if _center(component)[1] < image.height * 0.42]
    if len(navy) != 5:
        raise ValueError(f"expected five reference disks, found {len(navy)}")
    diameters = [max(c[2] - c[0] + 1, c[3] - c[1] + 1) for c in navy]
    if min(abs(a - b) for a, b in zip(diameters, diameters[1:])) < 6:
        raise ValueError("adjacent disk sizes are ambiguous")
    target = [int(diameters[i + 1] > diameters[i]) for i in range(4)]

    tags: list[tuple[float, str]] = []
    for name, color in COLORS.items():
        pieces = _components(_mask(array, color), minimum=300)
        if len(pieces) != 1:
            raise ValueError(f"expected one {name} tag")
        _, y = _center(pieces[0])
        if y < image.height * 0.50:
            raise ValueError("tag outside candidate region")
        tags.append((y, name))
    tags.sort()

    cells = []
    for bit, color in ((1, CYAN), (0, CREAM)):
        for component in _components(_mask(array, color), minimum=300):
            x, y = _center(component)
            if y > image.height * 0.50:
                cells.append((y, x, bit))
    if len(cells) != 12:
        raise ValueError(f"expected twelve code cells, found {len(cells)}")

    rows: list[list[int]] = []
    for tag_y, _ in tags:
        row = sorted((x, bit) for y, x, bit in cells if abs(y - tag_y) < 24)
        if len(row) != 4:
            raise ValueError("candidate row is incomplete")
        rows.append([bit for _, bit in row])
    matches = [name for (_, name), code in zip(tags, rows) if code == target]
    if len(matches) != 1:
        raise ValueError("no unique pixel-level match")
    return matches[0]
