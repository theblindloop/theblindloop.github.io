"""Deterministic renderer and analytic arm for adjacent-size slope codes."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

NAVY = (28, 48, 72)
GOLD = (230, 166, 38)
CYAN = (38, 190, 198)
CREAM = (245, 224, 170)
TAG_RGB = {"red": (219, 67, 66), "green": (52, 158, 94), "blue": (68, 105, 205)}
BACKGROUND = (250, 249, 245)


def _bits_from_radii(radii: list[int]) -> list[int]:
    return [int(radii[i + 1] > radii[i]) for i in range(4)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # A permutation yields varied global rankings while keeping every adjacent
    # diameter difference at least 14 px. Many permutations share one slope code.
    radii = [19, 26, 33, 40, 47]
    while True:
        rng.shuffle(radii)
        target = _bits_from_radii(radii)
        # Avoid the two visually trivial monotone cases.
        if 1 <= sum(target) <= 3:
            break

    answer = ("red", "green", "blue")[int(seed) % 3]
    distractors: list[list[int]] = []
    masks = list(range(1, 16))
    rng.shuffle(masks)
    for mask in masks:
        altered = [bit ^ ((mask >> i) & 1) for i, bit in enumerate(target)]
        if altered != target and altered not in distractors:
            distractors.append(altered)
        if len(distractors) == 2:
            break

    tags = ["red", "green", "blue"]
    rng.shuffle(tags)
    by_tag = {answer: target}
    remaining = [tag for tag in tags if tag != answer]
    by_tag[remaining[0]] = distractors[0]
    by_tag[remaining[1]] = distractors[1]
    codes = [by_tag[tag] for tag in tags]
    return {
        "radii": radii,
        "codes": codes,
        "tags": tags,
        "width": 800,
        "height": 800,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    radii = [int(value) for value in scene["radii"]]
    codes = [[int(bit) for bit in code] for code in scene["codes"]]
    tags = [str(tag) for tag in scene["tags"]]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # The small gold diamond labels the five detached disks as the reference.
    draw.polygon([(400, 128), (414, 142), (400, 156), (386, 142)], fill=GOLD)
    for x, radius in zip((200, 300, 400, 500, 600), radii, strict=True):
        draw.ellipse((x - radius, 225 - radius, x + radius, 225 + radius), fill=NAVY)

    for y, tag, code in zip((500, 575, 650), tags, codes, strict=True):
        draw.ellipse((244, y - 15, 274, y + 15), fill=TAG_RGB[tag])
        for x, bit in zip((330, 380, 430, 480), code, strict=True):
            color = CYAN if bit else CREAM
            draw.rounded_rectangle((x - 16, y - 16, x + 16, y + 16), radius=5, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    target = _bits_from_radii([int(value) for value in scene["radii"]])
    codes = [[int(bit) for bit in code] for code in scene["codes"]]
    tags = [str(tag) for tag in scene["tags"]]
    matches = [tag for tag, code in zip(tags, codes, strict=True) if code == target]
    if len(matches) != 1:
        raise ValueError("scene has no unique matching code")
    return matches[0]


def margin(scene: dict) -> float:
    radii = [int(value) for value in scene["radii"]]
    target = _bits_from_radii(radii)
    codes = [[int(bit) for bit in code] for code in scene["codes"]]
    size_clearance = min(2 * abs(a - b) for a, b in zip(radii, radii[1:]))
    mismatches = sorted(sum(a != b for a, b in zip(target, code)) for code in codes)
    uniqueness_clearance = 12 * mismatches[1] if len(mismatches) >= 2 and mismatches[0] == 0 else 0
    return float(min(size_clearance, uniqueness_clearance))


def is_quarantined(scene: dict) -> bool:
    radii = [int(value) for value in scene["radii"]]
    target = _bits_from_radii(radii)
    codes = [[int(bit) for bit in code] for code in scene["codes"]]
    matches = sum(code == target for code in codes)
    return matches != 1 or min(2 * abs(a - b) for a, b in zip(radii, radii[1:])) < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
