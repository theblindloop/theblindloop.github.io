from __future__ import annotations

import random
from typing import Dict, List, Tuple

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 480
WALL = 14
COLORS = {
    "red": (220, 52, 52),
    "green": (38, 158, 82),
    "blue": (48, 96, 205),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    order = ["red", "green", "blue"]
    rng.shuffle(order)
    return {"transform": rng.randrange(4), "color_order": order}


def _rot(point: Tuple[int, int], turns: int) -> Tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = WIDTH - y, x
    return x, y


def _line(draw: ImageDraw.ImageDraw, a: Tuple[int, int], b: Tuple[int, int], fill) -> None:
    draw.line([a, b], fill=fill, width=WALL)


def render(scene: dict) -> Image.Image:
    turns = int(scene["transform"]) % 4
    color_order = list(scene["color_order"])
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    black = (22, 25, 29)

    # A square floor plan split into three vertical chambers.  The short
    # horizontal wall at left is attached at one end but has a wide, visible
    # passage around its other end, so it does not split that chamber.
    walls = [
        ((60, 60), (420, 60)), ((420, 60), (420, 420)),
        ((420, 420), (60, 420)), ((60, 420), (60, 60)),
        ((180, 60), (180, 420)), ((300, 60), (300, 420)),
        ((60, 240), (150, 240)),
    ]
    for a, b in walls:
        _line(draw, _rot(a, turns), _rot(b, turns), black)

    # Roles: same incident chamber (attached slit), then two distinct-chamber
    # partitions.  All colored strokes have equal length and width.
    candidates = [
        ((72, 240), (136, 240)),
        ((180, 112), (180, 176)),
        ((300, 304), (300, 368)),
    ]
    for color, (a, b) in zip(color_order, candidates):
        _line(draw, _rot(a, turns), _rot(b, turns), COLORS[color])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["color_order"][0])


def margin(scene: dict) -> float:
    # Component-equality scores are 1, 0, and 0.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {"transform": int(scene["transform"]) + 4, "color_order": list(scene["color_order"])}
    return [("full_rotation_alias", twin)]
