"""Pixel-only inverse for two disconnected orthogonal route centerlines.

This module deliberately imports no renderer, scene, prompt, generator, or
verification code. It sees only the final RGB raster. The bright line palette
is segmented, its two connected components are recovered, and horizontal and
vertical straight-run cores are counted independently. A simple orthogonal
chain with S straight segments has S-1 bends.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "paired-bend-parity-oracle-0.1.0"

_CENTER_COLORS = np.asarray(
    [
        (55, 205, 232),
        (45, 194, 167),
    ],
    dtype=np.int16,
)


class OracleAbstention(ValueError):
    """Raised when the raster does not expose exactly two valid route chains."""


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            component = np.zeros_like(mask, dtype=bool)
            yy, xx = zip(*points)
            component[np.asarray(yy), np.asarray(xx)] = True
            output.append(component)
    return output


def _straight_core(mask: np.ndarray, horizontal: bool) -> np.ndarray:
    core = np.zeros_like(mask, dtype=bool)
    if horizontal:
        core[:, 2:-2] = mask[:, 2:-2] & mask[:, :-4] & mask[:, 4:]
    else:
        core[2:-2, :] = mask[2:-2, :] & mask[:-4, :] & mask[4:, :]
    return core


def _bend_count(component: np.ndarray) -> int:
    horizontal = len(_components(_straight_core(component, True), minimum=5))
    vertical = len(_components(_straight_core(component, False), minimum=5))
    segment_count = horizontal + vertical
    if segment_count not in {3, 4, 5, 6}:
        raise OracleAbstention(
            f"route has {horizontal} horizontal and {vertical} vertical runs"
        )
    if abs(horizontal - vertical) > 1:
        raise OracleAbstention("route run orientations do not alternate")
    return segment_count - 1


def _measure_bends(image: Image.Image) -> tuple[int, int]:
    # int32 avoids overflow when squared RGB differences exceed 181.
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - _CENTER_COLORS[None, None, :, :]) ** 2).sum(axis=3)
    centerline = distances.min(axis=2) <= 18 * 18
    routes = _components(centerline, minimum=24)
    if len(routes) != 2:
        raise OracleAbstention(f"expected two bright centerlines, found {len(routes)}")
    bends = sorted(_bend_count(route) for route in routes)
    return bends[0], bends[1]


def decision_from_image(image: Image.Image) -> str:
    bends_a, bends_b = _measure_bends(image)
    return "yes" if bends_a % 2 == bends_b % 2 else "no"
