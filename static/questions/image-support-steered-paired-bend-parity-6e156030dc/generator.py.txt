"""Independent latent sampler and raster renderer for paired bend parity."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "paired-bend-parity-renderer-0.1.0"
WIDTH = 768
HEIGHT = 768

PALETTES = {
    "ocean": {
        "route": (22, 45, 70),
        "center": (55, 205, 232),
        "start": (35, 105, 210),
        "finish": (238, 132, 38),
        "faint": (218, 224, 230),
    },
    "harbor": {
        "route": (35, 43, 68),
        "center": (45, 194, 167),
        "start": (42, 98, 190),
        "finish": (226, 116, 43),
        "faint": (224, 220, 230),
    },
}

_LENGTHS = {
    2: (21, 27, 21),
    3: (18, 17, 17, 18),
    4: (15, 14, 14, 14, 15),
    5: (15, 11, 11, 11, 11, 15),
}

# Each pair is two grid-cell centers separated by exactly two cells diagonally.
# This randomizes absolute position while keeping the joint evidence focal.
_ANCHOR_PAIRS = (
    ((240, 240), (432, 432)),
    ((240, 432), (432, 240)),
    ((336, 144), (528, 336)),
    ((144, 336), (336, 528)),
    ((336, 528), (528, 336)),
    ((432, 240), (240, 432)),
)


def _transform(point: tuple[int, int], code: int) -> tuple[int, int]:
    x, y = point
    if code >= 4:
        x = -x
    for _ in range(code % 4):
        x, y = -y, x
    return x, y


def _make_route(bends: int, transform_code: int) -> list[list[int]]:
    """Create a non-self-intersecting orthogonal chain with fixed total length."""
    lengths = _LENGTHS[bends]
    points = [(0, 0)]
    x = y = 0
    for index, length in enumerate(lengths):
        if index % 2 == 0:
            x += length
        else:
            y += length
        points.append((x, y))
    transformed = [_transform(point, transform_code) for point in points]
    min_x = min(point[0] for point in transformed)
    max_x = max(point[0] for point in transformed)
    min_y = min(point[1] for point in transformed)
    max_y = max(point[1] for point in transformed)
    center_x = (min_x + max_x) // 2
    center_y = (min_y + max_y) // 2
    return [[x - center_x, y - center_y] for x, y in transformed]


def _bend_count(points: Iterable[Iterable[int]]) -> int:
    points = [tuple(map(int, point)) for point in points]
    if len(points) < 3:
        raise ValueError("a route needs at least two segments")
    orientations: list[str] = []
    for first, second in zip(points, points[1:]):
        dx, dy = second[0] - first[0], second[1] - first[1]
        if (dx == 0) == (dy == 0):
            raise ValueError("route segments must be nonzero and orthogonal")
        orientations.append("h" if dy == 0 else "v")
    bends = sum(a != b for a, b in zip(orientations, orientations[1:]))
    if bends != len(orientations) - 1:
        raise ValueError("every interior vertex must be a square corner")
    return bends


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    same = int(seed) % 2 == 0
    bends_a = 2 + rng.randrange(4)
    compatible = [
        bends
        for bends in range(2, 6)
        if ((bends % 2) == (bends_a % 2)) == same
    ]
    bends_b = rng.choice(compatible)
    anchor_a, anchor_b = rng.choice(_ANCHOR_PAIRS)
    # Independently orient each route so endpoint direction and handedness are
    # balanced against the answer.
    route_a = _make_route(bends_a, rng.randrange(8))
    route_b = _make_route(bends_b, rng.randrange(8))

    decorations: list[list[int]] = []
    while len(decorations) < 7:
        x = rng.randrange(48, WIDTH - 48)
        y = rng.randrange(48, HEIGHT - 48)
        if min(math.dist((x, y), anchor_a), math.dist((x, y), anchor_b)) < 105:
            continue
        dx = rng.choice((-18, -14, 14, 18))
        dy = rng.choice((-5, 0, 5))
        decorations.append([x, y, x + dx, y + dy])
    return {
        "route_a": route_a,
        "route_b": route_b,
        "anchor_a": list(anchor_a),
        "anchor_b": list(anchor_b),
        "palette_id": rng.choice(sorted(PALETTES)),
        "decorations": decorations,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _place(points: Iterable[Iterable[int]], anchor: Iterable[int]) -> list[tuple[int, int]]:
    ax, ay = map(int, anchor)
    return [(ax + int(point[0]), ay + int(point[1])) for point in points]


def _draw_route(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], palette: dict) -> None:
    start, finish = points[0], points[-1]
    # Endpoint glyphs are drawn first so the full centerline remains visibly
    # continuous through them.
    sx, sy = start
    draw.ellipse((sx - 9, sy - 9, sx + 9, sy + 9), fill="white", outline=palette["start"], width=5)
    fx, fy = finish
    draw.rectangle((fx - 9, fy - 9, fx + 9, fy + 9), fill="white", outline=palette["finish"], width=5)
    draw.line(points, fill=palette["route"], width=11, joint="curve")
    draw.line(points, fill=palette["center"], width=3, joint="curve")
    # Drawing the line last leaves its centerline continuously traceable while
    # most of each endpoint outline remains exposed around it.


def render(scene: dict) -> Image.Image:
    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    image = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(image)
    palette = PALETTES[str(scene["palette_id"])]
    for x1, y1, x2, y2 in scene["decorations"]:
        draw.line((x1, y1, x2, y2), fill=palette["faint"], width=3)
    for anchor in (scene["anchor_a"], scene["anchor_b"]):
        ax, ay = map(int, anchor)
        draw.rounded_rectangle(
            (ax - 38, ay - 38, ax + 38, ay + 38),
            radius=10,
            outline=palette["faint"],
            width=2,
        )
    _draw_route(draw, _place(scene["route_a"], scene["anchor_a"]), palette)
    _draw_route(draw, _place(scene["route_b"], scene["anchor_b"]), palette)
    return image


def analytic_gold(scene: dict) -> str:
    parity_a = _bend_count(scene["route_a"]) % 2
    parity_b = _bend_count(scene["route_b"]) % 2
    return "yes" if parity_a == parity_b else "no"


def margin(scene: dict) -> float:
    lengths = []
    for key in ("route_a", "route_b"):
        points = scene[key]
        lengths.extend(
            abs(int(b[0]) - int(a[0])) + abs(int(b[1]) - int(a[1]))
            for a, b in zip(points, points[1:])
        )
    # Four pixels is the inverse arm's minimum resolved straight-run support.
    return float(min(lengths) - 4)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) <= 1.0 or any(
            _bend_count(scene[key]) not in {2, 3, 4, 5} for key in ("route_a", "route_b")
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The parameterization has no pixel-identical aliases.
    return []
