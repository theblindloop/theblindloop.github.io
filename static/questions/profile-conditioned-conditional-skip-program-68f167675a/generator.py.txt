"""Deterministic visual world for a tiny forward-only branching program."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1180, 520
INK = (48, 52, 59)
BLUE = (78, 121, 167)
MAGENTA = (225, 87, 143)
GREEN = (89, 161, 79)
ORANGE = (242, 142, 43)
PALE = (246, 247, 249)
GRAY = (150, 155, 164)
WHITE = (255, 255, 255)


def _replay(scene: dict) -> tuple[list[bool], list[int], list[int]]:
    states = [bool(value) for value in scene["initial"]]
    query = int(scene["query"])
    operations = scene["operations"]
    executed: list[int] = []
    skipped: list[int] = []
    pc = 0
    while pc < len(operations):
        executed.append(pc)
        operation = operations[pc]
        address = int(operation["address"]) - 1
        if operation["action"] == "flip":
            states[address] = not states[address]
            pc += 1
        elif operation["action"] == "skip":
            if states[address] and pc + 1 < len(operations):
                skipped.append(pc + 1)
                pc += 2
            else:
                pc += 1
        else:
            raise ValueError("unknown operation")
    return states, executed, skipped


def sample_scene(seed: int) -> dict:
    """Sample a balanced program; the seed parity selects the desired class."""
    rng = random.Random(int(seed))
    desired = bool(int(seed) & 1)
    for _ in range(10000):
        initial = [bool(rng.getrandbits(1)) for _ in range(3)]
        query = int(seed) % 3 + 1
        actions = ["flip"] * 4 + ["skip"] * 4
        rng.shuffle(actions)
        operations = [
            {"action": action, "address": rng.randint(1, 3)} for action in actions
        ]
        scene = {"initial": initial, "query": query, "operations": operations}
        final, executed, skipped = _replay(scene)
        executed_query_flip = any(
            operations[i]["action"] == "flip" and operations[i]["address"] == query
            for i in executed
        )
        if final[query - 1] == desired and skipped and executed_query_flip:
            return scene
    raise RuntimeError("could not sample a balanced conditional program")


def analytic_gold(scene: dict) -> str:
    final, _, _ = _replay(scene)
    return "ON" if final[int(scene["query"]) - 1] else "OFF"


def margin(scene: dict) -> float:
    """Runner-up margin: smallest designed stroke/radius clearance in pixels."""
    return 4.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_dots(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int) -> None:
    radius = 5
    spacing = 18
    start = cx - spacing * (count - 1) // 2
    for index in range(count):
        x = start + index * spacing
        draw.ellipse((x - radius, cy - radius, x + radius, cy + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    initial = scene["initial"]
    query = int(scene["query"])
    operations = scene["operations"]
    if len(initial) != 3 or len(operations) != 8:
        raise ValueError("expected three lamps and eight cards")

    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((30, 20), "START LAMPS", fill=INK, font=font)
    draw.text((370, 20), "READ CARDS LEFT TO RIGHT", fill=INK, font=font)
    draw.rounded_rectangle((25, 48, 335, 270), radius=12, fill=PALE, outline=INK, width=4)

    for index, state in enumerate(initial, start=1):
        cx, cy = 85 + (index - 1) * 95, 140
        if index == query:
            draw.ellipse((cx - 36, cy - 36, cx + 36, cy + 36), outline=MAGENTA, width=6)
        fill = BLUE if state else WHITE
        draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), fill=fill, outline=BLUE, width=5)
        _centered_dots(draw, cx, 220, index)
    draw.text((66, 180), "lamp", fill=GRAY, font=font)
    draw.text((161, 180), "lamp", fill=GRAY, font=font)
    draw.text((256, 180), "lamp", fill=GRAY, font=font)
    draw.text((75, 244), "solid = ON     hollow = OFF", fill=INK, font=font)

    card_y = 305
    card_w = 126
    gap = 17
    start_x = 28
    for index, operation in enumerate(operations):
        x = start_x + index * (card_w + gap)
        draw.rounded_rectangle(
            (x, card_y, x + card_w, card_y + 176), radius=9, fill=PALE, outline=INK, width=4
        )
        color = GREEN if operation["action"] == "flip" else ORANGE
        draw.rectangle((x + 11, card_y + 12, x + card_w - 11, card_y + 43), fill=color)
        word = "FLIP" if operation["action"] == "flip" else "SKIP"
        draw.text((x + 48, card_y + 22), word, fill=WHITE, font=font, anchor="mm")
        if operation["action"] == "flip":
            draw.arc((x + 45, card_y + 59, x + 81, card_y + 95), 30, 320, fill=GREEN, width=5)
        else:
            draw.line((x + 42, card_y + 77, x + 84, card_y + 77), fill=ORANGE, width=5)
            draw.polygon([(84 + x, card_y + 77), (73 + x, card_y + 68), (73 + x, card_y + 86)], fill=ORANGE)
        _centered_dots(draw, x + card_w // 2, card_y + 135, int(operation["address"]))
        draw.text((x + card_w // 2, card_y + 157), str(index + 1), fill=GRAY, font=font, anchor="mm")

    return image
