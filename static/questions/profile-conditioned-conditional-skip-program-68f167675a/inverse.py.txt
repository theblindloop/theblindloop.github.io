"""Independent PNG-only inverse for the conditional-skip lamp program."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

INK = np.array((48, 52, 59), dtype=np.int16)
BLUE = np.array((78, 121, 167), dtype=np.int16)
MAGENTA = np.array((225, 87, 143), dtype=np.int16)
GREEN = np.array((89, 161, 79), dtype=np.int16)
ORANGE = np.array((242, 142, 43), dtype=np.int16)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    values: list[dict] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        values.append({
            "area": len(points), "x0": min(xs), "x1": max(xs),
            "y0": min(ys), "y1": max(ys),
            "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys),
        })
    return values


def _dot_components(arr: np.ndarray) -> list[dict]:
    dots = []
    for comp in _components(_mask(arr, INK)):
        width = comp["x1"] - comp["x0"] + 1
        height = comp["y1"] - comp["y0"] + 1
        if 70 <= comp["area"] <= 120 and 9 <= width <= 13 and 9 <= height <= 13:
            dots.append(comp)
    return dots


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    blue = [
        c for c in _components(_mask(arr, BLUE))
        if 45 <= c["x1"] - c["x0"] + 1 <= 55 and 45 <= c["y1"] - c["y0"] + 1 <= 55
    ]
    if len(blue) != 3:
        raise ValueError(f"expected three lamp components, found {len(blue)}")
    blue.sort(key=lambda c: c["cx"])
    if max(c["cy"] for c in blue) - min(c["cy"] for c in blue) > 5:
        raise ValueError("lamp row is ambiguous")
    states = [c["area"] > 1000 for c in blue]

    rings = [
        c for c in _components(_mask(arr, MAGENTA))
        if 65 <= c["x1"] - c["x0"] + 1 <= 80 and 65 <= c["y1"] - c["y0"] + 1 <= 80
    ]
    if len(rings) != 1:
        raise ValueError("query ring is missing or ambiguous")
    query = min(range(3), key=lambda i: abs(blue[i]["cx"] - rings[0]["cx"]))
    if abs(blue[query]["cx"] - rings[0]["cx"]) > 5:
        raise ValueError("query ring does not bind a lamp")

    headers = []
    for action, color in (("flip", GREEN), ("skip", ORANGE)):
        for comp in _components(_mask(arr, color)):
            width = comp["x1"] - comp["x0"] + 1
            height = comp["y1"] - comp["y0"] + 1
            if width >= 90 and 25 <= height <= 40 and comp["area"] >= 2200:
                headers.append((comp["cx"], comp["cy"], action))
    headers.sort()
    if len(headers) != 8 or any(headers[i + 1][0] - headers[i][0] < 100 for i in range(7)):
        raise ValueError(f"expected eight separated action headers, found {len(headers)}")

    dots = _dot_components(arr)
    operations = []
    for cx, cy, action in headers:
        local = [d for d in dots if abs(d["cx"] - cx) < 32 and 75 < d["cy"] - cy < 135]
        if len(local) not in (1, 2, 3):
            raise ValueError("card address dots are missing or ambiguous")
        operations.append((action, len(local)))

    pc = 0
    while pc < len(operations):
        action, address = operations[pc]
        index = address - 1
        if action == "flip":
            states[index] = not states[index]
            pc += 1
        elif states[index] and pc + 1 < len(operations):
            pc += 2
        else:
            pc += 1
    return "ON" if states[query] else "OFF"
