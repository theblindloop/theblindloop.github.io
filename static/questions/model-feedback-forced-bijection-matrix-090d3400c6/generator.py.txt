"""Deterministic compatibility-matrix scenes for forced bijection reasoning."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 700
N = 8
GRID_X = 330
GRID_Y = 132
CELL = 62
ROW_BADGE_X = 269

BG = (244, 241, 233)
PAPER = (255, 255, 253)
INK = (35, 43, 54)
GRID = (166, 176, 186)
DOT = (33, 105, 164)
TARGET = (205, 65, 55)
AMBER = (232, 174, 54)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _permutation_with_value_at(rng: random.Random, value: int, position: int) -> list[int]:
    rest = [item for item in range(N) if item != value]
    rng.shuffle(rest)
    rest.insert(position, value)
    return rest


def _perfect_matchings(allowed: list[list[bool]]) -> list[tuple[int, ...]]:
    return [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(allowed[row][assignment[row]] for row in range(N))
    ]


def _propagation_depth(allowed: list[list[bool]], target_row: int) -> int:
    remaining_rows = set(range(N))
    remaining_cols = set(range(N))
    for depth in range(1, N + 2):
        singles = []
        for row in sorted(remaining_rows):
            choices = [column for column in remaining_cols if allowed[row][column]]
            if len(choices) == 1:
                singles.append((row, choices[0]))
        if not singles:
            return 0
        if any(row == target_row for row, _ in singles):
            return depth
        used = {column for _, column in singles}
        remaining_rows -= {row for row, _ in singles}
        remaining_cols -= used
    return 0


def sample_scene(seed: int) -> dict:
    """Create a permuted acyclic compatibility graph with one perfect matching."""

    rng = random.Random(seed ^ 0xB1E7C710)
    target_base = (seed * 5 + 1) % 6

    # The diagonal is a perfect matching. Every extra edge points forward in a
    # hidden topological order, so an alternating cycle (and hence a second
    # perfect matching) cannot exist. The order is not shown in the raster.
    base = [[False] * N for _ in range(N)]
    for row in range(N):
        base[row][row] = True
        if row + 1 < N:
            base[row][row + 1] = True
        if row + 2 < N:
            base[row][row + 2] = True
        if row + 3 < N and ((seed >> (row % 7)) & 1):
            base[row][row + 3] = True

    target_visual_row = (seed * 3 + 2) % N
    answer_visual_col = seed % N
    row_order = _permutation_with_value_at(rng, target_base, target_visual_row)
    col_order = _permutation_with_value_at(rng, target_base, answer_visual_col)
    allowed = [
        [bool(base[row_order[row]][col_order[column]]) for column in range(N)]
        for row in range(N)
    ]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "allowed": allowed,
        "target_row": target_visual_row,
    }
    # Sampling itself enforces the promised executable world.
    if len(_perfect_matchings(allowed)) != 1:
        raise RuntimeError("constructed matrix is not uniquely matchable")
    return scene


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    allowed = scene["allowed"]
    target_row = int(scene["target_row"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((58, 24, width - 58, height - 25), radius=24, fill=PAPER, outline=INK, width=3)
    title = "ONE-TO-ONE ASSIGNMENT"
    draw.text((width // 2, 53), title, font=_font(27, True), fill=INK, anchor="mm")
    draw.text((width // 2, 89), "DOT = ALLOWED   •   USE EACH COLUMN ONCE", font=_font(18, True), fill=(71, 82, 94), anchor="mm")

    # Column badges.
    for column in range(N):
        cx = GRID_X + column * CELL + CELL // 2
        draw.ellipse((cx - 21, GRID_Y - 44, cx + 21, GRID_Y - 2), fill=(245, 196, 80), outline=INK, width=2)
        draw.text((cx, GRID_Y - 23), chr(ord("A") + column), font=_font(23, True), fill=INK, anchor="mm")

    # Grid cells and compatibility dots.
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        draw.ellipse((ROW_BADGE_X - 21, cy - 21, ROW_BADGE_X + 21, cy + 21), fill=(239, 242, 245), outline=INK, width=2)
        if row == target_row:
            draw.ellipse((ROW_BADGE_X - 29, cy - 29, ROW_BADGE_X + 29, cy + 29), outline=TARGET, width=7)
            draw.polygon([(209, cy), (226, cy - 10), (226, cy + 10)], fill=TARGET)
            draw.text((185, cy), "TARGET", font=_font(16, True), fill=TARGET, anchor="rm")
        draw.text((ROW_BADGE_X, cy), str(row + 1), font=_font(21, True), fill=INK, anchor="mm")
        for column in range(N):
            x0 = GRID_X + column * CELL
            y0 = GRID_Y + row * CELL
            draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), outline=GRID, width=2)
            if bool(allowed[row][column]):
                cx = x0 + CELL // 2
                draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=DOT, outline=INK, width=2)
                draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=(226, 240, 249))

    draw.text((width // 2, 657), "Find the only complete assignment, then read the TARGET row's column.", font=_font(17), fill=(71, 82, 94), anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    allowed = scene["allowed"]
    target_row = int(scene["target_row"])
    matches = _perfect_matchings(allowed)
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one complete bijection")
    return chr(ord("A") + matches[0][target_row])


def margin(scene: dict) -> float:
    """Rounds of forced-singleton propagation before the target becomes fixed."""

    return float(_propagation_depth(scene["allowed"], int(scene["target_row"])))


def is_quarantined(scene: dict) -> bool:
    try:
        if len(_perfect_matchings(scene["allowed"])) != 1:
            return True
    except Exception:
        return True
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
