"""Independent fixed-layout PNG decoder and perfect-matching solver."""

from __future__ import annotations

import itertools

from PIL import Image

N = 8
GRID_X = 330
GRID_Y = 132
CELL = 62
ROW_BADGE_X = 269


def _is_blue(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return blue >= 120 and blue >= red + 35 and blue >= green + 20


def _is_red(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return red >= 150 and red >= green + 65 and red >= blue + 55


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (960, 700):
        raise ValueError("unexpected canvas size")
    pixels = raster.load()

    allowed: list[list[bool]] = []
    for row in range(N):
        decoded_row = []
        cy = GRID_Y + row * CELL + CELL // 2
        for column in range(N):
            cx = GRID_X + column * CELL + CELL // 2
            blue = sum(
                _is_blue(pixels[x, y])
                for y in range(cy - 12, cy + 13)
                for x in range(cx - 12, cx + 13)
            )
            decoded_row.append(blue > 260)
        allowed.append(decoded_row)

    ring_scores = []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        score = sum(
            _is_red(pixels[x, y])
            for y in range(cy - 31, cy + 32)
            for x in range(ROW_BADGE_X - 31, ROW_BADGE_X + 32)
            if 22 * 22 <= (x - ROW_BADGE_X) ** 2 + (y - cy) ** 2 <= 31 * 31
        )
        ring_scores.append(score)
    target_row = max(range(N), key=ring_scores.__getitem__)
    if ring_scores[target_row] < 110 or sum(score >= 110 for score in ring_scores) != 1:
        raise ValueError("target ring is missing or ambiguous")

    assignments = [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(allowed[row][assignment[row]] for row in range(N))
    ]
    if len(assignments) != 1:
        raise ValueError("visible matrix does not have exactly one complete assignment")
    return chr(ord("A") + assignments[0][target_row])
