"""Analytic sampler and raster renderer for calibrated run-length strips."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 760, 560
BACKGROUND = (248, 247, 242)
CYAN = (30, 160, 180)
GOLD = (228, 170, 45)
SCRAP = (112, 118, 124)
PALETTES = (
    ((35, 55, 90), (185, 211, 225)),
    ((92, 42, 65), (226, 193, 204)),
    ((35, 78, 55), (190, 220, 194)),
)
LABELS = ("up", "right", "down", "left")
BASE_UP = (
    "000010000",
    "000111000",
    "001111100",
    "011111110",
    "000111000",
    "000111000",
    "000111000",
    "000111000",
    "000111000",
)


def _rotate_clockwise(grid: list[list[int]]) -> list[list[int]]:
    return [list(row) for row in zip(*grid[::-1])]


def _templates() -> dict[str, list[list[int]]]:
    grid = [[int(v) for v in row] for row in BASE_UP]
    out = {}
    for label in LABELS:
        out[label] = [row[:] for row in grid]
        grid = _rotate_clockwise(grid)
    return out


def _to_runs(grid: list[list[int]]) -> list[list[list[int]]]:
    rows: list[list[list[int]]] = []
    for values in grid:
        runs: list[list[int]] = []
        for value in values:
            if runs and runs[-1][0] == value:
                runs[-1][1] += 1
            else:
                runs.append([value, 1])
        rows.append(runs)
    return rows


def _decode_runs_value(runs: Iterable[Iterable[Iterable[int]]]) -> list[list[int]]:
    grid = []
    for row in runs:
        values: list[int] = []
        for bit, length in row:
            if int(bit) not in (0, 1) or int(length) < 1:
                raise ValueError("invalid colored run")
            values.extend([int(bit)] * int(length))
        if len(values) != 9:
            raise ValueError("every expanded lane must contain nine tiles")
        grid.append(values)
    if len(grid) != 9:
        raise ValueError("nine lanes are required")
    return grid


def _decode_scene(scene: dict) -> list[list[int]]:
    return _decode_runs_value(scene["runs"])


def _classification(grid: list[list[int]]) -> tuple[str, str, int, dict[str, int]]:
    scored = {}
    for label, template in _templates().items():
        scored[label] = sum(
            int(grid[y][x] == template[y][x]) for y in range(9) for x in range(9)
        )
    order = sorted(scored, key=lambda label: (-scored[label], LABELS.index(label)))
    return order[0], order[1], scored[order[0]] - scored[order[1]], scored


def analytic_gold(scene: dict) -> str:
    return _classification(_decode_scene(scene))[0]


def margin(scene: dict) -> float:
    return float(_classification(_decode_scene(scene))[2])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def _make_scraps(count: int, rng: random.Random) -> list[list[int]]:
    if count < 1:
        raise ValueError("scrap equalizer needs at least one bar")
    total_width = 180
    widths = [total_width // count] * count
    for index in range(total_width % count):
        widths[index] += 1
    rng.shuffle(widths)
    scraps = []
    x, y = 510, 104
    for width in widths:
        if x + width > 742:
            x = 510
            y += 18
        scraps.append([x, y, width, 8])
        x += width + 4
    return scraps


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    direction_index = int(seed) % 4
    label = LABELS[direction_index]
    competitor = LABELS[(direction_index + 1 + (int(seed) // 24) % 3) % 4]
    templates = _templates()
    grid = [row[:] for row in templates[label]]
    different = [
        (y, x) for y in range(9) for x in range(9)
        if templates[label][y][x] != templates[competitor][y][x]
    ]
    rng.shuffle(different)
    mode = (int(seed) // 4) % 6
    wanted = {0: 0, 1: 1, 2: 2, 3: max(0, len(different) // 2 - 4),
              4: max(0, len(different) // 2 - 1), 5: 0}[mode]
    for y, x in different[:wanted]:
        grid[y][x] = templates[competitor][y][x]
    while _classification(grid)[0] != label or _classification(grid)[2] <= 0:
        wanted -= 1
        grid = [row[:] for row in templates[label]]
        for y, x in different[:max(0, wanted)]:
            grid[y][x] = templates[competitor][y][x]

    runs = _to_runs(grid)
    payload_count = sum(len(row) for row in runs)
    frame_x = 58 + rng.randrange(0, 68)
    frame_y = 62 + rng.randrange(0, 20)
    return {
        "runs": runs,
        "palette": [list(color) for color in PALETTES[(int(seed) // 4) % len(PALETTES)]],
        "unit_px": (12, 14, 16)[(int(seed) // 12) % 3],
        "frame_x": frame_x,
        "frame_y": frame_y,
        "row_offsets": [rng.randrange(0, 19) for _ in range(9)],
        "scraps": _make_scraps(56 - payload_count, rng),
    }


def render(scene: dict) -> Image.Image:
    runs = scene["runs"]
    dark, pale = (tuple(scene["palette"][0]), tuple(scene["palette"][1]))
    unit = int(scene["unit_px"])
    frame_x, frame_y = int(scene["frame_x"]), int(scene["frame_y"])
    offsets = [int(v) for v in scene["row_offsets"]]
    scraps = scene["scraps"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((frame_x, frame_y, frame_x + 330, frame_y + 428), outline=CYAN, width=4)
    draw.rectangle((frame_x + 18, frame_y - 27, frame_x + 18 + unit - 1, frame_y - 16), fill=GOLD)
    for row_index, row in enumerate(runs):
        y = frame_y + 18 + row_index * 44
        draw.line((frame_x + 7, y + 10, frame_x + 13, y + 10), fill=CYAN, width=3)
        x = frame_x + 25 + offsets[row_index]
        for bit, length in row:
            width = int(length) * unit
            draw.rectangle((x, y, x + width - 1, y + 21), fill=dark if int(bit) else pale)
            x += width + 7
    for x, y, width, height in scraps:
        draw.rectangle((int(x), int(y), int(x + width - 1), int(y + height - 1)), fill=SCRAP)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["scraps"] = list(reversed(twin["scraps"]))
    return [("permute_nonoverlapping_scrap_draw_order", twin)]
