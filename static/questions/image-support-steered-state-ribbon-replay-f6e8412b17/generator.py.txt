"""Deterministic connected state-ribbon renderer and analytic replay."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-ribbon-replay-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
SLOT_LABELS = ("A", "B", "C")
LANE_OFFSETS = (228, 298, 368)
STATION_X = (140, 218, 296, 374, 452)
SLAB_BOX = (28, 52, 484, 460)
OP_CENTER_OFFSET = 124
OP_COLORS = {
    -1: (244, 162, 97),   # down
    0: (123, 92, 172),    # hold
    1: (42, 157, 143),    # up
}
BACKGROUND = (248, 250, 252)
SLAB = (205, 233, 236)
SLAB_HIGHLIGHT = (231, 244, 245)
INK = (30, 55, 77)
TOKEN = (231, 111, 81)
DISTRACTOR = (108, 133, 145)


def _font(size: int):
    try:
        return ImageFont.load_default(size=size)
    except TypeError:  # Pillow versions without the size keyword
        return ImageFont.load_default()


def _scaled_box(box: tuple[float, ...], scale: int) -> list[int]:
    return [int(round(value * scale)) for value in box]


def _balanced_scene(seed: int) -> tuple[int, list[int], random.Random]:
    rng = random.Random(int(seed))
    initial = (int(seed) * 7 + 2) % 3
    desired = (int(seed) * 5 + int(seed) // 7 + 1) % 3
    first = [rng.choice((-1, 0, 1)) for _ in range(4)]
    needed = (desired - initial - sum(first)) % 3
    last = {0: 0, 1: 1, 2: -1}[needed]
    return initial, [*first, last], rng


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene; all label-bearing values are rendered."""

    initial, operations, rng = _balanced_scene(seed)
    shift_x = rng.choice((-8, 0, 8))
    shift_y = rng.choice((-6, 0, 6))
    token_radius = rng.choice((18, 22, 26))
    distractors = []
    for _ in range(2 + rng.randrange(4)):
        distractors.append(
            [
                round(rng.uniform(154, 430), 2),
                round(rng.uniform(246, 438), 2),
            ]
        )
    return {
        "initial_lane": initial,
        "operations": operations,
        "shift_x": shift_x,
        "shift_y": shift_y,
        "token_radius": token_radius,
        "distractors": distractors,
    }


def _draw_arrow(draw: ImageDraw.ImageDraw, cx: float, cy: float, op: int, ss: int) -> None:
    color = OP_COLORS[int(op)]
    if op == 1:
        points = [(cx, cy - 34), (cx - 23, cy - 7), (cx - 9, cy - 7),
                  (cx - 9, cy + 30), (cx + 9, cy + 30), (cx + 9, cy - 7),
                  (cx + 23, cy - 7)]
        draw.polygon([(int(round(x * ss)), int(round(y * ss))) for x, y in points], fill=color)
    elif op == -1:
        points = [(cx, cy + 34), (cx - 23, cy + 7), (cx - 9, cy + 7),
                  (cx - 9, cy - 30), (cx + 9, cy - 30), (cx + 9, cy + 7),
                  (cx + 23, cy + 7)]
        draw.polygon([(int(round(x * ss)), int(round(y * ss))) for x, y in points], fill=color)
    else:
        draw.rounded_rectangle(
            _scaled_box((cx - 27, cy - 10, cx + 27, cy + 10), ss),
            radius=8 * ss,
            fill=color,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one uninterrupted state slab and its ordered operation glyphs."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = int(scene["shift_x"]), int(scene["shift_y"])
    x0, y0, x1, y1 = (SLAB_BOX[0] + dx, SLAB_BOX[1] + dy,
                      SLAB_BOX[2] + dx, SLAB_BOX[3] + dy)

    draw.rounded_rectangle(
        _scaled_box((x0, y0, x1, y1), ss),
        radius=36 * ss,
        fill=SLAB,
        outline=INK,
        width=8 * ss,
    )

    font_big = _font(26 * ss)
    font_small = _font(22 * ss)
    # A single connected slab carries all lanes; these are rails, not separate panels.
    for index, offset in enumerate(LANE_OFFSETS):
        cy = y0 + offset
        draw.line(
            [(int((x0 + 90) * ss), int(cy * ss)),
             (int((x1 - 28) * ss), int(cy * ss))],
            fill=INK,
            width=7 * ss,
        )
        draw.text((int((x0 + 22) * ss), int((cy - 14) * ss)), SLOT_LABELS[index],
                  fill=INK, font=font_big)

    # Nuisance marks vary count and location but never share the token/operation colors.
    for px, py in scene["distractors"]:
        draw.ellipse(_scaled_box((px + dx - 7, py + dy - 7, px + dx + 7, py + dy + 7), ss),
                     fill=DISTRACTOR)

    # The only initial state marker is a ring plus a solid token; no final token is drawn.
    initial_y = y0 + LANE_OFFSETS[int(scene["initial_lane"])]
    token_x = x0 + 52
    radius = int(scene["token_radius"])
    draw.ellipse(
        _scaled_box((token_x - radius - 10, initial_y - radius - 10,
                     token_x + radius + 10, initial_y + radius + 10), ss),
        fill=SLAB_HIGHLIGHT,
        outline=INK,
        width=8 * ss,
    )
    draw.ellipse(
        _scaled_box((token_x - radius, initial_y - radius, token_x + radius, initial_y + radius), ss),
        fill=TOKEN,
    )

    # Numbered, separated glyphs sit on the same slab and are read left to right.
    op_y = y0 + OP_CENTER_OFFSET
    for number, (station, op) in enumerate(zip(STATION_X, scene["operations"], strict=True), start=1):
        cx = station + dx
        draw.rounded_rectangle(
            _scaled_box((cx - 32, op_y - 56, cx + 32, op_y + 56), ss),
            radius=16 * ss,
            outline=INK,
            width=5 * ss,
        )
        _draw_arrow(draw, cx, op_y, int(op), ss)
        draw.text((int((cx - 7) * ss), int((op_y - 89) * ss)), str(number),
                  fill=INK, font=font_small)

    if ss > 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def _replay_lane(initial_lane: int, operations: list[int] | tuple[int, ...]) -> int:
    lane = int(initial_lane)
    for operation in operations:
        lane = (lane + int(operation)) % 3
    return lane


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return only the terminal lane; the accessed fields are rendered inputs."""

    lane = _replay_lane(scene["initial_lane"], scene["operations"])
    return SLOT_LABELS[lane]


def margin(scene: dict[str, Any]) -> float:
    """Discrete glyph-separation margin; the sampled design has no near boundary."""

    return 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed: all sampled label inputs are visibly drawn."""

    return []
