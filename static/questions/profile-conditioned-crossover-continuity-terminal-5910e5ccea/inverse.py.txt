"""Independent pixel-only inverse arm for crossover-continuity tracing."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_RGB = {
    "green": (22, 163, 74),
    "cyan": (6, 182, 212),
    "orange": (249, 115, 22),
    "purple": (139, 92, 246),
    "blue": (37, 99, 235),
    "gray": (156, 163, 175),
}
_UNDERLAY = (71, 85, 105)
_CENTER = (17, 24, 39)
_ANSWERS = ("orange", "purple", "blue")


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float = 18.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) < radius * radius


def _components(mask: np.ndarray, minimum: int = 35) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            yy, xx = stack.pop()
            points.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def _line_support(mask: np.ndarray, p: tuple[float, float], q: tuple[float, float]) -> float:
    distance = math.dist(p, q)
    count = max(30, int(distance))
    hits = 0
    total = 0
    h, w = mask.shape
    trim = min(0.38, 19.0 / distance)
    for t in np.linspace(trim, 1.0 - trim, count):
        x = int(round(p[0] + t * (q[0] - p[0])))
        y = int(round(p[1] + t * (q[1] - p[1])))
        window = mask[max(0, y - 2):min(h, y + 3), max(0, x - 2):min(w, x + 3)]
        hits += bool(window.any())
        total += 1
    return hits / max(total, 1)


def _turn_angle(previous: tuple[float, float], current: tuple[float, float], nxt: tuple[float, float]) -> float:
    incoming = (current[0] - previous[0], current[1] - previous[1])
    outgoing = (nxt[0] - current[0], nxt[1] - current[1])
    cosine = (incoming[0] * outgoing[0] + incoming[1] * outgoing[1]) / (
        math.hypot(*incoming) * math.hypot(*outgoing)
    )
    return math.degrees(math.acos(max(-1.0, min(1.0, cosine))))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    nodes: list[tuple[float, float]] = []
    kinds: list[str] = []
    colors: dict[int, str] = {}

    specifications = [
        ("green", "root", 1, 45), ("cyan", "crossover", 2, 45),
        ("orange", "terminal", 1, 80), ("purple", "terminal", 1, 80),
        ("blue", "terminal", 1, 80), ("gray", "neutral", 3, 45),
    ]
    for color_name, kind, expected, minimum in specifications:
        components = _components(_near(arr, _RGB[color_name]), minimum)
        if len(components) != expected:
            raise ValueError(f"expected {expected} {color_name} markers, found {len(components)}")
        for x, y, _ in components:
            index = len(nodes)
            nodes.append((x, y))
            kinds.append(kind)
            if kind == "terminal":
                colors[index] = color_name

    route = _near(arr, _UNDERLAY, 18.0) | _near(arr, _CENTER, 18.0)
    adjacency = [set() for _ in nodes]
    for a in range(len(nodes)):
        for b in range(a + 1, len(nodes)):
            distance = math.dist(nodes[a], nodes[b])
            if 75.0 <= distance <= 195.0 and _line_support(route, nodes[a], nodes[b]) >= 0.92:
                adjacency[a].add(b)
                adjacency[b].add(a)
    edge_count = sum(map(len, adjacency)) // 2
    if edge_count != 8:
        raise ValueError(f"expected 8 route links, found {edge_count}")

    roots = [i for i, kind in enumerate(kinds) if kind == "root"]
    if len(roots) != 1 or len(adjacency[roots[0]]) != 1:
        raise ValueError("root route is ambiguous")
    previous = roots[0]
    current = next(iter(adjacency[previous]))
    for _ in range(12):
        if current in colors:
            return colors[current]
        choices = [node for node in adjacency[current] if node != previous]
        if kinds[current] == "crossover":
            if len(choices) != 3:
                raise ValueError("crossover does not have four spokes")
            ranked = sorted((_turn_angle(nodes[previous], nodes[current], nodes[nxt]), nxt) for nxt in choices)
            if ranked[0][0] > 5.0 or ranked[1][0] - ranked[0][0] < 45.0:
                raise ValueError("opposite spoke is not uniquely visible")
            nxt = ranked[0][1]
        elif len(choices) == 1:
            nxt = choices[0]
        else:
            raise ValueError("trace reached an unmarked choice or dead end")
        previous, current = current, nxt
    raise ValueError("trace did not reach a colored terminal")
