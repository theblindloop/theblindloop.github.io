"""Deterministic renderer and analytic arm for marked crossover tracing."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (255, 255, 255)
UNDERLAY = (71, 85, 105)
CENTERLINE = (17, 24, 39)
CYAN = (6, 182, 212)
GRAY = (156, 163, 175)
GREEN = (22, 163, 74)
COLORS = {
    "orange": (249, 115, 22),
    "purple": (139, 92, 246),
    "blue": (37, 99, 235),
}

_BASE_NODES = {
    "r": (42, 270),
    "x0": (160, 270),
    "t0": (160, 82),
    "t1": (160, 458),
    "b0": (268, 270),
    "x1": (350, 188),
    "t2": (450, 88),
    "q0": (450, 288),
    "q1": (250, 88),
}
_EDGE_PAIRS = [
    ("r", "x0"), ("x0", "t0"), ("x0", "t1"), ("x0", "b0"),
    ("b0", "x1"), ("x1", "t2"), ("x1", "q0"), ("x1", "q1"),
]


def _transform(point: tuple[int, int], turns: int, reflect: bool) -> list[int]:
    x, y = point[0] - 256, point[1] - 256
    if reflect:
        y = -y
    for _ in range(turns % 4):
        x, y = -y, x
    return [int(x + 256), int(y + 256)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    turns = int(seed) % 4
    reflect = bool((int(seed) // 4) % 2)
    nodes = {name: _transform(point, turns, reflect) for name, point in _BASE_NODES.items()}
    edges = [
        {"a": a, "b": b, "points": [list(nodes[a]), list(nodes[b])]}
        for a, b in _EDGE_PAIRS
    ]
    rng.shuffle(edges)
    winner = ("orange", "purple", "blue")[int(seed) % 3]
    others = [color for color in COLORS if color != winner]
    rng.shuffle(others)
    terminal_colors = {"t0": others[0], "t1": others[1], "t2": winner}
    return {
        "canvas": [CANVAS, CANVAS],
        "nodes": nodes,
        "node_kinds": {
            "r": "root", "x0": "crossover", "x1": "crossover", "b0": "bend",
            "t0": "terminal", "t1": "terminal", "t2": "terminal",
            "q0": "cap", "q1": "cap",
        },
        "edges": edges,
        "root_node": "r",
        "terminal_nodes": ["t0", "t1", "t2"],
        "terminal_colors": terminal_colors,
    }


def _adjacency(scene: dict) -> dict[str, list[str]]:
    adjacency = {name: [] for name in scene["nodes"]}
    for edge in scene["edges"]:
        a, b = edge["a"], edge["b"]
        if edge["points"][0] != scene["nodes"][a] or edge["points"][-1] != scene["nodes"][b]:
            raise ValueError("edge endpoints disagree with nodes")
        adjacency[a].append(b)
        adjacency[b].append(a)
    return adjacency


def _turn_angle(nodes: dict, previous: str, current: str, nxt: str) -> float:
    px, py = nodes[previous]
    cx, cy = nodes[current]
    nx, ny = nodes[nxt]
    incoming = (cx - px, cy - py)
    outgoing = (nx - cx, ny - cy)
    cosine = (incoming[0] * outgoing[0] + incoming[1] * outgoing[1]) / (
        math.hypot(*incoming) * math.hypot(*outgoing)
    )
    return math.degrees(math.acos(max(-1.0, min(1.0, cosine))))


def _trace(scene: dict) -> tuple[str, list[float]]:
    nodes = scene["nodes"]
    kinds = scene["node_kinds"]
    adjacency = _adjacency(scene)
    previous = scene["root_node"]
    if len(adjacency[previous]) != 1:
        raise ValueError("root must have one route")
    current = adjacency[previous][0]
    margins = []
    for _ in range(12):
        kind = kinds[current]
        if kind == "terminal":
            return current, margins
        choices = [node for node in adjacency[current] if node != previous]
        if kind == "bend":
            if len(choices) != 1:
                raise ValueError("bend must have one forward route")
            nxt = choices[0]
        elif kind == "crossover":
            if len(choices) != 3:
                raise ValueError("crossover must expose three forward spokes")
            ranked = sorted((_turn_angle(nodes, previous, current, node), node) for node in choices)
            margins.append(ranked[1][0] - ranked[0][0])
            if ranked[0][0] > 3.0 or ranked[1][0] < 45.0:
                raise ValueError("crossover has no unique opposite spoke")
            nxt = ranked[0][1]
        else:
            raise ValueError(f"trace ended at nonterminal {current}")
        previous, current = current, nxt
    raise ValueError("trace did not terminate")


def analytic_gold(scene: dict) -> str:
    terminal, _ = _trace(scene)
    return scene["terminal_colors"][terminal]


def margin(scene: dict) -> float:
    _, margins = _trace(scene)
    return float(min(margins))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    lengths = [math.dist(nodes[e["a"]], nodes[e["b"]]) for e in scene["edges"]]
    separations = [
        math.dist(nodes[a], nodes[b])
        for index, a in enumerate(nodes)
        for b in list(nodes)[index + 1:]
    ]
    return margin(scene) < 45.0 or min(lengths) < 80.0 or min(separations) < 55.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_storage_order", twin)]


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    root_name = scene["root_node"]
    if scene["node_kinds"].get(root_name) != "root":
        raise ValueError("root_node must name the visibly green root marker")
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for edge in scene["edges"]:
        draw.line([tuple(point) for point in edge["points"]], fill=UNDERLAY, width=11)
    for edge in scene["edges"]:
        draw.line([tuple(point) for point in edge["points"]], fill=CENTERLINE, width=3)

    nodes = scene["nodes"]
    for name, kind in scene["node_kinds"].items():
        x, y = nodes[name]
        if kind == "crossover":
            draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=BACKGROUND, outline=CYAN, width=6)
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=CENTERLINE)
        elif kind == "bend":
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=GRAY, outline=CENTERLINE, width=2)
        elif kind == "root":
            draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=BACKGROUND, outline=GREEN, width=6)
        elif kind == "terminal":
            color = COLORS[scene["terminal_colors"][name]]
            draw.rectangle((x - 14, y - 14, x + 14, y + 14), fill=color, outline=BACKGROUND, width=2)
        elif kind == "cap":
            draw.rectangle((x - 10, y - 10, x + 10, y + 10), fill=GRAY, outline=BACKGROUND, width=2)
    return image
