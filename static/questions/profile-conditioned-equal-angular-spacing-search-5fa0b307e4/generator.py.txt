"""Deterministic latent sampler and raster renderer for angular-gap search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 560
SUPERSAMPLE = 3
LABELS = ("A", "B", "C", "D")
PANEL_CENTERS = ((205.0, 160.0), (555.0, 160.0), (205.0, 400.0), (555.0, 400.0))
PANEL_BOXES = ((35, 35, 375, 275), (385, 35, 725, 275), (35, 285, 375, 525), (385, 285, 725, 525))
ORBIT_RADIUS = 70.0
DOT_RADIUS = 9.0
CENTER_RADIUS = 5.5
QUARANTINE_MARGIN_DEG = 8.0
PALETTES = (
    ("#173F5F", "#F4A261"),
    ("#6A1B9A", "#00A896"),
    ("#C1121F", "#3A86FF"),
)


def _angles_from_gaps(rotation: float, gaps: list[float]) -> list[float]:
    values = [rotation % 360.0]
    for gap in gaps[:-1]:
        values.append((values[-1] + gap) % 360.0)
    return values


def _spacing_error(angles: list[float]) -> float:
    ordered = sorted(value % 360.0 for value in angles)
    if len(ordered) != 6:
        raise ValueError("each panel must contain exactly six bearings")
    gaps = [
        (ordered[(index + 1) % 6] - ordered[index]) % 360.0
        for index in range(6)
    ]
    return math.sqrt(sum((gap - 60.0) ** 2 for gap in gaps) / 6.0)


def _panel_scores(angles: list[list[float]]) -> list[float]:
    if len(angles) != 4:
        raise ValueError("the world requires exactly four panels")
    return [_spacing_error(panel) for panel in angles]


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes; answer identity cycles exactly with the seed."""

    rng = random.Random(int(seed))
    target = int(seed) % 4
    near_boundary = int(seed) % 7 == 0
    amplitudes = [5.5 if near_boundary else 10.5, 14.0, 19.0]
    rng.shuffle(amplitudes)
    templates = [0, 1, 2]
    rng.shuffle(templates)
    loser_specs = list(zip(amplitudes, templates, strict=True))
    patterns: list[list[float]] = []
    loser_index = 0
    for panel_index in range(4):
        rotation = rng.uniform(0.0, 60.0)
        if panel_index == target:
            gaps = [60.0] * 6
        else:
            amplitude, template = loser_specs[loser_index]
            loser_index += 1
            if template == 0:
                gaps = [60.0 + amplitude, 60.0 - amplitude] * 3
            elif template == 1:
                gaps = [60.0 + amplitude, 60.0 + amplitude, 60.0 - 2.0 * amplitude] * 2
            else:
                gaps = [
                    60.0 + 1.5 * amplitude,
                    60.0 - 0.5 * amplitude,
                    60.0 - amplitude,
                ] * 2
        panel_angles = _angles_from_gaps(rotation, gaps)
        rng.shuffle(panel_angles)
        patterns.append(panel_angles)
    offsets = [[rng.randint(-4, 4), rng.randint(-4, 4)] for _ in range(4)]
    return {
        "angles": patterns,
        "palette": list(rng.choice(PALETTES)),
        "offsets": offsets,
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only declared scene fields; no answer-specific styling is used."""

    angles = scene["angles"]
    palette = scene["palette"]
    offsets = scene["offsets"]
    if len(angles) != 4 or len(offsets) != 4 or len(palette) != 2:
        raise ValueError("malformed scene")
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#FFFFFF")
    draw = ImageDraw.Draw(image)
    label_font = _font(24 * scale)
    for index, (box, base_center) in enumerate(zip(PANEL_BOXES, PANEL_CENTERS, strict=True)):
        scaled_box = tuple(int(value * scale) for value in box)
        draw.rounded_rectangle(
            scaled_box,
            radius=14 * scale,
            fill="#FAFBFC",
            outline="#AEB8C2",
            width=2 * scale,
        )
        draw.text(
            ((box[0] + 16) * scale, (box[1] + 12) * scale),
            LABELS[index],
            fill="#343A40",
            font=label_font,
        )
        cx = (base_center[0] + float(offsets[index][0])) * scale
        cy = (base_center[1] + float(offsets[index][1])) * scale
        center_r = CENTER_RADIUS * scale
        draw.ellipse(
            (cx - center_r, cy - center_r, cx + center_r, cy + center_r),
            fill=palette[1],
        )
        for angle in angles[index]:
            radians = math.radians(float(angle))
            x = cx + ORBIT_RADIUS * scale * math.cos(radians)
            y = cy + ORBIT_RADIUS * scale * math.sin(radians)
            radius = DOT_RADIUS * scale
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette[0])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene["angles"])
    order = sorted(range(4), key=lambda index: scores[index])
    if scores[order[1]] - scores[order[0]] <= 1e-9:
        raise ValueError("scene has no unique angular-spacing winner")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene["angles"]))
    return scores[1] - scores[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Angle-list order is latent; reversing it leaves every rendered dot fixed."""

    twin = copy.deepcopy(scene)
    twin["angles"] = [list(reversed(panel)) for panel in scene["angles"]]
    return [("reverse_unordered_angle_lists", twin)]
