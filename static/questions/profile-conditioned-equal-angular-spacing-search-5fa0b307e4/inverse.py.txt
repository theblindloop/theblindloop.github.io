"""Independent PNG-only inverse arm for equal angular-gap search.

The oracle knows only the public visual contract: four labeled regions read
left-to-right/top-to-bottom, each containing six larger colored dots around one
smaller colored center. It discovers components, centers, assignments, and
bearings directly from pixels and never imports the renderer or latent state.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
MIN_COMPONENT_AREA = 45
ABSTAIN_MARGIN_DEG = 2.0


def _components(mask: np.ndarray) -> list[dict[str, float]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[dict[str, float]] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if visited[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        visited[sy, sx] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x + 0.5
            sum_y += y + 0.5
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if count >= MIN_COMPONENT_AREA:
            components.append({"area": float(count), "cx": sum_x / count, "cy": sum_y / count})
    return components


def _ordered_centers(centers: list[dict[str, float]]) -> list[dict[str, float]]:
    by_y = sorted(centers, key=lambda item: item["cy"])
    top = sorted(by_y[:2], key=lambda item: item["cx"])
    bottom = sorted(by_y[2:], key=lambda item: item["cx"])
    if len(top) != 2 or len(bottom) != 2:
        raise ValueError("abstain: four panel centers were not recovered")
    return top + bottom


def _spacing_error(center: dict[str, float], dots: list[dict[str, float]]) -> float:
    if len(dots) != 6:
        raise ValueError("abstain: every panel must contain six dots")
    angles = sorted(
        math.degrees(math.atan2(dot["cy"] - center["cy"], dot["cx"] - center["cx"])) % 360.0
        for dot in dots
    )
    gaps = [(angles[(index + 1) % 6] - angles[index]) % 360.0 for index in range(6)]
    return math.sqrt(sum((gap - 60.0) ** 2 for gap in gaps) / 6.0)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_range = rgb.max(axis=2) - rgb.min(axis=2)
    colored = channel_range >= 62
    components = _components(colored)
    if len(components) != 28:
        raise ValueError(f"abstain: expected 28 colored marks, recovered {len(components)}")
    by_area = sorted(components, key=lambda item: item["area"])
    centers = by_area[:4]
    dots = by_area[4:]
    if centers[-1]["area"] >= dots[0]["area"] * 0.80:
        raise ValueError("abstain: center and dot sizes are not separable")
    centers = _ordered_centers(centers)
    assigned: list[list[dict[str, float]]] = [[] for _ in centers]
    for dot in dots:
        nearest = min(
            range(4),
            key=lambda index: (dot["cx"] - centers[index]["cx"]) ** 2
            + (dot["cy"] - centers[index]["cy"]) ** 2,
        )
        assigned[nearest].append(dot)
    if any(len(group) != 6 for group in assigned):
        raise ValueError("abstain: dot-to-center grouping is ambiguous")
    scores = [_spacing_error(center, group) for center, group in zip(centers, assigned, strict=True)]
    order = sorted(range(4), key=lambda index: scores[index])
    if scores[order[1]] - scores[order[0]] <= ABSTAIN_MARGIN_DEG:
        raise ValueError("abstain: angular-spacing winner lacks a safe runner-up margin")
    return LABELS[order[0]]
