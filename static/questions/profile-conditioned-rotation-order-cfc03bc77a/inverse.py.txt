"""Image-only inverse arm: recover the rosette's largest rotational order.

This module never sees the latent scene. From the PNG alone it

1. splits three pixel classes by background distance and darkness only: the
   near-black centre cross, the coloured dot ink, and the pale background;
2. takes the rotation centre as the centre of the dark cross's extent;
3. segments the dot ink into connected components (row-run union-find) and
   keeps each component's centroid;
4. scores every candidate order k by rotating the whole recovered point set
   about that centre and measuring the worst nearest-neighbour displacement,
   then returns the largest order whose score is sub-resolution.

Nothing local decides the answer: dot count, dot colour, dot size and any
single dot position are identical across answers, and the returned symbol is a
property of the global point set under rotation.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "rotation_order-oracle-0.1.0"

DARK_MAX = 80  # every channel below this => centre-cross ink
INK_MIN_DELTA = 55  # channel distance from the background => dot ink
MIN_DOT_PIXELS = 60
MIN_CROSS_PIXELS = 120
ORDERS = (2, 3, 4, 6)
MATCH_TOLERANCE_PX = 3.0  # sub-resolution: true symmetries score far below this


class OracleError(ValueError):
    """The image does not contain the structure the question declares."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Connected components (4-connectivity) via row runs and union-find."""
    height, width = mask.shape
    parent: list[int] = []

    def find(i: int) -> int:
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i: int, j: int) -> None:
        ri, rj = find(i), find(j)
        if ri != rj:
            parent[ri] = rj

    runs: list[tuple[int, int, int, int]] = []
    previous: list[tuple[int, int, int, int]] = []
    for row in range(height):
        line = mask[row]
        if not line.any():
            previous = []
            continue
        padded = np.concatenate(([False], line, [False]))
        edges = np.flatnonzero(padded[1:] != padded[:-1])
        current: list[tuple[int, int, int, int]] = []
        for start, end in zip(edges[0::2], edges[1::2]):
            index = len(parent)
            parent.append(index)
            entry = (row, int(start), int(end), index)
            current.append(entry)
            runs.append(entry)
            for _, p_start, p_end, p_index in previous:
                if p_start < end and start < p_end:
                    union(index, p_index)
        previous = current

    groups: dict[int, list[tuple[int, int, int]]] = {}
    for row, start, end, index in runs:
        groups.setdefault(find(index), []).append((row, start, end))

    out: list[np.ndarray] = []
    for pixel_runs in groups.values():
        columns: list[float] = []
        rows: list[float] = []
        for row, start, end in pixel_runs:
            for column in range(start, end):
                columns.append(column)
                rows.append(row)
        out.append(np.asarray([columns, rows], dtype=np.float64).T)
    return out


def measure(image: Image.Image) -> tuple[list[tuple[float, float]], tuple[float, float]]:
    """Return (dot centres, rotation centre) in continuous pixel coordinates."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    corners = np.array([arr[0, 0], arr[0, -1], arr[-1, 0], arr[-1, -1]], dtype=np.int16)
    background = np.median(corners, axis=0)

    dark = (arr < DARK_MAX).all(axis=-1)
    distance = np.abs(arr - background).max(axis=-1)
    ink = (distance >= INK_MIN_DELTA) & ~dark

    if int(dark.sum()) < MIN_CROSS_PIXELS:
        raise OracleError("no centre cross found in the image")
    dark_columns = np.flatnonzero(dark.any(axis=0))
    dark_rows = np.flatnonzero(dark.any(axis=1))
    centre = (
        float(dark_columns.min() + dark_columns.max() + 1) / 2.0,
        float(dark_rows.min() + dark_rows.max() + 1) / 2.0,
    )

    centres: list[tuple[float, float]] = []
    for component in _components(ink):
        if len(component) < MIN_DOT_PIXELS:
            continue
        centres.append(
            (float(component[:, 0].mean()) + 0.5, float(component[:, 1].mean()) + 0.5)
        )
    if len(centres) < 4:
        raise OracleError(f"only {len(centres)} dot components found")
    return centres, centre


def _score(points: list[tuple[float, float]], centre: tuple[float, float], order: int) -> float:
    angle = 2.0 * math.pi / order
    cos_a = math.cos(angle)
    sin_a = math.sin(angle)
    worst = 0.0
    for x, y in points:
        dx = x - centre[0]
        dy = y - centre[1]
        moved = (centre[0] + dx * cos_a - dy * sin_a, centre[1] + dx * sin_a + dy * cos_a)
        worst = max(worst, min(math.dist(moved, other) for other in points))
    return worst


def decision_from_image(image: Image.Image) -> str:
    """Largest rotational order of the dot set, recovered from pixels alone."""
    points, centre = measure(image)
    matched = [order for order in ORDERS if _score(points, centre, order) <= MATCH_TOLERANCE_PX]
    if not matched:
        raise OracleError("no declared rotational order maps the dot set onto itself")
    return str(max(matched))
