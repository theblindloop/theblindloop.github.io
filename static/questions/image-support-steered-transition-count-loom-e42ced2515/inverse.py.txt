"""Independent pixel-only inverse for the row-transition count loom."""

from __future__ import annotations

from collections import deque

from PIL import Image

BACKGROUND = (248, 246, 239)
CYAN = (69, 214, 208)
CREAM = (243, 229, 192)
TAGS = {
    "gold": (242, 193, 78),
    "red": (233, 93, 98),
    "green": (84, 182, 106),
    "blue": (86, 133, 216),
}


def _centroid(image: Image.Image, color: tuple[int, int, int]) -> tuple[float, float]:
    pixels = image.load()
    xs: list[int] = []
    ys: list[int] = []
    for y in range(image.height):
        for x in range(image.width):
            if pixels[x, y] == color:
                xs.append(x)
                ys.append(y)
    if len(xs) < 350:
        raise ValueError("missing or damaged frame")
    return sum(xs) / len(xs), sum(ys) / len(ys)


def _cell_components(image: Image.Image) -> list[tuple[float, float, int, int]]:
    pixels = image.load()
    width, height = image.size
    accepted = {CYAN: 1, CREAM: 0}
    visited: set[tuple[int, int]] = set()
    components: list[tuple[float, float, int, int]] = []
    for y in range(height):
        for x in range(width):
            color = pixels[x, y]
            if color not in accepted or (x, y) in visited:
                continue
            queue = deque([(x, y)])
            visited.add((x, y))
            count = total_x = total_y = 0
            while queue:
                cx, cy = queue.popleft()
                count += 1
                total_x += cx
                total_y += cy
                for nx, ny in ((cx - 1, cy), (cx + 1, cy), (cx, cy - 1), (cx, cy + 1)):
                    if 0 <= nx < width and 0 <= ny < height and (nx, ny) not in visited and pixels[nx, ny] == color:
                        visited.add((nx, ny))
                        queue.append((nx, ny))
            if count < 350:
                raise ValueError("incomplete cell")
            components.append((total_x / count, total_y / count, accepted[color], count))
    return components


def _decode_grid(cells: list[tuple[float, float, int, int]], rows: int, columns: int) -> list[list[int]]:
    if len(cells) != rows * columns:
        raise ValueError("wrong cell count")
    ordered = sorted(cells, key=lambda item: item[1])
    grid = []
    for row in range(rows):
        band = sorted(ordered[row * columns : (row + 1) * columns], key=lambda item: item[0])
        if max(item[1] for item in band) - min(item[1] for item in band) > 2:
            raise ValueError("damaged row alignment")
        grid.append([item[2] for item in band])
    if any(max(ordered[row * columns + column][3] for column in range(columns)) - min(ordered[row * columns + column][3] for column in range(columns)) > 1 for row in range(rows)):
        raise ValueError("inconsistent cell areas")
    return grid


def decision_from_image(image: Image.Image) -> str:
    picture = image.convert("RGB")
    centers = {name: _centroid(picture, color) for name, color in TAGS.items()}
    left = round(min(center[0] for center in centers.values()))
    right = round(max(center[0] for center in centers.values()))
    top = round(min(center[1] for center in centers.values()))
    bottom = round(max(center[1] for center in centers.values()))
    pixels = picture.load()
    if any(pixels[x, y] == BACKGROUND for y in range(top, bottom + 1) for x in range(left, right + 1)):
        raise ValueError("broken central loom")

    owned = {name: [] for name in TAGS}
    for cell in _cell_components(picture):
        x, y = cell[:2]
        owner = min(centers, key=lambda name: (x - centers[name][0]) ** 2 + (y - centers[name][1]) ** 2)
        owned[owner].append(cell)

    source = _decode_grid(owned["gold"], 5, 6)
    target = [sum(a != b for a, b in zip(row, row[1:])) for row in source]
    if sorted(target) != [1, 2, 3, 4, 5]:
        raise ValueError("invalid source boundary counts")

    codes: dict[str, list[int]] = {}
    for tag in ("red", "green", "blue"):
        grid = _decode_grid(owned[tag], 5, 5)
        code = []
        for column in range(5):
            values = [grid[row][column] for row in range(5)]
            height = sum(values)
            if values != [0] * (5 - height) + [1] * height or height < 1:
                raise ValueError("invalid unary column")
            code.append(height)
        if sorted(code) != [1, 2, 3, 4, 5]:
            raise ValueError("invalid code inventory")
        codes[tag] = code

    distances = {tag: sum(abs(a - b) for a, b in zip(target, code)) for tag, code in codes.items()}
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    if ordered[0][1] != 0 or ordered[0][1] == ordered[1][1]:
        raise ValueError("ambiguous transition-code match")
    return ordered[0][0]
