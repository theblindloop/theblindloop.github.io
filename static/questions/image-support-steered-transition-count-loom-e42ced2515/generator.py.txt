"""Deterministic row-transition loom sampler and Pillow renderer."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 640
BACKGROUND = (248, 246, 239)
NAVY = (20, 43, 69)
CYAN = (69, 214, 208)
CREAM = (243, 229, 192)
INNER_LINE = (61, 82, 107)
TAG_RGB = {
    "gold": (242, 193, 78),
    "red": (233, 93, 98),
    "green": (84, 182, 106),
    "blue": (86, 133, 216),
}
TAG_NAMES = ("red", "green", "blue")
SLOTS = ((0, 0), (1, 0), (0, 1), (1, 1))


def _row_with_transitions(count: int, start: int, rng: random.Random) -> list[int]:
    boundaries = set(rng.sample(range(5), count))
    row = [start]
    for boundary in range(5):
        row.append(row[-1] ^ (boundary in boundaries))
    return row


def _transition_code(source: list[list[int]]) -> list[int]:
    return [sum(a != b for a, b in zip(row, row[1:])) for row in source]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = [1, 2, 3, 4, 5]
    rng.shuffle(target)
    source = [_row_with_transitions(count, rng.randrange(2), rng) for count in target]

    correct_tag = TAG_NAMES[int(seed) % 3]
    codes: dict[str, list[int]] = {correct_tag: list(target)}
    other_tags = [tag for tag in TAG_NAMES if tag != correct_tag]
    rng.shuffle(other_tags)
    swap_pairs = [(0, 1), (2, 3)]
    rng.shuffle(swap_pairs)
    for tag, (first, second) in zip(other_tags, swap_pairs):
        altered = list(target)
        altered[first], altered[second] = altered[second], altered[first]
        codes[tag] = altered

    tags = ["gold", *TAG_NAMES]
    rng.shuffle(tags)
    return {
        "source": source,
        "codes": codes,
        "tags": tags,
        "cell_size": 27 + (int(seed) % 2),
        "atlas_origin": [53 + (int(seed) % 3), 52 + ((int(seed) // 3) % 3)],
    }


def _slot_center(origin: list[int], slot: tuple[int, int]) -> tuple[int, int]:
    ox, oy = origin
    column, row = slot
    return ox + 164 + column * 330, oy + 148 + row * 248


def _draw_grid(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    grid: list[list[int]],
    tag: str,
    cell_size: int,
) -> None:
    rows, columns = len(grid), len(grid[0])
    width, height = columns * cell_size, rows * cell_size
    left = center[0] - width // 2
    top = center[1] - height // 2
    draw.rectangle((left - 10, top - 10, left + width + 9, top + height + 9), outline=TAG_RGB[tag], width=6)
    for row in range(rows):
        for column in range(columns):
            x0 = left + column * cell_size + 3
            y0 = top + row * cell_size + 3
            x1 = left + (column + 1) * cell_size - 4
            y1 = top + (row + 1) * cell_size - 4
            draw.rectangle((x0, y0, x1, y1), fill=CYAN if grid[row][column] else CREAM)


def _code_grid(code: list[int]) -> list[list[int]]:
    return [[int(5 - row <= height) for height in code] for row in range(5)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    origin = list(scene["atlas_origin"])
    ox, oy = origin
    draw.rounded_rectangle((ox, oy, ox + 660, oy + 532), radius=34, fill=NAVY)
    draw.rounded_rectangle((ox + 18, oy + 18, ox + 642, oy + 514), radius=25, outline=INNER_LINE, width=4)
    cell_size = int(scene["cell_size"])
    for tag, slot in zip(scene["tags"], SLOTS):
        center = _slot_center(origin, slot)
        grid = scene["source"] if tag == "gold" else _code_grid(scene["codes"][tag])
        _draw_grid(draw, center, grid, tag, cell_size)
    return image


def analytic_gold(scene: dict) -> str:
    target = _transition_code(scene["source"])
    distances = {
        tag: sum(abs(a - b) for a, b in zip(target, scene["codes"][tag]))
        for tag in TAG_NAMES
    }
    return min(distances, key=distances.get)


def margin(scene: dict) -> float:
    target = _transition_code(scene["source"])
    distances = sorted(
        sum(abs(a - b) for a, b in zip(target, scene["codes"][tag]))
        for tag in TAG_NAMES
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    source = scene.get("source", [])
    codes = scene.get("codes", {})
    tags = scene.get("tags", [])
    if sorted(tags) != ["blue", "gold", "green", "red"]:
        return True
    if len(source) != 5 or any(len(row) != 6 or any(value not in (0, 1) for value in row) for row in source):
        return True
    target = _transition_code(source)
    if sorted(target) != [1, 2, 3, 4, 5]:
        return True
    if set(codes) != set(TAG_NAMES) or any(sorted(code) != [1, 2, 3, 4, 5] for code in codes.values()):
        return True
    distances = sorted(sum(abs(a - b) for a, b in zip(target, codes[tag])) for tag in TAG_NAMES)
    return distances[0] != 0 or distances[1] <= distances[0]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
