"""Independent inverse arm; receives only final RGB pixels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (37, 99, 205),
    "green": (18, 145, 86),
    "purple": (157, 67, 181),
}
_BAND = (229, 233, 238)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    work = mask.copy()
    height, width = work.shape
    output: list[tuple[int, int, int, int, int, float, float]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append((len(xs), min(xs), max(xs), min(ys), max(ys), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _longest_run(bits: list[bool]) -> int:
    best = current = 0
    for bit in bits:
        current = current + 1 if bit else 0
        best = max(best, current)
    return best


def _bands(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    mask = np.all(arr == np.asarray(_BAND, dtype=np.uint8), axis=2)
    # Checkpoint guide lines and colored circles interrupt the fill, so recover
    # each band from its long run of rows rather than assuming one component.
    good_rows = np.count_nonzero(mask, axis=1) > 450
    runs: list[tuple[int, int]] = []
    start: int | None = None
    for y, present in enumerate(good_rows.tolist() + [False]):
        if present and start is None:
            start = y
        elif not present and start is not None:
            if y - start > 40:
                runs.append((start, y - 1))
            start = None
    bands: list[tuple[int, int, int, int]] = []
    for y0, y1 in runs:
        _, xs = np.nonzero(mask[y0 : y1 + 1])
        bands.append((int(xs.min()), int(xs.max()), y0, y1))
    if len(bands) != 3:
        raise ValueError(f"expected three visible tolerance bands, recovered {len(bands)}")
    return bands


def _markers(arr: np.ndarray, rgb: tuple[int, int, int]) -> list[tuple[int, int, float, float]]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    markers: list[tuple[int, int, float, float]] = []
    for area, x0, x1, y0, y1, cx, cy in _components(mask):
        box_w, box_h = x1 - x0 + 1, y1 - y0 + 1
        if 180 <= area <= 360 and 16 <= box_w <= 21 and 16 <= box_h <= 21 and x0 > 120:
            markers.append((y0, y1, cx, cy))
    markers.sort(key=lambda item: item[2])
    if len(markers) != 7:
        raise ValueError(f"expected seven circular checkpoints, recovered {len(markers)}")
    if min(b[2] - a[2] for a, b in zip(markers, markers[1:])) < 45:
        raise ValueError("checkpoint order is not resolvable")
    return markers


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    bands = _bands(arr)
    scores: dict[str, int] = {}
    for color, rgb in _COLORS.items():
        markers = _markers(arr, rgb)
        median_y = float(np.median([marker[3] for marker in markers]))
        band = min(bands, key=lambda box: abs((box[2] + box[3]) / 2 - median_y))
        _, _, band_top, band_bottom = band
        bits: list[bool] = []
        for y0, y1, _, _ in markers:
            if abs(y0 - band_top) < 3 or abs(y1 - band_bottom) < 3:
                raise ValueError("checkpoint is too close to a band boundary")
            bits.append(band_top <= y0 and y1 <= band_bottom)
        if sum(bits) != 4:
            raise ValueError("row does not contain exactly four in-band checkpoints")
        scores[color] = _longest_run(bits)
    best = max(scores.values())
    winners = [color for color, score in scores.items() if score == best]
    if len(winners) != 1:
        raise ValueError("top consecutive-run length is tied")
    return winners[0]
