"""Deterministic Pillow world for ordered checkpoint runs inside visible bands."""

from __future__ import annotations

import itertools
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLOR_RGB = {
    "blue": (37, 99, 205),
    "green": (18, 145, 86),
    "purple": (157, 67, 181),
}
BACKGROUND = (248, 249, 247)
PANEL = (255, 255, 255)
BAND = (229, 233, 238)
GRID = (214, 219, 225)
INK = (39, 44, 52)
N_CHECKPOINTS = 7


def _longest_run(bits: list[bool]) -> int:
    best = current = 0
    for bit in bits:
        current = current + 1 if bit else 0
        best = max(best, current)
    return best


def _pattern_pool(run_length: int) -> list[list[bool]]:
    pool: list[list[bool]] = []
    for chosen in itertools.combinations(range(N_CHECKPOINTS), 4):
        bits = [index in chosen for index in range(N_CHECKPOINTS)]
        if _longest_run(bits) == run_length:
            pool.append(bits)
    return pool


PATTERNS = {length: _pattern_pool(length) for length in (1, 2, 3, 4)}


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    width, height = 720, 560
    marker_radius = 9
    checkpoint_xs = [150 + 80 * index for index in range(N_CHECKPOINTS)]
    colors = list(COLOR_RGB)

    # All three rows contain exactly four inside checkpoints. The winner cannot
    # be inferred from membership count, total colored ink, marker size, or row.
    run_sets = ((4, 3, 1), (4, 2, 1), (3, 2, 1))
    run_lengths = list(run_sets[(int(seed) // 3) % len(run_sets)])
    winner_color = colors[int(seed) % len(colors)]
    remaining_colors = [color for color in colors if color != winner_color]
    rng.shuffle(remaining_colors)
    by_color: dict[str, int] = {winner_color: run_lengths[0]}
    by_color[remaining_colors[0]] = run_lengths[1]
    by_color[remaining_colors[1]] = run_lengths[2]

    panel_colors = colors.copy()
    rng.shuffle(panel_colors)
    panel_tops = [66, 230, 394]
    panels: list[dict[str, Any]] = []
    for panel_top, color in zip(panel_tops, panel_colors, strict=True):
        band_top = panel_top + rng.choice((35, 39, 43))
        band_bottom = band_top + 58
        candidates = PATTERNS[by_color[color]]
        pattern = list(rng.choice(candidates))
        if rng.choice((False, True)):
            pattern.reverse()

        marker_ys: list[int] = []
        for inside in pattern:
            if inside:
                marker_ys.append(rng.randint(band_top + marker_radius + 5, band_bottom - marker_radius - 5))
            elif rng.choice((False, True)):
                marker_ys.append(rng.randint(panel_top + marker_radius + 5, band_top - marker_radius - 6))
            else:
                marker_ys.append(rng.randint(band_bottom + marker_radius + 6, panel_top + 142 - marker_radius - 5))
        panels.append(
            {
                "color": color,
                "panel_top": panel_top,
                "band_top": band_top,
                "band_bottom": band_bottom,
                "marker_ys": marker_ys,
            }
        )

    return {
        "width": width,
        "height": height,
        "marker_radius": marker_radius,
        "checkpoint_xs": checkpoint_xs,
        "panels": panels,
    }


def _inside_sequence(scene: dict[str, Any], panel: dict[str, Any]) -> list[bool]:
    radius = int(scene["marker_radius"])
    top, bottom = int(panel["band_top"]), int(panel["band_bottom"])
    return [top <= int(y) - radius and int(y) + radius <= bottom for y in panel["marker_ys"]]


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = {str(panel["color"]): _longest_run(_inside_sequence(scene, panel)) for panel in scene["panels"]}
    best = max(scores.values())
    winners = [color for color, score in scores.items() if score == best]
    return winners[0] if len(winners) == 1 else "abstain"


def margin(scene: dict[str, Any]) -> float:
    scores = sorted((_longest_run(_inside_sequence(scene, panel)) for panel in scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def _boundary_clearance(scene: dict[str, Any]) -> int:
    radius = int(scene["marker_radius"])
    clearances: list[int] = []
    for panel in scene["panels"]:
        top, bottom = int(panel["band_top"]), int(panel["band_bottom"])
        for y_value in panel["marker_ys"]:
            y = int(y_value)
            if top <= y - radius and y + radius <= bottom:
                clearances.append(min(y - radius - top, bottom - y - radius))
            elif y + radius < top:
                clearances.append(top - y - radius)
            elif y - radius > bottom:
                clearances.append(y - radius - bottom)
            else:
                clearances.append(0)
    return min(clearances)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1 or _boundary_clearance(scene) < 3


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    radius = int(scene["marker_radius"])
    checkpoint_xs = [int(value) for value in scene["checkpoint_xs"]]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 16)
        label_font = ImageFont.truetype("DejaVuSans.ttf", 12)
        color_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 13)
    except OSError:
        title_font = label_font = color_font = ImageFont.load_default()

    draw.text((74, 19), "Consecutive checkpoints fully inside the tolerance band", fill=INK, font=title_font)
    for panel in scene["panels"]:
        panel_top = int(panel["panel_top"])
        band_top, band_bottom = int(panel["band_top"]), int(panel["band_bottom"])
        color = str(panel["color"])
        rgb = COLOR_RGB[color]
        draw.rectangle((74, panel_top, 686, panel_top + 142), fill=PANEL, outline=GRID, width=2)
        draw.rectangle((120, band_top, 682, band_bottom), fill=BAND)
        draw.rectangle((88, panel_top + 58, 108, panel_top + 78), fill=rgb)
        draw.text((82, panel_top + 88), color, fill=INK, font=color_font)
        for index, x in enumerate(checkpoint_xs, start=1):
            draw.line((x, panel_top + 6, x, panel_top + 136), fill=GRID, width=1)
            draw.text((x - 4, panel_top + 5), str(index), fill=INK, font=label_font)
        for x, y in zip(checkpoint_xs, panel["marker_ys"], strict=True):
            y = int(y)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=rgb, outline=(255, 255, 255), width=1)
    return image
