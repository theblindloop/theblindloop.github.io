"""Renderer and analytic world for the marked-junction arm-count question.

The visible object is one non-crossing dark tree. Green and red disks identify
the route endpoints; a circle and a diamond outline identify two interior
junctions. The answer is the marker whose junction has larger graph degree.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "junction_arm_count_selection-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
BACKGROUND = (248, 248, 244)
INK = (28, 47, 78)
GREEN = (38, 160, 91)
RED = (211, 66, 61)
MARKER_COLOR = (143, 72, 180)
SHAPES = ("circle", "square")


def _transform(point: tuple[float, float], angle: float, scale: float, tx: float, ty: float) -> list[float]:
    x, y = point
    c, s = math.cos(angle), math.sin(angle)
    x *= scale
    y *= scale
    return [round(320.0 + tx + c * x - s * y, 5), round(320.0 + ty + s * x + c * y, 5)]


def _polyline(points: Iterable[Iterable[float]]) -> list[list[float]]:
    return [[float(x), float(y)] for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, transformed six-edge tree from an integer seed."""

    rng = random.Random(int(seed))
    angle = rng.uniform(0.0, math.tau)
    scale = rng.uniform(0.84, 0.96)
    tx = rng.uniform(-16.0, 16.0)
    ty = rng.uniform(-16.0, 16.0)
    topology_flip = rng.randrange(2)

    # The two junctions lie on a straight source-to-target trunk. One has
    # three incident arms and the other has four. The two topology layouts
    # swap which physical junction is the higher-degree one.
    source0 = (-230.0, 0.0)
    j10 = (-80.0, 0.0)
    j20 = (80.0, 0.0)
    target0 = (230.0, 0.0)
    if topology_flip == 0:
        branch0 = [(-80.0, -170.0)]
        branch1 = [(10.0, 175.0), (175.0, 135.0)]
    else:
        branch0 = [(-180.0, -130.0), (-10.0, -170.0)]
        branch1 = [(80.0, 170.0)]

    def t(point: tuple[float, float]) -> list[float]:
        return _transform(point, angle, scale, tx, ty)

    source = t(source0)
    j1 = t(j10)
    j2 = t(j20)
    target = t(target0)
    branch0_points = [t(point) for point in branch0]
    branch1_points = [t(point) for point in branch1]

    edges = [
        _polyline([source, j1]),
        _polyline([j1, j2]),
        _polyline([j2, target]),
    ]
    edges.extend(_polyline([j1, endpoint]) for endpoint in branch0_points)
    edges.extend(_polyline([j2, endpoint]) for endpoint in branch1_points)

    high_index = 1 if topology_flip == 0 else 0
    # The low bit of the encoded scene seed balances answer labels in every
    # generated dataset; it is independent of topology, orientation, and color.
    high_shape = "circle" if int(seed) % 2 == 0 else "square"
    low_shape = "square" if high_shape == "circle" else "circle"
    marker_shapes = [low_shape, low_shape]
    marker_shapes[high_index] = high_shape
    marker_shapes[1 - high_index] = low_shape

    marker_size = rng.uniform(21.0, 25.0)
    stroke = rng.uniform(8.0, 10.0)

    return {
        "canvas": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
        "palette": {
            "ink": list(INK),
            "source": list(GREEN),
            "target": list(RED),
            "marker": list(MARKER_COLOR),
        },
        "source": source,
        "target": target,
        "junctions": [
            {"id": "j1", "position": j1},
            {"id": "j2", "position": j2},
        ],
        "edges": edges,
        "marker_shapes": marker_shapes,
        "marker_colors": [list(MARKER_COLOR), list(MARKER_COLOR)],
        "stroke": round(stroke, 4),
        "marker_size": round(marker_size, 4),
        "orientation": round(angle, 8),
        "translation": [round(tx, 5), round(ty, 5)],
        "topology_flip": topology_flip,
    }


def _draw_round_polyline(draw: ImageDraw.ImageDraw, points: list[list[float]], fill, width: int) -> None:
    xy = [(int(round(x)), int(round(y))) for x, y in points]
    draw.line(xy, fill=fill, width=width, joint="curve")
    radius = width / 2.0
    for x, y in xy:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def _draw_marker(draw: ImageDraw.ImageDraw, center: list[float], shape: str, color, radius: float, width: int) -> None:
    x, y = center
    if shape == "circle":
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=width)
    elif shape == "square":
        draw.rectangle((x - radius, y - radius, x + radius, y + radius), outline=color, width=width)
    else:
        raise ValueError(f"unknown marker shape {shape!r}")


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields; no text or hidden payload is embedded."""

    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    ss = 3
    background = tuple(int(value) for value in scene["background"])
    palette = scene["palette"]
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)

    def scale_points(points: list[list[float]]) -> list[list[float]]:
        return [[x * ss, y * ss] for x, y in points]

    line_width = max(3, int(round(float(scene["stroke"]) * ss)))
    for edge in scene["edges"]:
        _draw_round_polyline(draw, scale_points(edge), tuple(palette["ink"]), line_width)

    endpoint_radius = 16.0 * ss
    for point, color_name in ((scene["source"], "source"), (scene["target"], "target")):
        x, y = float(point[0]) * ss, float(point[1]) * ss
        draw.ellipse((x - endpoint_radius, y - endpoint_radius, x + endpoint_radius, y + endpoint_radius), fill=tuple(palette[color_name]))

    marker_width = max(3, int(round(4.0 * ss)))
    marker_radius = float(scene["marker_size"]) * ss
    for junction, shape, color in zip(scene["junctions"], scene["marker_shapes"], scene["marker_colors"]):
        center = [float(junction["position"][0]) * ss, float(junction["position"][1]) * ss]
        _draw_marker(draw, center, shape, tuple(color), marker_radius, marker_width)

    return img.resize((width, height), Image.Resampling.LANCZOS)


def _incident_degree(scene: dict, position: list[float]) -> int:
    x, y = float(position[0]), float(position[1])
    count = 0
    for edge in scene["edges"]:
        endpoints = (edge[0], edge[-1])
        if any(math.hypot(float(point[0]) - x, float(point[1]) - y) < 0.01 for point in endpoints):
            count += 1
    return count


def analytic_gold(scene: dict) -> str:
    """Compute the marker label from the latent graph, not from rendered pixels."""

    junctions = scene["junctions"]
    edges = scene["edges"]
    marker_shapes = scene["marker_shapes"]
    degrees = []
    for junction in junctions:
        x, y = (float(value) for value in junction["position"])
        degree = 0
        for edge in edges:
            if any(math.hypot(float(point[0]) - x, float(point[1]) - y) < 0.01 for point in (edge[0], edge[-1])):
                degree += 1
        degrees.append(degree)
    if degrees[0] == degrees[1]:
        raise ValueError("junction arm-count tie")
    return str(marker_shapes[0] if degrees[0] > degrees[1] else marker_shapes[1])


def margin(scene: dict) -> float:
    degrees = [_incident_degree(scene, junction["position"]) for junction in scene["junctions"]]
    return float(abs(degrees[0] - degrees[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the alias that edge-list order is not visible in same-color ink."""

    twin = copy.deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("edge_order_permutation", twin)]
