"""Renderer and analytic gold for the state-handoff question world.

The image is a clean six-panel board.  Every panel contains the same four
identity glyphs in a shuffled row order and one categorical state swatch per
glyph.  The analytic decision is deliberately computed from the state table,
not from any layout field.  ``draw_order`` is a render-only latent alias: all
panel contents are disjoint, so changing the order in which they are painted
cannot change the raster.
"""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
STATE_COLORS = {
    0: (227, 89, 89),
    1: (79, 169, 112),
    2: (79, 125, 214),
}
BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 250, 253)
INK = (35, 39, 48)
MUTED = (133, 143, 155)

WIDTH = 1320
HEIGHT = 520
PANEL_COUNT = 6
PANEL_X0 = 20
PANEL_STEP = 215
PANEL_W = 195
PANEL_Y = 110
PANEL_H = 360
ROW_Y0 = 170
ROW_STEP = 72
GLYPH_DX = 36
SWATCH_DX0 = 82
SWATCH_DX1 = 154
SWATCH_HALF_H = 14


def _handoff_scores_from_states(states: dict[str, list[int]]) -> dict[str, int]:
    scores = {label: 0 for label in LABELS}
    for t in range(PANEL_COUNT - 1):
        old = {label: int(states[label][t]) for label in LABELS}
        new = {label: int(states[label][t + 1]) for label in LABELS}
        leaving = {
            label for label in LABELS if new[label] != old[label]
        }
        for recipient in LABELS:
            incoming = new[recipient]
            if incoming == old[recipient]:
                continue
            if any(
                sender != recipient
                and sender in leaving
                and old[sender] == incoming
                for sender in LABELS
            ):
                scores[recipient] += 1
    return scores


def _top_gap(scores: dict[str, int]) -> float:
    values = sorted(scores.values(), reverse=True)
    return float(values[0] - values[1])


def _shape_points(label: str, cx: int, cy: int) -> tuple[str, object]:
    if label == "A":
        return "ellipse", (cx - 12, cy - 12, cx + 12, cy + 12)
    if label == "B":
        return "polygon", ((cx, cy - 14), (cx - 13, cy + 12), (cx + 13, cy + 12))
    if label == "C":
        return "rectangle", (cx - 11, cy - 11, cx + 11, cy + 11)
    if label == "D":
        return "polygon", ((cx, cy - 14), (cx + 14, cy), (cx, cy + 14), (cx - 14, cy))
    raise ValueError(f"unknown identity label {label!r}")


def _draw_identity_glyph(draw: ImageDraw.ImageDraw, label: str, cx: int, cy: int) -> None:
    kind, payload = _shape_points(label, cx, cy)
    if kind == "ellipse":
        draw.ellipse(payload, fill=INK)
    elif kind == "rectangle":
        draw.rectangle(payload, fill=INK)
    else:
        draw.polygon(payload, fill=INK)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel_index: int,
    row_order: Iterable[str],
    states: dict[str, list[int]],
    font: ImageFont.ImageFont,
) -> None:
    x = PANEL_X0 + panel_index * PANEL_STEP
    draw.rectangle(
        (x, PANEL_Y, x + PANEL_W, PANEL_Y + PANEL_H),
        fill=PANEL_FILL,
        outline=INK,
        width=2,
    )
    title = f"P{panel_index + 1}"
    draw.text((x + 84, PANEL_Y + 8), title, fill=INK, font=font)
    for row_index, label in enumerate(row_order):
        cy = ROW_Y0 + row_index * ROW_STEP
        if row_index:
            draw.line((x + 12, cy - 36, x + PANEL_W - 12, cy - 36), fill=(221, 226, 232), width=1)
        _draw_identity_glyph(draw, label, x + GLYPH_DX, cy)
        state = int(states[label][panel_index])
        draw.rectangle(
            (x + SWATCH_DX0, cy - SWATCH_HALF_H, x + SWATCH_DX1, cy + SWATCH_HALF_H),
            fill=STATE_COLORS[state],
            outline=INK,
            width=1,
        )


def render(scene: dict) -> Image.Image:
    """Render one scene to a deterministic RGB image."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((24, 20), "IDENTITY KEY", fill=INK, font=font)
    for index, label in enumerate(LABELS):
        cx = 116 + index * 52
        _draw_identity_glyph(draw, label, cx, 52)
        draw.text((cx - 4, 70), label, fill=INK, font=font)

    draw.text((355, 20), "STATE KEY", fill=INK, font=font)
    for state, name in enumerate(STATE_NAMES):
        x = 438 + state * 98
        draw.rectangle((x, 39, x + 54, 65), fill=STATE_COLORS[state], outline=INK, width=1)
        draw.text((x + 60, 47), name, fill=INK, font=font)
    draw.text(
        (790, 38),
        "Read numbered panels left to right",
        fill=MUTED,
        font=font,
    )

    # draw_order is intentionally render-only.  Each panel occupies its own
    # rectangle, so every permutation produces exactly the same RGB raster.
    for panel_index in scene["draw_order"]:
        _draw_panel(
            draw,
            int(panel_index),
            scene["row_orders"][int(panel_index)],
            scene["states"],
            font,
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique top handoff recipient from the visible state table."""
    scores = _handoff_scores_from_states(scene["states"])
    ranked = sorted(LABELS, key=lambda label: (-scores[label], label))
    if scores[ranked[0]] == scores[ranked[1]]:
        raise ValueError("ambiguous state-handoff scene")
    return ranked[0]


def margin(scene: dict) -> float:
    """Continuous/one-sided verifiability margin: top score minus runner-up."""
    return _top_gap(_handoff_scores_from_states(scene["states"]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine one-count leads as finite-resolution near-boundary cases."""
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Return a nontrivial pixel-identical draw-order permutation."""
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_disjoint_panel_draw_order", twin)]


def _sample_states(rng: random.Random, target: str, minimum_gap: int) -> dict[str, list[int]]:
    for _ in range(20000):
        states = {
            label: [rng.randrange(3) for _ in range(PANEL_COUNT)]
            for label in LABELS
        }
        flattened = [value for trajectory in states.values() for value in trajectory]
        if len(set(flattened)) < 3:
            continue
        scores = _handoff_scores_from_states(states)
        ranked = sorted(LABELS, key=lambda label: (-scores[label], label))
        if ranked[0] == target and scores[ranked[0]] - scores[ranked[1]] >= minimum_gap:
            return states
    raise RuntimeError(f"unable to sample a handoff scene for {target} at gap {minimum_gap}")


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, balanced scene for one integer seed."""
    seed = int(seed)
    rng = random.Random(seed)
    target = LABELS[seed % len(LABELS)]
    minimum_gap = 1 if seed % 5 == 0 else 2
    states = _sample_states(rng, target, minimum_gap)
    row_orders: list[list[str]] = []
    for _ in range(PANEL_COUNT):
        order = list(LABELS)
        rng.shuffle(order)
        row_orders.append(order)
    draw_order = list(range(PANEL_COUNT))
    rng.shuffle(draw_order)
    return {
        "states": states,
        "row_orders": row_orders,
        "draw_order": draw_order,
    }
