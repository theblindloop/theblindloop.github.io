"""Independent image-only inverse for the state-handoff world.

This module intentionally duplicates the public visual contract rather than
importing any renderer or analytic code.  It finds each panel's identity glyph
by matching its dark binary silhouette against four public glyph templates and
reads the categorical state from the center of its colored swatch.  It then
replays the handoff rule over the recovered panel table.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

LABELS = ("A", "B", "C", "D")
STATE_COLORS = np.array(
    [
        (227, 89, 89),
        (79, 169, 112),
        (79, 125, 214),
    ],
    dtype=np.int16,
)
WIDTH = 1320
HEIGHT = 520
PANEL_X0 = 20
PANEL_STEP = 215
ROW_Y0 = 170
ROW_STEP = 72
GLYPH_DX = 36
SWATCH_CENTER_DX = 118


def _templates() -> dict[str, np.ndarray]:
    templates: dict[str, np.ndarray] = {}
    for label in LABELS:
        mask = Image.new("L", (37, 37), 0)
        draw = ImageDraw.Draw(mask)
        cx = cy = 18
        if label == "A":
            draw.ellipse((cx - 12, cy - 12, cx + 12, cy + 12), fill=255)
        elif label == "B":
            draw.polygon(((cx, cy - 14), (cx - 13, cy + 12), (cx + 13, cy + 12)), fill=255)
        elif label == "C":
            draw.rectangle((cx - 11, cy - 11, cx + 11, cy + 11), fill=255)
        else:
            draw.polygon(((cx, cy - 14), (cx + 14, cy), (cx, cy + 14), (cx - 14, cy)), fill=255)
        templates[label] = np.asarray(mask, dtype=np.uint8) > 0
    return templates


_GLYPH_TEMPLATES = _templates()


def _read_label(arr: np.ndarray, cx: int, cy: int) -> str:
    patch = arr[cy - 18 : cy + 19, cx - 18 : cx + 19]
    if patch.shape != (37, 37, 3):
        raise ValueError("identity glyph is clipped")
    dark = np.max(patch, axis=2) < 140
    scores = {
        label: int(np.logical_xor(dark, template).sum())
        for label, template in _GLYPH_TEMPLATES.items()
    }
    ranked = sorted(LABELS, key=lambda label: (scores[label], label))
    if scores[ranked[0]] > 42 or scores[ranked[1]] - scores[ranked[0]] < 4:
        raise ValueError(f"unresolved identity glyph: {scores}")
    return ranked[0]


def _read_state(arr: np.ndarray, x: int, y: int) -> int:
    pixel = arr[y, x].astype(np.int16)
    distances = np.linalg.norm(STATE_COLORS - pixel, axis=1)
    state = int(np.argmin(distances))
    if float(distances[state]) > 35.0:
        raise ValueError("state swatch is missing or corrupted")
    return state


def _handoff_scores(table: dict[str, list[int]]) -> dict[str, int]:
    scores = {label: 0 for label in LABELS}
    for t in range(5):
        old = {label: table[label][t] for label in LABELS}
        new = {label: table[label][t + 1] for label in LABELS}
        leaving = {label for label in LABELS if new[label] != old[label]}
        for recipient in LABELS:
            if new[recipient] == old[recipient]:
                continue
            if any(
                sender != recipient
                and sender in leaving
                and old[sender] == new[recipient]
                for sender in LABELS
            ):
                scores[recipient] += 1
    return scores


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected label from the PNG alone."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {rgb.size}")
    arr = np.asarray(rgb)
    table = {label: [] for label in LABELS}
    for panel in range(6):
        x0 = PANEL_X0 + panel * PANEL_STEP
        seen: set[str] = set()
        for row in range(4):
            cy = ROW_Y0 + row * ROW_STEP
            label = _read_label(arr, x0 + GLYPH_DX, cy)
            if label in seen:
                raise ValueError("duplicate identity in a panel")
            seen.add(label)
            table[label].append(_read_state(arr, x0 + SWATCH_CENTER_DX, cy))
        if seen != set(LABELS):
            raise ValueError("panel does not contain all four identities")
    scores = _handoff_scores(table)
    ranked = sorted(LABELS, key=lambda label: (-scores[label], label))
    if scores[ranked[0]] == scores[ranked[1]]:
        raise ValueError(f"ambiguous recovered handoff scores: {scores}")
    return ranked[0]
