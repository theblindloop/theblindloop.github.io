"""Renderer and analytic arm for the upper-left integral-XOR panel world."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
PANEL_SIZE = 208
FRAME = 12
INNER_SIZE = 184
CELL = 23
GRID = 8
TEMPLATE_GAP = 26.0

# The two output glyphs intentionally have equal output ink and equal inverse
# input ink.  Their inverse-prefix rasters are what is rendered.
OUTPUT_TEMPLATES: dict[str, tuple[str, ...]] = {
    "A": (
        "#.#.#.#.",
        "...#####",
        "#.#.#.#.",
        ".#.#.#.#",
        "#.#.#.#.",
        ".#.#.#.#",
        "#.#.#.#.",
        ".#.#.#..",
    ),
    "B": (
        "#..##.#.",
        "..####..",
        ".######.",
        "########",
        "#..##.#.",
        "...##...",
        "...##...",
        "...##...",
    ),
}

PALETTES = (
    {
        "background": (250, 248, 242),
        "frame": (25, 57, 90),
        "dark": (68, 76, 73),
        "light": (224, 232, 215),
        "marker": (211, 66, 60),
    },
    {
        "background": (247, 249, 246),
        "frame": (29, 52, 91),
        "dark": (76, 72, 66),
        "light": (226, 226, 216),
        "marker": (214, 73, 58),
    },
    {
        "background": (249, 247, 239),
        "frame": (20, 61, 94),
        "dark": (70, 80, 76),
        "light": (220, 231, 221),
        "marker": (206, 69, 61),
    },
    {
        "background": (248, 248, 244),
        "frame": (35, 50, 84),
        "dark": (72, 78, 71),
        "light": (230, 225, 211),
        "marker": (216, 76, 59),
    },
)


def _inverse_prefix(template: tuple[str, ...]) -> tuple[tuple[int, ...], ...]:
    output = [[1 if char == "#" else 0 for char in row] for row in template]
    source = [[0] * GRID for _ in range(GRID)]
    for row in range(GRID):
        for col in range(GRID):
            source[row][col] = (
                output[row][col]
                ^ (output[row - 1][col] if row else 0)
                ^ (output[row][col - 1] if col else 0)
                ^ (output[row - 1][col - 1] if row and col else 0)
            )
    return tuple(tuple(value for value in row) for row in source)


INPUT_TEMPLATES = {key: _inverse_prefix(value) for key, value in OUTPUT_TEMPLATES.items()}


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene; even and odd seeds counterbalance labels."""

    rng = random.Random(int(seed))
    return {
        "symbol": "A" if int(seed) % 2 == 0 else "B",
        "panel_left": 148 + 4 * rng.randrange(4),
        "panel_top": 148 + 4 * rng.randrange(4),
        "panel_size": PANEL_SIZE,
        "palette_id": rng.randrange(len(PALETTES)),
        "distractor_variant": rng.randrange(4),
    }


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    colors = ((190, 128, 55), (74, 139, 117), (151, 102, 157))
    layouts = (
        ((55, 61, 12), (421, 76, 18), (74, 431, 10)),
        ((82, 407, 16), (420, 411, 12), (56, 106, 10)),
        ((78, 75, 18), (428, 421, 14), (420, 116, 11)),
        ((57, 423, 11), (426, 66, 14), (84, 92, 15)),
    )
    for index, (cx, cy, radius) in enumerate(layouts[variant % len(layouts)]):
        color = colors[(index + variant) % len(colors)]
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH, HEIGHT), palette["background"])
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, int(scene["distractor_variant"]))

    left = int(scene["panel_left"])
    top = int(scene["panel_top"])
    size = int(scene["panel_size"])
    draw.rectangle([left, top, left + size - 1, top + size - 1], fill=palette["frame"])

    source = INPUT_TEMPLATES[str(scene["symbol"])]
    inner_left = left + FRAME
    inner_top = top + FRAME
    cell = (size - 2 * FRAME) // GRID
    if cell != CELL:
        raise ValueError("unsupported panel geometry")
    for row in range(GRID):
        for col in range(GRID):
            color = palette["dark"] if source[row][col] else palette["light"]
            x0 = inner_left + col * cell
            y0 = inner_top + row * cell
            draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=color)

    # The red 8-by-8 square is entirely inside the top and left frame bands.
    draw.rectangle([left + 2, top + 2, left + 9, top + 9], fill=palette["marker"])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    del scene
    return TEMPLATE_GAP


def is_quarantined(scene: dict) -> bool:
    del scene
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
