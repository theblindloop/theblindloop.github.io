"""Independent image-only inverse for the integral-XOR panel.

This module deliberately recovers the panel geometry, tones, and transform from
the raster.  It has no import path to the renderer or to the prompt/generation
arms and never accepts a latent scene.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_SIZE = 208
_FRAME = 12
_GRID = 8
_CELL = 23
# The red 8-by-8 fiducial overwrites 64 pixels in the otherwise solid border.
_EXPECTED_BORDER_PIXELS = 9344
_A = (
    "#.#.#.#.",
    "...#####",
    "#.#.#.#.",
    ".#.#.#.#",
    "#.#.#.#.",
    ".#.#.#.#",
    "#.#.#.#.",
    ".#.#.#..",
)
_B = (
    "#..##.#.",
    "..####..",
    ".######.",
    "########",
    "#..##.#.",
    "...##...",
    "...##...",
    "...##...",
)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for y, x in zip(*np.nonzero(mask)):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        count = 0
        min_y = max_y = y
        min_x = max_x = x
        while stack:
            cy, cx = stack.pop()
            count += 1
            min_y = min(min_y, cy)
            max_y = max(max_y, cy)
            min_x = min(min_x, cx)
            max_x = max(max_x, cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        components.append((count, min_x, min_y, max_x, max_y))
    return components


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        [array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)],
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _template_distance(bits: list[list[int]], template: tuple[str, ...]) -> int:
    return sum(
        int(bits[row][col] != (template[row][col] == "#"))
        for row in range(_GRID)
        for col in range(_GRID)
    )


def decision_from_image(image: Image.Image) -> str:
    """Return A/B from pixels, or raise ValueError when the raster is ambiguous."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise ValueError("image is too small")
    background = _background(array)

    # The frame is the only large blue-hued connected component.
    red = array.astype(np.int16)
    frame_mask = (
        (red[:, :, 2] - red[:, :, 0] > 18)
        & (red[:, :, 2] - red[:, :, 1] > 8)
        & (red.mean(axis=2) < 180)
    )
    components = _components(frame_mask)
    if not components:
        raise ValueError("framed panel is missing")
    count, left, top, right, bottom = max(components)
    if count != _EXPECTED_BORDER_PIXELS or right - left + 1 != _SIZE or bottom - top + 1 != _SIZE:
        raise ValueError("framed panel is incomplete or ambiguous")

    # Verify the visible top-left fiducial before reading the grid.
    marker_mask = (
        (red[:, :, 0] > 135)
        & (red[:, :, 0] - red[:, :, 1] > 55)
        & (red[:, :, 0] - red[:, :, 2] > 35)
    )
    markers = [
        item
        for item in _components(marker_mask)
        if item[0] == 64 and item[3] - item[1] + 1 == 8 and item[4] - item[2] + 1 == 8
    ]
    if not markers:
        raise ValueError("top-left fiducial is missing")
    marker_count, marker_left, marker_top, marker_right, marker_bottom = max(markers)
    if (
        marker_count != 64
        or abs(marker_left - (left + 2)) > 1
        or abs(marker_top - (top + 2)) > 1
        or marker_right - marker_left + 1 != 8
        or marker_bottom - marker_top + 1 != 8
    ):
        raise ValueError("top-left fiducial is ambiguous")

    # Every cell is required to be a complete, solid tone.  This makes evidence
    # erasure an observable abstention rather than a silent guessed label.
    medians = []
    for row in range(_GRID):
        row_values = []
        for col in range(_GRID):
            y0 = top + _FRAME + row * _CELL
            x0 = left + _FRAME + col * _CELL
            block = array[y0 : y0 + _CELL, x0 : x0 + _CELL].reshape(-1, 3).astype(np.float32)
            median = np.median(block, axis=0)
            close = np.linalg.norm(block - median, axis=1) < 4.0
            if float(np.mean(close)) < 0.995:
                raise ValueError("raster cell is incomplete")
            if float(np.linalg.norm(median - background)) < 20.0:
                raise ValueError("raster cell is missing")
            row_values.append(median)
        medians.append(row_values)

    luminances = np.array(
        [0.299 * value[0] + 0.587 * value[1] + 0.114 * value[2] for row in medians for value in row],
        dtype=np.float32,
    )
    low = float(np.min(luminances))
    high = float(np.max(luminances))
    if high - low < 25.0:
        raise ValueError("two cell tones are not separable")
    midpoint = (low + high) / 2.0
    bits = [[0] * _GRID for _ in range(_GRID)]
    for row in range(_GRID):
        for col in range(_GRID):
            bits[row][col] = int(
                0.299 * medians[row][col][0]
                + 0.587 * medians[row][col][1]
                + 0.114 * medians[row][col][2]
                < midpoint
            )

    # Execute the declared upper-left rectangle XOR independently.
    transformed = [[0] * _GRID for _ in range(_GRID)]
    for row in range(_GRID):
        for col in range(_GRID):
            transformed[row][col] = (
                bits[row][col]
                ^ (transformed[row - 1][col] if row else 0)
                ^ (transformed[row][col - 1] if col else 0)
                ^ (transformed[row - 1][col - 1] if row and col else 0)
            )

    distance_a = _template_distance(transformed, _A)
    distance_b = _template_distance(transformed, _B)
    gap = abs(distance_a - distance_b)
    if distance_a == distance_b or gap < 4:
        raise ValueError("integral-XOR template margin is ambiguous")
    return "A" if distance_a < distance_b else "B"
