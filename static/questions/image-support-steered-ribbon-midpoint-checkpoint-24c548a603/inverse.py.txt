"""Independent pixel-only inverse arm for Ribbon Midpoint Checkpoint."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _color_mask(array: np.ndarray, name: str) -> np.ndarray:
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    if name == "centerline":
        return (red < 55) & (green < 75) & (blue < 95)
    if name == "red":
        return (red > 175) & (green < 105) & (blue < 115)
    if name == "green":
        return (red < 95) & (green > 120) & (blue < 145)
    if name == "blue":
        return (red < 90) & (green < 145) & (blue > 165)
    if name == "orange":
        return (red > 185) & (green > 90) & (green < 185) & (blue < 95)
    raise ValueError(name)


def _centroid(mask: np.ndarray, label: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 90:
        raise ValueError(f"missing or damaged {label} marker")
    if len(points) > 500:
        raise ValueError(f"ambiguous {label} marker")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _nearest(mask: np.ndarray, point: tuple[int, int]) -> tuple[int, int]:
    points = np.argwhere(mask)
    if not len(points):
        raise ValueError("centerline is absent")
    delta = points - np.asarray(point)
    index = int(np.argmin(np.sum(delta * delta, axis=1)))
    nearest = points[index]
    if int(np.sum((nearest - np.asarray(point)) ** 2)) > 36:
        raise ValueError("marker is not centered on the route")
    return int(nearest[0]), int(nearest[1])


def _distances(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    distances = np.full((height, width), -1, dtype=np.int32)
    distances[start] = 0
    pending = deque([start])
    while pending:
        y, x = pending.popleft()
        next_distance = int(distances[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = next_distance
                pending.append((ny, nx))
    return distances


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centerline = _color_mask(array, "centerline")
    count = int(centerline.sum())
    if not 4800 <= count <= 9000:
        raise ValueError("centerline ink is incomplete or excessive")
    points = np.argwhere(centerline)
    span_y = int(points[:, 0].max() - points[:, 0].min())
    span_x = int(points[:, 1].max() - points[:, 1].min())
    if min(span_x, span_y) < 135 or max(span_x, span_y) > 205:
        raise ValueError("centerline extent is outside the declared compact ribbon")

    markers = {
        label: _centroid(_color_mask(array, label), label)
        for label in ("blue", "orange", "red", "green")
    }
    route_points = {label: _nearest(centerline, point) for label, point in markers.items()}
    distances = _distances(centerline, route_points["blue"])
    route_distance = int(distances[route_points["orange"]])
    if route_distance < 1000:
        raise ValueError("finish is disconnected or the trace has a shortcut")
    red_distance = int(distances[route_points["red"]])
    green_distance = int(distances[route_points["green"]])
    if red_distance < 0 or green_distance < 0:
        raise ValueError("a checkpoint is disconnected from the traced route")
    midpoint = route_distance / 2.0
    red_score = abs(red_distance - midpoint)
    green_score = abs(green_distance - midpoint)
    if abs(red_score - green_score) < 20.0:
        raise ValueError("pixel midpoint decision is ambiguous")
    return "red" if red_score < green_score else "green"
