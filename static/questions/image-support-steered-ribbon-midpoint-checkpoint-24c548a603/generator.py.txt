"""Deterministic renderer and analytic arm for Ribbon Midpoint Checkpoint."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

BACKGROUND = (248, 246, 240)
RIBBON = (111, 143, 166)
CENTERLINE = (25, 43, 61)
BLUE = (38, 105, 210)
ORANGE = (235, 137, 38)
RED = (220, 61, 67)
GREEN = (44, 158, 91)
PALE = (213, 211, 204)


def _segments(points: list[list[int]]) -> Iterable[tuple[list[int], list[int], float]]:
    for first, second in zip(points, points[1:]):
        yield first, second, math.dist(first, second)


def _route_length(points: list[list[int]]) -> float:
    return sum(length for _, _, length in _segments(points))


def _point_at(points: list[list[int]], station: float) -> tuple[float, float]:
    remaining = float(station)
    for first, second, length in _segments(points):
        if remaining <= length:
            ratio = remaining / length if length else 0.0
            return (
                first[0] + ratio * (second[0] - first[0]),
                first[1] + ratio * (second[1] - first[1]),
            )
        remaining -= length
    return float(points[-1][0]), float(points[-1][1])


def _transform_point(x: int, y: int, rotation: int, reflected: bool) -> tuple[int, int]:
    if reflected:
        x = -x
    for _ in range(rotation % 4):
        x, y = -y, x
    return x, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    base = [
        [-90, -75], [90, -75], [90, -45], [-90, -45],
        [-90, -15], [90, -15], [90, 15], [-90, 15],
        [-90, 45], [90, 45], [90, 75], [-90, 75],
    ]
    rotation = rng.randrange(4)
    reflected = bool(rng.randrange(2))
    cx = 256 + rng.randint(-12, 12)
    cy = 256 + rng.randint(-12, 12)
    points = []
    for x, y in base:
        tx, ty = _transform_point(x, y, rotation, reflected)
        points.append([tx + cx, ty + cy])
    if rng.randrange(2):
        points.reverse()

    total = _route_length(points)
    near_offset = 48 + rng.randrange(0, 25)
    far_offset = 168 + rng.randrange(0, 37)
    near_station = total / 2.0 + (-near_offset if rng.randrange(2) else near_offset)
    far_station = total / 2.0 + (-far_offset if rng.randrange(2) else far_offset)
    if seed % 2 == 0:
        red_t, green_t = near_station, far_station
    else:
        red_t, green_t = far_station, near_station

    decorations = []
    for _ in range(8):
        side = rng.randrange(4)
        along = rng.randint(-105, 105)
        away = rng.randint(116, 134)
        dx, dy = ((along, -away), (away, along), (along, away), (-away, along))[side]
        decorations.append([cx + dx, cy + dy, rng.choice([2, 3])])
    return {
        "points": points,
        "red_t": red_t,
        "green_t": green_t,
        "width": 512,
        "height": 512,
        "decorations": decorations,
    }


def _ring(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: int) -> None:
    x, y = center
    box = [round(x - radius), round(y - radius), round(x + radius), round(y + radius)]
    draw.ellipse(box, fill=color)
    inner = radius - 5
    inner_box = [round(x - inner), round(y - inner), round(x + inner), round(y + inner)]
    draw.ellipse(inner_box, fill=RIBBON)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    points = [[int(x), int(y)] for x, y in scene["points"]]
    red_t = float(scene["red_t"])
    green_t = float(scene["green_t"])
    decorations = scene["decorations"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, radius in decorations:
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=PALE)

    draw.line([tuple(point) for point in points], fill=RIBBON, width=27, joint="curve")
    for point in points:
        x, y = point
        draw.ellipse([x - 13, y - 13, x + 13, y + 13], fill=RIBBON)

    _ring(draw, tuple(points[0]), BLUE, 14)
    _ring(draw, tuple(points[-1]), ORANGE, 14)
    _ring(draw, _point_at(points, red_t), RED, 12)
    _ring(draw, _point_at(points, green_t), GREEN, 12)

    # The explicitly named centerline remains continuous through every ring.
    draw.line([tuple(point) for point in points], fill=CENTERLINE, width=5, joint="curve")
    for point in points:
        x, y = point
        draw.ellipse([x - 2, y - 2, x + 2, y + 2], fill=CENTERLINE)
    return image


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    red_t = float(scene["red_t"])
    green_t = float(scene["green_t"])
    halfway = _route_length(points) / 2.0
    red_distance = abs(red_t - halfway)
    green_distance = abs(green_t - halfway)
    return "red" if red_distance < green_distance else "green"


def margin(scene: dict) -> float:
    points = scene["points"]
    red_t = float(scene["red_t"])
    green_t = float(scene["green_t"])
    halfway = _route_length(points) / 2.0
    return abs(abs(red_t - halfway) - abs(green_t - halfway))


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    red_t = float(scene["red_t"])
    green_t = float(scene["green_t"])
    red_point = _point_at(points, red_t)
    green_point = _point_at(points, green_t)
    return margin(scene) < 28.0 or math.dist(red_point, green_point) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
