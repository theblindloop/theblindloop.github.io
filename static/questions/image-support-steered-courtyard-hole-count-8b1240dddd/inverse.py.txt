"""Independent image-only topology oracle.

The oracle segments teal pixels without access to latent geometry, then labels
four-connected components of the non-teal complement. Components touching the
image border are exterior; remaining sufficiently large components are holes.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_TEAL = np.array([20, 132, 145], dtype=np.int16)
_MAX_COLOR_DISTANCE = 42.0
_MIN_SHEET_PIXELS = 30000
_MIN_HOLE_PIXELS = 500
_MIN_VALID_SHEET_PIXELS = 76250
_MAX_VALID_SHEET_PIXELS = 76600


def _components(mask: np.ndarray) -> list[tuple[int, bool]]:
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    output: list[tuple[int, bool]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            area = 0
            border = False
            while queue:
                cy, cx = queue.popleft()
                area += 1
                if cy == 0 or cx == 0 or cy == height - 1 or cx == width - 1:
                    border = True
                if cy and mask[cy - 1, cx] and not seen[cy - 1, cx]:
                    seen[cy - 1, cx] = True
                    queue.append((cy - 1, cx))
                if cy + 1 < height and mask[cy + 1, cx] and not seen[cy + 1, cx]:
                    seen[cy + 1, cx] = True
                    queue.append((cy + 1, cx))
                if cx and mask[cy, cx - 1] and not seen[cy, cx - 1]:
                    seen[cy, cx - 1] = True
                    queue.append((cy, cx - 1))
                if cx + 1 < width and mask[cy, cx + 1] and not seen[cy, cx + 1]:
                    seen[cy, cx + 1] = True
                    queue.append((cy, cx + 1))
            output.append((area, border))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    distances = np.linalg.norm(array - _TEAL, axis=2)
    teal = distances <= _MAX_COLOR_DISTANCE
    if int(teal.sum()) < _MIN_SHEET_PIXELS:
        raise ValueError("no substantial teal sheet")
    teal_components = [area for area, _ in _components(teal) if area >= 100]
    if len(teal_components) != 1:
        raise ValueError("teal sheet is not one connected region")
    complement = ~teal
    holes = [area for area, touches_border in _components(complement) if not touches_border and area >= _MIN_HOLE_PIXELS]
    count = len(holes)
    sheet_pixels = int(teal.sum())
    if not (_MIN_VALID_SHEET_PIXELS <= sheet_pixels <= _MAX_VALID_SHEET_PIXELS):
        raise ValueError("teal sheet has a missing or extraneous area")
    if count not in (1, 2, 3):
        # Destructive edits may produce a valid but out-of-contract topology.
        return str(count)
    return str(count)
