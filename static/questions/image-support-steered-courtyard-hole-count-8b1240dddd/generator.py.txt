"""Deterministic latent sampler and raster renderer for courtyard hole counting."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 320
HEIGHT = 320
TEAL = [20, 132, 145]
WHITE = [248, 248, 244]
SHEET = [18, 18, 302, 302]
QUARANTINE_WALL = 18


def _wall_margin(sheet: list[int], holes: list[list[int]]) -> float:
    sx0, sy0, sx1, sy1 = sheet
    values: list[float] = []
    for x0, y0, x1, y1 in holes:
        values.extend((x0 - sx0, y0 - sy0, sx1 - x1, sy1 - y1))
    for i, (ax0, ay0, ax1, ay1) in enumerate(holes):
        for bx0, by0, bx1, by1 in holes[i + 1 :]:
            dx = max(bx0 - ax1, ax0 - bx1, 0)
            dy = max(by0 - ay1, ay0 - by1, 0)
            values.append(math.hypot(dx, dy))
    return float(min(values))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    count = 1 + (int(seed) % 3)
    # Total courtyard area is always 4,800 px, counterbalancing total teal ink.
    # Partitions overlap in individual sizes so count is not encoded by one
    # special shape. Rectangle bounds below use inclusive PIL coordinates.
    dimension_sets = {
        1: [[60, 80]],
        2: rng.choice([[[40, 50], [40, 70]], [[48, 50], [48, 50]], [[40, 40], [40, 80]]]),
        3: rng.choice([[[30, 40], [40, 40], [40, 50]], [[20, 40], [40, 40], [48, 50]]]),
    }
    dimensions = [list(item) for item in dimension_sets[count]]
    rng.shuffle(dimensions)
    # Six remote cells counterbalance position and keep thick walls.
    cells = [(44, 42), (130, 42), (218, 42), (44, 184), (130, 184), (218, 184)]
    rng.shuffle(cells)
    selected = cells[:count]
    holes: list[list[int]] = []
    for (base_x, base_y), (w, h) in zip(selected, dimensions, strict=True):
        x0 = base_x + rng.randint(-2, 2)
        y0 = base_y + rng.randint(-2, 2)
        holes.append([x0, y0, x0 + w - 1, y0 + h - 1])
    holes.sort(key=lambda b: (b[1], b[0]))
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "sheet_bounds": list(SHEET),
        "holes": holes,
        "hole_count": count,
        "teal": list(TEAL),
        "background": list(WHITE),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    draw.rectangle(tuple(scene["sheet_bounds"]), fill=tuple(scene["teal"]))
    # Reading hole_count makes explicit that the rendered hole list realizes the label.
    if len(scene["holes"]) != int(scene["hole_count"]):
        raise ValueError("hole_count does not match rendered holes")
    for bounds in scene["holes"]:
        draw.rectangle(tuple(bounds), fill=tuple(scene["background"]))
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict[str, Any]) -> float:
    return _wall_margin(scene["sheet_bounds"], scene["holes"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_WALL


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
