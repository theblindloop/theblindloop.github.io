"""Sequence-order correspondence world renderer (paper-semantic family).

A cross-panel structural-correspondence task (linear *order* family). The top
panel shows a ``reference`` row of four distinctly colored square tiles placed
left to right in some fixed order. Below a divider sits a column of three
``candidate`` rows labeled A (top), B (middle), C (bottom); each candidate row
shows the *same* four colored tiles (the same multiset) but in a different
left-to-right permutation. Exactly one candidate reproduces the reference's
exact left-to-right color order. The question asks which candidate (A, B, or C)
preserves the reference's color sequence.

The pixel-only inverse arm must recover each tile as a local colored part, read
each row's left-to-right color order (a 1-D linear sequence), and then solve the
cross-panel correspondence by total order equality.

Latent state z (all influencing fields are rendered -- nothing hidden):

    ref_seq    : left-to-right color ordering (4-tuple) of the reference row.
    cand_seq   : length-3 list of candidate color orderings (4-tuples). Exactly
                 one equals ``ref_seq`` (the structural match); the rest are
                 distinct permutations of the same 4 colors.
    tile_colors: the fixed 4-color palette used by every tile.
    width, height, divider_y, ref_center, cand_centers, tile_x_offsets :
                 fixed render geometry, drawn verbatim.

The analytic gold is the label (A/B/C) of the candidate row whose color order
equals the reference's. Because every such field (tile colors in each row) is
rendered into the raster, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import random
from itertools import permutations

from PIL import Image, ImageDraw

RENDERER_VERSION = "sequence_order_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (80, 80, 80)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for counterbalance of ink).
WIDTH = 900
HEIGHT = 760
DIVIDER_Y = 200
DIVIDER_THICK = 3
TILE_HALF = 28          # half-side of a square tile
REF_CENTER = (450, 120)
CAND_Y = [300, 470, 640]   # candidate rows A, B, C top to bottom
TILE_X_OFFSETS = [-150, -50, 50, 150]

_MARGIN_FLOOR = 2.0  # require distractors to differ in >= 2 order positions


def _tile_positions(center) -> list[tuple[float, float]]:
    cx, cy = center
    return [(cx + ox, cy) for ox in TILE_X_OFFSETS]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    names = list(PALETTE)
    rng.shuffle(names)
    ref_seq = list(names)

    base_tiles = [[name for name in names]]
    other = [list(p) for p in permutations(names) if list(p) != ref_seq]
    rng.shuffle(other)
    for perm in other[:2]:
        base_tiles.append(perm)
    rng.shuffle(base_tiles)

    return {
        "ref_seq": ref_seq,
        "cand_seq": base_tiles,
        "tile_colors": list(PALETTE.keys()),
        "graph_center_desc": "reference row",
        "ref_center": list(REF_CENTER),
        "cand_centers": [[450, y] for y in CAND_Y],
        "tile_x_offsets": list(TILE_X_OFFSETS),
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_row(draw, center, order):
    for (x, y), color in zip(_tile_positions(center), order, strict=True):
        draw.rectangle(
            [x - TILE_HALF, y - TILE_HALF, x + TILE_HALF, y + TILE_HALF],
            fill=PALETTE[color],
        )


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution (crisp edges)."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_row(draw, tuple(scene["ref_center"]), scene["ref_seq"])
    for center, order in zip(scene["cand_centers"], scene["cand_seq"], strict=True):
        _draw_row(draw, tuple(center), order)
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate row with the reference order."""
    return "ABC"[scene["cand_seq"].index(scene["ref_seq"])]


def _hamming(a, b):
    return sum(1 for x, y in zip(a, b) if x != y)


def margin(scene: dict) -> float:
    """Runner-up margin: minimum number of differing order positions between the
    reference sequence and any distractor candidate row. Explicit Hamming distance."""
    ref = list(scene["ref_seq"])
    diffs = [
        _hamming(ref, list(m))
        for m in scene["cand_seq"]
        if list(m) != ref
    ]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene: never produced, because every distinct
    permutation of four distinct colors differs from the reference in at least two
    order positions, so distractors always sit at least two steps away."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (the color order of the
    reference and each candidate row) is rendered verbatim, so the raster
    uniquely determines the gold; no pixel-identical transform is declared."""
    return []
