"""Renderer, analytic gold and margin for deal_collect_shuffle.

The picture draws only a starting hand of nine discs, one of them ringed, a row
of coloured bins standing for the piles in dealing order, and a row of coloured
chips standing for the pick-up order.  No pile and no final row is ever drawn:
the answer is the place the ringed disc reaches after the declared deal and the
declared pick-up are replayed.

Everything the answer depends on is a drawn stroke: the drawn ring position (the
ringed disc is the one whose centre is nearest the ring centre), the number of
drawn bins, and the drawn left-to-right order of the chips matched to the bins
by colour.  Disc colours are decorative and never enter the gold.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "deal_collect_shuffle-renderer-0.1.0"

# ---------------------------------------------------------------- geometry
CANVAS = (900, 430)
BG = (250, 246, 236)
GREY = (150, 150, 150)
DARK = (20, 20, 20)

N_DISCS = 9
DISC_R = 15
DISC_Y = 72
DISC_X0 = 74
DISC_PITCH = 94

RING_R = 34
RING_W = 4

BIN_W, BIN_H, BIN_GAP, BIN_Y0 = 126, 58, 40, 176
CHIP_S, CHIP_GAP, CHIP_Y0 = 46, 44, 286

QUARANTINE_MARGIN = 6.0

DISC_COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (214, 45, 45),
    "orange": (238, 132, 26),
    "yellow": (235, 203, 40),
    "green": (46, 160, 70),
    "teal": (32, 164, 168),
    "blue": (48, 92, 206),
    "purple": (138, 64, 190),
    "pink": (232, 110, 170),
    "brown": (150, 96, 52),
}
BIN_COLOURS: dict[str, tuple[int, int, int]] = {
    "rust": (198, 64, 32),
    "steel": (58, 120, 196),
    "olive": (120, 168, 48),
    "orchid": (150, 80, 180),
}
BIN_NAMES = tuple(BIN_COLOURS)

POSITION_WORDS = (
    "one",
    "two",
    "three",
    "four",
    "five",
    "six",
    "seven",
    "eight",
    "nine",
)


def _disc_cx(slot: int) -> int:
    return DISC_X0 + DISC_PITCH * slot


def _bin_x0(n_piles: int, index: int) -> int:
    total = n_piles * BIN_W + (n_piles - 1) * BIN_GAP
    return (CANVAS[0] - total) // 2 + index * (BIN_W + BIN_GAP)


def _chip_x0(n_piles: int, index: int) -> int:
    total = n_piles * CHIP_S + (n_piles - 1) * CHIP_GAP
    return (CANVAS[0] - total) // 2 + index * (CHIP_S + CHIP_GAP)


# ---------------------------------------------------------------- scene ops
def _ordered(bag: list[dict], key: str) -> list[dict]:
    return sorted(bag, key=lambda rec: rec[key])


def ringed_slot(scene: dict) -> int:
    """The drawn ring names the disc whose centre is nearest the ring centre."""
    ring_x = float(scene["ring_x"])
    return min(range(N_DISCS), key=lambda s: abs(_disc_cx(s) - ring_x))


def _pile_sizes(n_piles: int) -> list[int]:
    sizes = [0] * n_piles
    for j in range(N_DISCS):
        sizes[j % n_piles] += 1
    return sizes


def final_position(scene: dict) -> int:
    """1-based place of the ringed disc in the collected row."""
    n_piles = int(scene["n_piles"])
    slot = ringed_slot(scene)
    pile = slot % n_piles
    rank = slot // n_piles
    order = [rec["bin"] for rec in _ordered(scene["collect"], "rank")]
    sizes = _pile_sizes(n_piles)
    before = 0
    for b in order:
        if b == pile:
            break
        before += sizes[b]
    return before + rank + 1


def analytic_gold(scene: dict) -> str:
    return POSITION_WORDS[final_position(scene) - 1]


def margin(scene: dict) -> float:
    """Clearance, in pixels, of the drawn ring-to-disc assignment."""
    ring_x = float(scene["ring_x"])
    dists = sorted(abs(_disc_cx(s) - ring_x) for s in range(N_DISCS))
    return round(dists[1] - dists[0], 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Bag permutations: every record carries its own drawn index."""
    out: list[tuple[str, dict]] = []
    discs = list(scene["discs"])
    out.append(("discs_reversed", dict(scene, discs=list(reversed(discs)))))
    out.append(("discs_rotated", dict(scene, discs=discs[3:] + discs[:3])))
    out.append(("bins_reversed", dict(scene, bins=list(reversed(scene["bins"])))))
    out.append(
        ("collect_reversed", dict(scene, collect=list(reversed(scene["collect"]))))
    )
    return out


# ---------------------------------------------------------------- sampling
def _sample_layout(rng: random.Random, target: int) -> tuple[int, list[int], int] | None:
    """Find (n_piles, collect order, ringed slot) whose replay lands on target."""
    for _ in range(400):
        n_piles = rng.choice((2, 3, 4))
        order = list(range(n_piles))
        rng.shuffle(order)
        slot = rng.randrange(N_DISCS)
        probe = {
            "n_piles": n_piles,
            "ring_x": float(_disc_cx(slot)),
            "collect": [{"rank": r, "bin": b} for r, b in enumerate(order)],
        }
        pos = final_position(probe)
        if pos != target:
            continue
        # over-correction: the drawn starting place never predicts the answer
        if pos == slot + 1:
            continue
        return n_piles, order, slot
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = (seed % N_DISCS) + 1
    layout = None
    for _ in range(64):
        layout = _sample_layout(rng, target)
        if layout is not None:
            break
        target = rng.randrange(1, N_DISCS + 1)
    assert layout is not None, "layout sampling failed"
    n_piles, order, slot = layout

    colours = list(DISC_COLOURS)
    rng.shuffle(colours)
    discs = [{"slot": j, "colour": colours[j]} for j in range(N_DISCS)]
    rng.shuffle(discs)

    bin_names = list(BIN_NAMES)
    rng.shuffle(bin_names)
    bins = [{"index": b, "colour": bin_names[b]} for b in range(n_piles)]
    rng.shuffle(bins)

    collect = [{"rank": r, "bin": b} for r, b in enumerate(order)]
    rng.shuffle(collect)

    near = seed % 7 == 3
    cx = _disc_cx(slot)
    if near and 0 < slot < N_DISCS - 1:
        want = rng.uniform(3.0, 16.0)
        offset = (DISC_PITCH - want) / 2.0
        offset *= rng.choice((-1.0, 1.0))
    else:
        offset = rng.uniform(-12.0, 12.0)
    ring_x = round(cx + offset, 3)

    return {
        "canvas": list(CANVAS),
        "n_discs": N_DISCS,
        "discs": discs,
        "ring_x": ring_x,
        "n_piles": n_piles,
        "bins": bins,
        "collect": collect,
    }


# ---------------------------------------------------------------- rendering
def _arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    draw.line([(x0, y), (x1 - 16, y)], fill=GREY, width=4)
    draw.polygon([(x1, y), (x1 - 18, y - 9), (x1 - 18, y + 9)], fill=GREY)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)
    n_piles = int(scene["n_piles"])

    for rec in _ordered(scene["discs"], "slot"):
        cx = _disc_cx(rec["slot"])
        rgb = DISC_COLOURS[rec["colour"]]
        d.ellipse(
            [cx - DISC_R, DISC_Y - DISC_R, cx + DISC_R, DISC_Y + DISC_R], fill=rgb
        )

    rx = int(round(float(scene["ring_x"])))
    d.ellipse(
        [rx - RING_R, DISC_Y - RING_R, rx + RING_R, DISC_Y + RING_R],
        outline=DARK,
        width=RING_W,
    )

    _arrow(d, 60, 840, 128)

    for rec in _ordered(scene["bins"], "index"):
        x0 = _bin_x0(n_piles, rec["index"])
        d.rectangle(
            [x0, BIN_Y0, x0 + BIN_W, BIN_Y0 + BIN_H], fill=BIN_COLOURS[rec["colour"]]
        )

    by_index = {rec["index"]: rec["colour"] for rec in scene["bins"]}
    for rec in _ordered(scene["collect"], "rank"):
        x0 = _chip_x0(n_piles, rec["rank"])
        d.rectangle(
            [x0, CHIP_Y0, x0 + CHIP_S, CHIP_Y0 + CHIP_S],
            fill=BIN_COLOURS[by_index[rec["bin"]]],
        )

    _arrow(d, _chip_x0(n_piles, 0) - 20, _chip_x0(n_piles, n_piles - 1) + CHIP_S + 20, 366)
    return img
