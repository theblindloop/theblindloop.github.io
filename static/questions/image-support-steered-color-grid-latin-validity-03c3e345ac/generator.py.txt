"""Deterministic renderer for a visible 4-by-4 color Latin-square check."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
QUARANTINE_MARGIN = 10.0

BACKGROUNDS = (
    (249, 245, 235),
    (241, 248, 246),
    (248, 241, 247),
    (242, 245, 251),
)
BOARD_FILLS = (
    (202, 220, 232),
    (227, 211, 190),
    (214, 225, 201),
    (222, 207, 229),
)
GRID_COLORS = (
    (42, 52, 63),
    (57, 47, 43),
    (42, 57, 48),
    (55, 44, 61),
)
PALETTE_SETS = (
    ((205, 55, 67), (42, 111, 211), (35, 151, 94), (235, 166, 38)),
    ((177, 61, 190), (24, 151, 169), (230, 105, 39), (73, 83, 207)),
    ((217, 71, 42), (49, 132, 194), (120, 67, 188), (42, 157, 116)),
    ((193, 54, 116), (30, 145, 178), (225, 137, 34), (65, 139, 76)),
)


def _permuted_values(rng: random.Random, counterfactual: bool) -> list[list[int]]:
    values = [[(row + column) % 4 for column in range(4)] for row in range(4)]
    if counterfactual:
        # A two-cell swap preserves the four-per-color global histogram while
        # producing duplicates in two rows and two columns.
        values[0][0], values[1][1] = values[1][1], values[0][0]
    row_order = list(range(4))
    column_order = list(range(4))
    symbol_order = list(range(4))
    rng.shuffle(row_order)
    rng.shuffle(column_order)
    rng.shuffle(symbol_order)
    return [
        [symbol_order[values[row_order[row]][column_order[column]]] for column in range(4)]
        for row in range(4)
    ]


def sample_scene(seed: int) -> dict:
    """Return adjacent matched control/counterfactual scenes.

    Seeds 2k and 2k+1 share all visual nuisances.  The even scene is a valid
    Latin square; the odd scene differs only by a count-preserving value swap.
    """

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x4C415449)
    palette_code = pair % len(PALETTE_SETS)
    palette = list(PALETTE_SETS[palette_code])
    rng.shuffle(palette)
    cell_size = (92, 94, 96)[(pair // 4) % 3]
    board_size = 4 * cell_size
    return {
        "background": list(BACKGROUNDS[(pair // 2) % len(BACKGROUNDS)]),
        "board_fill": list(BOARD_FILLS[(pair // 8) % len(BOARD_FILLS)]),
        "board_left": 256 - board_size // 2 + 2 * rng.randrange(-3, 4),
        "board_top": 256 - board_size // 2 + 2 * rng.randrange(-3, 4),
        "cell_size": cell_size,
        "grid_color": list(GRID_COLORS[(pair // 16) % len(GRID_COLORS)]),
        "palette": [list(color) for color in palette],
        "radius": (24, 26, 28)[(pair // 12) % 3],
        "rotation_quarters": rng.randrange(4),
        "values": _permuted_values(rng, counterfactual=bool(seed % 2)),
    }


def _rotated_values(scene: dict) -> list[list[int]]:
    values = [list(map(int, row)) for row in scene["values"]]
    for _ in range(int(scene["rotation_quarters"]) % 4):
        values = [list(row) for row in zip(*values[::-1], strict=True)]
    return values


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def render(scene: dict) -> Image.Image:
    ss = SUPERSAMPLE
    background = tuple(map(int, scene["background"]))
    board_fill = tuple(map(int, scene["board_fill"]))
    grid_color = tuple(map(int, scene["grid_color"]))
    palette = [tuple(map(int, color)) for color in scene["palette"]]
    left = int(scene["board_left"])
    top = int(scene["board_top"])
    cell = int(scene["cell_size"])
    radius = int(scene["radius"])
    right = left + 4 * cell
    bottom = top + 4 * cell

    image = Image.new("RGB", (CANVAS_SIZE * ss, CANVAS_SIZE * ss), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle(_scaled_box((left, top, right, bottom), ss), fill=board_fill)

    line_width = 6 * ss
    for index in range(5):
        x = (left + index * cell) * ss
        y = (top + index * cell) * ss
        draw.line([(x, top * ss), (x, bottom * ss)], fill=grid_color, width=line_width)
        draw.line([(left * ss, y), (right * ss, y)], fill=grid_color, width=line_width)

    values = _rotated_values(scene)
    for row in range(4):
        for column in range(4):
            cx = left + (column + 0.5) * cell
            cy = top + (row + 0.5) * cell
            color = palette[values[row][column]]
            draw.ellipse(
                _scaled_box((cx - radius, cy - radius, cx + radius, cy + radius), ss),
                fill=color,
                outline=tuple(max(0, channel - 24) for channel in color),
                width=2 * ss,
            )

    return image.resize((CANVAS_SIZE, CANVAS_SIZE), Image.Resampling.LANCZOS)


def _valid(values: list[list[int]]) -> bool:
    if len(values) != 4 or any(len(row) != 4 for row in values):
        return False
    return all(len(set(row)) == 4 for row in values) and all(
        len({values[row][column] for row in range(4)}) == 4 for column in range(4)
    )


def analytic_gold(scene: dict) -> str:
    values = [list(map(int, row)) for row in scene["values"]]
    return "yes" if _valid(values) else "no"


def _distance(a: list[int], b: list[int]) -> float:
    return math.sqrt(sum((int(x) - int(y)) ** 2 for x, y in zip(a, b, strict=True)))


def margin(scene: dict) -> float:
    palette = list(scene["palette"])
    color_margin = min(
        _distance(palette[first], palette[second])
        for first in range(4)
        for second in range(first + 1, 4)
    ) - 64.0
    token_clearance = float(scene["cell_size"]) - 2.0 * float(scene["radius"]) - 20.0
    panel_contrast = _distance(scene["background"], scene["board_fill"]) - 30.0
    return round(min(color_margin, token_clearance, panel_contrast), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["rotation_quarters"] = int(twin["rotation_quarters"]) + 4
    return [("one_full_board_turn", twin)]
