"""Independent pixel-only inverse arm for marked-vertex connectivity."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        pending = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while pending:
            cy, cx = pending.popleft()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        pending.append((ny, nx))
        found.append(np.asarray(pixels, dtype=float))
    return found


def _junctions(rgb: np.ndarray) -> tuple[list[tuple[float, float]], int]:
    high = rgb.max(axis=2).astype(int)
    low = rgb.min(axis=2).astype(int)
    saturated = (high - low > 45) & (high < 238) & (low < 195)
    components = [component for component in _components(saturated) if 400 <= len(component) <= 1100]
    if len(components) != 16:
        raise ValueError(f"expected 16 visible circular junctions, found {len(components)}")
    centers = []
    warmth = []
    for component in components:
        y, x = component.mean(axis=0)
        centers.append((float(x), float(y)))
        yy = component[:, 0].astype(int)
        xx = component[:, 1].astype(int)
        mean = rgb[yy, xx].astype(float).mean(axis=0)
        warmth.append(float(mean[0] - 0.5 * (mean[1] + mean[2])))
    order = np.argsort(warmth)
    target = int(order[-1])
    if warmth[target] - warmth[int(order[-2])] < 60.0:
        raise ValueError("orange marked junction is missing or not unique")
    return centers, target


def _dark_fraction(rgb: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    x1, y1 = first
    x2, y2 = second
    dx = x2 - x1
    dy = y2 - y1
    length = math.hypot(dx, dy)
    px = -dy / length
    py = dx / length
    hits = 0
    total = 0
    for step in range(20, 81):
        t = step / 100.0
        bx = x1 + t * dx
        by = y1 + t * dy
        for offset in (-3, -2, -1, 0, 1, 2, 3):
            x = int(round(bx + px * offset))
            y = int(round(by + py * offset))
            if 0 <= y < rgb.shape[0] and 0 <= x < rgb.shape[1]:
                pixel = rgb[y, x].astype(int)
                total += 1
                if int(pixel.max()) < 130 and int(pixel.max() - pixel.min()) < 38:
                    hits += 1
    return hits / max(total, 1)


def _recover_graph(rgb: np.ndarray) -> tuple[list[set[int]], int]:
    centers, target = _junctions(rgb)
    pairs = []
    for left in range(len(centers)):
        for right in range(left + 1, len(centers)):
            pairs.append((math.dist(centers[left], centers[right]), left, right))
    nearest = min(distance for distance, _, _ in pairs)
    adjacency = [set() for _ in centers]
    for distance, left, right in pairs:
        if distance > nearest * 1.20:
            continue
        if _dark_fraction(rgb, centers[left], centers[right]) > 0.70:
            adjacency[left].add(right)
            adjacency[right].add(left)
    return adjacency, target


def decision_from_image(image: Image.Image) -> str:
    """Recover graph incidence and the orange vertex, then delete that vertex."""
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.shape[0] < 200 or rgb.shape[1] < 200:
        raise ValueError("raster is too small")
    adjacency, target = _recover_graph(rgb)
    edge_count = sum(map(len, adjacency)) // 2
    if edge_count != 17 or len(adjacency[target]) != 2:
        raise ValueError(f"unstable graph recovery: edges={edge_count}, target_degree={len(adjacency[target])}")
    remaining = set(range(len(adjacency))) - {target}
    start = next(iter(remaining))
    seen = {start}
    pending = [start]
    while pending:
        for neighbor in adjacency[pending.pop()]:
            if neighbor != target and neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return "yes" if seen != remaining else "no"
