"""Deterministic renderer and analytic arm for marked-vertex connectivity."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 450
HEIGHT = 450
NODE_COUNT = 16
GRID_EDGES = tuple(
    (4 * row + col, 4 * row + col + 1)
    for row in range(4)
    for col in range(3)
) + tuple(
    (4 * row + col, 4 * (row + 1) + col)
    for row in range(3)
    for col in range(4)
)

NODE_PALETTES = (
    (29, 122, 174),
    (32, 148, 111),
    (86, 92, 178),
)
TARGET_PALETTES = (
    (218, 83, 50),
    (190, 70, 110),
    (220, 132, 35),
)
EDGE_PALETTES = ((35, 40, 48), (43, 39, 47), (31, 47, 45))
BACKGROUNDS = ((249, 248, 244), (247, 249, 250), (250, 247, 242))


def _adjacency(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> list[set[int]]:
    adjacency = [set() for _ in range(NODE_COUNT)]
    for left, right in edges:
        adjacency[int(left)].add(int(right))
        adjacency[int(right)].add(int(left))
    return adjacency


def _degrees(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> list[int]:
    return [len(neighbors) for neighbors in _adjacency(edges)]


def _component_count(edges: list[list[int]] | tuple[tuple[int, int], ...], skip: int | None = None) -> int:
    adjacency = _adjacency(edges)
    remaining = [index for index in range(NODE_COUNT) if index != skip]
    unseen = set(remaining)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        pending = [start]
        while pending:
            for neighbor in adjacency[pending.pop()]:
                if neighbor != skip and neighbor in unseen:
                    unseen.remove(neighbor)
                    pending.append(neighbor)
    return count


def _targets(edges: list[list[int]]) -> tuple[list[int], list[int]]:
    degrees = _degrees(edges)
    articulation = []
    ordinary = []
    for node, degree in enumerate(degrees):
        if degree != 2:
            continue
        if _component_count(edges, node) > 1:
            articulation.append(node)
        else:
            ordinary.append(node)
    return articulation, ordinary


def _transform_point(x: int, y: int, pose: int) -> tuple[int, int]:
    """Apply one of the eight square symmetries about the canvas center."""
    x -= 225
    y -= 225
    if pose >= 4:
        x = -x
    for _ in range(pose % 4):
        x, y = -y, x
    return x + 225, y + 225


def sample_scene(seed: int) -> dict:
    """Sample matched graph pairs with equal node/link counts and target degree."""
    seed = int(seed)
    want_yes = seed % 2 == 0
    graph_rng = random.Random((seed // 2) ^ 0xA471C)
    edges: list[list[int]] | None = None
    yes_targets: list[int] = []
    no_targets: list[int] = []
    for _ in range(10000):
        trial = [list(edge) for edge in graph_rng.sample(GRID_EDGES, 17)]
        if _component_count(trial) != 1:
            continue
        yes_targets, no_targets = _targets(trial)
        if yes_targets and no_targets:
            edges = sorted(trial)
            break
    if edges is None:
        raise RuntimeError(f"could not sample a counterbalanced graph for seed {seed}")
    target_pool = yes_targets if want_yes else no_targets
    target = int(target_pool[graph_rng.randrange(len(target_pool))])

    # A seed pair shares every nuisance and even the graph; only which degree-two
    # vertex is orange changes, yielding opposite analytic decisions.
    rng = random.Random((seed // 2) ^ 0xC07F17)
    pose = rng.randrange(8)
    dx = rng.randint(-9, 9)
    dy = rng.randint(-9, 9)
    canonical = [(90 + 90 * col, 90 + 90 * row) for row in range(4) for col in range(4)]
    nodes = []
    for x, y in canonical:
        tx, ty = _transform_point(x, y, pose)
        nodes.append([tx + dx, ty + dy])

    decoration_slots = [(35, 40), (415, 38), (35, 410), (415, 412), (224, 34), (224, 416)]
    rng.shuffle(decoration_slots)
    decorations = [[x + rng.randint(-4, 4), y + rng.randint(-4, 4)] for x, y in decoration_slots[:4]]
    return {
        "nodes": nodes,
        "edges": edges,
        "target": target,
        "node_radius": rng.choice((14, 15, 16)),
        "edge_width": rng.choice((9, 10, 11)),
        "node_rgb": list(rng.choice(NODE_PALETTES)),
        "target_rgb": list(rng.choice(TARGET_PALETTES)),
        "edge_rgb": list(rng.choice(EDGE_PALETTES)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    """Render all public fields as antialiased raster geometry."""
    scale = 3
    background = tuple(scene["background_rgb"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)

    for x, y in scene["decorations"]:
        radius = 5 * scale
        pale = tuple((3 * int(channel) + 210) // 4 for channel in background)
        draw.ellipse(
            [x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius],
            fill=pale,
        )

    nodes = scene["nodes"]
    edge_color = tuple(scene["edge_rgb"])
    for left, right in scene["edges"]:
        x1, y1 = nodes[int(left)]
        x2, y2 = nodes[int(right)]
        draw.line(
            (x1 * scale, y1 * scale, x2 * scale, y2 * scale),
            fill=edge_color,
            width=int(scene["edge_width"]) * scale,
        )

    radius = int(scene["node_radius"]) * scale
    ordinary = tuple(scene["node_rgb"])
    marked = tuple(scene["target_rgb"])
    target = int(scene["target"])
    for index, (x, y) in enumerate(nodes):
        color = marked if index == target else ordinary
        draw.ellipse(
            [x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius],
            fill=color,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    target = int(scene["target"])
    return "yes" if _component_count(edges, target) > 1 else "no"


def margin(scene: dict) -> float:
    """Signed half-integer distance between one and multiple remaining pieces."""
    pieces = _component_count(scene["edges"], int(scene["target"]))
    return float(pieces) - 1.5


def is_quarantined(scene: dict) -> bool:
    """Reject any latent record outside the thick, separated incidence regime."""
    try:
        if len(scene["nodes"]) != NODE_COUNT or len(scene["edges"]) != 17:
            return True
        if _component_count(scene["edges"]) != 1:
            return True
        if _degrees(scene["edges"])[int(scene["target"])] != 2:
            return True
        allowed = {tuple(sorted(edge)) for edge in GRID_EDGES}
        if any(tuple(sorted(map(int, edge))) not in allowed for edge in scene["edges"]):
            return True
        if int(scene["node_radius"]) < 13 or int(scene["edge_width"]) < 8:
            return True
        background = scene["background_rgb"]
        for color, floor in ((scene["node_rgb"], 90), (scene["target_rgb"], 90), (scene["edge_rgb"], 150)):
            contrast = math.sqrt(sum((int(a) - int(b)) ** 2 for a, b in zip(color, background)))
            if contrast < floor:
                return True
    except (KeyError, IndexError, TypeError, ValueError):
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse unordered same-color primitives without changing pixels or gold."""
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_primitive_serialization", twin)]
