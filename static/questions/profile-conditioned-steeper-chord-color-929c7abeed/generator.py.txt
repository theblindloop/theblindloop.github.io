"""Intra-colour chord-steepness measurement world.

A light panel carries exactly two red dots and exactly two blue dots. For each
colour the two dots are the endpoints of that colour's chord; the chord's
steepness is its deviation from horizontal, measured as the angle away from the
horizontal axis: atan2(|dy|, |dx|) in degrees. The decision is a recovered
measurement: recover each colour's two dot centres from its coloured disks,
form the unique within-colour chord, measure its steepness angle, and decide
which colour's chord is STEEPER (larger deviation from horizontal).

The analysed quantity is an intra-class orientation extremum recovered from the
raster: neither the total coloured ink, the dot count (both colours have exactly
two dots), chord length, dot positions alone, nor a colour-area proxy determines
which colour's chord is steeper. Ties are resolved by a clear angular margin and
any near-tie scene is quarantined. The oracle and analytic gold share no
geometry and never read latent scene data.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "steeper_chord_color-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
DOT_R = 8
RED_COUNT = 2
BLUE_COUNT = 2
EDGE = 55
MIN_ANGLE = 12.0          # min chord steepness angle (degrees from horizontal)
MAX_ANGLE = 78.0          # max chord steepness angle (degrees from horizontal)
CHORD_MIN = 36.0          # min chord length (px) for a stable angle measurement
CHORD_MAX = 150.0         # max chord length (px)
MIN_SEP = 26.0            # min distance between any red and any blue dot
ANGLE_MARGIN = 10.0       # min angular gap between the two chords
QUARANTINE = 6.0
MIN_MARGIN = 9.0


def _chord_angle(dots):
    (x1, y1), (x2, y2) = dots
    dx = abs(x2 - x1)
    dy = abs(y2 - y1)
    return math.degrees(math.atan2(dy, dx))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 977 + 17)
    while True:
        def pair():
            while True:
                x1 = rng.uniform(EDGE, WIDTH - EDGE)
                y1 = rng.uniform(EDGE, HEIGHT - EDGE)
                angle = rng.uniform(MIN_ANGLE, MAX_ANGLE)
                length = rng.uniform(CHORD_MIN, CHORD_MAX)
                dx = length * math.cos(math.radians(angle))
                dy = length * math.sin(math.radians(angle))
                sx = rng.choice((-1.0, 1.0))
                sy = rng.choice((-1.0, 1.0))
                x2 = x1 + sx * dx
                y2 = y1 + sy * dy
                if EDGE <= x2 <= WIDTH - EDGE and EDGE <= y2 <= HEIGHT - EDGE:
                    return [
                        [round(x1, 3), round(y1, 3)],
                        [round(x2, 3), round(y2, 3)],
                    ]

        red = pair()
        blue = pair()
        all_dots = red + blue
        ok = True
        for i in range(len(all_dots)):
            for j in range(i + 1, len(all_dots)):
                if (i < 2) == (j < 2):
                    continue
                if _pair_distance_frame(all_dots[i], all_dots[j]) < MIN_SEP:
                    ok = False
        if not ok:
            continue
        a_red = _chord_angle(red)
        a_blue = _chord_angle(blue)
        if abs(a_red - a_blue) < ANGLE_MARGIN:
            continue
        scene = {
            "red_dots": red,
            "blue_dots": blue,
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        return scene


def _pair_distance_frame(a, b):
    return math.hypot(b[0] - a[0], b[1] - a[1])


def _angles(scene: dict):
    return _chord_angle(scene["red_dots"]), _chord_angle(scene["blue_dots"])


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for x, y in scene["red_dots"]:
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=COLOR_RGB["red"])
    for x, y in scene["blue_dots"]:
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=COLOR_RGB["blue"])
    return img


def analytic_gold(scene: dict) -> str:
    a_red, a_blue = _angles(scene)
    return "red" if a_red > a_blue else "blue"


def margin(scene: dict) -> float:
    a_red, a_blue = _angles(scene)
    return float(abs(a_red - a_blue))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE


def latent_symmetries(scene: dict) -> list:
    return []
