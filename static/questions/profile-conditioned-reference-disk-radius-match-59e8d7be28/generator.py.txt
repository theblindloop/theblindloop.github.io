"""Reference-guided disk-radius visual search world.

A dark reference disk is drawn in a box at the top of the image. Below, the
image is split into four quadrants, each holding one dark filled disk. Exactly
one quadrant's disk has the SAME radius as the reference disk; each of the
other three disks has a radius that clearly differs from the reference radius.
The decision is the quadrant containing the disk whose radius matches the
reference. All rendering is deterministic and every decision-relevant radius is
visible in the raster.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SIZE = 480
BG = (255, 255, 255)
SEG = (20, 20, 20)
GRAY = (196, 196, 196)
BOX = (170, 170, 170)

REF_CX = SIZE // 2
REF_CY = 70

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, 250),
    "TR": (3 * SIZE // 4, 250),
    "BL": (SIZE // 4, 410),
    "BR": (3 * SIZE // 4, 410),
}

RMIN = 14
RMAX = 46
# Every distractor radius must differ from the reference radius by at least
# this many pixels, giving a guaranteed runner-up decision margin.
MIN_RADIUS_GAP = 4
QUARANTINE_THRESH = 3.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_radius = int(rng.integers(RMIN, RMAX + 1))

    radii: dict[str, int] = {}
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    radii[winner_sector] = ref_radius

    for sector in SECTORS:
        if sector in radii:
            continue
        for _ in range(800):
            candidate = int(rng.integers(RMIN, RMAX + 1))
            if abs(candidate - ref_radius) < MIN_RADIUS_GAP:
                continue
            if candidate in radii.values():
                continue
            radii[sector] = candidate
            break
        else:  # pragma: no cover - extremely unlikely at this radius range
            radii[sector] = int((ref_radius + MIN_RADIUS_GAP) % RMAX) or RMIN

    candidates = [
        {
            "cx": SECTOR_CENTERS[sector][0],
            "cy": SECTOR_CENTERS[sector][1],
            "radius": int(radii[sector]),
            "sector": sector,
        }
        for sector in SECTORS
    ]
    return {
        "reference": {
            "cx": REF_CX,
            "cy": REF_CY,
            "radius": int(ref_radius),
        },
        "candidates": candidates,
    }


def _stamp_disk(arr: np.ndarray, cx: int, cy: int, r: int) -> None:
    ys, xs = np.ogrid[:SIZE, :SIZE]
    mask = (xs - cx) ** 2 + (ys - cy) ** 2 <= r * r
    arr[mask] = SEG


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG

    # Light-gray reference-panel box outline and quadrant dividers. All are
    # strictly lighter than the dark-threshold so the oracle never confuses
    # them with a disk.
    y0, y1 = 14, 126
    arr[y0:y1, REF_CX - 170 : REF_CX - 170 + 6] = BOX
    arr[y0:y1, REF_CX + 170 - 6 : REF_CX + 170] = BOX
    arr[y0 : y0 + 4, REF_CX - 170 : REF_CX + 170] = BOX
    arr[y1 - 4 : y1, REF_CX - 170 : REF_CX + 170] = BOX
    # horizontal divider at y=310, vertical divider at x=240 below the box.
    arr[309:311, :] = GRAY
    arr[:, 240:242] = GRAY
    ref = scene["reference"]
    _stamp_disk(arr, ref["cx"], ref["cy"], ref["radius"])
    for cand in scene["candidates"]:
        _stamp_disk(arr, cand["cx"], cand["cy"], cand["radius"])

    return Image.fromarray(arr, "RGB")


def _decision(scene: dict) -> tuple[str, float]:
    ref_radius = scene["reference"]["radius"]
    gaps = []
    for cand in scene["candidates"]:
        gaps.append((abs(cand["radius"] - ref_radius), cand["sector"]))
    gaps.sort(key=lambda item: (item[0], item[1]))
    best = gaps[0]
    second = gaps[1]
    return best[1], float(second[0] - best[0])


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
