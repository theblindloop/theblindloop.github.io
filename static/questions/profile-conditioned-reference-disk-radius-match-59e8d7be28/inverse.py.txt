"""Independent pixel-only inverse arm for reference_disk_radius_match.

Receives only a PIL image and the public question contract. It separates the
dark reference disk (top box) from the four dark candidate disks (quadrants
below), measures each disk's radius as half of its dark-pixel bounding-box
width, and selects the quadrant whose disk radius best matches (equals) the
reference disk's radius.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np


def _radius_of(sel_x: np.ndarray, sel_y: np.ndarray) -> float | None:
    if sel_x.size < 8:
        return None
    width = int(sel_x.max()) - int(sel_x.min())
    return width / 2.0


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ys, xs = np.nonzero(dark)
    if xs.size == 0:
        return "TL"

    x_split = w * 0.5
    y_ref = h * 0.31
    y_upper = h * 0.416
    y_split = h * 0.645

    ref_sel = ys < y_ref
    ref_radius = _radius_of(xs[ref_sel], ys[ref_sel])
    if ref_radius is None:
        return "TL"

    best_sector = None
    best_gap = float("inf")
    for sector in ("TL", "TR", "BL", "BR"):
        if sector in ("TL", "BL"):
            left = xs < x_split
        else:
            left = xs >= x_split
        if sector in ("TL", "TR"):
            band = (ys >= y_upper) & (ys < y_split)
        else:
            band = ys >= y_split
        sel = band & left
        radius = _radius_of(xs[sel], ys[sel])
        if radius is None:
            continue
        gap = abs(radius - ref_radius)
        if gap < best_gap:
            best_gap = gap
            best_sector = sector

    return best_sector if best_sector is not None else "TL"
