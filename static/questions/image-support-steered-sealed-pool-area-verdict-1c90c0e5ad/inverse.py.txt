"""Pixel-only inverse arm for `sealed_pool_area_verdict`.

Input is one rasterised page and nothing else. The arm segments reef ink by
darkness, recovers the reef's block lattice from the ink itself, checks the
blocky configuration that the question declares, groups the carved blocks into
pools, keeps only the pools that reef ink shuts in, and compares the largest
shut pool with the sum of the other shut pools. Any raster that does not show
the declared configuration - a single blocky slab of uniform blocks whose pools
are separated rectangles - is answered with `abstain`, because pool closure is
undefined once the slab has been broken.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK_LEVEL = 70.0
MIN_BLOCK_PX = 14
MAX_BLOCK_PX = 44
MIN_BLOCKS = 5
MAX_BLOCKS = 20
ABSTAIN = "abstain"


def _dark_mask(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float64)
    luma = rgb[:, :, 0] * 0.299 + rgb[:, :, 1] * 0.587 + rgb[:, :, 2] * 0.114
    return luma < DARK_LEVEL


def _slab_box(dark: np.ndarray):
    rows = np.flatnonzero(dark.any(axis=1))
    cols = np.flatnonzero(dark.any(axis=0))
    if not len(rows) or not len(cols):
        return None
    top, bottom = int(rows[0]), int(rows[-1]) + 1
    left, right = int(cols[0]), int(cols[-1]) + 1
    height, width = dark.shape
    if top == 0 or left == 0 or bottom == height or right == width:
        return None
    if (bottom - top) != (right - left):
        return None
    return top, left, bottom, right


def _lattice(patch: np.ndarray):
    """Recover the block size and the block occupancy of the slab patch."""
    side = patch.shape[0]
    for blocks in range(MIN_BLOCKS, MAX_BLOCKS + 1):
        if side % blocks:
            continue
        step = side // blocks
        if step < MIN_BLOCK_PX or step > MAX_BLOCK_PX:
            continue
        tiles = patch.reshape(blocks, step, blocks, step)
        filled = tiles.sum(axis=(1, 3))
        if not np.all((filled == 0) | (filled == step * step)):
            continue
        return step, (filled > 0)
    return None


def _pool_groups(occupied: np.ndarray):
    blocks = occupied.shape[0]
    carved = {
        (row, col)
        for row in range(blocks)
        for col in range(blocks)
        if not occupied[row, col]
    }
    seen: set[tuple[int, int]] = set()
    groups = []
    for start in sorted(carved):
        if start in seen:
            continue
        stack = [start]
        seen.add(start)
        group = []
        touches = False
        while stack:
            row, col = stack.pop()
            group.append((row, col))
            if row in (0, blocks - 1) or col in (0, blocks - 1):
                touches = True
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nxt = (row + dy, col + dx)
                if nxt in carved and nxt not in seen:
                    seen.add(nxt)
                    stack.append(nxt)
        groups.append((group, touches))
    return groups, carved


def _rectangular(group) -> bool:
    rows = [cell[0] for cell in group]
    cols = [cell[1] for cell in group]
    height = max(rows) - min(rows) + 1
    width = max(cols) - min(cols) + 1
    return height * width == len(group)


def _slab_connected(occupied: np.ndarray) -> bool:
    blocks = occupied.shape[0]
    dark = {
        (row, col)
        for row in range(blocks)
        for col in range(blocks)
        if occupied[row, col]
    }
    if not dark:
        return False
    start = min(dark)
    seen = {start}
    stack = [start]
    while stack:
        row, col = stack.pop()
        for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nxt = (row + dy, col + dx)
            if nxt in dark and nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return len(seen) == len(dark)


def decision_from_image(image: Image.Image) -> str:
    dark = _dark_mask(image)
    box = _slab_box(dark)
    if box is None:
        return ABSTAIN
    top, left, bottom, right = box
    patch = dark[top:bottom, left:right]
    lattice = _lattice(patch)
    if lattice is None:
        return ABSTAIN
    _step, occupied = lattice
    if not _slab_connected(occupied):
        return ABSTAIN
    groups, carved = _pool_groups(occupied)
    if not groups:
        return ABSTAIN
    shut = []
    for group, touches in groups:
        if not _rectangular(group):
            return ABSTAIN
        if not touches:
            # A shut pool must keep one whole block of ink between itself and
            # every other pool, so no carved block may be a diagonal neighbour.
            for row, col in group:
                for dy in (-1, 0, 1):
                    for dx in (-1, 0, 1):
                        near = (row + dy, col + dx)
                        if near in carved and near not in group:
                            return ABSTAIN
            shut.append(len(group))
    if not shut:
        return ABSTAIN
    shut.sort(reverse=True)
    if len(shut) == 1:
        return "biggest"
    return "biggest" if shut[0] > sum(shut[1:]) else "others"
