"""Renderer and analytic scene definition for the polyphase quilt world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw


RENDERER_VERSION = "polyphase-quilt-restore-0.1.0"
WIDTH, HEIGHT = 640, 480
TILE_SIZES = (224, 240, 256, 272)
MARK_SIZE = 18
MARK_GAP = 7
FRAME = (74, 71, 66)
BACKGROUND = (246, 242, 233)
MARKER_COLORS = (
    (221, 55, 95),
    (35, 135, 202),
    (54, 164, 91),
    (176, 76, 194),
    (235, 143, 34),
)
SYMBOLS = ("arch", "bolt", "fork", "crown")
SMOOTH_SIZE = 7


def _glyph_points(symbol: str) -> tuple[tuple[float, float], ...]:
    if symbol == "arch":
        return ((0.22, 0.70), (0.27, 0.54), (0.37, 0.41), (0.50, 0.36),
                (0.63, 0.41), (0.73, 0.54), (0.78, 0.70))
    if symbol == "bolt":
        return ((0.58, 0.20), (0.36, 0.49), (0.51, 0.49), (0.40, 0.80),
                (0.68, 0.45), (0.53, 0.45))
    if symbol == "fork":
        return ((0.50, 0.78), (0.50, 0.49), (0.29, 0.25),
                (0.50, 0.49), (0.71, 0.25))
    if symbol == "crown":
        return ((0.24, 0.73), (0.24, 0.34), (0.39, 0.55),
                (0.50, 0.27), (0.61, 0.55), (0.76, 0.34),
                (0.76, 0.73), (0.24, 0.73))
    raise ValueError(f"unknown symbol {symbol!r}")


def _glyph_mask(symbol: str, size: int, dx: int, dy: int, scale: float) -> np.ndarray:
    mask = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(mask)
    cx = size * 0.5 + float(dx)
    cy = size * 0.5 + float(dy)
    points = [
        (cx + (x - 0.5) * size * scale, cy + (y - 0.5) * size * scale)
        for x, y in _glyph_points(symbol)
    ]
    width = max(7, int(round(size * 0.055 * scale)))
    draw.line(points, fill=255, width=width, joint="curve")
    return np.asarray(mask, dtype=np.float32) / 255.0


def _carrier(scene: dict, size: int) -> np.ndarray:
    yy, xx = np.indices((size, size), dtype=np.int32)
    phase = int(scene["carrier_phase"])
    mode = int(scene["carrier_mode"]) % 3
    if mode == 0:
        field = xx + phase
        signs = np.where(np.mod(field, 4) < 2, 1.0, -1.0)
    elif mode == 1:
        field = xx + yy + phase
        signs = np.where(np.mod(field, 4) < 2, 1.0, -1.0)
    else:
        field = 2 * xx + yy + phase
        signs = np.where(np.mod(field, 8) < 4, 1.0, -1.0)
    return signs * float(scene["carrier_amplitude"])


def _canonical(scene: dict) -> np.ndarray:
    size = int(scene["tile_size"])
    glyph = _glyph_mask(
        str(scene["symbol"]),
        size,
        int(scene["symbol_dx"]),
        int(scene["symbol_dy"]),
        float(scene["glyph_scale"]),
    )
    values = (
        float(scene["base_level"])
        + _carrier(scene, size)
        - float(scene["signal_strength"]) * glyph
    )
    return np.rint(np.clip(values, 0.0, 255.0)).astype(np.uint8)


def _polyphase_weave(canonical: np.ndarray) -> np.ndarray:
    size = canonical.shape[0]
    half = size // 2
    recorded = np.empty_like(canonical)
    recorded[0::2, 0::2] = canonical[:half, :half]
    recorded[0::2, 1::2] = canonical[:half, half:]
    recorded[1::2, 0::2] = canonical[half:, :half]
    recorded[1::2, 1::2] = canonical[half:, half:]
    return recorded


def _marker_boxes(tile_x: int, tile_y: int, size: int) -> list[tuple[int, int, int, int]]:
    left = tile_x - MARK_GAP - MARK_SIZE
    top = tile_y - MARK_GAP - MARK_SIZE
    right = tile_x + size + MARK_GAP
    bottom = tile_y + size + MARK_GAP
    return [
        (left, top, left + MARK_SIZE - 1, top + MARK_SIZE - 1),
        (right, top, right + MARK_SIZE - 1, top + MARK_SIZE - 1),
        (left, bottom, left + MARK_SIZE - 1, bottom + MARK_SIZE - 1),
        (right, bottom, right + MARK_SIZE - 1, bottom + MARK_SIZE - 1),
    ]


def sample_scene(seed: int) -> dict:
    """Sample a JSON-serializable scene with label/nuisance separation."""
    seed = int(seed)
    rng = random.Random(seed * 130363 + 7919)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    signal_levels = (34, 44, 54, 64)
    return {
        "symbol": symbol,
        "tile_size": TILE_SIZES[rng.randrange(len(TILE_SIZES))],
        "tile_x": 88 + rng.randrange(0, 232),
        "tile_y": 62 + rng.randrange(0, 92),
        "marker_color": list(MARKER_COLORS[rng.randrange(len(MARKER_COLORS))]),
        "carrier_phase": rng.randrange(8),
        "carrier_mode": rng.randrange(3),
        "carrier_amplitude": (20, 24, 28)[rng.randrange(3)],
        "base_level": (108, 128, 148)[rng.randrange(3)],
        "signal_strength": signal_levels[(seed // 4) % len(signal_levels)],
        "symbol_dx": rng.randrange(-18, 19),
        "symbol_dy": rng.randrange(-18, 19),
        "glyph_scale": (0.90, 1.00, 1.10)[rng.randrange(3)],
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["tile_size"])
    if size % 2 or size not in TILE_SIZES:
        raise ValueError(f"unsupported even tile size {size}")
    tile_x, tile_y = int(scene["tile_x"]), int(scene["tile_y"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    recorded = _polyphase_weave(_canonical(scene))
    tile = Image.fromarray(recorded, mode="L").convert("RGB")
    image.paste(tile, (tile_x, tile_y))
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (tile_x - 2, tile_y - 2, tile_x + size + 1, tile_y + size + 1),
        outline=FRAME,
        width=2,
    )
    color = tuple(int(value) for value in scene["marker_color"])
    for box in _marker_boxes(tile_x, tile_y, size):
        draw.rectangle(box, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return float(scene["signal_strength"]) - 1.15 * float(scene["carrier_amplitude"]) - 22.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 8
    return [("eight_phase_period_alias", twin)]
