"""Gap-seal world: which open coloured doorway, if bricked up, seals off a pocket.

A rectangular compound of thick black walls is drawn on white paper. Some
lattice edges carry no ink at all (plain gaps), three carry a coloured bar
(open doorways), and the rest are solid black wall. Every part of the floor can
currently be walked out to the paper outside the compound. Exactly one of the
three coloured doorways is the only remaining way out of some pocket of floor:
bricking that one up - and only that one - would leave floor with no path to
the outside.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 480
SUPERSAMPLE = 3
ROWS = 4
COLS = 4

COLORS = ("amber", "magenta", "azure")
RGB = {
    "amber": (245, 170, 20),
    "magenta": (205, 30, 150),
    "azure": (30, 120, 235),
}
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

QUARANTINE_MARGIN = 8.0
SAMPLE_MIN_MARGIN = 9.0


# ---------------------------------------------------------------- lattice ---


def _all_edges(rows: int, cols: int):
    edges = []
    for i in range(rows + 1):
        for j in range(cols):
            edges.append(("h", i, j))
    for i in range(rows):
        for j in range(cols + 1):
            edges.append(("v", i, j))
    return edges


def _is_rim(edge, rows: int, cols: int) -> bool:
    kind, i, j = edge
    if kind == "h":
        return i == 0 or i == rows
    return j == 0 or j == cols


def _sides(edge, rows: int, cols: int):
    """The two graph nodes an edge separates; "OUT" is the paper outside."""
    kind, i, j = edge
    if kind == "h":
        a = (i - 1, j) if i > 0 else "OUT"
        b = (i, j) if i < rows else "OUT"
    else:
        a = (i, j - 1) if j > 0 else "OUT"
        b = (i, j) if j < cols else "OUT"
    return a, b


def _nodes_of(edge):
    """Lattice corner points touched by an edge, as (row, col) lattice indices."""
    kind, i, j = edge
    if kind == "h":
        return {(i, j), (i, j + 1)}
    return {(i, j), (i + 1, j)}


def _sealed_cells(rows, cols, blocked, cell_list):
    """Cells with no path to OUT when `blocked` edges are solid."""
    passable = {}
    for edge in _all_edges(rows, cols):
        if edge in blocked:
            continue
        a, b = _sides(edge, rows, cols)
        passable.setdefault(a, []).append(b)
        passable.setdefault(b, []).append(a)
    seen = {"OUT"}
    stack = ["OUT"]
    while stack:
        node = stack.pop()
        for nxt in passable.get(node, ()):
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return [c for c in cell_list if c not in seen]


# ---------------------------------------------------------------- sampling --


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    rows, cols = ROWS, COLS
    cells = [(i, j) for i in range(rows) for j in range(cols)]
    edges = _all_edges(rows, cols)
    target_color = COLORS[seed % 3]

    for _ in range(4000):
        cell = rng.randint(84, 92)
        thickness = rng.randint(9, 15)
        interior_wall_p = rng.uniform(0.46, 0.66)

        walls = set()
        for edge in edges:
            rim = _is_rim(edge, rows, cols)
            p = 0.86 if rim else interior_wall_p
            if rng.random() < p:
                walls.add(edge)
        open_edges = [e for e in edges if e not in walls]
        if not open_edges:
            continue
        if not any(_is_rim(e, rows, cols) for e in open_edges):
            continue
        # The compound must start fully open to the outside: no pre-sealed pocket.
        if _sealed_cells(rows, cols, walls, cells):
            continue

        winners, neutrals = [], []
        for edge in open_edges:
            pocket = _sealed_cells(rows, cols, walls | {edge}, cells)
            if 2 <= len(pocket) <= 8:
                winners.append((edge, pocket))
            elif not pocket:
                neutrals.append(edge)
        if not winners or len(neutrals) < 2:
            continue

        win_edge, pocket = winners[rng.randrange(len(winners))]
        win_nodes = _nodes_of(win_edge)
        pool = [e for e in neutrals if not (_nodes_of(e) & win_nodes)]
        rim_pool = [e for e in pool if _is_rim(e, rows, cols)]
        inner_pool = [e for e in pool if not _is_rim(e, rows, cols)]
        if not rim_pool or not inner_pool:
            continue
        # One distractor always leads straight outside and one is interior, so
        # neither "the rim doorway" nor "the inner doorway" is a usable shortcut.
        d1 = rim_pool[rng.randrange(len(rim_pool))]
        inner_ok = [e for e in inner_pool if not (_nodes_of(e) & _nodes_of(d1))]
        if not inner_ok:
            continue
        d2 = inner_ok[rng.randrange(len(inner_ok))]

        others = [c for c in COLORS if c != target_color]
        rng.shuffle(others)
        door_map = {target_color: win_edge, others[0]: d1, others[1]: d2}

        margin = float(min(thickness, cell - thickness))
        if margin < SAMPLE_MIN_MARGIN:
            continue

        origin = (CANVAS - cols * cell) // 2
        scene = {
            "seed": seed,
            "canvas": CANVAS,
            "supersample": SUPERSAMPLE,
            "rows": rows,
            "cols": cols,
            "cell": cell,
            "origin": origin,
            "wall_thickness": thickness,
            "walls": sorted([list(e) for e in walls]),
            "doors": [
                {"color": c, "edge": list(door_map[c])} for c in COLORS
            ],
        }
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


# ----------------------------------------------------------------- reading --


def _walls(scene):
    return {tuple(e) for e in scene["walls"]}


def _doors(scene):
    return {d["color"]: tuple(d["edge"]) for d in scene["doors"]}


def _cells(scene):
    return [(i, j) for i in range(scene["rows"]) for j in range(scene["cols"])]


def analytic_gold(scene: dict) -> str:
    rows, cols = scene["rows"], scene["cols"]
    walls, doors = _walls(scene), _doors(scene)
    cells = _cells(scene)
    sealers = [
        c
        for c in COLORS
        if _sealed_cells(rows, cols, walls | {doors[c]}, cells)
    ]
    if len(sealers) != 1:
        raise ValueError("scene does not have a unique sealing doorway")
    return sealers[0]


def margin(scene: dict) -> float:
    return float(min(scene["wall_thickness"], scene["cell"] - scene["wall_thickness"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    """Pure bookkeeping relabellings; the renderer canonicalises before drawing."""
    out = []
    flipped_walls = dict(scene)
    flipped_walls["walls"] = list(reversed(scene["walls"]))
    out.append(("reverse_wall_list", flipped_walls))
    flipped_doors = dict(scene)
    flipped_doors["doors"] = list(reversed(scene["doors"]))
    out.append(("reverse_door_list", flipped_doors))
    both = dict(scene)
    both["walls"] = list(reversed(scene["walls"]))
    both["doors"] = list(reversed(scene["doors"]))
    out.append(("reverse_both_lists", both))
    return out


# --------------------------------------------------------------- rendering --


def _edge_rect(edge, origin, cell, half):
    kind, i, j = edge
    x0 = origin + j * cell
    y0 = origin + i * cell
    if kind == "h":
        return (x0 - half, y0 - half, origin + (j + 1) * cell + half, y0 + half)
    return (x0 - half, y0 - half, x0 + half, origin + (i + 1) * cell + half)


def render(scene: dict) -> Image.Image:
    s = scene["supersample"]
    size = scene["canvas"] * s
    cell = scene["cell"] * s
    origin = scene["origin"] * s
    half = (scene["wall_thickness"] * s) // 2

    img = Image.new("RGB", (size, size), WHITE)
    draw = ImageDraw.Draw(img)

    doors = _doors(scene)
    walls = _walls(scene)
    # Doorways first, solid wall on top: the coloured bar therefore spans exactly
    # the gap between two wall faces and never lays a seam across a wall.
    for color in COLORS:
        draw.rectangle(_edge_rect(doors[color], origin, cell, half), fill=RGB[color])
    for edge in _all_edges(scene["rows"], scene["cols"]):
        if edge in walls:
            draw.rectangle(_edge_rect(edge, origin, cell, half), fill=BLACK)

    return img.resize((scene["canvas"], scene["canvas"]), Image.BOX)
