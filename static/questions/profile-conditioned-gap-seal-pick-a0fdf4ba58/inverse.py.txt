"""Pixel-only inverse arm for the gap-seal world.

Receives nothing but the rendered RGB image. It segments the palette, then for
each coloured doorway in turn treats that doorway as bricked up and floods the
walkable paper inwards from the border of the picture. The doorway that leaves
white floor unreachable from the outside is the answer; anything other than
exactly one such doorway abstains.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_COLORS = ("amber", "magenta", "azure")
_REF = {
    "white": (255, 255, 255),
    "amber": (245, 170, 20),
    "magenta": (205, 30, 150),
    "azure": (30, 120, 235),
}
_DARK = 90
_MIN_POCKET_PIXELS = 200
_MIN_DOOR_PIXELS = 60
ABSTAIN = "abstain"


def _labels(arr: np.ndarray) -> np.ndarray:
    """0 = solid black ink, 1 = white paper, 2..4 = the three doorway colours."""
    dark = np.all(arr < _DARK, axis=2)
    keys = ("white",) + _COLORS
    dist = np.stack(
        [np.sum((arr.astype(np.int32) - np.array(_REF[k], dtype=np.int32)) ** 2, axis=2)
         for k in keys],
        axis=0,
    )
    lab = np.argmin(dist, axis=0).astype(np.int8) + 1
    lab[dark] = 0
    return lab


def _flood_from_border(passable: np.ndarray) -> np.ndarray:
    """Scanline flood fill of `passable` seeded from every border pixel."""
    h, w = passable.shape
    reached = np.zeros_like(passable)
    stack = []
    for x in range(w):
        stack.append((x, 0))
        stack.append((x, h - 1))
    for y in range(h):
        stack.append((0, y))
        stack.append((w - 1, y))

    while stack:
        x, y = stack.pop()
        if not passable[y, x] or reached[y, x]:
            continue
        x1 = x
        while x1 >= 0 and passable[y, x1] and not reached[y, x1]:
            x1 -= 1
        x1 += 1
        x2 = x
        while x2 < w and passable[y, x2] and not reached[y, x2]:
            x2 += 1
        x2 -= 1
        reached[y, x1 : x2 + 1] = True
        for ny in (y - 1, y + 1):
            if 0 <= ny < h:
                row = passable[ny, x1 : x2 + 1] & ~reached[ny, x1 : x2 + 1]
                idx = np.nonzero(row)[0]
                if idx.size:
                    starts = idx[np.concatenate(([True], np.diff(idx) > 1))]
                    for sx in starts:
                        stack.append((x1 + int(sx), ny))
    return reached


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    lab = _labels(arr)
    solid = lab == 0
    white = lab == 1

    sealers = []
    for index, color in enumerate(_COLORS, start=2):
        door = lab == index
        if int(door.sum()) < _MIN_DOOR_PIXELS:
            return ABSTAIN
        passable = ~solid & ~door
        reached = _flood_from_border(passable)
        pocket = white & ~reached
        if int(pocket.sum()) >= _MIN_POCKET_PIXELS:
            sealers.append(color)

    # Sanity: with nothing bricked up, all white paper must already be reachable.
    open_reached = _flood_from_border(~solid)
    if int((white & ~open_reached).sum()) >= _MIN_POCKET_PIXELS:
        return ABSTAIN

    if len(sealers) != 1:
        return ABSTAIN
    return sealers[0]
