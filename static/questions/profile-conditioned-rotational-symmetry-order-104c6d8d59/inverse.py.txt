"""Independent pixel-only inverse arm for rotational_symmetry_order.

The arm accepts only the final RGB raster. It segments the gray center ring and
all 60 dark components, measures their centroids, and compares the complete set
with copies rotated by 1/3, 1/4, 1/5, and 1/6 of a full turn.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_DARK = np.array((38.0, 46.0, 57.0), dtype=np.float64)
_CENTER_GRAY = np.array((118.0, 126.0, 138.0), dtype=np.float64)
_ORDERS: tuple[int, ...] = (3, 4, 5, 6)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = set(zip(*np.nonzero(mask), strict=True))
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        queue: deque[tuple[int, int]] = deque([start])
        group = [start]
        while queue:
            y, x = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        queue.append(neighbor)
                        group.append(neighbor)
        groups.append(np.asarray(group, dtype=np.float64))
    return groups


def _rotation_score(points: np.ndarray, center: np.ndarray, order: int) -> float:
    shifted = points - center
    angle = 2.0 * math.pi / float(order)
    cosine, sine = math.cos(angle), math.sin(angle)
    rotated = np.column_stack(
        (
            center[0] + cosine * shifted[:, 0] - sine * shifted[:, 1],
            center[1] + sine * shifted[:, 0] + cosine * shifted[:, 1],
        )
    )
    distances = np.linalg.norm(rotated[:, None, :] - points[None, :, :], axis=2)
    residuals = np.sort(distances.min(axis=1))[::-1]
    return float(np.mean(residuals[: max(1, len(residuals) // 4)]))


def measure_image(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    if arr.ndim != 3 or arr.shape[2] != 3:
        raise ValueError("abstain: expected an RGB raster")

    dark_mask = np.linalg.norm(arr - _DARK, axis=2) < 48.0
    dark_groups = [group for group in _components(dark_mask) if len(group) >= 45]
    if len(dark_groups) != 60 or any(not 55 <= len(group) <= 150 for group in dark_groups):
        raise ValueError("abstain: expected exactly 60 intact dark-dot components")
    points = np.asarray(
        [
            [float(group[:, 1].mean() + 0.5), float(group[:, 0].mean() + 0.5)]
            for group in dark_groups
        ],
        dtype=np.float64,
    )

    center_mask = np.linalg.norm(arr - _CENTER_GRAY, axis=2) < 52.0
    center_groups = sorted(_components(center_mask), key=len, reverse=True)
    if not center_groups or not 180 <= len(center_groups[0]) <= 650:
        raise ValueError("abstain: the hollow gray center ring is absent or malformed")
    ring = center_groups[0]
    center = np.array(
        [float(ring[:, 1].mean() + 0.5), float(ring[:, 0].mean() + 0.5)],
        dtype=np.float64,
    )

    scores = {order: _rotation_score(points, center, order) for order in _ORDERS}
    if scores[6] < 3.0:
        decision = 6
        winner_score = max(scores[3], scores[6])
        runner_score = min(scores[4], scores[5])
    else:
        decision = min((3, 4, 5), key=lambda order: (scores[order], order))
        winner_score = scores[decision]
        runner_score = min(scores[order] for order in _ORDERS if order != decision)
    runner_up_gap = float(runner_score - winner_score)
    if winner_score > 2.5 or runner_up_gap < 5.0:
        raise ValueError("abstain: rotational repeat order is raster-ambiguous")
    return {
        "decision": str(decision),
        "winner_score": float(winner_score),
        "runner_up_gap": runner_up_gap,
        "center": center.tolist(),
        "dark_component_count": len(dark_groups),
        "scores": {str(order): float(scores[order]) for order in _ORDERS},
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure_image(image)["decision"])
