"""Sampler, renderer, and analytic arm for rotational_symmetry_order."""

from __future__ import annotations

import copy
import math
import random

import numpy as np
from PIL import Image, ImageDraw

ORDERS: tuple[int, ...] = (3, 4, 5, 6)
BACKGROUND = (248, 247, 242)
DARK = (38, 46, 57)
CENTER_GRAY = (118, 126, 138)
POINT_COUNT = 60
QUARANTINE_MARGIN_PX = 7.0


def _rotation_score(points: list[list[float]], center: list[float], order: int) -> float:
    """Mean of the worst quarter of rotate-to-set residuals, in pixels."""

    array = np.asarray(points, dtype=np.float64)
    origin = np.asarray(center, dtype=np.float64)
    shifted = array - origin
    angle = 2.0 * math.pi / float(order)
    cosine, sine = math.cos(angle), math.sin(angle)
    rotated = np.column_stack(
        (
            origin[0] + cosine * shifted[:, 0] - sine * shifted[:, 1],
            origin[1] + sine * shifted[:, 0] + cosine * shifted[:, 1],
        )
    )
    distances = np.linalg.norm(rotated[:, None, :] - array[None, :, :], axis=2)
    residuals = np.sort(distances.min(axis=1))[::-1]
    return float(np.mean(residuals[: max(1, len(residuals) // 4)]))


def _scores(scene: dict) -> dict[int, float]:
    return {
        order: _rotation_score(scene["points"], scene["center"], order)
        for order in ORDERS
    }


def _decision_and_margin(scene: dict) -> tuple[str, float]:
    scores = _scores(scene)
    # Sixfold structure is also invariant under a 1/3 turn. The public rule asks
    # for the smallest positive turn (equivalently the greatest repeat count),
    # so a near-zero 1/6 score takes precedence over the nested 1/3 match.
    if scores[6] < 1.0:
        winner = 6
        winner_score = max(scores[3], scores[6])
        runner_score = min(scores[4], scores[5])
    else:
        winner = min((3, 4, 5), key=lambda order: (scores[order], order))
        winner_score = scores[winner]
        runner_score = min(scores[order] for order in ORDERS if order != winner)
    return str(winner), float(runner_score - winner_score)


def _far_enough(points: list[list[float]], candidate: list[float], clearance: float) -> bool:
    return all(
        math.hypot(candidate[0] - point[0], candidate[1] - point[1]) >= clearance
        for point in points
    )


def sample_scene(seed: int) -> dict:
    """Sample one intact k-fold constellation with a fixed 60-dot ink budget.

    Consecutive quartets share center, canvas, scale range, dot count, and style;
    only the randomly realized global repeat order cycles through 3, 4, 5, 6.
    """

    geometry_id = int(seed) // len(ORDERS)
    order = ORDERS[int(seed) % len(ORDERS)]
    base_seed = 0x51A77 + geometry_id * 104729
    width = height = 640

    for layout_attempt in range(120):
        common_rng = random.Random(base_seed + layout_attempt * 15485863)
        center = [float(common_rng.randint(306, 334)), float(common_rng.randint(306, 334))]
        # Class-specific randomness starts only after the shared visible framing
        # has been fixed, preventing order from controlling center or scale.
        rng = random.Random(base_seed + layout_attempt * 15485863 + order * 1009)
        points: list[list[float]] = []
        orbit_count = POINT_COUNT // order
        completed = True
        for _orbit in range(orbit_count):
            accepted = False
            for _ in range(5000):
                radius = rng.uniform(62.0, 270.0)
                phase = rng.uniform(0.0, 2.0 * math.pi / float(order))
                orbit = [
                    [
                        round(center[0] + radius * math.cos(phase + turn * 2.0 * math.pi / order), 5),
                        round(center[1] + radius * math.sin(phase + turn * 2.0 * math.pi / order), 5),
                    ]
                    for turn in range(order)
                ]
                if not all(
                    20.0 <= point[0] <= width - 20.0
                    and 20.0 <= point[1] <= height - 20.0
                    for point in orbit
                ):
                    continue
                trial = list(points)
                if all(_far_enough(trial + orbit[:index], point, 18.0) for index, point in enumerate(orbit)):
                    points.extend(orbit)
                    accepted = True
                    break
            if not accepted:
                completed = False
                break
        if not completed or len(points) != POINT_COUNT:
            continue
        rng.shuffle(points)
        scene = {
            "width": width,
            "height": height,
            "center": center,
            "points": points,
            "dot_radius": 5,
            "center_radius": 14,
            "center_stroke": 4,
            "background": list(BACKGROUND),
            "dark": list(DARK),
            "center_gray": list(CENTER_GRAY),
        }
        decision, decision_margin = _decision_and_margin(scene)
        if decision == str(order) and decision_margin >= 9.0:
            return scene
    raise RuntimeError(f"could not sample an unambiguous rotational order for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Render only visible point and center geometry with supersampling."""

    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    dot_radius = int(scene["dot_radius"]) * scale
    for px, py in scene["points"]:
        x, y = int(round(px * scale)), int(round(py * scale))
        draw.ellipse(
            [x - dot_radius, y - dot_radius, x + dot_radius, y + dot_radius],
            fill=tuple(scene["dark"]),
        )

    cx, cy = scene["center"]
    center_radius = int(scene["center_radius"]) * scale
    center_stroke = int(scene["center_stroke"]) * scale
    draw.ellipse(
        [
            cx * scale - center_radius,
            cy * scale - center_radius,
            cx * scale + center_radius,
            cy * scale + center_radius,
        ],
        outline=tuple(scene["center_gray"]),
        width=center_stroke,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _decision_and_margin(scene)[0]


def margin(scene: dict) -> float:
    return _decision_and_margin(scene)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_unordered_dark_point_list", twin)]
