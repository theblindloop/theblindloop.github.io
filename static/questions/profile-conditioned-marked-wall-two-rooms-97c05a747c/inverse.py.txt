"""Independent pixel-only inverse for marked wall / bounded-face incidence.

Only the supplied raster is inspected. No renderer, latent scene, coordinates,
templates, prompt module, metadata, or analytic answer is imported or accepted.
"""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image


class AmbiguousRasterError(ValueError):
    """Raised when the PNG does not expose a robust instance of the public rule."""


def _flood_exterior(free: np.ndarray) -> np.ndarray:
    height, width = free.shape
    exterior = np.zeros_like(free, dtype=bool)
    queue: deque[tuple[int, int]] = deque()

    def add(y: int, x: int) -> None:
        if free[y, x] and not exterior[y, x]:
            exterior[y, x] = True
            queue.append((y, x))

    for x in range(width):
        add(0, x)
        add(height - 1, x)
    for y in range(1, height - 1):
        add(y, 0)
        add(y, width - 1)
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and free[ny, nx] and not exterior[ny, nx]:
                exterior[ny, nx] = True
                queue.append((ny, nx))
    return exterior


def _component_map(free: np.ndarray, exterior: np.ndarray) -> tuple[np.ndarray, list[int]]:
    height, width = free.shape
    labels = np.zeros((height, width), dtype=np.int16)
    labels[exterior] = -1
    available = free & ~exterior
    sizes: list[int] = []
    label = 0
    for sy, sx in np.argwhere(available):
        if labels[sy, sx]:
            continue
        label += 1
        labels[sy, sx] = label
        queue: deque[tuple[int, int]] = deque([(int(sy), int(sx))])
        size = 0
        while queue:
            y, x = queue.popleft()
            size += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and available[ny, nx]
                    and labels[ny, nx] == 0
                ):
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        sizes.append(size)
    return labels, sizes


def _dominant_label(values: np.ndarray) -> int:
    useful = [int(value) for value in values.ravel() if int(value) != 0]
    if len(useful) < 80:
        raise AmbiguousRasterError("too few visible free-space pixels beside the orange wall")
    label, count = Counter(useful).most_common(1)[0]
    if count / len(useful) < 0.92:
        raise AmbiguousRasterError("a long side of the orange wall meets multiple regions")
    return label


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.shape[0] < 180 or rgb.shape[1] < 180:
        raise AmbiguousRasterError("image is too small for stable topology")

    free = np.min(rgb, axis=2) > 220
    orange = (
        (rgb[:, :, 0] > 155)
        & (rgb[:, :, 0] - rgb[:, :, 1] > 55)
        & (rgb[:, :, 1] - rgb[:, :, 2] > 25)
    )
    points = np.argwhere(orange)
    area = len(points)
    if not 750 <= area <= 1900:
        raise AmbiguousRasterError(f"orange wall area is ambiguous: {area}")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height = int(y1 - y0 + 1)
    width = int(x1 - x0 + 1)
    if max(height, width) < 80 or max(height, width) < 4 * min(height, width):
        raise AmbiguousRasterError(f"orange component is not one long wall: {(width, height)}")

    exterior = _flood_exterior(free)
    labels, sizes = _component_map(free, exterior)
    if len(sizes) != 5 or min(sizes, default=0) < 5000:
        raise AmbiguousRasterError(f"expected five robust enclosed rooms, found {sizes}")

    if height > width:
        trim = max(12, height // 4)
        ys = slice(int(y0 + trim), int(y1 - trim + 1))
        left = labels[ys, max(0, int(x0) - 20) : max(0, int(x0) - 3)]
        right = labels[ys, min(labels.shape[1], int(x1) + 4) : min(labels.shape[1], int(x1) + 21)]
        first, second = _dominant_label(left), _dominant_label(right)
    else:
        trim = max(12, width // 4)
        xs = slice(int(x0 + trim), int(x1 - trim + 1))
        above = labels[max(0, int(y0) - 20) : max(0, int(y0) - 3), xs]
        below = labels[min(labels.shape[0], int(y1) + 4) : min(labels.shape[0], int(y1) + 21), xs]
        first, second = _dominant_label(above), _dominant_label(below)
    if first == second:
        raise AmbiguousRasterError("the two long sides do not resolve to distinct regions")
    return "yes" if first > 0 and second > 0 else "no"
