"""Deterministic sampler, analytic arm, and renderer for marked wall/face incidence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

SIZE = 420
XS = (60, 160, 260, 360)
YS = (100, 210, 320)

_PALETTES = (
    ((35, 43, 55), (250, 249, 246), (221, 117, 39)),
    ((48, 39, 63), (248, 250, 247), (204, 93, 42)),
    ((28, 50, 57), (250, 248, 244), (224, 133, 35)),
)

_TRANSPOSE = (
    None,
    Image.Transpose.ROTATE_90,
    Image.Transpose.ROTATE_180,
    Image.Transpose.ROTATE_270,
    Image.Transpose.FLIP_LEFT_RIGHT,
    Image.Transpose.FLIP_TOP_BOTTOM,
    Image.Transpose.TRANSPOSE,
    Image.Transpose.TRANSVERSE,
)


def _normalized(edge: list[int]) -> tuple[int, int, int, int]:
    a = (int(edge[0]), int(edge[1]))
    b = (int(edge[2]), int(edge[3]))
    if b < a:
        a, b = b, a
    return a[0], a[1], b[0], b[1]


def _all_edges(rows: int, cols: int) -> list[list[int]]:
    edges: list[list[int]] = []
    for x in range(cols + 1):
        for y in range(rows):
            edges.append([x, y, x, y + 1])
    for y in range(rows + 1):
        for x in range(cols):
            edges.append([x, y, x + 1, y])
    return edges


def _internal_edges(rows: int, cols: int) -> list[list[int]]:
    edges: list[list[int]] = []
    for x in range(1, cols):
        for y in range(rows):
            edges.append([x, y, x, y + 1])
    for y in range(1, rows):
        for x in range(cols):
            edges.append([x, y, x + 1, y])
    return edges


def _perimeter_edges(rows: int, cols: int) -> list[list[int]]:
    return [
        edge
        for edge in _all_edges(rows, cols)
        if edge[1] == edge[3] == 0
        or edge[1] == edge[3] == rows
        or edge[0] == edge[2] == 0
        or edge[0] == edge[2] == cols
    ]


def _adjacent_cells(edge: list[int], rows: int, cols: int) -> tuple[int, int]:
    x1, y1, x2, y2 = _normalized(edge)
    if x1 == x2 and 0 < x1 < cols and y2 == y1 + 1:
        return y1 * cols + x1 - 1, y1 * cols + x1
    if y1 == y2 and 0 < y1 < rows and x2 == x1 + 1:
        return (y1 - 1) * cols + x1, y1 * cols + x1
    raise ValueError("marked edge is not one internal unit wall")


def _opening_cell(edge: list[int], rows: int, cols: int) -> int:
    x1, y1, x2, y2 = _normalized(edge)
    if y1 == y2 == 0 and x2 == x1 + 1:
        return x1
    if y1 == y2 == rows and x2 == x1 + 1:
        return (rows - 1) * cols + x1
    if x1 == x2 == 0 and y2 == y1 + 1:
        return y1 * cols
    if x1 == x2 == cols and y2 == y1 + 1:
        return y1 * cols + cols - 1
    raise ValueError("opening edge is not one perimeter unit wall")


def sample_scene(seed: int) -> dict:
    """Sample paired, class-balanced scenes with matched counts and local target ink."""

    rows, cols = 2, 3
    pair = seed // 2
    rng = random.Random((pair * 0x9E3779B1) ^ 0x51DEFACE)
    marked = rng.choice(_internal_edges(rows, cols))
    adjacent = set(_adjacent_cells(marked, rows, cols))
    positive = seed % 2 == 0
    eligible = []
    for edge in _perimeter_edges(rows, cols):
        cell = _opening_cell(edge, rows, cols)
        if (cell not in adjacent) == positive:
            eligible.append(edge)
    opening = list(rng.choice(eligible))
    if rng.choice((False, True)):
        opening = [opening[2], opening[3], opening[0], opening[1]]

    wall_rgb, paper_rgb, marker_rgb = _PALETTES[pair % len(_PALETTES)]
    return {
        "rows": rows,
        "cols": cols,
        "marked_edge": list(marked),
        "opening_edge": opening,
        "transform": pair % len(_TRANSPOSE),
        "wall_width": (10, 12, 14)[pair % 3],
        "wall_rgb": list(wall_rgb),
        "paper_rgb": list(paper_rgb),
        "marker_rgb": list(marker_rgb),
    }


def _pixel_endpoints(edge: list[int]) -> tuple[int, int, int, int]:
    x1, y1, x2, y2 = edge
    return XS[int(x1)], YS[int(y1)], XS[int(x2)], YS[int(y2)]


def render(scene: dict) -> Image.Image:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    marked = _normalized(scene["marked_edge"])
    opening = _normalized(scene["opening_edge"])
    wall_width = int(scene["wall_width"])
    wall = tuple(int(v) for v in scene["wall_rgb"])
    paper = tuple(int(v) for v in scene["paper_rgb"])
    orange = tuple(int(v) for v in scene["marker_rgb"])
    transform = int(scene["transform"])

    scale = 3
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), paper)
    draw = ImageDraw.Draw(image)
    for edge in _all_edges(rows, cols):
        norm = _normalized(edge)
        if norm in (opening, marked):
            continue
        endpoints = tuple(value * scale for value in _pixel_endpoints(edge))
        draw.line(endpoints, fill=wall, width=wall_width * scale)
    endpoints = tuple(value * scale for value in _pixel_endpoints(list(marked)))
    draw.line(endpoints, fill=orange, width=wall_width * scale)
    image = image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)
    operation = _TRANSPOSE[transform]
    return image if operation is None else image.transpose(operation)


def analytic_gold(scene: dict) -> str:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    adjacent = set(_adjacent_cells(scene["marked_edge"], rows, cols))
    opened = _opening_cell(scene["opening_edge"], rows, cols)
    return "yes" if opened not in adjacent else "no"


def _side_room_count(scene: dict) -> int:
    return 2 if analytic_gold(scene) == "yes" else 1


def margin(scene: dict) -> float:
    """Runner-up gap around the threshold of two enclosed side faces."""

    return float(abs(_side_room_count(scene) - 1.5))


def is_quarantined(scene: dict) -> bool:
    try:
        rows = int(scene["rows"])
        cols = int(scene["cols"])
        adjacent = _adjacent_cells(scene["marked_edge"], rows, cols)
        _opening_cell(scene["opening_edge"], rows, cols)
    except (KeyError, TypeError, ValueError, IndexError):
        return True
    return bool(
        (rows, cols) != (2, 3)
        or len(set(adjacent)) != 2
        or int(scene["wall_width"]) < 10
        or int(scene["transform"]) not in range(8)
        or margin(scene) < 0.5
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    x1, y1, x2, y2 = twin["marked_edge"]
    twin["marked_edge"] = [x2, y2, x1, y1]
    return [("reverse_undirected_marked_edge", twin)]
