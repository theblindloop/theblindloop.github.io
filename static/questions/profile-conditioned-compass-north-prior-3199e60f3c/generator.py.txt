"""Deterministic compass renderer and analytic label function.

The image contains one clearly marked N target and one one-ended triangular
needle.  The target bearing and needle bearing are sampled independently so a
compass convention is not a reliable shortcut.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 540
SUPERSAMPLE = 4
SECTOR_HALF_ANGLE = 16.0
QUARANTINE_MARGIN = 8.0

# These palettes are deliberately answer-independent.  The oracle duplicates
# the public swatches instead of importing this module.
MARKER_SWATCHES = (
    (24, 135, 213),
    (171, 61, 174),
    (228, 123, 24),
    (28, 163, 112),
)
NEEDLE_SWATCHES = (
    (202, 45, 55),
    (19, 86, 166),
    (176, 65, 22),
    (95, 45, 160),
)
BACKGROUND_SWATCHES = (
    (232, 238, 243),
    (247, 247, 241),
    (232, 241, 235),
)
DIAL_SWATCHES = (
    (250, 251, 248),
    (244, 248, 252),
    (252, 247, 239),
)
OUTLINE_SWATCHES = (
    (67, 81, 97),
    (79, 91, 99),
    (63, 101, 88),
)
HUB_SWATCHES = (
    (255, 255, 255),
    (242, 245, 244),
    (250, 247, 241),
)
TICK_SWATCHES = (
    (118, 129, 141),
    (105, 120, 116),
    (132, 126, 119),
)
MARKER_TEXT = (38, 45, 55)


def _as_rgb(value: list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    return int(value[0]), int(value[1]), int(value[2])


def _unit(angle: float) -> tuple[float, float]:
    """Return a screen-space unit vector, with zero degrees pointing up."""

    theta = math.radians(float(angle))
    return math.sin(theta), -math.cos(theta)


def _point(center: tuple[float, float], distance: float, angle: float) -> tuple[float, float]:
    ux, uy = _unit(angle)
    return center[0] + distance * ux, center[1] + distance * uy


def _angular_distance(first: float, second: float) -> float:
    delta = (float(first) - float(second) + 180.0) % 360.0 - 180.0
    return abs(delta)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one presentation-balanced compass scene from an integer seed."""

    rng = random.Random(int(seed))

    radius = round(rng.uniform(142.0, 158.0), 3)
    center = [
        round(rng.uniform(radius + 42.0, WIDTH - radius - 42.0), 3),
        round(rng.uniform(radius + 42.0, HEIGHT - radius - 42.0), 3),
    ]
    background = rng.choice(BACKGROUND_SWATCHES)
    dial_fill = rng.choice(DIAL_SWATCHES)
    dial_outline = rng.choice(OUTLINE_SWATCHES)
    marker_fill = rng.choice(MARKER_SWATCHES)
    needle_fill = rng.choice(NEEDLE_SWATCHES)
    hub_fill = rng.choice(HUB_SWATCHES)
    tick_color = rng.choice(TICK_SWATCHES)
    north_angle = rng.choice((0.0, 90.0, 180.0, 270.0)) + rng.uniform(-5.0, 5.0)

    # The presentation variables above are drawn before the answer-bearing
    # bearing.  The yes/no class is balanced by each generated pair, and the
    # relative bearing is otherwise independent of style, position, and size.
    if rng.random() < 0.5:
        relative = rng.uniform(-4.5, 4.5)
    else:
        relative = rng.choice((90.0, 180.0, 270.0)) + rng.uniform(-4.5, 4.5)
    needle_angle = north_angle + relative

    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "background": list(background),
        "dial_fill": list(dial_fill),
        "dial_outline": list(dial_outline),
        "marker_fill": list(marker_fill),
        "needle_fill": list(needle_fill),
        "hub_fill": list(hub_fill),
        "marker_text_fill": list(MARKER_TEXT),
        "tick_color": list(tick_color),
        "center": center,
        "radius": radius,
        "north_angle": round(north_angle, 3),
        "needle_angle": round(needle_angle, 3),
        "marker_radius": round(radius * rng.uniform(0.80, 0.84), 3),
        "marker_size": round(rng.uniform(27.0, 31.0), 3),
        "ring_width": rng.choice((4, 5)),
        "tick_width": rng.choice((3, 4)),
        "needle_width": round(rng.uniform(14.0, 18.0), 3),
    }


def counterfactual_scene(scene: dict[str, Any]) -> dict[str, Any]:
    """Flip only the needle bearing, preserving the complete presentation."""

    twin = copy.deepcopy(scene)
    north = float(scene["north_angle"])
    if _angular_distance(float(scene["needle_angle"]), north) <= SECTOR_HALF_ANGLE:
        twin["needle_angle"] = round(north + 90.0, 3)
    else:
        twin["needle_angle"] = round(north, 3)
    return twin


def _scaled_points(points: list[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def _draw_n_badge(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    scene: dict[str, Any],
    scale: int,
) -> None:
    marker_center = _point(
        center,
        float(scene["marker_radius"]),
        float(scene["north_angle"]),
    )
    radial = _unit(float(scene["north_angle"]))
    tangent = (-radial[1], radial[0])
    size = float(scene["marker_size"])
    diamond = [
        (marker_center[0] + tangent[0] * size, marker_center[1] + tangent[1] * size),
        (marker_center[0] + radial[0] * size, marker_center[1] + radial[1] * size),
        (marker_center[0] - tangent[0] * size, marker_center[1] - tangent[1] * size),
        (marker_center[0] - radial[0] * size, marker_center[1] - radial[1] * size),
    ]
    draw.polygon(
        _scaled_points(diamond, scale),
        fill=_as_rgb(scene["marker_fill"]),
    )

    # A hand-drawn N avoids any dependency on installed font files.  The
    # letter remains upright so the badge is semantically legible at every
    # rotated bearing.
    x, y = marker_center
    stroke = _as_rgb(scene["marker_text_fill"])
    draw.line(
        _scaled_points(
            [(x - 8.0, y + 9.0), (x - 8.0, y - 9.0), (x + 8.0, y + 9.0), (x + 8.0, y - 9.0)],
            scale,
        ),
        fill=stroke,
        width=3 * scale,
        joint="curve",
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a single-page, antialiased compass image."""

    scale = SUPERSAMPLE
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width * scale, height * scale), _as_rgb(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx, cy = (float(value) for value in scene["center"])
    center = (cx, cy)
    radius = float(scene["radius"])
    bounds = [
        round((cx - radius) * scale),
        round((cy - radius) * scale),
        round((cx + radius) * scale),
        round((cy + radius) * scale),
    ]
    draw.ellipse(
        bounds,
        fill=_as_rgb(scene["dial_fill"]),
        outline=_as_rgb(scene["dial_outline"]),
        width=int(scene["ring_width"]) * scale,
    )

    # Tick marks rotate with the N bearing.  Four long ticks and twelve short
    # ticks provide an orientation cue without encoding the answer.
    tick_fill = _as_rgb(scene["tick_color"])
    for index in range(16):
        angle = float(scene["north_angle"]) + index * 22.5
        long_tick = index % 4 == 0
        inner = radius - (25.0 if long_tick else 18.0)
        outer = radius - 8.0
        first = _point(center, inner, angle)
        second = _point(center, outer, angle)
        draw.line(
            _scaled_points([first, second], scale),
            fill=tick_fill,
            width=int(scene["tick_width"]) * scale,
        )

    _draw_n_badge(draw, center, scene, scale)

    # A filled triangle makes the forward tip directly visible.  The hub is
    # painted last and never participates in the analytic label.
    radial = _unit(float(scene["needle_angle"]))
    tangent = (-radial[1], radial[0])
    needle_base_distance = 14.0
    base_half_width = float(scene["needle_width"]) * 0.5
    base_center = _point(center, needle_base_distance, float(scene["needle_angle"]))
    tip = _point(center, radius * 0.74, float(scene["needle_angle"]))
    triangle = [
        (
            base_center[0] + tangent[0] * base_half_width,
            base_center[1] + tangent[1] * base_half_width,
        ),
        (
            base_center[0] - tangent[0] * base_half_width,
            base_center[1] - tangent[1] * base_half_width,
        ),
        tip,
    ]
    draw.polygon(_scaled_points(triangle, scale), fill=_as_rgb(scene["needle_fill"]))
    hub_radius = 13.0
    draw.ellipse(
        [
            round((cx - hub_radius) * scale),
            round((cy - hub_radius) * scale),
            round((cx + hub_radius) * scale),
            round((cy + hub_radius) * scale),
        ],
        fill=_as_rgb(scene["hub_fill"]),
        outline=_as_rgb(scene["needle_fill"]),
        width=2 * scale,
    )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible angular predicate as a constrained answer string."""

    delta = _angular_distance(scene["needle_angle"], scene["north_angle"])
    return "yes" if delta <= SECTOR_HALF_ANGLE else "no"


def margin(scene: dict[str, Any]) -> float:
    """Return the angular runner-up margin to the visible marker boundary."""

    delta = _angular_distance(scene["needle_angle"], scene["north_angle"])
    return round(abs(delta - SECTOR_HALF_ANGLE), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No render-invariant latent transform is claimed for this scene."""

    return []
