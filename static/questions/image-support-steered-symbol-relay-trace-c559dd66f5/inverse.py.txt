"""Independent pixel-only inverse arm for the two-panel symbol relay."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

NAVY = np.array((29, 53, 87), dtype=np.int16)
BLUE = np.array((35, 102, 224), dtype=np.int16)
VIOLET = np.array((126, 87, 194), dtype=np.int16)
RED = np.array((218, 57, 61), dtype=np.int16)
GOLD = np.array((232, 166, 35), dtype=np.int16)


class OracleAbstention(ValueError):
    pass


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 52.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int64) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        stack = [(y0, x0)]
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            components.append(np.asarray(points, dtype=np.int32))
    return components


def _describe(points: np.ndarray) -> dict:
    ys = points[:, 0]
    xs = points[:, 1]
    return {
        "points": points,
        "cx": float(xs.mean()),
        "cy": float(ys.mean()),
        "width": int(xs.max() - xs.min() + 1),
        "height": int(ys.max() - ys.min() + 1),
        "fill": float(len(points) / ((xs.max() - xs.min() + 1) * (ys.max() - ys.min() + 1))),
    }


def _distance(component: dict, marker: dict) -> float:
    points = component["points"]
    dx = points[:, 1].astype(np.float64) - marker["cx"]
    dy = points[:, 0].astype(np.float64) - marker["cy"]
    return float(np.sqrt(np.min(dx * dx + dy * dy)))


def _one_marker(array: np.ndarray, color: np.ndarray, name: str) -> dict:
    parts = [_describe(p) for p in _components(_mask(array, color), 80)]
    if len(parts) != 1:
        raise OracleAbstention(f"expected one intact {name} marker, found {len(parts)}")
    return parts[0]


def _shape(marker: dict) -> str:
    if marker["width"] < 35 or marker["height"] < 35:
        raise OracleAbstention("violet symbol is too damaged to classify")
    return "square" if marker["fill"] > 0.88 else "circle"


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer from RGB pixels with no latent-scene input."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if width < 600 or height < 300:
        raise OracleAbstention("canvas is too small")

    navy = [_describe(p) for p in _components(_mask(array, NAVY), 240)]
    violet = [_describe(p) for p in _components(_mask(array, VIOLET), 250)]
    if len(navy) < 3 or len(violet) < 2:
        raise OracleAbstention("insufficient intact route or shape components")

    blue = _one_marker(array, BLUE, "blue")
    left_route = min(navy, key=lambda part: _distance(part, blue))
    if _distance(left_route, blue) > 32.0:
        raise OracleAbstention("blue ring is not attached to a navy route")

    left_shapes = [part for part in violet if part["cx"] < width / 2]
    if not left_shapes:
        raise OracleAbstention("left endpoint symbol is missing")
    key_marker = min(left_shapes, key=lambda part: _distance(left_route, part))
    if _distance(left_route, key_marker) > 32.0:
        raise OracleAbstention("blue route does not reach an intact violet symbol")
    key_shape = _shape(key_marker)

    right_shapes = [part for part in violet if part["cx"] > width / 2 and _shape(part) == key_shape]
    if len(right_shapes) != 1:
        raise OracleAbstention(f"expected one right {key_shape} symbol, found {len(right_shapes)}")
    selected_shape = right_shapes[0]
    right_route = min(navy, key=lambda part: _distance(part, selected_shape))
    if _distance(right_route, selected_shape) > 32.0:
        raise OracleAbstention("matching right symbol is not attached to a navy route")

    terminals: list[tuple[str, dict]] = []
    for name, color in (("red", RED), ("gold", GOLD)):
        parts = [_describe(p) for p in _components(_mask(array, color), 80)]
        terminals.extend((name, part) for part in parts)
    if not terminals:
        raise OracleAbstention("no terminal ring is visible")
    ranked = sorted(((_distance(right_route, part), name) for name, part in terminals))
    if ranked[0][0] > 32.0:
        raise OracleAbstention("selected right route reaches no terminal")
    if len(ranked) > 1 and ranked[1][0] - ranked[0][0] < 18.0:
        raise OracleAbstention("terminal ownership is ambiguous")
    return ranked[0][1]
