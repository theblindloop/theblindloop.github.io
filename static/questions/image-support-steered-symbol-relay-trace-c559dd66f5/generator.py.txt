"""Deterministic renderer and analytic arm for the symbol relay world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 1200
HEIGHT = 600
SS = 2

WHITE = (250, 250, 247)
NAVY = (29, 53, 87)
BLUE = (35, 102, 224)
VIOLET = (126, 87, 194)
RED = (218, 57, 61)
GOLD = (232, 166, 35)
PALE = (205, 211, 219)
FRAME = (174, 183, 194)

SHAPES = ("circle", "square")
COLORS = ("red", "gold")


def sample_scene(seed: int) -> dict:
    """Sample a counterbalanced, well-separated pair of route panels."""
    rng = random.Random(int(seed) ^ 0x51A7B0)
    desired = COLORS[int(seed) & 1]
    key_shape = SHAPES[(int(seed) // 2) & 1]

    left_shape_order = list(SHAPES)
    if (int(seed) // 4) & 1:
        left_shape_order.reverse()
    right_shape_order = list(SHAPES)
    if (int(seed) // 8) & 1:
        right_shape_order.reverse()

    selected_row = right_shape_order.index(key_shape)
    terminal_order = [None, None]
    terminal_order[selected_row] = desired
    terminal_order[1 - selected_row] = COLORS[1 - COLORS.index(desired)]

    return {
        "key_shape": key_shape,
        "left_shape_order": left_shape_order,
        "right_shape_order": right_shape_order,
        "terminal_order": terminal_order,
        "left_variant": rng.randrange(4),
        "right_variants": [rng.randrange(4), rng.randrange(4)],
        "left_shift": rng.choice([-8, -4, 0, 4, 8]),
        "right_shift": rng.choice([-8, -4, 0, 4, 8]),
    }


def _s(point: tuple[float, float]) -> tuple[int, int]:
    return (round(point[0] * SS), round(point[1] * SS))


def _line(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], color: tuple[int, int, int], width: int) -> None:
    draw.line([_s(p) for p in points], fill=color, width=width * SS, joint="curve")
    radius = width * SS // 2
    for x, y in [_s(p) for p in points[1:-1]]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _ring(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: int = 23) -> None:
    x, y = _s(center)
    r = radius * SS
    draw.ellipse((x - r, y - r, x + r, y + r), fill=color)
    inner = (radius - 9) * SS
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=WHITE)


def _symbol(draw: ImageDraw.ImageDraw, center: tuple[float, float], shape: str, radius: int = 23) -> None:
    x, y = _s(center)
    r = radius * SS
    if shape == "circle":
        draw.ellipse((x - r, y - r, x + r, y + r), fill=VIOLET)
    elif shape == "square":
        draw.rounded_rectangle((x - r, y - r, x + r, y + r), radius=3 * SS, fill=VIOLET)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def _dashes(draw: ImageDraw.ImageDraw) -> None:
    for x in range(115, 335, 42):
        _line(draw, [(x, 82), (x + 20, 82)], PALE, 5)
    for x in range(845, 1090, 44):
        _line(draw, [(x, 300), (x + 22, 300)], PALE, 5)
    for y in range(120, 500, 55):
        _line(draw, [(680, y), (680, y + 24)], PALE, 5)


def _left_selected_points(target_y: int, variant: int, shift: int) -> list[tuple[float, float]]:
    mid = 300 + shift
    offsets = (-42, -24, 24, 42)
    bend = offsets[variant]
    approach = target_y - 38 if target_y < mid else target_y + 38
    return [(72, mid), (135, mid), (135, mid + bend), (235, mid + bend), (235, approach), (310, approach), (310, target_y), (380, target_y)]


def _left_decoy_points(other_y: int, variant: int, shift: int) -> list[tuple[float, float]]:
    direction = 1 if other_y < 300 + shift else -1
    jog = 22 + 7 * variant
    return [(205, other_y + direction * 62), (265, other_y + direction * 62), (265, other_y + direction * jog), (330, other_y + direction * jog), (330, other_y), (380, other_y)]


def _right_points(row_y: int, variant: int, shift: int) -> list[tuple[float, float]]:
    y = row_y + shift
    sign = -1 if row_y < 300 else 1
    levels = (20, 31, 42, 53)
    jog = sign * levels[variant]
    return [(820, y), (875, y), (875, y + jog), (955, y + jog), (955, y - jog), (1035, y - jog), (1035, y), (1125, y)]


def render(scene: dict) -> Image.Image:
    """Render only from the public latent scene dictionary."""
    key_shape = scene["key_shape"]
    left_order = scene["left_shape_order"]
    right_order = scene["right_shape_order"]
    terminal_order = scene["terminal_order"]
    left_variant = int(scene["left_variant"])
    right_variants = scene["right_variants"]
    left_shift = int(scene["left_shift"])
    right_shift = int(scene["right_shift"])

    image = Image.new("RGB", (WIDTH * SS, HEIGHT * SS), WHITE)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((_s((24, 35)), _s((438, 565))), radius=18 * SS, outline=FRAME, width=3 * SS)
    draw.rounded_rectangle((_s((762, 35)), _s((1176, 565))), radius=18 * SS, outline=FRAME, width=3 * SS)
    _dashes(draw)

    left_rows = [158 + left_shift, 442 + left_shift]
    target_row = left_order.index(key_shape)
    other_row = 1 - target_row
    _line(draw, _left_selected_points(left_rows[target_row], left_variant, left_shift), NAVY, 14)
    _line(draw, _left_decoy_points(left_rows[other_row], (left_variant + 1) % 4, left_shift), NAVY, 14)

    right_rows = [158, 442]
    for row in (0, 1):
        _line(draw, _right_points(right_rows[row], int(right_variants[row]), right_shift), NAVY, 14)

    _ring(draw, (72, 300 + left_shift), BLUE)
    for row, shape in enumerate(left_order):
        _symbol(draw, (380, left_rows[row]), shape)
    for row, shape in enumerate(right_order):
        _symbol(draw, (820, right_rows[row] + right_shift), shape)
    for row, color_name in enumerate(terminal_order):
        color = RED if color_name == "red" else GOLD
        _ring(draw, (1125, right_rows[row] + right_shift), color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    key_shape = scene["key_shape"]
    selected_row = scene["right_shape_order"].index(key_shape)
    return scene["terminal_order"][selected_row]


def margin(scene: dict) -> float:
    variants = [int(scene["left_variant"]), *[int(v) for v in scene["right_variants"]]]
    return float(42 - max(variants) * 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
