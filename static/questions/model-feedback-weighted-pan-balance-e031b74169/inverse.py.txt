"""Independent PNG-only inverse arm for the weighted-pan balance world.

Recovers, from the raster alone, the per-colour weights (number of dark dots
beside each key swatch), counts the coloured disks in the left and right boxes,
forms the two weighted sums, and returns which box has the larger total.  Uses
no renderer geometry, scene parameters, prompt, or gold code; only chromatic
masks, connected components, and centroid grouping.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RGB = {
    "crimson": (204, 44, 64),
    "teal": (0, 140, 150),
    "amber": (232, 158, 32),
}
MID_X = 390
MIN_SWATCH = 150
MIN_DISK = 500
MIN_DOT = 40


def _color_mask(arr: np.ndarray, color: tuple, tol: int = 40) -> np.ndarray:
    r, g, b = color
    return (
        (np.abs(arr[:, :, 0] - r) <= tol)
        & (np.abs(arr[:, :, 1] - g) <= tol)
        & (np.abs(arr[:, :, 2] - b) <= tol)
    )


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    mean = arr.mean(axis=2).astype(np.int16)
    return (chroma <= 20) & (mean < 130)


def _components(mask: np.ndarray, minsize: int) -> list[dict]:
    pixels = set(zip(*np.nonzero(mask)))
    found = []
    while pixels:
        start = pixels.pop()
        stack = [start]
        cells = []
        while stack:
            y, x = stack.pop()
            cells.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    stack.append((ny, nx))
        if len(cells) >= minsize:
            cy = sum(y for y, _ in cells) / len(cells)
            cx = sum(x for _, x in cells) / len(cells)
            found.append({"cx": cx, "cy": cy, "size": len(cells)})
    return found


def _read_weights(arr: np.ndarray) -> dict:
    weights = {}
    band_y0, band_y1 = 40, 200
    for color in RGB:
        mask = _color_mask(arr, RGB[color])
        mask[:band_y0, :] = False
        mask[band_y1:, :] = False
        comps = _components(mask, MIN_SWATCH)
        if not comps:
            raise ValueError(f"missing key swatch for {color}")
        comps.sort(key=lambda c: c["cx"])
        swatch = comps[0]
        cx = swatch["cx"]
        dark = _dark_mask(arr)
        window = np.zeros_like(dark)
        window[80:160, max(0, int(cx) + 25): int(cx) + 118] = dark[
            80:160, max(0, int(cx) + 25): int(cx) + 118
        ]
        dots = _components(window, MIN_DOT)
        w = len(dots)
        if w < 1 or w > 3:
            raise ValueError(f"invalid weight {w} for {color}")
        weights[color] = w
    return weights


def _count_disks(arr: np.ndarray, weights: dict) -> tuple[int, int]:
    total_a = 0
    total_b = 0
    for color, w in weights.items():
        mask = _color_mask(arr, RGB[color])
        comps = _components(mask, MIN_DISK)
        for comp in comps:
            if comp["cy"] < 240:
                continue
            if comp["cx"] < MID_X:
                total_a += w
            else:
                total_b += w
    return total_a, total_b


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    weights = _read_weights(arr)
    total_a, total_b = _count_disks(arr, weights)
    if total_a == total_b:
        raise ValueError("weighted totals are indistinguishable")
    return "A" if total_a > total_b else "B"
