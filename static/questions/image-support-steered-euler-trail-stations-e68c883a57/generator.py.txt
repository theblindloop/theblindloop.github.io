"""Renderer and analytic arm for the one-stroke trail world."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 640
ROAD_COUNT = 10
NODE_COUNT = 8

# All templates are connected planar subgraphs of the same visible 2x4 station
# arrangement.  Each has ten roads and exactly two odd-degree stations.
TEMPLATES = (
    ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6), (6, 7), (0, 4), (1, 5), (3, 7), (0, 5)),
    ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6), (6, 7), (0, 4), (1, 5), (3, 7), (1, 6)),
    ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6), (6, 7), (0, 4), (2, 6), (3, 7), (1, 6)),
    ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6), (6, 7), (0, 4), (2, 6), (3, 7), (2, 7)),
    ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6), (0, 4), (1, 5), (2, 6), (3, 7), (2, 7)),
    ((0, 1), (2, 3), (4, 5), (5, 6), (6, 7), (0, 4), (1, 5), (2, 6), (3, 7), (1, 6)),
)

PALETTES = (
    {"plain": "#f4c95d", "start": "#e56b2f", "finish": "#2474b5", "road": "#202830", "background": "#faf9f4"},
    {"plain": "#e8c55b", "start": "#d95f32", "finish": "#2b79b8", "road": "#202830", "background": "#faf9f4"},
    {"plain": "#f0c34e", "start": "#df6935", "finish": "#216fae", "road": "#202830", "background": "#faf9f4"},
)


def _degrees(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> list[int]:
    values = [0] * NODE_COUNT
    for a, b in edges:
        values[a] += 1
        values[b] += 1
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    edges = [list(edge) for edge in TEMPLATES[rng.randrange(len(TEMPLATES))]]
    base_x = [174, 314, 454, 594]
    # Keep the connected graph in one deliberately elongated regional strip.
    # This makes multiple interior road portions—not opposite image edges—the
    # decisive visual support.
    y_top, y_bottom = 285, 355
    dx = rng.randint(-24, 24)
    nodes = []
    for row in range(2):
        for col in range(4):
            x = base_x[col] + dx + rng.randint(-7, 7)
            y = (y_top if row == 0 else y_bottom) + rng.randint(-3, 3)
            nodes.append([x, y])

    # Reflecting the complete embedding diversifies position while retaining
    # the explicit node identities used by the graph.
    if rng.random() < 0.5:
        nodes = [[WIDTH - x, y] for x, y in nodes]
    if rng.random() < 0.5:
        nodes = [[x, HEIGHT - y] for x, y in nodes]

    odd = [index for index, degree in enumerate(_degrees(edges)) if degree % 2]
    want_yes = (int(seed) % 2 == 0)
    if want_yes:
        start, finish = odd
    else:
        pairs = [
            (a, b)
            for a in range(NODE_COUNT)
            for b in range(a + 1, NODE_COUNT)
            if {a, b} != set(odd)
        ]
        start, finish = pairs[rng.randrange(len(pairs))]
    if rng.random() < 0.5:
        start, finish = finish, start
    return {
        "nodes": nodes,
        "edges": edges,
        "start": start,
        "finish": finish,
        "palette": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    nodes = [tuple(point) for point in scene["nodes"]]
    edges = scene["edges"]
    start = int(scene["start"])
    finish = int(scene["finish"])
    colors = PALETTES[int(scene["palette"])]
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), colors["background"])
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        p, q = nodes[int(a)], nodes[int(b)]
        draw.line(
            [(p[0] * scale, p[1] * scale), (q[0] * scale, q[1] * scale)],
            fill=colors["road"],
            width=4 * scale,
        )
    radius = 23 * scale
    for index, (x, y) in enumerate(nodes):
        color = colors["start"] if index == start else colors["finish"] if index == finish else colors["plain"]
        box = (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius)
        draw.ellipse(box, fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    # Access to nodes is intentional: the declared public world requires eight
    # visible stations, and malformed latent graphs are not silently labelled.
    if len(scene["nodes"]) != NODE_COUNT:
        raise ValueError("scene must contain eight stations")
    edges = scene["edges"]
    start = int(scene["start"])
    finish = int(scene["finish"])
    adjacency = [set() for _ in range(NODE_COUNT)]
    for a, b in edges:
        adjacency[int(a)].add(int(b))
        adjacency[int(b)].add(int(a))
    seen = {0}
    stack = [0]
    while stack:
        for neighbor in adjacency[stack.pop()]:
            if neighbor not in seen:
                seen.add(neighbor)
                stack.append(neighbor)
    odd = {index for index, neighbors in enumerate(adjacency) if len(neighbors) % 2}
    return "yes" if len(seen) == NODE_COUNT and odd == {start, finish} else "no"


def margin(scene: dict) -> float:
    points = [tuple(point) for point in scene["nodes"]]
    edge_clearance = min(math.dist(points[a], points[b]) for a, b in scene["edges"]) - 70.0
    mismatch = 2.0 if analytic_gold(scene) == "yes" else 1.0
    return round(mismatch + max(0.0, edge_clearance) / 100.0, 6)


def is_quarantined(scene: dict) -> bool:
    points = [tuple(point) for point in scene["nodes"]]
    if len(points) != NODE_COUNT or len(scene["edges"]) != ROAD_COUNT:
        return True
    if min(math.dist(points[a], points[b]) for a, b in scene["edges"]) < 55.0:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
