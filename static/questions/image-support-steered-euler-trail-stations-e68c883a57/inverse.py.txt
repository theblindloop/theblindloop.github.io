"""Independent pixel-only inverse arm for the trail network."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

NODE_COUNT = 8
ROAD_COUNT = 10
ROAD = np.array((32, 40, 48), dtype=np.int16)
PLAIN = tuple(np.array(color, dtype=np.int16) for color in ((244, 201, 93), (232, 197, 91), (240, 195, 78)))
START = tuple(np.array(color, dtype=np.int16) for color in ((229, 107, 47), (217, 95, 50), (223, 105, 53)))
FINISH = tuple(np.array(color, dtype=np.int16) for color in ((36, 116, 181), (43, 121, 184), (33, 111, 174)))


def _near(arr: np.ndarray, colors: tuple[np.ndarray, ...], tolerance: int = 34) -> np.ndarray:
    return np.logical_or.reduce([np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance for color in colors])


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = [(y, x)]
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centroid(component: np.ndarray) -> np.ndarray:
    return component.mean(axis=0)[::-1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    plain_mask = _near(arr, PLAIN)
    start_mask = _near(arr, START)
    finish_mask = _near(arr, FINISH)
    node_components = _components(plain_mask | start_mask | finish_mask, 220)
    if len(node_components) != NODE_COUNT:
        raise ValueError(f"expected {NODE_COUNT} visible stations, found {len(node_components)}")
    centers = np.asarray([_centroid(component) for component in node_components])

    start_components = _components(start_mask, 180)
    finish_components = _components(finish_mask, 180)
    if len(start_components) != 1 or len(finish_components) != 1:
        raise ValueError("start and finish stations must each be uniquely visible")
    start_center = _centroid(start_components[0])
    finish_center = _centroid(finish_components[0])
    start = int(np.argmin(np.linalg.norm(centers - start_center, axis=1)))
    finish = int(np.argmin(np.linalg.norm(centers - finish_center, axis=1)))
    if start == finish:
        raise ValueError("marked stations merged")

    road_mask = np.max(np.abs(arr.astype(np.int16) - ROAD), axis=2) <= 40
    road_components = _components(road_mask, 50)
    if len(road_components) != ROAD_COUNT:
        raise ValueError(f"expected {ROAD_COUNT} complete visible roads, found {len(road_components)}")

    edges = set()
    for component in road_components:
        # A complete road approaches exactly two station centers.  Use
        # component-to-center distance and demand a clear two-endpoint attachment.
        xy = component[:, ::-1]
        distances = np.sqrt(((xy[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2)).min(axis=0)
        attached = np.flatnonzero(distances < 25.0)
        if len(attached) != 2:
            raise ValueError("a road does not join exactly two visible stations")
        a, b = sorted((int(attached[0]), int(attached[1])))
        if a == b or (a, b) in edges:
            raise ValueError("duplicate or self-loop road")
        edges.add((a, b))

    adjacency = [set() for _ in range(NODE_COUNT)]
    for a, b in edges:
        adjacency[a].add(b)
        adjacency[b].add(a)
    seen = {0}
    queue = deque([0])
    while queue:
        for neighbor in adjacency[queue.popleft()]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    if len(seen) != NODE_COUNT:
        raise ValueError("road network is visibly disconnected")
    odd = {index for index, neighbors in enumerate(adjacency) if len(neighbors) % 2}
    return "yes" if odd == {start, finish} else "no"
