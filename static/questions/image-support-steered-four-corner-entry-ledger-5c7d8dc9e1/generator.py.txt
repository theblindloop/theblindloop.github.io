"""Deterministic latent sampler and renderer for the four-corner entry ledger."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
COLORS = ("cyan", "amber", "rose")

_BASE_CORNERS = ((150, 150), (650, 150), (150, 650), (650, 650))
_PALETTES = (
    {
        "cyan": (24, 169, 190),
        "amber": (235, 157, 36),
        "rose": (217, 72, 122),
        "region": (217, 235, 242),
        "card": (251, 248, 240),
    },
    {
        "cyan": (34, 151, 181),
        "amber": (218, 139, 27),
        "rose": (196, 61, 113),
        "region": (207, 229, 239),
        "card": (248, 246, 238),
    },
)
_RANK_INK = (103, 66, 153)
_OUTLINE = (48, 55, 64)
_BACKGROUND = (255, 255, 255)
_REGION_RADIUS = 62
_TOKEN_RADIUS = 11

# All non-winner histories have exactly one inward transition. The winner has
# two, so the top-two count margin is always one while the visual arrangements
# remain varied.
_RUNNER_HISTORIES = (
    (False, False, False, True),
    (False, False, True, True),
    (False, True, True, True),
    (False, True, True, False),
    (True, False, False, True),
    (True, False, True, True),
    (True, False, True, False),
    (True, True, False, True),
)


def _entry_counts(histories: dict[str, list[bool]]) -> dict[str, int]:
    return {
        color: sum((not states[index - 1]) and states[index] for index in range(1, 4))
        for color, states in histories.items()
    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, unambiguous scene from an integer seed."""

    rng = random.Random(int(seed))
    winner = COLORS[int(seed) % len(COLORS)]
    histories: dict[str, list[bool]] = {winner: [False, True, False, True]}
    pool = list(_RUNNER_HISTORIES)
    rng.shuffle(pool)
    runner_index = 0
    for color in COLORS:
        if color != winner:
            histories[color] = list(pool[runner_index])
            runner_index += 1

    corner_for_rank = list(range(4))
    rng.shuffle(corner_for_rank)
    slot_orders = []
    for _ in range(4):
        order = list(COLORS)
        rng.shuffle(order)
        slot_orders.append(order)
    corner_jitter = [[rng.randint(-12, 12), rng.randint(-12, 12)] for _ in range(4)]

    return {
        "histories": histories,
        "corner_for_rank": corner_for_rank,
        "slot_orders": slot_orders,
        "corner_jitter": corner_jitter,
        "style_variant": int(seed) % len(_PALETTES),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique color with the largest number of inward transitions."""

    counts = _entry_counts(scene["histories"])
    best = max(counts.values())
    winners = [color for color in COLORS if counts[color] == best]
    if len(winners) != 1:
        raise ValueError("entry-count winner is not unique")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    counts = sorted(_entry_counts(scene["histories"]).values(), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    histories = scene.get("histories", {})
    if set(histories) != set(COLORS) or any(len(value) != 4 for value in histories.values()):
        return True
    return margin(scene) < 1.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No pixel-identical, label-relevant transform is declared."""

    return []


def _ellipse(draw: ImageDraw.ImageDraw, cx: int, cy: int, radius: int, **kwargs: Any) -> None:
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), **kwargs)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render four disconnected snapshot cards on a large white canvas."""

    histories = scene["histories"]
    corner_for_rank = scene["corner_for_rank"]
    slot_orders = scene["slot_orders"]
    corner_jitter = scene["corner_jitter"]
    palette = _PALETTES[int(scene["style_variant"])]

    if sorted(corner_for_rank) != [0, 1, 2, 3]:
        raise ValueError("corner_for_rank must be a permutation")
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)

    inside_slots = ((-28, -14), (28, -14), (0, 31))
    outside_slots = ((-85, 0), (85, 0), (0, 87))

    for rank_index in range(4):
        physical_corner = int(corner_for_rank[rank_index])
        base_x, base_y = _BASE_CORNERS[physical_corner]
        jitter_x, jitter_y = corner_jitter[physical_corner]
        card_x = int(base_x + jitter_x)
        card_y = int(base_y + jitter_y)
        draw.rounded_rectangle(
            (card_x - 112, card_y - 112, card_x + 112, card_y + 112),
            radius=16,
            fill=palette["card"],
            outline=_OUTLINE,
            width=4,
        )

        region_x, region_y = card_x, card_y + 12
        _ellipse(
            draw,
            region_x,
            region_y,
            _REGION_RADIUS,
            fill=palette["region"],
            outline=_OUTLINE,
            width=4,
        )

        rank = rank_index + 1
        pip_y = card_y - 82
        pip_spacing = 19
        pip_start = card_x - ((rank - 1) * pip_spacing) // 2
        for pip_index in range(rank):
            _ellipse(draw, pip_start + pip_index * pip_spacing, pip_y, 6, fill=_RANK_INK)

        order = list(slot_orders[rank_index])
        if sorted(order) != sorted(COLORS):
            raise ValueError("each slot order must contain every color")
        for state, positions in ((True, inside_slots), (False, outside_slots)):
            group = [color for color in order if bool(histories[color][rank_index]) is state]
            for slot_index, color in enumerate(group):
                dx, dy = positions[slot_index]
                token_x, token_y = region_x + dx, region_y + dy
                _ellipse(draw, token_x, token_y, _TOKEN_RADIUS + 2, fill=_OUTLINE)
                _ellipse(draw, token_x, token_y, _TOKEN_RADIUS, fill=palette[color])

    return image
