"""Independent PNG-only inverse arm for the four-corner entry ledger.

The implementation knows only the public visual vocabulary: two rows by two
columns of detached cards, purple pip counts for temporal rank, three named
token colors, and pale-blue circular regions. It receives neither latent data
nor analytic labels.
"""

from __future__ import annotations

import math
from typing import Any

import numpy as np
from PIL import Image

_PROTOTYPES: dict[str, tuple[tuple[int, int, int], ...]] = {
    "cyan": ((24, 169, 190), (34, 151, 181)),
    "amber": ((235, 157, 36), (218, 139, 27)),
    "rose": ((217, 72, 122), (196, 61, 113)),
    "region": ((217, 235, 242), (207, 229, 239)),
    "rank": ((103, 66, 153),),
}
_COLORS = ("cyan", "amber", "rose")


class OracleAbstention(ValueError):
    """Raised when the raster no longer admits one unambiguous replay."""


def _mask(array: np.ndarray, name: str, tolerance: float = 8.0) -> np.ndarray:
    pixels = array.astype(np.int32)
    distances = []
    for prototype in _PROTOTYPES[name]:
        target = np.asarray(prototype, dtype=np.int32)
        distances.append(np.sum((pixels - target) ** 2, axis=2))
    return np.minimum.reduce(distances) <= tolerance * tolerance


def _runs(active: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(active)
    if len(indices) == 0:
        return []
    output: list[tuple[int, int]] = []
    start = previous = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value != previous + 1:
            output.append((start, previous + 1))
            start = value
        previous = value
    output.append((start, previous + 1))
    return output


def _region_centers(region_mask: np.ndarray) -> list[tuple[float, float]]:
    x_runs = _runs(region_mask.any(axis=0))
    y_runs = _runs(region_mask.any(axis=1))
    if len(x_runs) != 2 or len(y_runs) != 2:
        raise OracleAbstention(
            f"expected two separated region columns and rows, found {len(x_runs)} and {len(y_runs)}"
        )
    centers: list[tuple[float, float]] = []
    for y0, y1 in y_runs:
        for x0, x1 in x_runs:
            local = region_mask[y0:y1, x0:x1]
            ys, xs = np.nonzero(local)
            area = len(xs)
            if not 8000 <= area <= 11200:
                raise OracleAbstention(f"damaged or missing pale-blue region (area={area})")
            width = int(xs.max() - xs.min() + 1)
            height = int(ys.max() - ys.min() + 1)
            if not 112 <= width <= 121 or not 112 <= height <= 121:
                raise OracleAbstention(f"region boundary is incomplete ({width}x{height})")
            cx = x0 + (float(xs.min()) + float(xs.max())) / 2.0
            cy = y0 + (float(ys.min()) + float(ys.max())) / 2.0
            centers.append((cx, cy))
    if len(centers) != 4:
        raise OracleAbstention("four regions were not recovered")
    return centers


def _rank_for_panel(rank_mask: np.ndarray, cx: float, cy: float) -> int:
    height, width = rank_mask.shape
    x0, x1 = max(0, int(cx - 104)), min(width, int(cx + 105))
    y0, y1 = max(0, int(cy - 108)), min(height, int(cy - 68))
    area = int(rank_mask[y0:y1, x0:x1].sum())
    rank = int(round(area / 129.0))
    if rank not in (1, 2, 3, 4) or abs(area - 129 * rank) > 8:
        raise OracleAbstention(f"rank pips are ambiguous (area={area})")
    return rank


def _token_state(
    token_mask: np.ndarray, cx: float, cy: float, *, color: str
) -> bool:
    height, width = token_mask.shape
    x0, x1 = max(0, int(cx - 104)), min(width, int(cx + 105))
    y0, y1 = max(0, int(cy - 76)), min(height, int(cy + 106))
    local = token_mask[y0:y1, x0:x1]
    ys, xs = np.nonzero(local)
    area = len(xs)
    if not 400 <= area <= 425:
        raise OracleAbstention(f"{color} token is missing or damaged (area={area})")
    tx = x0 + float(xs.mean())
    ty = y0 + float(ys.mean())
    radial = math.hypot(tx - cx, ty - cy)
    if radial <= 48.0:
        return True
    if radial >= 72.0:
        return False
    raise OracleAbstention(f"{color} token center is too near the region rim")


def decision_from_image(image: Image.Image) -> str:
    """Recover panel order, replay membership changes, and return the winner."""

    if image.size[0] < 300 or image.size[1] < 300:
        raise OracleAbstention("image is too small")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    region_mask = _mask(array, "region")
    rank_mask = _mask(array, "rank")
    token_masks = {color: _mask(array, color) for color in _COLORS}

    panels: dict[int, dict[str, bool]] = {}
    for cx, cy in _region_centers(region_mask):
        rank = _rank_for_panel(rank_mask, cx, cy)
        if rank in panels:
            raise OracleAbstention(f"duplicate panel rank {rank}")
        panels[rank] = {
            color: _token_state(token_masks[color], cx, cy, color=color) for color in _COLORS
        }
    if set(panels) != {1, 2, 3, 4}:
        raise OracleAbstention(f"panel ranks are incomplete: {sorted(panels)}")

    entry_counts: dict[str, int] = {}
    for color in _COLORS:
        states = [panels[rank][color] for rank in (1, 2, 3, 4)]
        entry_counts[color] = sum(
            (not states[index - 1]) and states[index] for index in range(1, 4)
        )
    best = max(entry_counts.values())
    winners = [color for color in _COLORS if entry_counts[color] == best]
    if len(winners) != 1:
        raise OracleAbstention(f"entry-count winner is tied: {entry_counts}")
    return winners[0]
