"""Deterministic renderer and analytic arm for enclosed-cell search.

The semantic decision is discrete topology.  In each 6x6 mosaic, white cells
are connected only through shared sides.  A white cell is enclosed when no
such path reaches a mosaic edge.  Every sampled panel has exactly sixteen dark
cells; exactly one panel has one enclosed white cell and the others have none.
"""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw, ImageFont

GRID = 6
LOCAL_GRID = 5
DARK_COUNT = 16
LABELS = ("A", "B", "C", "D")

CANVAS = (960, 300)
BACKGROUND = (247, 244, 237)
PANEL_WHITE = (255, 255, 252)
DARK = (31, 41, 55)
FRAME = (70, 105, 140)
ACCENT = (204, 105, 70)
MUTED = (95, 101, 108)


def _neighbors(cell: tuple[int, int], size: int) -> list[tuple[int, int]]:
    row, col = cell
    return [
        (nr, nc)
        for nr, nc in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1))
        if 0 <= nr < size and 0 <= nc < size
    ]


def _enclosed_cells(cells: list[list[int]], size: int = GRID) -> list[tuple[int, int]]:
    dark = {tuple(value) for value in cells}
    white = {(row, col) for row in range(size) for col in range(size)} - dark
    reached: set[tuple[int, int]] = set()
    queue: deque[tuple[int, int]] = deque(
        sorted(
            cell
            for cell in white
            if cell[0] in (0, size - 1) or cell[1] in (0, size - 1)
        )
    )
    reached.update(queue)
    while queue:
        cell = queue.popleft()
        for neighbor in _neighbors(cell, size):
            if neighbor in white and neighbor not in reached:
                reached.add(neighbor)
                queue.append(neighbor)
    return sorted(white - reached)


def _is_dark_connected(cells: list[list[int]]) -> bool:
    dark = {tuple(value) for value in cells}
    if not dark:
        return False
    reached = {next(iter(dark))}
    queue = deque(reached)
    while queue:
        for neighbor in _neighbors(queue.popleft(), GRID):
            if neighbor in dark and neighbor not in reached:
                reached.add(neighbor)
                queue.append(neighbor)
    return reached == dark


def _bbox(cells: set[tuple[int, int]]) -> tuple[int, int, int, int]:
    rows = [cell[0] for cell in cells]
    cols = [cell[1] for cell in cells]
    return min(rows), min(cols), max(rows), max(cols)


def _local_holes(cells: set[tuple[int, int]]) -> list[tuple[int, int]]:
    return _enclosed_cells([list(cell) for cell in sorted(cells)], LOCAL_GRID)


def _grow_local_pattern(rng: random.Random, enclosed: bool) -> set[tuple[int, int]]:
    """Grow a count-balanced connected near-ring on a 5x5 support."""

    ring = {
        (row, col)
        for row in range(1, 4)
        for col in range(1, 4)
        if (row, col) != (2, 2)
    }
    cardinal = [(1, 2), (2, 3), (3, 2), (2, 1)]
    for _ in range(3000):
        cells = set(ring)
        forbidden = {(2, 2)}
        if not enclosed:
            gate = rng.choice(cardinal)
            cells.remove(gate)
            forbidden.add(gate)
        while len(cells) < DARK_COUNT:
            frontier = sorted(
                {
                    neighbor
                    for cell in cells
                    for neighbor in _neighbors(cell, LOCAL_GRID)
                    if neighbor not in cells and neighbor not in forbidden
                }
            )
            if not frontier:
                break
            # Prefer expanding the support before filling its interior.  The
            # small random term keeps the sampled silhouettes varied.
            extremes = [
                cell
                for cell in frontier
                if cell[0] in (0, LOCAL_GRID - 1) or cell[1] in (0, LOCAL_GRID - 1)
            ]
            pool = extremes if extremes and rng.random() < 0.58 else frontier
            cells.add(rng.choice(pool))
        if len(cells) != DARK_COUNT or _bbox(cells) != (0, 0, 4, 4):
            continue
        holes = _local_holes(cells)
        if enclosed and holes == [(2, 2)]:
            return cells
        if not enclosed and not holes:
            return cells
    raise RuntimeError("could not sample a balanced near-ring")


def _dihedral(cell: tuple[int, int], transform: int) -> tuple[int, int]:
    row, col = cell
    if transform >= 4:
        col = LOCAL_GRID - 1 - col
    for _ in range(transform % 4):
        row, col = col, LOCAL_GRID - 1 - row
    return row, col


def _place_local(
    rng: random.Random, cells: set[tuple[int, int]]
) -> list[list[int]]:
    transform = rng.randrange(8)
    row_offset = rng.randrange(2)
    col_offset = rng.randrange(2)
    placed = [
        [row + row_offset, col + col_offset]
        for row, col in (_dihedral(cell, transform) for cell in cells)
    ]
    rng.shuffle(placed)  # coordinate order is an explicit latent alias
    return placed


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = _place_local(rng, _grow_local_pattern(rng, enclosed=True))
    distractors: list[list[list[int]]] = []
    seen = {frozenset(map(tuple, target))}
    while len(distractors) < 3:
        panel = _place_local(rng, _grow_local_pattern(rng, enclosed=False))
        signature = frozenset(map(tuple, panel))
        if signature not in seen:
            seen.add(signature)
            distractors.append(panel)
    target_index = int(seed) % len(LABELS)
    panels = list(distractors)
    panels.insert(target_index, target)
    return {"panels": panels}


def _panel_scores(panels: list[list[list[int]]]) -> list[int]:
    return [len(_enclosed_cells(panel)) for panel in panels]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene["panels"])
    winners = [index for index, score in enumerate(scores) if score == 1]
    if len(winners) != 1 or any(score not in (0, 1) for score in scores):
        raise ValueError(f"scene has no unique one-cell enclosure: {scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene["panels"]), reverse=True)
    return float(scores[0] - scores[1]) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    panels = scene.get("panels")
    if not isinstance(panels, list) or len(panels) != 4:
        return True
    try:
        inventories = [len({tuple(value) for value in panel}) for panel in panels]
        valid_cells = all(
            all(len(value) == 2 and 0 <= value[0] < GRID and 0 <= value[1] < GRID for value in panel)
            for panel in panels
        )
        scores = _panel_scores(panels)
        connected = all(_is_dark_connected(panel) for panel in panels)
    except (TypeError, ValueError):
        return True
    return not (
        valid_cells
        and inventories == [DARK_COUNT] * 4
        and connected
        and sorted(scores) == [0, 0, 0, 1]
        and margin(scene) == 1.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return [
        (
            "reverse_coordinate_serialization",
            {"panels": [list(reversed(panel)) for panel in scene["panels"]]},
        )
    ]


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    title = "SIDE-ADJACENT WHITE-CELL REACHABILITY"
    title_box = draw.textbbox((0, 0), title, font=font)
    title_width = title_box[2] - title_box[0]
    draw.text(((CANVAS[0] - title_width) // 2, 22), title, fill=MUTED, font=font)

    origins = (72, 298, 524, 750)
    top = 65
    step = 23
    span = GRID * step
    for index, (origin, cells) in enumerate(zip(origins, panels)):
        draw.rectangle((origin, top, origin + span, top + span), fill=PANEL_WHITE)
        dark = {tuple(value) for value in cells}
        for row, col in dark:
            left = origin + col * step + 2
            upper = top + row * step + 2
            draw.rectangle(
                (left, upper, origin + (col + 1) * step - 2, top + (row + 1) * step - 2),
                fill=DARK,
            )
        for offset in range(GRID + 1):
            x = origin + offset * step
            y = top + offset * step
            draw.line((x, top, x, top + span), fill=FRAME, width=2)
            draw.line((origin, y, origin + span, y), fill=FRAME, width=2)

        label_center = origin + span // 2
        draw.rounded_rectangle(
            (label_center - 18, 221, label_center + 18, 257),
            radius=8,
            fill=ACCENT,
        )
        label = LABELS[index]
        box = draw.textbbox((0, 0), label, font=font)
        draw.text(
            (label_center - (box[2] - box[0]) // 2, 233),
            label,
            fill=(255, 255, 255),
            font=font,
        )

    draw.text(
        (312, 278),
        "White moves through shared sides only",
        fill=MUTED,
        font=font,
    )
    return image
