"""Renderer for the socket-notch / plug-fit correspondence world.

A dark board is littered with decorative tile clusters. Somewhere on the board
sits one small cream panel. The panel carries exactly three disconnected
tile glyphs, separated by plain cream background:

* a gray SOCKET: a 5 x 5 block of tiles with a connected hole bitten out of its
  strict interior (so the socket's outer 5 x 5 outline is always complete);
* two coloured PLUGS below it, one left and one right, each a single colour and
  each built from exactly as many tiles as the hole.

Exactly one plug is the hole's shape under pure translation. The other plug is
the hole after a 180 degree turn or a mirror flip, so it has the same tile
count and the same bounding-box dimensions but a different cell set. The answer
is the colour of the plug that fits.

All decisive evidence therefore lives inside one small panel, yet it is split
across three disconnected glyphs, each of which is individually necessary.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "socket_notch_plug_fit-0.1.0"

CANVAS_W = 640
CANVAS_H = 480

TILE = 7
PITCH = 10
GAP = 36

SOCKET_N = 5
PLUG_BOX = 3

# panel-inner-relative anchors (all multiples of PITCH)
SOCKET_OX = 20
SOCKET_OY = 0
PLUG_ROW_OY = SOCKET_OY + (SOCKET_N - 1) * PITCH + TILE + GAP  # 47 + 36 = 83
PLUG_LEFT_OX = 0
PLUG_RIGHT_OX = (PLUG_BOX - 1) * PITCH + TILE + GAP  # 27 + 36 = 63

CONTENT_W = PLUG_RIGHT_OX + (PLUG_BOX - 1) * PITCH + TILE  # 90
CONTENT_H = PLUG_ROW_OY + (PLUG_BOX - 1) * PITCH + TILE  # 110
PANEL_PAD = 10
PANEL_W = CONTENT_W + 2 * PANEL_PAD  # 110
PANEL_H = CONTENT_H + 2 * PANEL_PAD  # 130

BOARD = (26, 32, 44)
CREAM = (240, 232, 208)
SOCKET_GRAY = (96, 104, 118)

COLOR_NAMES = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (196, 52, 52),
    "blue": (44, 92, 196),
    "green": (36, 140, 80),
    "purple": (132, 60, 168),
}

# Every decorative colour, including gray, gets exactly this many clutter tiles,
# so no colour can be picked out by total ink outside the panel.
CLUTTER_TILES_PER_COLOR = 14
CLUTTER_CLUSTER_SIZES = (3, 4, 3, 4)

QUARANTINE_MARGIN = 3.0

_INTERIOR = tuple((r, c) for r in range(1, SOCKET_N - 1) for c in range(1, SOCKET_N - 1))


def _normalise(cells) -> tuple[tuple[int, int], ...]:
    """Translate a cell set so its bounding box starts at (0, 0)."""
    rows = [r for r, _ in cells]
    cols = [c for _, c in cells]
    r0, c0 = min(rows), min(cols)
    return tuple(sorted((r - r0, c - c0) for r, c in cells))


def _connected(cells) -> bool:
    """4-connectivity of a cell set."""
    remaining = set(cells)
    if not remaining:
        return False
    stack = [next(iter(remaining))]
    remaining.discard(stack[0])
    while stack:
        r, c = stack.pop()
        for nb in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if nb in remaining:
                remaining.discard(nb)
                stack.append(nb)
    return not remaining


def _sample_hole(rng: random.Random):
    """A connected hole of 3-5 cells inside the socket's strict 3 x 3 interior."""
    while True:
        size = rng.choice((3, 4, 5))
        cells = set(rng.sample(_INTERIOR, size))
        if not _connected(cells):
            continue
        return tuple(sorted(cells))


def _variants(hole) -> dict[str, tuple[tuple[int, int], ...]]:
    """Dimension-preserving re-poses of the hole: half turn and two mirrors."""
    norm = _normalise(hole)
    height = max(r for r, _ in norm) + 1
    width = max(c for _, c in norm) + 1
    return {
        "half_turn": _normalise([(height - 1 - r, width - 1 - c) for r, c in norm]),
        "mirror_lr": _normalise([(r, width - 1 - c) for r, c in norm]),
        "mirror_ud": _normalise([(height - 1 - r, c) for r, c in norm]),
    }


def _clutter(rng: random.Random, panel_box) -> list[dict]:
    """Non-overlapping decorative tile clusters, ink-balanced across colours."""
    px0, py0, px1, py1 = panel_box
    keep_out = (px0 - 12, py0 - 12, px1 + 12, py1 + 12)
    placed: list[tuple[int, int, int, int]] = []
    items: list[dict] = []
    names = list(COLOR_NAMES) + ["gray"]
    for name in names:
        for size in CLUTTER_CLUSTER_SIZES:
            for _ in range(400):
                cols = rng.randint(1, 3)
                rows = (size + cols - 1) // cols
                w = (cols - 1) * PITCH + TILE
                h = (rows - 1) * PITCH + TILE
                x = rng.randint(6, CANVAS_W - w - 6)
                y = rng.randint(6, CANVAS_H - h - 6)
                box = (x - 6, y - 6, x + w + 6, y + h + 6)
                if box[0] < keep_out[2] and keep_out[0] < box[2] and box[1] < keep_out[3] and keep_out[1] < box[3]:
                    continue
                if any(box[0] < o[2] and o[0] < box[2] and box[1] < o[3] and o[1] < box[3] for o in placed):
                    continue
                placed.append(box)
                offsets = [(i // cols, i % cols) for i in range(size)]
                items.append({"color": name, "x": x, "y": y, "cells": [list(o) for o in offsets]})
                break
    return items


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    # The answer colour cycles on seed % 4 and the near-boundary flag cycles on
    # (seed // 4) % 3, so answer identity is independent of the decision margin.
    want_near_boundary = (seed // len(COLOR_NAMES)) % 3 == 0

    hole = _sample_hole(rng)
    poses: dict[str, tuple[tuple[int, int], ...]] = {}
    pool: list[str] = []
    for _ in range(120):
        hole = _sample_hole(rng)
        norm_hole = _normalise(hole)
        poses = {k: v for k, v in _variants(hole).items() if v != norm_hole}
        if not poses:
            continue
        gaps = {k: len(set(norm_hole) ^ set(v)) for k, v in poses.items()}
        pool = sorted(k for k, gap in gaps.items() if (gap < QUARANTINE_MARGIN) == want_near_boundary)
        if pool:
            break
    if not poses:
        raise RuntimeError("no asymmetric hole was found for this seed")
    if not pool:
        pool = sorted(poses)
    pose_name = pool[rng.randrange(len(pool))]
    decoy = poses[pose_name]

    # Balanced answer colour: seeds arrive as base*1000 + index, so this cycles.
    match_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    others = [name for name in COLOR_NAMES if name != match_color]
    other_color = others[rng.randrange(len(others))]
    match_side = "left" if rng.random() < 0.5 else "right"

    panel_x = rng.randint(12, CANVAS_W - PANEL_W - 12)
    panel_y = rng.randint(12, CANVAS_H - PANEL_H - 12)
    clutter = _clutter(rng, (panel_x, panel_y, panel_x + PANEL_W, panel_y + PANEL_H))

    return {
        "panel": [panel_x, panel_y],
        "hole": [list(cell) for cell in hole],
        "decoy": [list(cell) for cell in decoy],
        "decoy_pose": pose_name,
        "match_color": match_color,
        "other_color": other_color,
        "match_side": match_side,
        "clutter": clutter,
    }


def _plug_cells(scene: dict, side: str):
    if side == scene["match_side"]:
        return _normalise([tuple(c) for c in scene["hole"]]), scene["match_color"]
    return _normalise([tuple(c) for c in scene["decoy"]]), scene["other_color"]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BOARD)
    draw = ImageDraw.Draw(image)

    for item in sorted(scene["clutter"], key=lambda d: (d["y"], d["x"], d["color"])):
        fill = SOCKET_GRAY if item["color"] == "gray" else PALETTE[item["color"]]
        for r, c in sorted(tuple(cell) for cell in item["cells"]):
            x = item["x"] + c * PITCH
            y = item["y"] + r * PITCH
            draw.rectangle([x, y, x + TILE - 1, y + TILE - 1], fill=fill)

    px, py = scene["panel"]
    draw.rectangle([px, py, px + PANEL_W - 1, py + PANEL_H - 1], fill=CREAM)
    ox, oy = px + PANEL_PAD, py + PANEL_PAD

    hole = {tuple(cell) for cell in scene["hole"]}
    for r in range(SOCKET_N):
        for c in range(SOCKET_N):
            if (r, c) in hole:
                continue
            x = ox + SOCKET_OX + c * PITCH
            y = oy + SOCKET_OY + r * PITCH
            draw.rectangle([x, y, x + TILE - 1, y + TILE - 1], fill=SOCKET_GRAY)

    for side, base_x in (("left", PLUG_LEFT_OX), ("right", PLUG_RIGHT_OX)):
        cells, name = _plug_cells(scene, side)
        fill = PALETTE[name]
        for r, c in cells:
            x = ox + base_x + c * PITCH
            y = oy + PLUG_ROW_OY + r * PITCH
            draw.rectangle([x, y, x + TILE - 1, y + TILE - 1], fill=fill)

    return image


def analytic_gold(scene: dict) -> str:
    """Colour of the plug whose tile set equals the hole under translation."""
    hole = _normalise([tuple(c) for c in scene["hole"]])
    decoy = _normalise([tuple(c) for c in scene["decoy"]])
    if decoy == hole:
        raise ValueError("decoy plug is not distinguishable from the hole")
    return scene["match_color"]


def margin(scene: dict) -> float:
    """Runner-up gap: tiles by which the wrong plug fails to fill the hole."""
    hole = set(_normalise([tuple(c) for c in scene["hole"]]))
    decoy = set(_normalise([tuple(c) for c in scene["decoy"]]))
    return float(len(hole ^ decoy))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Stored cell and clutter lists are sets: their order is a pixel-identical alias."""
    return [
        ("hole_cell_order", {**scene, "hole": list(reversed(scene["hole"]))}),
        ("decoy_cell_order", {**scene, "decoy": list(reversed(scene["decoy"]))}),
        ("clutter_order", {**scene, "clutter": list(reversed(scene["clutter"]))}),
    ]
