"""Independent pixel-only inverse for thread-terminal search.

This module deliberately duplicates the public palette and sees only an RGB PNG.
It enumerates every terminal color, extracts the black connected component touching
the cyan start bead, and selects the uniquely nearest terminal with a runner-up gap.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BLACK = np.array((28, 31, 36), dtype=np.int16)
_START = np.array((20, 184, 196), dtype=np.int16)
_COLORS = {
    "red": np.array((220, 58, 61), dtype=np.int16),
    "gold": np.array((225, 164, 25), dtype=np.int16),
    "violet": np.array((132, 76, 184), dtype=np.int16),
    "green": np.array((45, 158, 91), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    components: list[np.ndarray] = []
    height, width = work.shape
    while work.any():
        y0, x0 = np.argwhere(work)[0]
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _minimum_distance(a: np.ndarray, b: np.ndarray) -> float:
    # Components are small (< 700 pixels), so exact chunked point distance is cheap.
    best = float("inf")
    for start in range(0, len(a), 128):
        delta = a[start : start + 128, None, :] - b[None, :, :]
        value = float(np.sqrt(np.sum(delta.astype(np.float64) ** 2, axis=2)).min())
        best = min(best, value)
    return best


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    start_points = np.argwhere(_mask(array, _START))
    if len(start_points) < 60:
        raise ValueError("cyan start bead missing or damaged")
    black_components = [part for part in _components(_mask(array, _BLACK)) if len(part) >= 20]
    if not black_components:
        raise ValueError("black thread missing")
    start_distances = [_minimum_distance(part, start_points) for part in black_components]
    touching = [i for i, distance in enumerate(start_distances) if distance <= 2.0]
    if len(touching) != 1:
        raise ValueError("start does not identify one black component")
    thread = black_components[touching[0]]
    if len(thread) < 180:
        raise ValueError("start-connected thread is broken or too short")

    distances: dict[str, float] = {}
    for name, color in _COLORS.items():
        points = np.argwhere(_mask(array, color))
        if len(points) < 60:
            # A destructive crop of a distractor is nondiagnostic: it must not
            # make the oracle depend on the entire terminal cluster. A missing
            # reached terminal still fails below because no remaining candidate
            # is in visible contact with the start-connected thread.
            distances[name] = float("inf")
        else:
            distances[name] = _minimum_distance(thread, points)
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    winner, best = ordered[0]
    runner = ordered[1][1]
    if best > 2.0:
        raise ValueError("thread touches no terminal")
    gap = runner - best
    if gap < 6.0:
        raise ValueError("ambiguous terminal contact")
    return winner, float(gap)


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
