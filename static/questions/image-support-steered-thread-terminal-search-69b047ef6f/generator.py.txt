"""Deterministic latent sampler and renderer for a local endpoint-tracing task."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

BACKGROUND = (250, 249, 246)
BLACK = (28, 31, 36)
START = (20, 184, 196)
COLORS = {
    "red": (220, 58, 61),
    "gold": (225, 164, 25),
    "violet": (132, 76, 184),
    "green": (45, 158, 91),
}
COLOR_NAMES = tuple(COLORS)
TERMINAL_RADIUS = 6
THREAD_WIDTH = 5


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    # The complete decisive inset is under 100 x 58 px and moves between scenes.
    ox = rng.randint(80, 330)
    offsets = [-24, -8, 8, 24]
    order = list(COLOR_NAMES)
    rng.shuffle(order)
    answer = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    target_offset = offsets[order.index(answer)]
    # Put the decisive route well inside one of the public 8x8 ablation rows.
    # Its absolute row still varies, and the terminal order independently varies.
    target_y = rng.choice((96, 160, 224, 288, 352, 416))
    oy = target_y - target_offset
    sx, sy = ox, target_y
    tx = ox + 86
    # Alternating elbows prevent the task from reducing to a straight pointing cue.
    elbow_x = ox + rng.choice((27, 31, 35))
    middle_y = sy + rng.choice((-5, 5))
    points = [
        [sx + 7, sy],
        [elbow_x, sy],
        [elbow_x, middle_y],
        [elbow_x + 18, middle_y],
        [elbow_x + 18, target_y],
        [tx - TERMINAL_RADIUS + 1, target_y],
    ]
    return {
        "width": width,
        "height": height,
        "origin": [ox, oy],
        "terminal_order": order,
        "route_points": points,
        "answer_color": answer,
        "nonce": rng.randrange(1_000_000),
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    ox, oy = map(int, scene["origin"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    tx = ox + 86
    offsets = (-24, -8, 8, 24)

    # The thread is drawn first so the opaque terminal bead supplies a crisp,
    # visible contact boundary rather than leaking black ink through its center.
    points = [tuple(map(int, point)) for point in scene["route_points"]]
    answer = str(scene["answer_color"])
    if answer not in scene["terminal_order"]:
        raise ValueError("answer color is not a rendered terminal")
    answer_y = oy + offsets[list(scene["terminal_order"]).index(answer)]
    if points[-1] != (tx - TERMINAL_RADIUS + 1, answer_y):
        raise ValueError("answer color disagrees with the visibly contacted terminal")
    draw.line(points, fill=BLACK, width=THREAD_WIDTH, joint="curve")
    r0 = 6
    start_x = points[0][0] - 7
    start_y = points[0][1]
    draw.ellipse((start_x - r0, start_y - r0, start_x + r0, start_y + r0), fill=START)
    for name, dy in zip(scene["terminal_order"], offsets, strict=True):
        y = oy + dy
        r = TERMINAL_RADIUS
        draw.ellipse((tx - r, y - r, tx + r, y + r), fill=COLORS[name])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["answer_color"])


def _point_segment_distance(px: float, py: float, a: list[int], b: list[int]) -> float:
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    if vx == 0 and vy == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / (vx * vx + vy * vy)))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def margin(scene: dict) -> float:
    ox, oy = map(int, scene["origin"])
    tx = ox + 86
    offsets = (-24, -8, 8, 24)
    route = scene["route_points"]
    distances = {}
    for name, dy in zip(scene["terminal_order"], offsets, strict=True):
        cy = oy + dy
        distances[name] = min(
            _point_segment_distance(tx, cy, route[i], route[i + 1])
            for i in range(len(route) - 1)
        ) - TERMINAL_RADIUS - THREAD_WIDTH / 2
    answer = analytic_gold(scene)
    competitors = [distance for name, distance in distances.items() if name != answer]
    return float(min(competitors) - distances[answer])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["nonce"] = int(twin["nonce"]) + 1
    return [("render_inert_nonce_increment", twin)]
