"""Deterministic renderer and analytic stack replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 600
BACKGROUND = (255, 255, 255)
CARD = (242, 245, 249)
INK = (35, 49, 66)
GOLD = (236, 178, 42)
PALETTE = {
    "red": (214, 68, 76),
    "green": (48, 153, 103),
    "blue": (55, 105, 205),
}
CARD_BOXES = ((20, 20, 240, 190), (560, 20, 780, 190), (20, 410, 240, 580), (560, 410, 780, 580))


def _replay(scene: dict) -> list[str]:
    stack = list(scene["initial_stack"])
    for operation in scene["operations"]:
        if operation["kind"] == "push":
            stack.append(operation["color"])
        elif operation["kind"] == "pop":
            if not stack:
                raise ValueError("stack underflow")
            stack.pop()
        else:
            raise ValueError("unknown operation")
    return stack


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(PALETTE)
    target = colors[seed % len(colors)]
    mode = seed % 6
    initial = [rng.choice(colors) for _ in range(3)]
    push_a = rng.choice(colors)
    push_b = rng.choice(colors)

    if mode in (0, 1):
        # Push then remove it, then expose the initial middle item.
        initial[1] = target
        operations = [
            {"kind": "push", "color": push_a},
            {"kind": "pop"},
            {"kind": "pop"},
        ]
    elif mode in (2, 3):
        # Remove the initial top, briefly push/remove, leaving initial middle.
        initial[1] = target
        operations = [
            {"kind": "pop"},
            {"kind": "push", "color": push_a},
            {"kind": "pop"},
        ]
    elif mode == 4:
        # Two removals expose the bottom, then a final push controls the answer.
        operations = [
            {"kind": "pop"},
            {"kind": "pop"},
            {"kind": "push", "color": target},
        ]
    else:
        # Two pushes followed by a pop leave the first pushed color.
        push_a = target
        operations = [
            {"kind": "push", "color": push_a},
            {"kind": "push", "color": push_b},
            {"kind": "pop"},
        ]
    scene = {"initial_stack": initial, "operations": operations}
    if _replay(scene)[-1] != target:
        raise AssertionError("balanced constructor failed")
    return scene


def analytic_gold(scene: dict) -> str:
    final = _replay(scene)
    if not final:
        raise ValueError("empty final stack")
    return final[-1]


def margin(scene: dict) -> float:
    # One discrete stack item separates the winner from any runner-up state.
    _replay(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        return not bool(_replay(scene))
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int]) -> None:
    draw.rounded_rectangle(box, radius=14, fill=CARD, outline=INK, width=5)


def _disk(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color, outline=INK, width=3)


def _triangle(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]) -> None:
    x, y = center
    draw.polygon(((x, y - 48), (x - 54, y + 43), (x + 54, y + 43)), fill=color, outline=INK, width=4)


def _cross(draw: ImageDraw.ImageDraw, center: tuple[int, int]) -> None:
    x, y = center
    draw.line((x - 47, y - 47, x + 47, y + 47), fill=INK, width=15)
    draw.line((x + 47, y - 47, x - 47, y + 47), fill=INK, width=15)


def _order_dots(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], count: int) -> None:
    x0, y0, x1, _ = box
    cx = (x0 + x1) // 2
    for index in range(count):
        x = cx + (index - (count - 1) / 2) * 30
        draw.ellipse((x - 8, y0 + 14, x + 8, y0 + 30), fill=GOLD)


def render(scene: dict) -> Image.Image:
    initial = list(scene["initial_stack"])
    operations = list(scene["operations"])
    if len(initial) != 3 or len(operations) != 3:
        raise ValueError("renderer expects three start items and three operations")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for box in CARD_BOXES:
        _card(draw, box)

    # START is the only card with three vertically ordered round stack items.
    start_box = CARD_BOXES[0]
    sx = (start_box[0] + start_box[2]) // 2
    draw.line((sx - 42, 39, sx - 42, 176, sx + 42, 176, sx + 42, 39), fill=INK, width=5)
    for index, color in enumerate(reversed(initial)):
        _disk(draw, (sx, 57 + index * 53), 21, PALETTE[color])

    for order, (operation, box) in enumerate(zip(operations, CARD_BOXES[1:]), start=1):
        _order_dots(draw, box, order)
        center = ((box[0] + box[2]) // 2, (box[1] + box[3]) // 2 + 20)
        if operation["kind"] == "push":
            _triangle(draw, center, PALETTE[operation["color"]])
        elif operation["kind"] == "pop":
            _cross(draw, center)
        else:
            raise ValueError("unknown operation")
    return image
