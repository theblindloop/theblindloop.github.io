"""Independent image-only inverse for the four detached stack cards."""

from __future__ import annotations

import numpy as np
from PIL import Image

CARD = np.array((242, 245, 249), dtype=np.int16)
INK = np.array((35, 49, 66), dtype=np.int16)
GOLD = np.array((236, 178, 42), dtype=np.int16)
COLORS = {
    "red": np.array((214, 68, 76), dtype=np.int16),
    "green": np.array((48, 153, 103), dtype=np.int16),
    "blue": np.array((55, 105, 205), dtype=np.int16),
}


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(indices) - 1]
    return [(int(indices[a]), int(indices[b]) + 1) for a, b in zip(starts, ends)]


def _components(mask: np.ndarray, minimum: int = 30) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    found = []
    height, width = work.shape
    for y, x in np.argwhere(work):
        y, x = int(y), int(x)
        if not work[y, x]:
            continue
        stack = [(y, x)]
        work[y, x] = False
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    stack.append((ny, nx))
        if len(points) >= minimum:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            found.append((min(xs), min(ys), max(xs) + 1, max(ys) + 1, len(points)))
    return found


def _panel_boxes(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    fill = _near(array, CARD)
    x_runs = _runs(fill.sum(axis=0) > 100)
    y_runs = _runs(fill.sum(axis=1) > 180)
    if len(x_runs) != 2 or len(y_runs) != 2:
        raise ValueError("four complete detached cards were not recovered")
    boxes = []
    for y0, y1 in y_runs:
        for x0, x1 in x_runs:
            if x1 - x0 < 190 or y1 - y0 < 140:
                raise ValueError("card extent is incomplete")
            boxes.append((x0, y0, x1, y1))
    return boxes


def _color_bands(crop: np.ndarray) -> list[tuple[int, str, int]]:
    union = np.zeros(crop.shape[:2], dtype=bool)
    masks = {}
    for name, rgb in COLORS.items():
        masks[name] = _near(crop, rgb)
        union |= masks[name]
    bands = _runs(union.sum(axis=1) > 6)
    output = []
    for y0, y1 in bands:
        counts = {name: int(mask[y0:y1].sum()) for name, mask in masks.items()}
        name, amount = max(counts.items(), key=lambda item: item[1])
        output.append(((y0 + y1) // 2, name, amount))
    return output


def _gold_count(crop: np.ndarray) -> int:
    mask = _near(crop, GOLD)
    components = _components(mask, minimum=90)
    if any(not (180 <= item[4] <= 260) for item in components):
        raise ValueError("damaged order dot")
    return len(components)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("RGB image required")
    boxes = _panel_boxes(array)
    start_stack = None
    ordered: dict[int, dict] = {}
    for x0, y0, x1, y1 in boxes:
        crop = array[y0:y1, x0:x1]
        gold_count = _gold_count(crop)
        bands = _color_bands(crop)
        if gold_count == 0:
            if start_stack is not None or len(bands) != 3:
                raise ValueError("unique three-item START card not recovered")
            if any(amount < 1000 for _, _, amount in bands):
                raise ValueError("damaged START token")
            # Bands arrive top-to-bottom; stack representation is bottom-to-top.
            start_stack = [name for _, name, _ in reversed(bands)]
        elif gold_count in (1, 2, 3):
            if gold_count in ordered:
                raise ValueError("duplicate card number")
            if len(bands) == 1:
                _, name, amount = bands[0]
                if amount < 3600:
                    raise ValueError("damaged push triangle")
                ordered[gold_count] = {"kind": "push", "color": name}
            elif len(bands) == 0:
                ink = _near(crop, INK)
                # Ignore the border and inspect the central operation-glyph zone.
                inner = ink[38:-12, 38:-38]
                if int(inner.sum()) < 2200:
                    raise ValueError("damaged pop X")
                ordered[gold_count] = {"kind": "pop"}
            else:
                raise ValueError("operation card has ambiguous colored glyphs")
        else:
            raise ValueError("invalid card order badge")
    if start_stack is None or set(ordered) != {1, 2, 3}:
        raise ValueError("complete ordered replay not recovered")
    stack = list(start_stack)
    for index in (1, 2, 3):
        operation = ordered[index]
        if operation["kind"] == "push":
            stack.append(operation["color"])
        else:
            if not stack:
                raise ValueError("stack underflow")
            stack.pop()
    if not stack:
        raise ValueError("empty final stack")
    return stack[-1]
