"""Pixel-only decision arm for the even-spacing world.

Receives one RGB image and nothing else. It finds the three coloured frames,
segments the dark dots that lie inside each frame, reduces every panel to its
mean nearest-neighbour distance, and names the frame colour of the panel whose
dots are spread the most evenly. No renderer, prompt, generator, verifier or
analytic-gold code is imported, and no latent state is accepted.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

# Colour vocabulary of the public question contract.
REFERENCE_COLORS = {
    "red": (206, 44, 44),
    "green": (26, 132, 60),
    "blue": (40, 68, 196),
}

SATURATION_CUT = 60
COLOR_TOLERANCE = 110
DARK_SATURATION_CUT = 40
DARK_VALUE_CUT = 150
MIN_BLOB_PIXELS = 6
MIN_DOTS_PER_PANEL = 20
INTERIOR_TRIM = 7
MIN_DECISION_GAP_PX = 1.5


def _frame_boxes(arr: np.ndarray) -> dict[str, tuple[int, int, int, int]]:
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    saturated = (high - low) >= SATURATION_CUT
    boxes: dict[str, tuple[int, int, int, int]] = {}
    for name, rgb in REFERENCE_COLORS.items():
        distance = np.sqrt(((arr - np.array(rgb, dtype=np.float64)) ** 2).sum(axis=2))
        mask = saturated & (distance <= COLOR_TOLERANCE)
        ys, xs = np.nonzero(mask)
        if xs.size < 200:
            continue
        boxes[name] = (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
    return boxes


def _blob_centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    """Four-connected components of a small binary mask, as centroids."""
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    centroids: list[tuple[float, float]] = []
    for sy, sx in zip(*np.nonzero(mask)):
        if visited[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        visited[sy, sx] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= MIN_BLOB_PIXELS:
            centroids.append(
                (
                    sum(p[1] for p in pixels) / len(pixels),
                    sum(p[0] for p in pixels) / len(pixels),
                )
            )
    return centroids


def _mean_nearest_neighbour(points: list[tuple[float, float]]) -> float:
    total = 0.0
    for i, point in enumerate(points):
        best = float("inf")
        for j, other in enumerate(points):
            if i == j:
                continue
            distance = math.hypot(point[0] - other[0], point[1] - other[1])
            if distance < best:
                best = distance
        total += best
    return total / len(points)


def panel_spacings(image: Image.Image) -> dict[str, float]:
    """Mean nearest-neighbour dot distance inside each coloured frame."""
    arr = np.asarray(image.convert("RGB")).astype(np.float64)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    dark = ((high - low) < DARK_SATURATION_CUT) & (arr.mean(axis=2) < DARK_VALUE_CUT)
    spacings: dict[str, float] = {}
    for name, (x0, y0, x1, y1) in _frame_boxes(arr).items():
        window = dark[
            y0 + INTERIOR_TRIM : y1 - INTERIOR_TRIM + 1,
            x0 + INTERIOR_TRIM : x1 - INTERIOR_TRIM + 1,
        ]
        if window.size == 0:
            continue
        centroids = _blob_centroids(window)
        if len(centroids) < MIN_DOTS_PER_PANEL:
            continue
        spacings[name] = _mean_nearest_neighbour(centroids)
    return spacings


def decision_from_image(image: Image.Image) -> str:
    spacings = panel_spacings(image)
    if len(spacings) < 2:
        return ABSTAIN
    ranked = sorted(spacings.items(), key=lambda item: (-item[1], item[0]))
    if ranked[0][1] - ranked[1][1] < MIN_DECISION_GAP_PX:
        return ABSTAIN
    return ranked[0][0]
