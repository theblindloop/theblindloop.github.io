"""Scene sampling, rastering and analytic gold for the even-spacing world.

Three framed panels each carry the same number of identical dots. The dots of
one panel are laid out so that no dot crowds another and no gap is left open,
while the dots of the other two panels either clump together or fall where they
like. Nothing local separates the panels: the dot count, the dot radius, the
dot colour and therefore the ink of each panel are identical, and the frame
colours are permuted independently of the answer. Only a whole-panel statistic
of the arrangement - the typical distance from a dot to its nearest neighbour -
tells the panels apart.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

ABSTAIN = "abstain"

WIDTH = 808
HEIGHT = 284
PANEL_SIDE = 240
PANEL_GAP = 22
OUTER_MARGIN = 22
FRAME_WIDTH = 4
FIELD_INSET = 18

BG_COLOR = (247, 247, 243)
DOT_COLOR = (54, 54, 58)
DOT_RADIUS = 3.0
DOT_COUNT = 49
GRID_SIDE = 7
MIN_SEPARATION = 8.0

CLASS_NAMES = ("red", "green", "blue")
FRAME_COLORS = {
    "red": (206, 44, 44),
    "green": (26, 132, 60),
    "blue": (40, 68, 196),
}

QUARANTINE_PX = 3.0
SAMPLING_FLOOR_PX = 5.0
MAX_RESAMPLE = 60


def _panel_rects() -> list[tuple[int, int, int, int]]:
    rects = []
    for index in range(3):
        x0 = OUTER_MARGIN + index * (PANEL_SIDE + PANEL_GAP)
        y0 = OUTER_MARGIN
        rects.append((x0, y0, x0 + PANEL_SIDE, y0 + PANEL_SIDE))
    return rects


def _far_enough(point: tuple[float, float], placed: list[tuple[float, float]]) -> bool:
    for other in placed:
        if math.hypot(point[0] - other[0], point[1] - other[1]) < MIN_SEPARATION:
            return False
    return True


def _even_dots(rng: random.Random, span: float) -> list[tuple[float, float]]:
    """Jittered square lattice: every dot keeps its own share of the panel."""
    cell = span / GRID_SIDE
    jitter = cell * rng.uniform(0.14, 0.26)
    points = []
    for row in range(GRID_SIDE):
        for col in range(GRID_SIDE):
            cx = (col + 0.5) * cell + rng.uniform(-jitter, jitter)
            cy = (row + 0.5) * cell + rng.uniform(-jitter, jitter)
            points.append((cx, cy))
    return points


def _loose_dots(rng: random.Random, span: float) -> list[tuple[float, float]]:
    """Dots dropped where they like, subject only to not touching."""
    points: list[tuple[float, float]] = []
    while len(points) < DOT_COUNT:
        candidate = (rng.uniform(0.0, span), rng.uniform(0.0, span))
        if _far_enough(candidate, points):
            points.append(candidate)
    return points


def _clumped_dots(rng: random.Random, span: float) -> list[tuple[float, float]]:
    """Dots gathered into a few tight clumps with wide empty ground between."""
    clump_count = rng.choice((4, 5, 6))
    centers: list[tuple[float, float]] = []
    for _ in range(400):
        if len(centers) == clump_count:
            break
        candidate = (rng.uniform(0.18 * span, 0.82 * span), rng.uniform(0.18 * span, 0.82 * span))
        if all(math.hypot(candidate[0] - c[0], candidate[1] - c[1]) > 0.26 * span for c in centers):
            centers.append(candidate)
    sigma = span * rng.uniform(0.070, 0.095)
    points: list[tuple[float, float]] = []
    index = 0
    for _ in range(20000):
        if len(points) == DOT_COUNT:
            break
        cx, cy = centers[index % len(centers)]
        candidate = (
            min(max(rng.gauss(cx, sigma), 0.0), span),
            min(max(rng.gauss(cy, sigma), 0.0), span),
        )
        if _far_enough(candidate, points):
            points.append(candidate)
            index += 1
    while len(points) < DOT_COUNT:  # exhausted attempts: fall back to loose fill
        candidate = (rng.uniform(0.0, span), rng.uniform(0.0, span))
        if _far_enough(candidate, points):
            points.append(candidate)
    return points


def _mean_nearest_neighbour(dots: list) -> float:
    """Whole-panel spacing statistic: the mean nearest-neighbour distance."""
    total = 0.0
    for i, point in enumerate(dots):
        best = float("inf")
        for j, other in enumerate(dots):
            if i == j:
                continue
            distance = math.hypot(point[0] - other[0], point[1] - other[1])
            if distance < best:
                best = distance
        total += best
    return total / len(dots)


def _panel_dots(rng: random.Random, kind: str, rect: tuple[int, int, int, int]) -> list[list[float]]:
    span = float(PANEL_SIDE - 2 * FIELD_INSET)
    if kind == "even":
        raw = _even_dots(rng, span)
    elif kind == "clumped":
        raw = _clumped_dots(rng, span)
    else:
        raw = _loose_dots(rng, span)
    rng.shuffle(raw)
    x0 = rect[0] + FIELD_INSET
    y0 = rect[1] + FIELD_INSET
    return [[round(x0 + px, 3), round(y0 + py, 3)] for px, py in raw]


def _sample_once(rng: random.Random) -> dict:
    rects = _panel_rects()
    names = list(CLASS_NAMES)
    rng.shuffle(names)
    even_slot = rng.randrange(3)
    kinds = [rng.choice(("loose", "clumped")) for _ in range(3)]
    kinds[even_slot] = "even"
    panels = []
    for slot, rect in enumerate(rects):
        panels.append(
            {
                "name": names[slot],
                "rect": list(rect),
                "dots": _panel_dots(rng, kinds[slot], rect),
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "bg_color": list(BG_COLOR),
        "dot_color": list(DOT_COLOR),
        "dot_radius": DOT_RADIUS,
        "frame_width": FRAME_WIDTH,
        "panels": panels,
    }


def sample_scene(seed: int) -> dict:
    """Rejection-sample until the winning panel leads by the sampling floor."""
    rng = random.Random(seed)
    scene = _sample_once(rng)
    for _ in range(MAX_RESAMPLE):
        if margin(scene) >= SAMPLING_FLOOR_PX and analytic_gold(scene) != ABSTAIN:
            return scene
        scene = _sample_once(rng)
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"])
    for panel in scene["panels"]:
        x0, y0, x1, y1 = [int(v) for v in panel["rect"]]
        draw.rectangle(
            [x0, y0, x1, y1],
            outline=FRAME_COLORS[panel["name"]],
            width=int(scene["frame_width"]),
        )
    for panel in scene["panels"]:
        for x, y in panel["dots"]:
            draw.ellipse(
                [x - radius, y - radius, x + radius, y + radius],
                fill=tuple(scene["dot_color"]),
            )
    return image


def _spacings(scene: dict) -> list[tuple[float, str]]:
    return sorted(
        ((_mean_nearest_neighbour(panel["dots"]), panel["name"]) for panel in scene["panels"]),
        key=lambda item: (-item[0], item[1]),
    )


def analytic_gold(scene: dict) -> str:
    ranked = _spacings(scene)
    if ranked[0][0] - ranked[1][0] < QUARANTINE_PX:
        return ABSTAIN
    return ranked[0][1]


def margin(scene: dict) -> float:
    ranked = _spacings(scene)
    return float(ranked[0][0] - ranked[1][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings: dot lists and the panel list are unordered."""
    panels = [dict(panel) for panel in scene["panels"]]
    alias_a = dict(scene)
    alias_a["panels"] = [
        {**panel, "dots": panel["dots"][3:] + panel["dots"][:3]} for panel in panels
    ]
    alias_b = dict(scene)
    alias_b["panels"] = list(reversed(panels))
    alias_c = dict(scene)
    alias_c["panels"] = [
        {**panel, "dots": list(reversed(panel["dots"]))} for panel in reversed(panels)
    ]
    return [
        ("dot_lists_cycled", alias_a),
        ("panel_list_reversed", alias_b),
        ("panels_reversed_and_dots_reversed", alias_c),
    ]
