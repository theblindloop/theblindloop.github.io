"""Independent pixel-only inverse for fuse-wire continuity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_CHROMA = 58
_MIN_FRAGMENT_AREA = 18
_MIN_TOTAL_AREA = 430
_MAX_TOTAL_AREA = 920
_MIN_TRACE_WIDTH = 100
_MAX_TRACE_WIDTH = 112
_MIN_TRACE_HEIGHT = 13
_MAX_TRACE_HEIGHT = 28


class OracleAbstention(ValueError):
    """The raster is not a supported, structurally complete fuse trace."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _trace_components(array: np.ndarray) -> list[np.ndarray]:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    chroma = high - low
    mask = (chroma >= _MIN_CHROMA) & (high >= 105)
    pieces = []
    speckle = 0
    for points in _components(mask):
        if len(points) < _MIN_FRAGMENT_AREA:
            speckle += len(points)
        else:
            pieces.append(points)
    if speckle > 20:
        raise OracleAbstention(f"too much chromatic speckle ({speckle} pixels)")
    return pieces


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise OracleAbstention("raster is too small")
    pieces = _trace_components(array)
    if len(pieces) not in (1, 2):
        raise OracleAbstention(f"expected one or two substantial trace components, found {len(pieces)}")

    all_points = np.concatenate(pieces, axis=0)
    total_area = len(all_points)
    ys, xs = all_points[:, 0], all_points[:, 1]
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if not _MIN_TOTAL_AREA <= total_area <= _MAX_TOTAL_AREA:
        raise OracleAbstention(f"unsupported trace area {total_area}")
    if not _MIN_TRACE_WIDTH <= width <= _MAX_TRACE_WIDTH:
        raise OracleAbstention(f"unsupported end-to-end trace width {width}")
    if not _MIN_TRACE_HEIGHT <= height <= _MAX_TRACE_HEIGHT or width / height < 3.8:
        raise OracleAbstention(f"trace is not a thin local route ({width}x{height})")

    if len(pieces) == 1:
        return "yes"

    # A valid broken fuse has two substantial left/right pieces separated only by
    # one small visible gap; gross erasures create undersized or widely separated remnants.
    ordered = sorted(pieces, key=lambda points: float(points[:, 1].mean()))
    if min(len(points) for points in ordered) < 145:
        raise OracleAbstention("a broken-trace component is too small")
    left_x_max = int(ordered[0][:, 1].max())
    right_x_min = int(ordered[1][:, 1].min())
    horizontal_gap = right_x_min - left_x_max - 1
    if not 1 <= horizontal_gap <= 14:
        raise OracleAbstention(f"unsupported component gap {horizontal_gap}")
    return "no"
