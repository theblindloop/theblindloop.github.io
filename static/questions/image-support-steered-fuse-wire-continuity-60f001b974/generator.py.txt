"""Deterministic renderer for visible fuse-wire continuity."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
CELL_SIZE = 64
SUPERSAMPLE = 4

WIRE_PALETTE = (
    (35, 101, 221),
    (14, 148, 126),
    (177, 57, 196),
    (224, 82, 31),
)
BACKGROUND_TONES = ((248, 247, 243), (243, 247, 248), (248, 244, 247))
HOUSING_TONES = ((191, 199, 202), (184, 195, 199), (199, 193, 201))
CAP_TONES = ((74, 78, 80), (82, 84, 86), (68, 73, 77))


def sample_scene(seed: int) -> dict:
    """Return adjacent matched control/counterfactual fuse scenes."""

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0xF053C017)
    return {
        "intact": seed % 2 == 0,
        "canvas_size": CANVAS_SIZE,
        "cell_col": rng.randrange(1, 7),
        "cell_row": rng.randrange(1, 7),
        "jitter_x": rng.randrange(-4, 5),
        # Keep the whole thin trace away from horizontal fine-grid boundaries.
        "jitter_y": rng.randrange(0, 4),
        "wire_color_code": pair % len(WIRE_PALETTE),
        "stroke_width": 5 + rng.randrange(2),
        "gap_half_width": 9 + rng.randrange(2),
        "bridge_half_span": 16 + rng.randrange(3),
        "bridge_shift_side": -1 if rng.randrange(2) == 0 else 1,
        "bridge_shift_distance": 20 + 2 * rng.randrange(2),
        "arch_height": 10 + rng.randrange(2),
        "housing_tone": rng.randrange(len(HOUSING_TONES)),
        "decor_variant": rng.randrange(16),
    }


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in values)


def _rounded_polyline(
    draw: ImageDraw.ImageDraw,
    points: list[tuple[float, float]],
    *,
    fill: tuple[int, int, int],
    width: int,
    scale: int,
) -> None:
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    draw.line(scaled, fill=fill, width=width * scale, joint="curve")
    radius = width * scale / 2.0
    for x, y in (scaled[0], scaled[-1]):
        draw.ellipse(
            (round(x - radius), round(y - radius), round(x + radius), round(y + radius)),
            fill=fill,
        )


def _quadratic_bridge(
    cx: float, cy: float, half_span: float, arch_height: float
) -> list[tuple[float, float]]:
    points = []
    for index in range(25):
        t = index / 24.0
        x = cx - half_span + 2.0 * half_span * t
        # Endpoints lie on the base wire; the middle rises visibly upward.
        y = cy - 4.0 * arch_height * t * (1.0 - t)
        points.append((x, y))
    return points


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    intact = bool(scene["intact"])
    col = int(scene["cell_col"])
    row = int(scene["cell_row"])
    center_x = col * CELL_SIZE + CELL_SIZE // 2 + int(scene["jitter_x"])
    center_y = row * CELL_SIZE + CELL_SIZE // 2 + int(scene["jitter_y"])
    color = WIRE_PALETTE[int(scene["wire_color_code"]) % len(WIRE_PALETTE)]
    stroke = int(scene["stroke_width"])
    gap = int(scene["gap_half_width"])
    half_span = int(scene["bridge_half_span"])
    shift_side = int(scene["bridge_shift_side"])
    shift_distance = int(scene["bridge_shift_distance"])
    arch = int(scene["arch_height"])
    housing_code = int(scene["housing_tone"]) % len(HOUSING_TONES)
    decor = int(scene["decor_variant"])

    background = BACKGROUND_TONES[housing_code % len(BACKGROUND_TONES)]
    housing = HOUSING_TONES[housing_code]
    cap = CAP_TONES[housing_code]
    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Remote achromatic marks are deliberately unrelated to wire state.
    remote = tuple(max(0, channel - 17) for channel in background)
    for index in range(3):
        x = 27 + ((decor * 53 + index * 179) % 458)
        y = 28 if (decor + index) % 2 == 0 else size - 29
        draw.line(
            [(round((x - 9) * ss), round(y * ss)), (round((x + 9) * ss), round(y * ss))],
            fill=remote,
            width=2 * ss,
        )

    # A small familiar cartridge-fuse housing. It is visible context, not oracle input.
    glass_fill = tuple(min(255, channel + 42) for channel in housing)
    outline = tuple(max(0, channel - 48) for channel in housing)
    draw.rounded_rectangle(
        _box((center_x - 61, center_y - 20, center_x + 61, center_y + 20), ss),
        radius=18 * ss,
        fill=glass_fill,
        outline=outline,
        width=2 * ss,
    )
    for side in (-1, 1):
        cap_center = center_x + side * 58
        draw.rounded_rectangle(
            _box((cap_center - 8, center_y - 17, cap_center + 8, center_y + 17), ss),
            radius=3 * ss,
            fill=cap,
        )
        shine_x = cap_center - 2 if side < 0 else cap_center + 2
        draw.line(
            [(round(shine_x * ss), round((center_y - 12) * ss)),
             (round(shine_x * ss), round((center_y + 12) * ss))],
            fill=tuple(min(255, value + 42) for value in cap),
            width=1 * ss,
        )

    # The two base halves are identical in both answer classes.
    left = [
        (center_x - 52, center_y),
        (center_x - 39, center_y - 2),
        (center_x - 25, center_y + 2),
        (center_x - gap, center_y),
    ]
    right = [
        (center_x + gap, center_y),
        (center_x + 25, center_y - 2),
        (center_x + 39, center_y + 2),
        (center_x + 52, center_y),
    ]
    _rounded_polyline(draw, left, fill=color, width=stroke, scale=ss)
    _rounded_polyline(draw, right, fill=color, width=stroke, scale=ss)

    # Equal-span bypass in both classes. Centered = touches both halves; shifted =
    # touches one half and visibly stops short of the other.
    bridge_center = center_x if intact else center_x + shift_side * shift_distance
    bridge = _quadratic_bridge(bridge_center, center_y, half_span, arch)
    _rounded_polyline(draw, bridge, fill=color, width=stroke, scale=ss)

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["intact"]) else "no"


def margin(scene: dict) -> float:
    gap = float(scene["gap_half_width"])
    half_span = float(scene["bridge_half_span"])
    if bool(scene["intact"]):
        return half_span - gap
    shift = float(scene["bridge_shift_distance"])
    stroke = float(scene["stroke_width"])
    return gap + shift - half_span - stroke


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["wire_color_code"] = int(twin["wire_color_code"]) + len(WIRE_PALETTE)
    return [("one_full_wire_palette_cycle", twin)]
