"""Deterministic renderer for a visibly edge-matched strip chain."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 560
CARD_W, CARD_H = 94, 430
CARD_X = [40 + 124 * i for i in range(7)]
BODY_X0, BODY_Y0, CELL_W, CELL_H = 31, 166, 32, 32

LABELS = ("plus", "X", "V", "T")


def _blank() -> list[list[int]]:
    return [[0 for _ in range(7)] for _ in range(7)]


def _templates() -> dict[str, list[list[int]]]:
    out = {}
    g = _blank()
    for i in range(7):
        g[3][i] = g[i][3] = 1
    out["plus"] = g
    g = _blank()
    for i in range(7):
        g[i][i] = g[i][6 - i] = 1
    out["X"] = g
    g = _blank()
    for r, cols in enumerate(((0, 6), (0, 6), (1, 5), (1, 5), (2, 4), (2, 4), (3,))):
        for c in cols:
            g[r][c] = 1
    out["V"] = g
    g = _blank()
    for c in range(7):
        g[0][c] = 1
    for r in range(1, 7):
        g[r][3] = 1
    out["T"] = g
    assert {sum(map(sum, grid)) for grid in out.values()} == {13}
    return out


TEMPLATES = _templates()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = LABELS[int(seed) % len(LABELS)]
    permutation = list(range(7))
    rng.shuffle(permutation)
    joint_codes = rng.sample(range(1, 15), 6)
    return {
        "symbol": symbol,
        "grid": copy.deepcopy(TEMPLATES[symbol]),
        "permutation": permutation,
        "joint_codes": joint_codes,
        "card_y": [rng.randint(56, 78) for _ in range(7)],
        "palette_variant": rng.randrange(2),
        "subpixel_jitter": 0.10,
    }


def _bits(code: int) -> tuple[int, int, int, int]:
    return tuple((int(code) >> shift) & 1 for shift in (3, 2, 1, 0))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 244))
    draw = ImageDraw.Draw(image)
    frame_colors = ((28, 145, 105), (35, 115, 165))
    frame = frame_colors[int(scene["palette_variant"])]
    jitter = float(scene["subpixel_jitter"])
    grid = scene["grid"]

    for slot, logical_col in enumerate(scene["permutation"]):
        x0 = int(round(CARD_X[slot] + jitter))
        y0 = int(round(scene["card_y"][slot] + jitter))
        x1, y1 = x0 + CARD_W - 1, y0 + CARD_H - 1
        draw.rectangle((x0, y0, x1, y1), fill=frame)
        draw.rectangle((x0 + 4, y0 + 4, x1 - 4, y1 - 4), fill=(255, 255, 255))

        # Endpoint caps replace the missing code on the outside chain edges.
        if logical_col == 0:
            draw.rounded_rectangle((x0 + 7, y0 + 61, x0 + 19, y0 + 143), radius=4, fill=(0, 185, 220))
        if logical_col == 6:
            draw.rounded_rectangle((x1 - 19, y0 + 61, x1 - 7, y0 + 143), radius=4, fill=(215, 45, 160))

        for side, code in (("left", None if logical_col == 0 else scene["joint_codes"][logical_col - 1]),
                           ("right", None if logical_col == 6 else scene["joint_codes"][logical_col])):
            if code is None:
                continue
            cx = x0 + (13 if side == "left" else CARD_W - 14)
            for k, bit in enumerate(_bits(code)):
                cy = y0 + 70 + 23 * k
                draw.ellipse((cx - 6, cy - 6, cx + 6, cy + 6), fill=(25, 27, 29))
                if not bit:
                    draw.ellipse((cx - 3, cy - 3, cx + 3, cy + 3), fill=(255, 255, 255))

        # The bound glyph column. Every logical cell is visibly delimited.
        for row in range(7):
            xa = x0 + BODY_X0
            ya = y0 + BODY_Y0 + row * CELL_H
            draw.rectangle((xa, ya, xa + CELL_W - 1, ya + CELL_H - 1), fill=(205, 211, 213), outline=(125, 132, 135), width=1)
            inset = 4
            fill = (28, 31, 34) if int(grid[row][logical_col]) else (239, 242, 241)
            draw.rectangle((xa + inset, ya + inset, xa + CELL_W - 1 - inset, ya + CELL_H - 1 - inset), fill=fill)

    # A short visible legend reinforces that the four edge marks are binary codes.
    draw.text((31, 516), "MATCH IDENTICAL 4-DOT EDGES   • = 1   ◦ = 0", fill=(55, 59, 61))
    return image


def analytic_gold(scene: dict) -> str:
    grid = [[int(v) for v in row] for row in scene["grid"]]
    distances = {name: sum(grid[r][c] != templ[r][c] for r in range(7) for c in range(7))
                 for name, templ in TEMPLATES.items()}
    ordered = sorted(distances.items(), key=lambda item: (item[1], LABELS.index(item[0])))
    if ordered[0][1] == ordered[1][1]:
        raise ValueError("ambiguous analytic symbol")
    return ordered[0][0]


def margin(scene: dict) -> float:
    grid = [[int(v) for v in row] for row in scene["grid"]]
    scores = sorted(sum(grid[r][c] != templ[r][c] for r in range(7) for c in range(7))
                    for templ in TEMPLATES.values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        valid_perm = sorted(map(int, scene["permutation"])) == list(range(7))
        codes = list(map(int, scene["joint_codes"]))
        return (not valid_perm or len(codes) != 6 or len(set(codes)) != 6
                or any(not 1 <= code <= 14 for code in codes) or margin(scene) <= 0)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["subpixel_jitter"] = 0.20 if float(scene["subpixel_jitter"]) < 0.15 else 0.10
    return [("within_raster_bin_subpixel_shift", twin)]
