"""Independent PNG-only inverse for visible edge-code strip chains.

This module deliberately duplicates its four public symbol templates and does
not import renderer, prompts, generation, verification, scene data, or gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("plus", "X", "V", "T")


def _templates() -> dict[str, np.ndarray]:
    out = {}
    g = np.zeros((7, 7), dtype=np.uint8)
    g[3, :] = 1; g[:, 3] = 1; out["plus"] = g
    g = np.zeros((7, 7), dtype=np.uint8)
    for i in range(7): g[i, i] = g[i, 6 - i] = 1
    out["X"] = g
    g = np.zeros((7, 7), dtype=np.uint8)
    for r, cols in enumerate(((0, 6), (0, 6), (1, 5), (1, 5), (2, 4), (2, 4), (3,))):
        for c in cols: g[r, c] = 1
    out["V"] = g
    g = np.zeros((7, 7), dtype=np.uint8)
    g[0, :] = 1; g[1:, 3] = 1; out["T"] = g
    return out


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    comps = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))]); seen[y, x] = True
        xs, ys = [], []
        while q:
            yy, xx = q.popleft(); xs.append(xx); ys.append(yy)
            for ny, nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; q.append((ny, nx))
        comps.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return comps


def _dark_fraction(arr: np.ndarray, cx: int, cy: int, radius: int = 2) -> float:
    patch = arr[max(0, cy-radius):cy+radius+1, max(0, cx-radius):cx+radius+1]
    if patch.size == 0:
        return 0.0
    return float(np.mean(np.max(patch, axis=2) < 90))


def _read_code(arr: np.ndarray, x: int, y0: int) -> tuple[int, int, int, int]:
    return tuple(int(_dark_fraction(arr, x, y0 + 70 + 23*k, 2) > 0.55) for k in range(4))


def _cap_present(arr: np.ndarray, box: tuple[int, int, int, int], kind: str) -> bool:
    x0, y0, x1, _ = box
    if kind == "cyan":
        patch = arr[y0+61:y0+144, x0+6:x0+21]
        mask = (patch[..., 0] < 70) & (patch[..., 1] > 130) & (patch[..., 2] > 160)
    else:
        patch = arr[y0+61:y0+144, x1-20:x1-5]
        mask = (patch[..., 0] > 150) & (patch[..., 1] < 100) & (patch[..., 2] > 100)
    return int(mask.sum()) > 400


def _measure(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    frame = (r < 80) & (g > 85) & (b < 190) & ((np.maximum(g, b).astype(int) - r.astype(int)) > 45)
    boxes = [(x0,y0,x1,y1) for x0,y0,x1,y1,n in _components(frame)
             if n > 2500 and 88 <= x1-x0+1 <= 100 and 420 <= y1-y0+1 <= 440]
    boxes.sort()
    if len(boxes) != 7:
        raise ValueError(f"expected seven framed strips, found {len(boxes)}")

    cards = []
    for box in boxes:
        x0, y0, x1, _ = box
        cyan = _cap_present(arr, box, "cyan")
        magenta = _cap_present(arr, box, "magenta")
        left = None if cyan else _read_code(arr, x0 + 13, y0)
        right = None if magenta else _read_code(arr, x1 - 13, y0)
        column = []
        for row in range(7):
            cx, cy = x0 + 47, y0 + 166 + row * 32 + 16
            patch = arr[cy-8:cy+9, cx-8:cx+9]
            column.append(int(float(np.mean(patch)) < 120.0))
        cards.append({"cyan": cyan, "magenta": magenta, "left": left, "right": right, "column": column})

    starts = [i for i, card in enumerate(cards) if card["cyan"]]
    ends = [i for i, card in enumerate(cards) if card["magenta"]]
    if len(starts) != 1 or len(ends) != 1:
        raise ValueError("endpoint caps are missing or ambiguous")
    chain, used = [], set()
    current = starts[0]
    while True:
        if current in used:
            raise ValueError("edge-code chain contains a cycle")
        used.add(current); chain.append(current)
        if cards[current]["magenta"]:
            break
        code = cards[current]["right"]
        matches = [i for i, card in enumerate(cards) if i not in used and card["left"] == code]
        if len(matches) != 1:
            raise ValueError("edge-code chain is not unique")
        current = matches[0]
    if len(chain) != 7 or chain[-1] != ends[0]:
        raise ValueError("edge-code chain does not span all strips")

    grid = np.array([[cards[idx]["column"][row] for idx in chain] for row in range(7)], dtype=np.uint8)
    scores = sorted(((int(np.count_nonzero(grid != templ)), name) for name, templ in TEMPLATES.items()),
                    key=lambda item: (item[0], LABELS.index(item[1])))
    gap = float(scores[1][0] - scores[0][0])
    if gap < 4:
        raise ValueError(f"ambiguous symbol template, gap={gap}")
    return scores[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return _measure(image)[0]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    return _measure(image)
