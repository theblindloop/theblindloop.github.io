"""Independent image-only inverse for the flip-splice tile deck."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (900, 650)
DECK_SIZE, TURNS = 6, 8
CELL, GAP, FIRST_X = 58, 10, 360
START_TOP = 68
OPTION_TOPS = (244, 322, 400, 478, 556)
INK = np.array((24, 59, 86), dtype=np.float32)
PAPER = np.array((248, 246, 238), dtype=np.float32)
NAVY = np.array((51, 77, 92), dtype=np.float32)
IDENTITIES = np.array(
    ((209, 73, 91), (224, 122, 31), (42, 157, 112), (37, 137, 189), (115, 87, 178), (198, 78, 139)),
    dtype=np.float32,
)


class OracleError(ValueError):
    pass


def _read_identity(array: np.ndarray, column: int, top: int) -> int:
    left = FIRST_X + column * (CELL + GAP)
    patch = array[top + 2 : top + 7, left + 17 : left + 42].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    distances = np.linalg.norm(IDENTITIES - mean, axis=1)
    order = np.argsort(distances)
    if float(distances[order[0]]) > 10.0 or float(distances[order[1]] - distances[order[0]]) < 30.0:
        raise OracleError("missing or ambiguous identity frame")
    return int(order[0])


def _read_bit(array: np.ndarray, column: int, top: int) -> int:
    left = FIRST_X + column * (CELL + GAP)
    patches = [
        array[top + 13 : top + 22, left + 13 : left + 22],
        array[top + 13 : top + 22, left + 36 : left + 45],
        array[top + 36 : top + 45, left + 13 : left + 22],
        array[top + 36 : top + 45, left + 36 : left + 45],
    ]
    mean = np.concatenate([patch.reshape(-1, 3) for patch in patches]).astype(np.float32).mean(axis=0)
    ink_distance = float(np.linalg.norm(mean - INK))
    paper_distance = float(np.linalg.norm(mean - PAPER))
    if ink_distance < 10.0 and ink_distance + 30.0 < paper_distance:
        return 1
    if paper_distance < 10.0 and paper_distance + 30.0 < ink_distance:
        return 0
    raise OracleError("tile has no clean binary center")


def _read_deck(array: np.ndarray, top: int) -> list[tuple[int, int]]:
    result = [(_read_identity(array, column, top), _read_bit(array, column, top)) for column in range(DECK_SIZE)]
    if {identity for identity, _ in result} != set(range(DECK_SIZE)):
        raise OracleError("a row does not contain each identity exactly once")
    return result


def _validate_labels(array: np.ndarray) -> None:
    for top in OPTION_TOPS:
        cy = top + CELL // 2
        patch = array[cy - 18 : cy + 19, 157:194].astype(np.float32)
        navy_count = int((np.linalg.norm(patch - NAVY, axis=2) < 18.0).sum())
        white_count = int((patch.mean(axis=2) > 235).sum())
        if navy_count < 700 or white_count < 35:
            raise OracleError("an option label disk or glyph is missing")


def _advance(deck: list[tuple[int, int]]) -> list[tuple[int, int]]:
    current = list(deck)
    identity, old = current.pop(0)
    current.insert(2 if old == 0 else 3, (identity, 1 - old))
    return current


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    deck = _read_deck(array, START_TOP)
    options = [_read_deck(array, top) for top in OPTION_TOPS]
    signatures = {tuple(option) for option in options}
    if len(signatures) != 5:
        raise OracleError("option decks are not distinct")
    for _ in range(TURNS):
        deck = _advance(deck)
    matches = [index for index, option in enumerate(options) if option == deck]
    if len(matches) != 1:
        raise OracleError("recovered deck does not match exactly one option")
    return "ABCDE"[matches[0]]
