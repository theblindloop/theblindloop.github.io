"""Deterministic sampler, renderer, and analytic flip-splice deck program."""

from __future__ import annotations

import random
from typing import Iterable, NamedTuple

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 650
DECK_SIZE, TURNS = 6, 8
CELL, GAP, GRID_X = 58, 10, 360
START_TOP = 68
OPTION_TOPS = (244, 322, 400, 478, 556)
DARK = "#183b56"
LIGHT = "#f8f6ee"
NAVY = "#334d5c"
GOLD = "#d58d16"
BACKGROUND = "#ffffff"
IDENTITY_COLORS = ("#d1495b", "#e07a1f", "#2a9d70", "#2589bd", "#7357b2", "#c64e8b")
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


class Tile(NamedTuple):
    identity: int
    bit: int


Deck = tuple[Tile, ...]


def _tile(value: object) -> Tile:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError("a tile needs exactly an identity-bit pair")
    identity, bit = int(value[0]), int(value[1])
    if identity not in range(DECK_SIZE) or bit not in (0, 1):
        raise ValueError("invalid tile")
    return Tile(identity, bit)


def _deck(values: Iterable[object]) -> Deck:
    result = tuple(_tile(value) for value in values)
    if len(result) != DECK_SIZE or {tile.identity for tile in result} != set(range(DECK_SIZE)):
        raise ValueError("a deck must contain each of six identities exactly once")
    return result


def _as_json(deck: Deck) -> list[list[int]]:
    return [[tile.identity, tile.bit] for tile in deck]


def _turn(deck: Deck) -> Deck:
    current = list(_deck(_as_json(deck)))
    front = current.pop(0)
    flipped = Tile(front.identity, 1 - front.bit)
    # Five tiles remain. Index 2 is after the second; index 3 is before
    # the second-from-right.
    current.insert(2 if front.bit == 0 else 3, flipped)
    return tuple(current)


def _evolve(deck: Deck, turns: int = TURNS) -> Deck:
    result = _deck(_as_json(deck))
    for _ in range(int(turns)):
        result = _turn(result)
    return result


def _evolve_variant(deck: Deck, mode: str) -> Deck:
    result = _deck(_as_json(deck))
    for _ in range(TURNS):
        current = list(result)
        front = current.pop(0)
        if mode == "no_flip":
            moved = front
            site = 2 if front.bit == 0 else 3
        elif mode == "swap_sites":
            moved = Tile(front.identity, 1 - front.bit)
            site = 3 if front.bit == 0 else 2
        elif mode == "always_after_second":
            moved = Tile(front.identity, 1 - front.bit)
            site = 2
        elif mode == "after_third":
            moved = Tile(front.identity, 1 - front.bit)
            site = 3 if front.bit == 0 else 4
        else:
            raise ValueError("unknown variant")
        current.insert(site, moved)
        result = tuple(current)
    return result


def _trace(deck: Deck) -> list[Deck]:
    history = [_deck(_as_json(deck))]
    for _ in range(TURNS):
        history.append(_turn(history[-1]))
    return history


def _valid_scene(scene: dict) -> tuple[Deck, list[Deck]]:
    if set(scene) != {"start_tiles", "options"}:
        raise ValueError("scene fields do not match the declared label inputs")
    start = _deck(scene["start_tiles"])
    if not isinstance(scene["options"], list):
        raise ValueError("options must be a list")
    options = [_deck(option) for option in scene["options"]]
    if len(options) != 5 or len(set(options)) != 5:
        raise ValueError("five distinct complete deck options are required")
    return start, options


def _distance(first: Deck, second: Deck) -> int:
    return sum(
        int(left.identity != right.identity) + int(left.bit != right.bit)
        for left, right in zip(first, second)
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(1200):
        identities = list(range(DECK_SIZE))
        rng.shuffle(identities)
        bits = [rng.randrange(2) for _ in range(DECK_SIZE)]
        start = tuple(Tile(identity, bit) for identity, bit in zip(identities, bits))
        history = _trace(start)
        reads = [state[0].bit for state in history[:-1]]
        front_ids = {state[0].identity for state in history[:-1]}
        target = history[-1]
        if not (
            2 <= sum(bits) <= 4
            and reads.count(0) >= 3
            and reads.count(1) >= 2
            and len(front_ids) >= 4
            and tuple(tile.identity for tile in target) != tuple(identities)
        ):
            continue

        preferred = [
            history[-2],
            _evolve(start, TURNS + 1),
            _evolve_variant(start, "swap_sites"),
            _evolve_variant(start, "no_flip"),
            _evolve_variant(start, "always_after_second"),
            _evolve_variant(start, "after_third"),
        ]
        distractors: list[Deck] = []
        for option in preferred:
            if option != target and option not in distractors:
                distractors.append(option)
        while len(distractors) < 4:
            variant = list(target)
            if rng.randrange(2):
                a, b = rng.sample(range(DECK_SIZE), 2)
                variant[a], variant[b] = variant[b], variant[a]
            else:
                index = rng.randrange(DECK_SIZE)
                tile = variant[index]
                variant[index] = Tile(tile.identity, 1 - tile.bit)
            candidate = tuple(variant)
            if candidate != target and candidate not in distractors:
                distractors.append(candidate)
        rng.shuffle(distractors)
        options = distractors[:4]
        options.insert(int(seed) % 5, target)
        return {"start_tiles": _as_json(start), "options": [_as_json(option) for option in options]}
    raise RuntimeError("could not sample a feedback-rich flip-splice deck")


def analytic_gold(scene: dict) -> str:
    start, options = _valid_scene(scene)
    target = _evolve(start)
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("the final deck must occur exactly once")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, options = _valid_scene(scene)
    target = _evolve(start)
    distances = [_distance(target, option) for option in options if option != target]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, options = _valid_scene(scene)
        target = _evolve(start)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _valid_scene(scene)
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _draw_deck(draw: ImageDraw.ImageDraw, deck: Deck, top: int, *, start: bool = False) -> None:
    checked = _deck(_as_json(deck))
    for column, tile in enumerate(checked):
        left = GRID_X + column * (CELL + GAP)
        color = IDENTITY_COLORS[tile.identity]
        draw.rounded_rectangle((left, top, left + CELL, top + CELL), radius=7, fill=color)
        draw.rounded_rectangle(
            (left + 8, top + 8, left + CELL - 8, top + CELL - 8),
            radius=3,
            fill=DARK if tile.bit else LIGHT,
            outline=GOLD if start and column == 0 else NAVY,
            width=2,
        )
        draw.text(
            (left + CELL // 2, top + CELL // 2),
            str(tile.identity + 1),
            fill="white" if tile.bit else NAVY,
            font=_font(20),
            anchor="mm",
        )


def render(scene: dict) -> Image.Image:
    start, options = _valid_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 20), "FRAME + NUMBER = IDENTITY   •   CENTER = FACE", fill=NAVY, font=_font(18), anchor="mm")
    draw.text((175, START_TOP + CELL / 2), "START", fill=GOLD, font=_font(25), anchor="mm")
    _draw_deck(draw, start, START_TOP, start=True)
    draw.text((235, 146), "LIGHT  →  FLIP  •  INSERT AFTER 2nd", fill=NAVY, font=_font(17), anchor="lm")
    draw.rounded_rectangle((205, 134, 225, 158), radius=3, fill=LIGHT, outline=NAVY, width=2)
    draw.text((235, 176), "DARK  →  FLIP  •  INSERT BEFORE 2nd FROM RIGHT", fill=NAVY, font=_font(17), anchor="lm")
    draw.rounded_rectangle((205, 164, 225, 188), radius=3, fill=DARK, outline=NAVY, width=2)
    draw.line((60, 205, WIDTH - 60, 205), fill="#d8dee3", width=2)
    draw.text((60, 226), "OPTIONS — complete deck after 8 turns", fill=NAVY, font=_font(17), anchor="lm")
    for label, option, top in zip("ABCDE", options, OPTION_TOPS):
        centre = (175, top + CELL // 2)
        draw.ellipse((centre[0] - 22, centre[1] - 22, centre[0] + 22, centre[1] + 22), fill=NAVY)
        draw.text(centre, label, fill="white", font=_font(25), anchor="mm")
        _draw_deck(draw, option, top)
    return image
