"""Independent pixel-only inverse arm for the checkpoint-gap decision.

This module deliberately contains no plot coordinates, checkpoint count from
renderer state, scene data, or imports from the forward world. It locates marker
columns as peaks in color-mask column density and measures marker row centroids.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_CYAN = np.array((0, 148, 198), dtype=np.int32)
_MAGENTA = np.array((205, 55, 135), dtype=np.int32)


def _mask(rgb: np.ndarray, target: np.ndarray) -> np.ndarray:
    # int32 avoids overflow while squaring 8-bit channel differences.
    delta = rgb.astype(np.int32) - target
    return np.sum(delta * delta, axis=2) <= 38 * 38


def _runs(indices: np.ndarray) -> list[np.ndarray]:
    if indices.size == 0:
        return []
    splits = np.flatnonzero(np.diff(indices) > 1) + 1
    return [part for part in np.split(indices, splits) if part.size]


def _plot_rows(rgb: np.ndarray) -> tuple[int, int]:
    """Infer the plot's horizontal frame lines from long dark raster runs."""

    dark = np.all(rgb < 105, axis=2)
    dense_rows = np.flatnonzero(dark.sum(axis=1) > rgb.shape[1] * 0.45)
    bands = _runs(dense_rows)
    if len(bands) < 2:
        raise ValueError("plot frame was not recovered")
    return int(bands[0].mean()), int(bands[-1].mean())


def _marker_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    column_ink = mask.sum(axis=0)
    # Three-pixel polylines occupy only a few rows per column. Circular markers
    # create broad, high-density peaks; the threshold is derived from the raster.
    positive = column_ink[column_ink > 0]
    if positive.size == 0:
        raise ValueError("colored trace is absent")
    peak_cut = max(10, int(np.percentile(positive, 88)))
    candidate_columns = np.flatnonzero(column_ink >= peak_cut)
    groups = [group for group in _runs(candidate_columns) if group.size >= 3]

    # If a steep line contributes a short extra peak, keep broad/high-ink marker
    # groups and then require an equal five-marker layout for both colors.
    scored: list[tuple[float, np.ndarray]] = []
    for group in groups:
        score = float(column_ink[group].sum())
        scored.append((score, group))
    if len(scored) < 5:
        raise ValueError(f"only {len(scored)} marker peaks recovered")
    chosen = [group for _, group in sorted(scored, key=lambda item: item[0], reverse=True)[:5]]
    chosen.sort(key=lambda group: float(group.mean()))

    centers: list[tuple[float, float]] = []
    for group in chosen:
        center_x = int(round(float(group.mean())))
        left = max(0, center_x - 7)
        right = min(mask.shape[1], center_x + 8)
        rows, cols = np.nonzero(mask[:, left:right])
        # Exclude thin line tails by taking pixels near the dominant row mode.
        if rows.size < 35:
            raise ValueError("marker peak has insufficient colored area")
        hist = np.bincount(rows, minlength=mask.shape[0])
        modal_y = int(hist.argmax())
        keep = np.abs(rows - modal_y) <= 9
        if int(keep.sum()) < 30:
            raise ValueError("marker centroid is unstable")
        centers.append((float((cols[keep] + left).mean()), float(rows[keep].mean())))
    return centers


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    top, bottom = _plot_rows(rgb)
    if bottom - top < rgb.shape[0] * 0.45:
        raise ValueError("recovered plot frame is too small")
    cyan_mask = _mask(rgb, _CYAN)
    magenta_mask = _mask(rgb, _MAGENTA)
    cyan_mask[: top + 2] = False
    cyan_mask[bottom:] = False
    magenta_mask[: top + 2] = False
    magenta_mask[bottom:] = False
    cyan = _marker_centers(cyan_mask)
    magenta = _marker_centers(magenta_mask)
    if len(cyan) != 5 or len(magenta) != 5:
        raise ValueError("expected five markers per colored trace")
    if any(abs(a[0] - b[0]) > 5 for a, b in zip(cyan, magenta)):
        raise ValueError("colored checkpoints do not align")
    gaps = [abs(a[1] - b[1]) for a, b in zip(cyan, magenta)]
    order = sorted(range(5), key=gaps.__getitem__, reverse=True)
    if gaps[order[0]] - gaps[order[1]] < 6:
        raise ValueError("ambiguous top-two pixel gap")
    return f"C{order[0] + 1}"
