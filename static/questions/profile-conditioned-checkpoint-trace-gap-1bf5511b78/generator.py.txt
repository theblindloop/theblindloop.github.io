"""Deterministic paired-trace checkpoint plot renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 480
X_POSITIONS = (100, 220, 340, 460, 580)
CYAN = (0, 148, 198)
MAGENTA = (205, 55, 135)
INK = (37, 43, 52)
GRID = (222, 226, 232)
BACKGROUND = (250, 251, 253)
MARKER_RADIUS = 9
QUARANTINE_PX = 8.0


def _top_two(values: Iterable[float]) -> tuple[float, float]:
    ordered = sorted((float(value) for value in values), reverse=True)
    return ordered[0], ordered[1]


def sample_scene(seed: int) -> dict:
    """Sample five paired heights with a counterbalanced unique winner."""

    rng = random.Random(int(seed))
    winner = int(seed) % 5
    runner = (winner + 1 + rng.randrange(4)) % 5
    delta = 10 + 3 * (abs(int(seed)) % 5)
    gaps = [rng.randrange(30, 91) for _ in range(5)]
    gaps[winner] = 142
    gaps[runner] = 142 - delta
    # Keep every non-runner distinctly below the two leaders.
    for index in range(5):
        if index not in (winner, runner):
            gaps[index] = min(gaps[index], 104)

    cyan_y: list[int] = []
    magenta_y: list[int] = []
    for index, gap in enumerate(gaps):
        midpoint = rng.randrange(155, 276)
        sign = -1 if rng.randrange(2) else 1
        # Alternate one sign deterministically to avoid answer identity becoming
        # tied to which color is higher at the winning checkpoint.
        if index == winner and (int(seed) // 5) % 2:
            sign *= -1
        cyan_y.append(midpoint - sign * gap // 2)
        magenta_y.append(midpoint + sign * (gap - gap // 2))
    return {"cyan_y": cyan_y, "magenta_y": magenta_y}


def render(scene: dict) -> Image.Image:
    cyan_y = [int(value) for value in scene["cyan_y"]]
    magenta_y = [int(value) for value in scene["magenta_y"]]
    if len(cyan_y) != 5 or len(magenta_y) != 5:
        raise ValueError("each trace must have five checkpoint heights")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    # Plot frame and visibly shared checkpoint guides.
    draw.rectangle((70, 48, 610, 380), fill=(255, 255, 255), outline=INK, width=2)
    for x in X_POSITIONS:
        draw.line((x, 49, x, 379), fill=GRID, width=1)
    for y in (100, 180, 260, 340):
        draw.line((71, y, 609, y), fill=(237, 239, 243), width=1)

    cyan_points = list(zip(X_POSITIONS, cyan_y))
    magenta_points = list(zip(X_POSITIONS, magenta_y))
    draw.line(cyan_points, fill=CYAN, width=3, joint="curve")
    draw.line(magenta_points, fill=MAGENTA, width=3, joint="curve")
    for color, points in ((CYAN, cyan_points), (MAGENTA, magenta_points)):
        for x, y in points:
            draw.ellipse(
                (x - MARKER_RADIUS, y - MARKER_RADIUS,
                 x + MARKER_RADIUS, y + MARKER_RADIUS),
                fill=color,
                outline=(255, 255, 255),
                width=2,
            )

    for index, x in enumerate(X_POSITIONS, start=1):
        label = f"C{index}"
        box = draw.textbbox((0, 0), label, font=font)
        draw.text((x - (box[2] - box[0]) / 2, 394), label, fill=INK, font=font)

    draw.text((70, 20), "Paired measurements by checkpoint", fill=INK, font=font)
    draw.line((454, 25, 477, 25), fill=CYAN, width=3)
    draw.ellipse((462, 20, 472, 30), fill=CYAN)
    draw.text((482, 20), "cyan", fill=INK, font=font)
    draw.line((536, 25, 559, 25), fill=MAGENTA, width=3)
    draw.ellipse((544, 20, 554, 30), fill=MAGENTA)
    draw.text((564, 20), "magenta", fill=INK, font=font)
    return image


def _gaps(scene: dict) -> list[float]:
    cyan_y = scene["cyan_y"]
    magenta_y = scene["magenta_y"]
    return [abs(float(a) - float(b)) for a, b in zip(cyan_y, magenta_y)]


def analytic_gold(scene: dict) -> str:
    cyan_y = scene["cyan_y"]
    magenta_y = scene["magenta_y"]
    gaps = [abs(float(a) - float(b)) for a, b in zip(cyan_y, magenta_y)]
    return f"C{max(range(5), key=gaps.__getitem__) + 1}"


def margin(scene: dict) -> float:
    first, second = _top_two(_gaps(scene))
    return first - second


def is_quarantined(scene: dict) -> bool:
    gaps = _gaps(scene)
    if len(gaps) != 5 or any(value < 22 for value in gaps):
        return True
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Visible arrays are the complete label-bearing latent state. Palette swap
    # and vertical reflection are tested as answer-preserving (not pixel-identical)
    # metamorphisms in tests, so they are not latent aliases.
    return []
