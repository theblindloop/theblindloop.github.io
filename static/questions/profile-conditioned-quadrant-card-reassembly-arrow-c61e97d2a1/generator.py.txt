"""Renderer for a visibly keyed quarter-turn card reassembly task."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 620
BACKGROUND = (247, 249, 252)
PAPER = (255, 255, 255)
FRAME = (91, 105, 121)
INK = (20, 38, 63)
GOLD = (242, 171, 35)
DISTRACTOR = (207, 216, 225)

LABELS = ("up", "right", "down", "left")
PALETTES = (
    ((218, 64, 82), (39, 121, 208), (38, 164, 105), (143, 78, 190)),
    ((196, 54, 117), (24, 143, 175), (72, 151, 70), (111, 92, 201)),
    ((229, 74, 70), (50, 105, 196), (21, 155, 132), (156, 72, 167)),
)

CARD_SIZE = 120
CELL = 24
CARD_X = (25, 235, 445, 655)
CARD_Y = 430
MAP_X, MAP_Y, MAP_SIZE = 304, 40, 192


def _blank() -> list[list[int]]:
    return [[0 for _ in range(10)] for _ in range(10)]


def _up_signal() -> list[list[int]]:
    grid = _blank()
    for x in (4, 5):
        grid[1][x] = 1
    for x in range(3, 7):
        grid[2][x] = 1
    for x in range(2, 8):
        grid[3][x] = 1
    for y in range(4, 9):
        grid[y][4] = grid[y][5] = 1
    return grid


def _up_full() -> list[list[int]]:
    grid = _up_signal()
    fillers = (
        (0, 0), (1, 4), (9, 0), (8, 4),
        (0, 5), (2, 5), (1, 7), (0, 9), (2, 9),
        (9, 5), (7, 5), (8, 7), (9, 9), (7, 9),
    )
    for x, y in fillers:
        grid[y][x] = 1
    return grid


def _rotate_cw(grid: list[list[int]], turns: int) -> list[list[int]]:
    out = [row[:] for row in grid]
    for _ in range(turns % 4):
        out = [list(row) for row in zip(*out[::-1])]
    return out


def signal_template(label: str) -> list[list[int]]:
    if label not in LABELS:
        raise ValueError(f"unknown glyph {label!r}")
    return _rotate_cw(_up_signal(), LABELS.index(label))


def full_template(label: str) -> list[list[int]]:
    if label not in LABELS:
        raise ValueError(f"unknown glyph {label!r}")
    return _rotate_cw(_up_full(), LABELS.index(label))


def _hamming(a: list[list[int]], b: list[list[int]]) -> int:
    return sum(x != y for ar, br in zip(a, b, strict=True) for x, y in zip(ar, br, strict=True))


def sample_scene(seed: int) -> dict:
    """Sample a four-way counterbalanced scene.

    Consecutive groups of four seeds share all nuisances and rotate only the
    answer. Thus color, placement, turn, palette, and distractors are exactly
    counterbalanced against answer identity over each complete group.
    """
    label = LABELS[int(seed) % 4]
    rng = random.Random((int(seed) // 4) * 7919 + 1701)
    target_colors = list(range(4))
    source_order = list(range(4))
    draw_order = list(range(4))
    rng.shuffle(target_colors)
    rng.shuffle(source_order)
    rng.shuffle(draw_order)
    turns = [rng.randrange(4) for _ in range(4)]
    palette = [list(rgb) for rgb in PALETTES[rng.randrange(len(PALETTES))]]
    rng.shuffle(palette)
    shift = [rng.randint(-8, 8), rng.randint(-8, 8)]
    anchors = ((54, 70), (746, 72), (58, 286), (742, 292), (204, 298), (596, 302))
    distractors = [
        [x + rng.randint(-9, 9), y + rng.randint(-9, 9), rng.choice((4, 5, 6))]
        for x, y in anchors
    ]
    distractor_order = list(range(len(distractors)))
    rng.shuffle(distractor_order)
    return {
        "glyph": label,
        "target_colors": target_colors,
        "source_order": source_order,
        "turns": turns,
        "palette": palette,
        "draw_order": draw_order,
        "shift": shift,
        "distractors": distractors,
        "distractor_order": distractor_order,
    }


def _fragment(full: list[list[int]], quadrant: int) -> list[list[int]]:
    row0 = 5 * (quadrant // 2)
    col0 = 5 * (quadrant % 2)
    return [row[col0 : col0 + 5] for row in full[row0 : row0 + 5]]


def _draw_badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: Iterable[int]) -> None:
    cx, cy = center
    draw.ellipse((cx - 14, cy - 14, cx + 14, cy + 14), fill=tuple(color))


def _draw_notch(draw: ImageDraw.ImageDraw, x0: int, y0: int, turn: int) -> None:
    x1, y1 = x0 + CARD_SIZE, y0 + CARD_SIZE
    cx, cy = x0 + CARD_SIZE // 2, y0 + CARD_SIZE // 2
    points = {
        0: ((cx - 10, y0 - 5), (cx + 10, y0 - 5), (cx, y0 - 18)),
        1: ((x1 + 5, cy - 10), (x1 + 5, cy + 10), (x1 + 18, cy)),
        2: ((cx - 10, y1 + 5), (cx + 10, y1 + 5), (cx, y1 + 18)),
        3: ((x0 - 5, cy - 10), (x0 - 5, cy + 10), (x0 - 18, cy)),
    }[turn % 4]
    draw.polygon(points, fill=GOLD)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    x0: int,
    y0: int,
    matrix: list[list[int]],
    badge: Iterable[int],
    turn: int,
) -> None:
    draw.rectangle((x0, y0, x0 + CARD_SIZE, y0 + CARD_SIZE), fill=PAPER)
    draw.rectangle((x0, y0, x0 + CARD_SIZE, y0 + CARD_SIZE), outline=FRAME, width=4)
    for index in range(1, 5):
        draw.line((x0 + index * CELL, y0, x0 + index * CELL, y0 + CARD_SIZE), fill=FRAME, width=3)
        draw.line((x0, y0 + index * CELL, x0 + CARD_SIZE, y0 + index * CELL), fill=FRAME, width=3)
    for row in range(5):
        for col in range(5):
            if matrix[row][col]:
                left = x0 + col * CELL + 6
                top = y0 + row * CELL + 6
                draw.rectangle((left, top, left + 12, top + 12), fill=INK)
    _draw_badge(draw, (x0 + CARD_SIZE // 2, y0 - 42), badge)
    _draw_notch(draw, x0, y0, turn)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = (int(value) for value in scene["shift"])

    for index in scene["distractor_order"]:
        x, y, radius = scene["distractors"][index]
        x, y, radius = int(x) + dx, int(y) + dy, int(radius)
        draw.polygon(((x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)), fill=DISTRACTOR)

    map_x, map_y = MAP_X + dx, MAP_Y + dy
    draw.rectangle((map_x, map_y, map_x + MAP_SIZE, map_y + MAP_SIZE), fill=PAPER)
    draw.rectangle((map_x, map_y, map_x + MAP_SIZE, map_y + MAP_SIZE), outline=FRAME, width=5)
    draw.line((map_x + MAP_SIZE // 2, map_y, map_x + MAP_SIZE // 2, map_y + MAP_SIZE), fill=FRAME, width=4)
    draw.line((map_x, map_y + MAP_SIZE // 2, map_x + MAP_SIZE, map_y + MAP_SIZE // 2), fill=FRAME, width=4)
    palette = scene["palette"]
    for quadrant, color_id in enumerate(scene["target_colors"]):
        qx = map_x + MAP_SIZE // 4 + (quadrant % 2) * (MAP_SIZE // 2)
        qy = map_y + MAP_SIZE // 4 + (quadrant // 2) * (MAP_SIZE // 2)
        _draw_badge(draw, (qx, qy), palette[color_id])

    full = full_template(scene["glyph"])
    for slot in scene["draw_order"]:
        color_id = scene["source_order"][slot]
        quadrant = scene["target_colors"].index(color_id)
        canonical = _fragment(full, quadrant)
        turn = int(scene["turns"][color_id])
        displayed = _rotate_cw(canonical, turn)
        _draw_card(
            draw,
            CARD_X[slot] + dx,
            CARD_Y + dy,
            displayed,
            palette[color_id],
            turn,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    label = str(scene["glyph"])
    true = signal_template(label)
    return float(min(_hamming(true, signal_template(other)) for other in LABELS if other != label))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    twin["distractor_order"] = list(reversed(twin["distractor_order"]))
    return [("reverse_disjoint_draw_order", twin)]
