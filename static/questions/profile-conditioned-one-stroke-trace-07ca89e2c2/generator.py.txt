"""One-stroke trace: three grey wire networks, exactly one of which is Eulerian.

Latent scene, deterministic raster, analytic gold and clearance margin.
"""

from __future__ import annotations

import math
import random
from typing import Dict, List, Sequence, Tuple

from PIL import Image, ImageDraw

CANVAS = (960, 300)
SUPERSAMPLE = 3
BACKGROUND = (255, 255, 255)
SEPARATOR_COLOR = (203, 203, 208)
WIRE_COLOR = (106, 106, 114)
HUB_COLOR = (30, 30, 36)
WIRE_WIDTH = 6.0
HUB_RADIUS = 10.0

GRID_COLS = 4
GRID_ROWS = 3
SPACING = 72.0
PANEL_SLOT = 320.0
PANEL_ORIGIN_X = 52.0
PANEL_ORIGIN_Y = 78.0
JITTER = 14.0

SLOT_NAMES = ("left", "centre", "right")
QUARANTINE_MARGIN = 7.0


# --------------------------------------------------------------------------
# graph helpers
# --------------------------------------------------------------------------
def _lattice_edges() -> List[Tuple[int, int]]:
    edges: List[Tuple[int, int]] = []
    for row in range(GRID_ROWS):
        for col in range(GRID_COLS):
            index = row * GRID_COLS + col
            if col + 1 < GRID_COLS:
                edges.append((index, index + 1))
            if row + 1 < GRID_ROWS:
                edges.append((index, index + GRID_COLS))
    return edges


LATTICE_EDGES = _lattice_edges()


def _degrees(edges: Sequence[Sequence[int]]) -> Dict[int, int]:
    degree: Dict[int, int] = {}
    for a, b in edges:
        degree[a] = degree.get(a, 0) + 1
        degree[b] = degree.get(b, 0) + 1
    return degree


def _connected(edges: Sequence[Sequence[int]]) -> bool:
    if not edges:
        return False
    adjacency: Dict[int, List[int]] = {}
    for a, b in edges:
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    start = next(iter(adjacency))
    seen = {start}
    stack = [start]
    while stack:
        node = stack.pop()
        for other in adjacency[node]:
            if other not in seen:
                seen.add(other)
                stack.append(other)
    return len(seen) == len(adjacency)


def _odd_count(edges: Sequence[Sequence[int]]) -> int:
    return sum(1 for value in _degrees(edges).values() if value % 2 == 1)


def _is_one_stroke(edges: Sequence[Sequence[int]]) -> bool:
    """A connected network is traceable in one stroke iff it has 0 or 2 odd hubs."""
    return _connected(edges) and _odd_count(edges) in (0, 2)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _panel_pool(rng: random.Random, n_edges: int) -> Dict[Tuple[int, int, int], Dict[str, list]]:
    pool: Dict[Tuple[int, int, int], Dict[str, list]] = {}
    for _ in range(900):
        edges = tuple(sorted(rng.sample(LATTICE_EDGES, n_edges)))
        if not _connected(edges):
            continue
        degree = _degrees(edges)
        used = sorted(degree)
        signature = (
            len(used),
            sum(1 for value in degree.values() if value == 1),
            max(degree.values()),
        )
        bucket = pool.setdefault(signature, {"one_stroke": [], "blocked": []})
        key = "one_stroke" if _odd_count(edges) in (0, 2) else "blocked"
        if edges not in bucket[key]:
            bucket[key].append(edges)
    return pool


def _jittered_nodes(rng: random.Random, slot: int, used: Sequence[int]) -> Dict[int, List[float]]:
    nodes: Dict[int, List[float]] = {}
    for index in used:
        col = index % GRID_COLS
        row = index // GRID_COLS
        x = slot * PANEL_SLOT + PANEL_ORIGIN_X + col * SPACING + rng.uniform(-JITTER, JITTER)
        y = PANEL_ORIGIN_Y + row * SPACING + rng.uniform(-JITTER, JITTER)
        nodes[index] = [round(x, 3), round(y, 3)]
    return nodes


def sample_scene(seed: int) -> dict:
    rng = random.Random(7919 * int(seed) + 13)
    for _attempt in range(60):
        n_edges = rng.choice([12, 13, 14])
        pool = _panel_pool(rng, n_edges)
        keys = sorted(
            key
            for key, bucket in pool.items()
            if len(bucket["one_stroke"]) >= 1 and len(bucket["blocked"]) >= 2
        )
        if not keys:
            continue
        signature = keys[rng.randrange(len(keys))]
        bucket = pool[signature]
        answer_slot = rng.randrange(3)
        traceable = bucket["one_stroke"][rng.randrange(len(bucket["one_stroke"]))]
        blocked = rng.sample(bucket["blocked"], 2)
        chosen: List[Tuple[Tuple[Tuple[int, int], ...], bool]] = []
        blocked_iter = iter(blocked)
        for slot in range(3):
            if slot == answer_slot:
                chosen.append((traceable, True))
            else:
                chosen.append((next(blocked_iter), False))

        scene = {
            "canvas": list(CANVAS),
            "supersample": SUPERSAMPLE,
            "background": list(BACKGROUND),
            "separator_color": list(SEPARATOR_COLOR),
            "wire_color": list(WIRE_COLOR),
            "hub_color": list(HUB_COLOR),
            "wire_width": WIRE_WIDTH,
            "hub_radius": HUB_RADIUS,
            "panels": [],
        }
        for slot, (edges, _flag) in enumerate(chosen):
            used = sorted(_degrees(edges))
            scene["panels"].append(
                {
                    "slot": slot,
                    "nodes": _jittered_nodes(rng, slot, used),
                    "edges": [list(edge) for edge in edges],
                }
            )
        if margin(scene) >= QUARANTINE_MARGIN + 1.5:
            return scene
        for _retry in range(40):
            for panel in scene["panels"]:
                panel["nodes"] = _jittered_nodes(
                    rng, panel["slot"], sorted(_degrees(panel["edges"]))
                )
            if margin(scene) >= QUARANTINE_MARGIN + 1.5:
                return scene
        return scene
    raise RuntimeError("scene sampling failed")


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _point_segment_distance(p, a, b) -> float:
    ax, ay = a
    bx, by = b
    px, py = p
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom <= 1e-12:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _segment_distance(a1, a2, b1, b2) -> float:
    return min(
        _point_segment_distance(a1, b1, b2),
        _point_segment_distance(a2, b1, b2),
        _point_segment_distance(b1, a1, a2),
        _point_segment_distance(b2, a1, a2),
    )


def _stub_point(center, other, radius):
    dx, dy = other[0] - center[0], other[1] - center[1]
    length = math.hypot(dx, dy) or 1.0
    return (center[0] + dx / length * radius, center[1] + dy / length * radius)


def margin(scene: dict) -> float:
    """Smallest visible clearance in pixels; the decision boundary is a fused or lost wire."""
    hub_r = float(scene["hub_radius"])
    half = float(scene["wire_width"]) / 2.0
    worst = float("inf")
    all_hubs: List[Tuple[int, int, List[float]]] = []
    for panel in scene["panels"]:
        nodes = {int(k): v for k, v in panel["nodes"].items()}
        edges = [tuple(edge) for edge in panel["edges"]]
        for key, position in nodes.items():
            all_hubs.append((panel["slot"], key, position))
        for i, (a, b) in enumerate(edges):
            pa, pb = nodes[a], nodes[b]
            worst = min(worst, math.hypot(pb[0] - pa[0], pb[1] - pa[1]) - 2.0 * hub_r)
            for key, position in nodes.items():
                if key not in (a, b):
                    worst = min(
                        worst, _point_segment_distance(position, pa, pb) - hub_r - half
                    )
            for j in range(i + 1, len(edges)):
                c, d = edges[j]
                pc, pd = nodes[c], nodes[d]
                if {a, b} & {c, d}:
                    shared = ({a, b} & {c, d}).pop()
                    far_a = b if shared == a else a
                    far_b = d if shared == c else c
                    s1 = _stub_point(nodes[shared], nodes[far_a], hub_r + 3.0)
                    s2 = _stub_point(nodes[shared], nodes[far_b], hub_r + 3.0)
                    worst = min(worst, math.hypot(s1[0] - s2[0], s1[1] - s2[1]) - 2.0 * half)
                else:
                    worst = min(worst, _segment_distance(pa, pb, pc, pd) - 2.0 * half)
    for i in range(len(all_hubs)):
        for j in range(i + 1, len(all_hubs)):
            _si, _ki, pi = all_hubs[i]
            _sj, _kj, pj = all_hubs[j]
            worst = min(worst, math.hypot(pi[0] - pj[0], pi[1] - pj[1]) - 2.0 * hub_r)
    return float(round(worst, 4))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


# --------------------------------------------------------------------------
# gold
# --------------------------------------------------------------------------
def _panels_left_to_right(scene: dict) -> List[dict]:
    def mean_x(panel: dict) -> float:
        values = [position[0] for position in panel["nodes"].values()]
        return sum(values) / len(values)

    return sorted(scene["panels"], key=mean_x)


def analytic_gold(scene: dict) -> str:
    """Name the panel, in painted left-to-right order, that can be drawn in one stroke."""
    hits = [
        SLOT_NAMES[index]
        for index, panel in enumerate(_panels_left_to_right(scene))
        if _is_one_stroke([tuple(edge) for edge in panel["edges"]])
    ]
    if len(hits) != 1:
        raise ValueError("scene must contain exactly one one-stroke network")
    return hits[0]


def structural_margin(scene: dict) -> int:
    """Runner-up decision margin: extra odd hubs the nearest distractor still carries."""
    gaps = []
    for panel in _panels_left_to_right(scene):
        edges = [tuple(edge) for edge in panel["edges"]]
        if not _is_one_stroke(edges):
            gaps.append(_odd_count(edges) - 2)
    return int(min(gaps))


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    scale = int(scene["supersample"])
    width, height = scene["canvas"]
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    for index in (1, 2):
        x = index * PANEL_SLOT * scale
        draw.rectangle(
            [x - 1.0 * scale, 18.0 * scale, x + 1.0 * scale, (height - 18.0) * scale],
            fill=tuple(scene["separator_color"]),
        )

    wire_width = max(1, int(round(scene["wire_width"] * scale)))
    for panel in scene["panels"]:
        nodes = {int(k): v for k, v in panel["nodes"].items()}
        for a, b in sorted(tuple(sorted(edge)) for edge in panel["edges"]):
            pa, pb = nodes[a], nodes[b]
            draw.line(
                [pa[0] * scale, pa[1] * scale, pb[0] * scale, pb[1] * scale],
                fill=tuple(scene["wire_color"]),
                width=wire_width,
            )
    radius = scene["hub_radius"] * scale
    for panel in scene["panels"]:
        for _key, position in sorted(
            ((int(k), v) for k, v in panel["nodes"].items()), key=lambda item: item[0]
        ):
            cx, cy = position[0] * scale, position[1] * scale
            draw.ellipse(
                [cx - radius, cy - radius, cx + radius, cy + radius],
                fill=tuple(scene["hub_color"]),
            )
    return image.resize((width, height), Image.LANCZOS)


# --------------------------------------------------------------------------
# latent aliases
# --------------------------------------------------------------------------
def _copy(scene: dict) -> dict:
    clone = dict(scene)
    clone["panels"] = [
        {
            "slot": panel["slot"],
            "nodes": {k: list(v) for k, v in panel["nodes"].items()},
            "edges": [list(edge) for edge in panel["edges"]],
        }
        for panel in scene["panels"]
    ]
    return clone


def latent_symmetries(scene: dict) -> List[Tuple[str, dict]]:
    """Pixel-identical latent permutations that must leave the answer untouched."""
    out: List[Tuple[str, dict]] = []

    rotated = _copy(scene)
    rotated["panels"] = rotated["panels"][1:] + rotated["panels"][:1]
    out.append(("panel_list_rotation", rotated))

    reversed_edges = _copy(scene)
    for panel in reversed_edges["panels"]:
        panel["edges"] = list(reversed(panel["edges"]))
    out.append(("edge_order_reversal", reversed_edges))

    swapped = _copy(scene)
    for panel in swapped["panels"]:
        panel["edges"] = [[b, a] for a, b in panel["edges"]]
    out.append(("edge_endpoint_swap", swapped))

    reordered = _copy(scene)
    for panel in reordered["panels"]:
        items = list(panel["nodes"].items())
        panel["nodes"] = dict(reversed(items))
    out.append(("hub_draw_order_permutation", reordered))
    return out
