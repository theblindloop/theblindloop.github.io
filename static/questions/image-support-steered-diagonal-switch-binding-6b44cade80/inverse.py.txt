"""Independent PNG-only inverse for diagonal side-switch binding."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


LABEL_COLORS = {
    "amber": np.asarray((225, 150, 35), dtype=np.float32),
    "teal": np.asarray((25, 150, 145), dtype=np.float32),
    "violet": np.asarray((135, 75, 185), dtype=np.float32),
}
LABELS = ("amber", "teal", "violet")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    found: list[np.ndarray] = []
    while points:
        start = points.pop()
        queue = deque([start])
        component = [start]
        while queue:
            y, x = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in points:
                        points.remove(neighbor)
                        queue.append(neighbor)
                        component.append(neighbor)
        found.append(np.asarray(component, dtype=np.int32))
    return found


def _color_centroids(array: np.ndarray) -> dict[str, list[tuple[float, float]]]:
    result: dict[str, list[tuple[float, float]]] = {}
    pixels = array.astype(np.float32)
    spread = pixels.max(axis=2) - pixels.min(axis=2)
    for label, reference in LABEL_COLORS.items():
        distance = np.linalg.norm(pixels - reference[None, None, :], axis=2)
        mask = (distance < 105.0) & (spread > 45.0)
        comps = [item for item in _components(mask) if len(item) >= 45]
        if len(comps) != 2:
            raise ValueError(f"expected two {label} samples, recovered {len(comps)}")
        centroids = []
        for component in comps:
            y, x = component.mean(axis=0)
            centroids.append((float(x), float(y)))
        result[label] = sorted(centroids)
    return result


def _line_fits(array: np.ndarray) -> list[tuple[float, float, float, float]]:
    pixels = array.astype(np.float32)
    spread = pixels.max(axis=2) - pixels.min(axis=2)
    intensity = pixels.mean(axis=2)
    mask = (spread < 42.0) & (intensity < 178.0)
    candidates = []
    for component in _components(mask):
        if len(component) < 18:
            continue
        ys = component[:, 0].astype(np.float64)
        xs = component[:, 1].astype(np.float64)
        if xs.max() - xs.min() < 5 or ys.max() - ys.min() < 5:
            continue
        candidates.append(component)
    if len(candidates) < 2:
        raise ValueError(f"expected two diagonal line components, recovered {len(candidates)}")
    midpoint = array.shape[1] / 2.0
    groups = [
        np.concatenate([item for item in candidates if float(item[:, 1].mean()) < midpoint], axis=0),
        np.concatenate([item for item in candidates if float(item[:, 1].mean()) >= midpoint], axis=0),
    ]
    if any(len(group) < 80 for group in groups):
        raise ValueError("one diagonal has too little visible support")
    fits = []
    for group in groups:
        ys = group[:, 0].astype(np.float64)
        xs = group[:, 1].astype(np.float64)
        slope, intercept = np.polyfit(xs, ys, 1)
        fits.append((float(slope), float(intercept), float(xs.mean()), float(len(group))))
    return fits


def _side(line: tuple[float, float, float, float], point: tuple[float, float]) -> str:
    slope, intercept, _, _ = line
    return "below" if point[1] > slope * point[0] + intercept else "above"


def decision_from_image(image: Image.Image) -> str:
    """Recover color-keyed half-plane status from raster pixels only."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centroids = _color_centroids(array)
    lines = _line_fits(array)
    midpoint = (lines[0][2] + lines[1][2]) / 2.0
    left_line, right_line = (lines if lines[0][2] < lines[1][2] else lines[::-1])
    statuses: dict[str, tuple[str, str]] = {}
    for label in LABELS:
        left_points = [point for point in centroids[label] if point[0] < midpoint]
        right_points = [point for point in centroids[label] if point[0] >= midpoint]
        if len(left_points) != 1 or len(right_points) != 1:
            raise ValueError(f"could not bind exactly one {label} sample to each card")
        statuses[label] = (_side(left_line, left_points[0]), _side(right_line, right_points[0]))
    winners = [label for label, sides in statuses.items() if sides == ("below", "above")]
    if len(winners) != 1:
        raise ValueError(f"expected one side switch, recovered {winners}")
    return winners[0]
