"""Renderer and analytic reference for diagonal side-switch binding."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 512
HEIGHT = 512
LABELS = ("amber", "teal", "violet")
LABEL_COLORS = {
    "amber": (225, 150, 35),
    "teal": (25, 150, 145),
    "violet": (135, 75, 185),
}
LINE_COLORS = ((42, 53, 64), (49, 58, 69), (37, 48, 59))
CARD_COLORS = ((247, 250, 252), (252, 250, 245), (248, 247, 252))
BACKGROUNDS = ((255, 255, 255), (249, 251, 253), (252, 250, 246))


def _line_y(line: list[list[int]], x: float) -> float:
    (x0, y0), (x1, y1) = line
    if x1 == x0:
        return (float(y0) + float(y1)) / 2.0
    return float(y0) + (float(y1) - float(y0)) * ((float(x) - x0) / (x1 - x0))


def _side(line: list[list[int]], sample: dict[str, Any]) -> str:
    return "below" if float(sample["y"]) > _line_y(line, float(sample["x"])) else "above"


def _distance_to_line(line: list[list[int]], sample: dict[str, Any]) -> float:
    (x0, y0), (x1, y1) = line
    x = float(sample["x"])
    y = float(sample["y"])
    numerator = abs((y1 - y0) * x - (x1 - x0) * y + x1 * y0 - y1 * x0)
    denominator = math.hypot(y1 - y0, x1 - x0)
    return numerator / denominator


def _switching_label(scene: dict[str, Any]) -> str:
    left = {str(item["color"]): _side(scene["left_line"], item) for item in scene["left_samples"]}
    right = {str(item["color"]): _side(scene["right_line"], item) for item in scene["right_samples"]}
    winners = [label for label in LABELS if left.get(label) == "below" and right.get(label) == "above"]
    if len(winners) != 1:
        raise ValueError("scene does not contain a unique below-to-above switch")
    return winners[0]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced paired-card scene from one integer seed."""

    rng = random.Random(int(seed))
    winner = int(seed) % 3
    near_boundary = int(seed) % 11 == 0
    card_x = rng.choice((120, 126, 132, 138))
    card_y = rng.choice((142, 154, 166, 178, 190))
    card_width = rng.choice((76, 80, 84))
    card_height = rng.choice((148, 152, 156))
    card_gap = rng.choice((104, 110, 116))
    right_x = card_x + card_width + card_gap
    # Endpoint margins leave a generous visible line segment for the inverse fit.
    left_tilt = rng.choice((-10, -6, 0, 6, 10))
    right_tilt = rng.choice((-10, -6, 0, 6, 10))
    left_line = [
        [card_x + 14, card_y + card_height - 52 + left_tilt],
        [card_x + card_width - 14, card_y + 52 - left_tilt],
    ]
    right_line = [
        [right_x + 14, card_y + card_height - 52 + right_tilt],
        [right_x + card_width - 14, card_y + 52 - right_tilt],
    ]

    x_fractions = [0.24, 0.50, 0.76]
    left_labels = list(LABELS)
    right_labels = list(LABELS)
    rng.shuffle(left_labels)
    rng.shuffle(right_labels)
    left_slots = list(range(3))
    right_slots = list(range(3))
    rng.shuffle(left_slots)
    rng.shuffle(right_slots)
    signs = {
        winner: (1, -1),
        (winner + 1) % 3: (-1, -1),
        (winner + 2) % 3: (1, 1),
    }
    regular_offsets = [28, 32, 36, 40, 44]
    offset_by_label = {label: rng.choice(regular_offsets) for label in LABELS}
    if near_boundary:
        offset_by_label[rng.choice(LABELS)] = rng.choice((11, 12, 13))
    dot_radius = rng.choice((6, 7, 8))
    jitter = [-4, 0, 4]
    left_samples: list[dict[str, Any]] = []
    right_samples: list[dict[str, Any]] = []
    for label_index, label in enumerate(left_labels):
        slot = left_slots[label_index]
        x = int(round(card_x + 14 + x_fractions[slot] * (card_width - 28) + jitter[rng.randrange(3)]))
        base_y = _line_y(left_line, x)
        sign = signs[LABELS.index(label)][0]
        left_samples.append({"color": label, "x": x, "y": int(round(base_y + sign * offset_by_label[label]))})
    for label_index, label in enumerate(right_labels):
        slot = right_slots[label_index]
        x = int(round(right_x + 14 + x_fractions[slot] * (card_width - 28) + jitter[rng.randrange(3)]))
        base_y = _line_y(right_line, x)
        sign = signs[LABELS.index(label)][1]
        right_samples.append({"color": label, "x": x, "y": int(round(base_y + sign * offset_by_label[label]))})
    return {
        "left_line": left_line,
        "right_line": right_line,
        "left_samples": left_samples,
        "right_samples": right_samples,
        "card_x": int(card_x),
        "card_y": int(card_y),
        "card_width": int(card_width),
        "card_height": int(card_height),
        "card_gap": int(card_gap),
        "line_color": list(rng.choice(LINE_COLORS)),
        "card_color": list(rng.choice(CARD_COLORS)),
        "background": list(rng.choice(BACKGROUNDS)),
        "dot_radius": int(dot_radius),
        "stroke_width": int(rng.choice((4, 5))),
    }


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the two cards with supersampling for stable pixel geometry."""

    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    card_x = int(scene["card_x"])
    card_y = int(scene["card_y"])
    card_w = int(scene["card_width"])
    card_h = int(scene["card_height"])
    card_gap = int(scene["card_gap"])
    right_x = card_x + card_w + card_gap
    card_fill = tuple(scene["card_color"])
    line_color = tuple(scene["line_color"])
    for x, letter in ((card_x, "P"), (right_x, "Q")):
        draw.rounded_rectangle(
            (x * scale, card_y * scale, (x + card_w) * scale, (card_y + card_h) * scale),
            radius=8 * scale,
            fill=card_fill,
            outline=(187, 198, 208),
            width=2 * scale,
        )
        draw.text(((x + 7) * scale, (card_y + 5) * scale), letter, fill=line_color, font=_font())
    for line in (scene["left_line"], scene["right_line"]):
        draw.line(
            [(int(point[0]) * scale, int(point[1]) * scale) for point in line],
            fill=line_color,
            width=int(scene["stroke_width"]) * scale,
        )
    radius = int(scene["dot_radius"]) * scale
    for sample in [*scene["left_samples"], *scene["right_samples"]]:
        x = int(sample["x"]) * scale
        y = int(sample["y"]) * scale
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=LABEL_COLORS[str(sample["color"])])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return _switching_label(scene)


def margin(scene: dict[str, Any]) -> float:
    distances = [
        _distance_to_line(scene["left_line"], item) for item in scene["left_samples"]
    ] + [
        _distance_to_line(scene["right_line"], item) for item in scene["right_samples"]
    ]
    return float(min(distances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        _switching_label(scene)
    except (KeyError, TypeError, ValueError, IndexError):
        return True
    return margin(scene) < 14.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
