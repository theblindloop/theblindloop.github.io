"""Independent scene sampler, analytic answer, and raster renderer."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 340
SCALE = 3
LABELS = ("A", "B", "C", "D")
NODE_COLORS = ((18, 151, 147), (207, 52, 129), (54, 105, 208), (236, 177, 35))
ORANGE = (232, 91, 24)
INK = (30, 38, 51)
LIGHT = (164, 174, 187)
PANEL_X = (25, 245, 465, 685)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 4
    target_direction = rng.randrange(2)
    panels: list[list[int]] = []
    distractor_patterns = []
    for ones in (1, 2, 3):
        bits = [1] * ones + [0] * (4 - ones)
        rng.shuffle(bits)
        distractor_patterns.append(bits)
    rng.shuffle(distractor_patterns)
    cursor = 0
    for panel in range(4):
        if panel == target:
            panels.append([target_direction] * 4)
        else:
            panels.append(distractor_patterns[cursor])
            cursor += 1
    color_orders = []
    jitters = []
    for _ in range(4):
        order = list(range(4))
        rng.shuffle(order)
        color_orders.append(order)
        jitters.append([rng.randint(-3, 3), rng.randint(-2, 2)])
    return {"directions": panels, "color_orders": color_orders, "jitters": jitters}


def _scores(directions: list[list[int]]) -> list[int]:
    return [max(sum(bits), 4 - sum(bits)) if len(bits) == 4 else -99 for bits in directions]


def analytic_gold(scene: dict) -> str:
    directions = scene["directions"]
    winners = [i for i, bits in enumerate(directions) if len(bits) == 4 and len(set(bits)) == 1]
    if len(winners) != 1:
        raise ValueError("scene does not contain exactly one directed cycle")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene["directions"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) <= 0 or analytic_gold(scene) not in LABELS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[float, float], end: tuple[float, float]) -> None:
    sx, sy = start
    ex, ey = end
    dx, dy = ex - sx, ey - sy
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    line_a = (sx + ux * 20 * SCALE, sy + uy * 20 * SCALE)
    line_b = (ex - ux * 20 * SCALE, ey - uy * 20 * SCALE)
    draw.line((*line_a, *line_b), fill=LIGHT, width=5 * SCALE)
    cx, cy = sx + dx * 0.70, sy + dy * 0.70
    tip = (cx + ux * 10 * SCALE, cy + uy * 10 * SCALE)
    back = (cx - ux * 8 * SCALE, cy - uy * 8 * SCALE)
    left = (back[0] + px * 8 * SCALE, back[1] + py * 8 * SCALE)
    right = (back[0] - px * 8 * SCALE, back[1] - py * 8 * SCALE)
    draw.polygon((tip, left, right), fill=ORANGE)


def render(scene: dict) -> Image.Image:
    directions = scene["directions"]
    color_orders = scene["color_orders"]
    jitters = scene["jitters"]
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (250, 250, 248))
    draw = ImageDraw.Draw(canvas)
    draw.text(
        (WIDTH * SCALE // 2, 18 * SCALE),
        "Which panel forms one directed loop through all four circles?",
        font=_font(21 * SCALE), fill=INK, anchor="ma"
    )
    draw.text(
        (WIDTH * SCALE // 2, 48 * SCALE),
        "Follow the orange arrowheads; visit each circle once and return.",
        font=_font(14 * SCALE), fill=(75, 83, 96), anchor="ma"
    )
    for index, base_x in enumerate(PANEL_X):
        jx, jy = jitters[index]
        left, top = (base_x + jx) * SCALE, (86 + jy) * SCALE
        right, bottom = (base_x + 190 + jx) * SCALE, (292 + jy) * SCALE
        draw.rounded_rectangle((left, top, right, bottom), radius=10 * SCALE, outline=(120, 130, 145), width=2 * SCALE, fill=(255, 255, 255))
        points = [
            ((base_x + 43 + jx) * SCALE, (128 + jy) * SCALE),
            ((base_x + 147 + jx) * SCALE, (128 + jy) * SCALE),
            ((base_x + 147 + jx) * SCALE, (235 + jy) * SCALE),
            ((base_x + 43 + jx) * SCALE, (235 + jy) * SCALE),
        ]
        edges = ((0, 1), (1, 2), (2, 3), (3, 0))
        for bit, (a, b) in zip(directions[index], edges):
            source, dest = (a, b) if bit else (b, a)
            _arrow(draw, points[source], points[dest])
        for node, point in enumerate(points):
            radius = 13 * SCALE
            color = NODE_COLORS[color_orders[index][node]]
            draw.ellipse((point[0] - radius, point[1] - radius, point[0] + radius, point[1] + radius), fill=color)
        draw.text(((base_x + 95 + jx) * SCALE, (307 + jy) * SCALE), LABELS[index], font=_font(18 * SCALE), fill=INK, anchor="ma")
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def palette_permutation(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["color_orders"] = [row[1:] + row[:1] for row in twin["color_orders"]]
    return twin


def distractor_permutation(scene: dict) -> dict:
    twin = deepcopy(scene)
    winner = LABELS.index(analytic_gold(scene))
    distractors = [i for i in range(4) if i != winner]
    a, b = distractors[:2]
    for key in ("directions", "color_orders", "jitters"):
        twin[key][a], twin[key][b] = twin[key][b], twin[key][a]
    return twin
