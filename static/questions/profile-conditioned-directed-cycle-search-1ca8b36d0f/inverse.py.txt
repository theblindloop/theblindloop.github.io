"""Pixel-only inverse arm. This module intentionally imports no world code."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
NODE_COLORS = np.array(((18, 151, 147), (207, 52, 129), (54, 105, 208), (236, 177, 35)), dtype=np.int16)
ORANGE = np.array((232, 91, 24), dtype=np.int16)


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return found


def _nearest_color_mask(rgb: np.ndarray, colors: np.ndarray, max_distance: float) -> np.ndarray:
    pixels = rgb[:, :, None, :].astype(np.int32)
    palette = colors[None, None, :, :].astype(np.int32)
    distances = ((pixels - palette) ** 2).sum(axis=3)
    return distances.min(axis=2) <= max_distance * max_distance


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    node_mask = _nearest_color_mask(rgb, NODE_COLORS, 46.0)
    orange_mask = _nearest_color_mask(rgb, ORANGE.reshape(1, 3), 42.0)
    nodes = _components(node_mask, 260)
    heads = _components(orange_mask, 65)
    if len(nodes) != 16 or len(heads) != 16:
        raise ValueError(f"expected 16 nodes and arrowheads, found {len(nodes)} and {len(heads)}")
    nodes.sort(key=lambda item: item[0])
    panels = [nodes[i * 4:(i + 1) * 4] for i in range(4)]
    winners = []
    for panel_index, panel_nodes in enumerate(panels):
        xs = [p[0] for p in panel_nodes]
        panel_heads = [h for h in heads if min(xs) - 15 <= h[0] <= max(xs) + 15]
        if len(panel_heads) != 4:
            raise ValueError("could not assign four arrowheads to each panel")
        cx = sum(p[0] for p in panel_nodes) / 4
        cy = sum(p[1] for p in panel_nodes) / 4
        ordered = [None, None, None, None]
        for node in panel_nodes:
            slot = (0 if node[1] < cy else 3) if node[0] < cx else (1 if node[1] < cy else 2)
            ordered[slot] = node
        if any(node is None for node in ordered):
            raise ValueError("node quadrant assignment failed")
        edges = {frozenset((0, 1)), frozenset((1, 2)), frozenset((2, 3)), frozenset((3, 0))}
        indegree = [0, 0, 0, 0]
        outdegree = [0, 0, 0, 0]
        used = set()
        for head in panel_heads:
            ranked = sorted(range(4), key=lambda i: (head[0] - ordered[i][0]) ** 2 + (head[1] - ordered[i][1]) ** 2)
            dest, source = ranked[0], ranked[1]
            edge = frozenset((source, dest))
            if edge not in edges or edge in used:
                raise ValueError("arrowhead does not identify one unique perimeter edge")
            used.add(edge)
            outdegree[source] += 1
            indegree[dest] += 1
        if indegree == [1, 1, 1, 1] and outdegree == [1, 1, 1, 1]:
            winners.append(panel_index)
    if len(winners) != 1:
        raise ValueError("pixel evidence does not yield one directed-cycle panel")
    return LABELS[winners[0]]
