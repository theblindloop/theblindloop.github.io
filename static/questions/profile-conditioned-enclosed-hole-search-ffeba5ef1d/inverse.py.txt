"""Independent pixel-only inverse for the enclosed-hole search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANVAS_SIZE = (960, 360)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (16, 18, 244, 342),
    (252, 18, 480, 342),
    (488, 18, 716, 342),
    (724, 18, 952, 342),
)


class OracleAmbiguous(ValueError):
    """Raised when a raster does not expose one measurable hole winner."""


def _dark_crop(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    crop = array[y0 + 50 : y1 - 16, x0 + 20 : x1 - 20]
    dark = np.max(crop, axis=2) < 80
    ys, xs = np.nonzero(dark)
    if len(ys) == 0:
        raise OracleAmbiguous("panel contains no dark candidate shape")
    return dark[int(ys.min()) : int(ys.max()) + 1, int(xs.min()) : int(xs.max()) + 1]


def _connected_components(mask: np.ndarray) -> int:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            count += 1
            visited[y, x] = True
            queue: deque[tuple[int, int]] = deque([(x, y)])
            while queue:
                cx, cy = queue.popleft()
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((nx, ny))
    return count


def _hole_count(mask: np.ndarray) -> int:
    """Count background components not connected to the crop boundary."""

    background = np.pad(~mask, 1, constant_values=True)
    height, width = background.shape
    visited = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if background[y, x] and not visited[y, x]:
                visited[y, x] = True
                queue.append((x, y))
    for y in range(height):
        for x in (0, width - 1):
            if background[y, x] and not visited[y, x]:
                visited[y, x] = True
                queue.append((x, y))
    while queue:
        x, y = queue.popleft()
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nx, ny = x + dx, y + dy
            if 0 <= nx < width and 0 <= ny < height and background[ny, nx] and not visited[ny, nx]:
                visited[ny, nx] = True
                queue.append((nx, ny))
    holes = 0
    for y in range(height):
        for x in range(width):
            if not background[y, x] or visited[y, x]:
                continue
            holes += 1
            visited[y, x] = True
            queue.append((x, y))
            while queue:
                cx, cy = queue.popleft()
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if 0 <= nx < width and 0 <= ny < height and background[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((nx, ny))
    return holes


def decision_from_image(image: Image.Image) -> str:
    """Measure all four panels from the final RGB pixels and select the hole."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    scores: list[int] = []
    for box in PANEL_BOXES:
        shape = _dark_crop(array, box)
        if _connected_components(shape) != 1:
            raise OracleAmbiguous("candidate silhouette is not one connected component")
        scores.append(_hole_count(shape))
    best = max(scores)
    if scores.count(best) != 1 or best <= 0:
        raise OracleAmbiguous(f"hole scores are ambiguous: {scores}")
    return LABELS[scores.index(best)]
