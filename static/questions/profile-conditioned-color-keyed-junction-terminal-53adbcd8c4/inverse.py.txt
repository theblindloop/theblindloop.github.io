from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

GRAY = (92, 98, 106)
GREEN = (34, 155, 84)
LINK_RGB = {"cyan": (0, 151, 190), "magenta": (202, 54, 143)}
TERMINAL_RGB = {
    "orange": (232, 126, 34), "purple": (126, 78, 181),
    "blue": (48, 103, 206), "red": (213, 62, 55),
}


def _components(mask: np.ndarray, minimum: int = 1):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))]); seen[y, x] = True; pts = []
        while q:
            cy, cx = q.popleft(); pts.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; q.append((ny, nx))
        if len(pts) >= minimum:
            out.append(np.asarray(pts, dtype=np.int32))
    return out


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 3) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - np.array(rgb, dtype=np.int16)), axis=2) <= tolerance


def _center(component: np.ndarray) -> tuple[float, float]:
    return float(component[:, 1].mean()), float(component[:, 0].mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    nodes: dict[str, tuple[float, float]] = {}
    gray = _components(_mask(arr, GRAY), 500)
    if len(gray) != 3:
        raise ValueError(f"abstain: expected 3 junctions, found {len(gray)}")
    for i, comp in enumerate(gray):
        nodes[f"j{i}"] = _center(comp)
    roots = _components(_mask(arr, GREEN), 500)
    if len(roots) != 1:
        raise ValueError("abstain: missing or ambiguous green start ring")
    nodes["start"] = _center(roots[0])
    terminal_name: dict[str, str] = {}
    for name, rgb in TERMINAL_RGB.items():
        comps = _components(_mask(arr, rgb), 500)
        if len(comps) != 1:
            raise ValueError(f"abstain: terminal {name} missing or ambiguous")
        node = f"t_{name}"
        nodes[node] = _center(comps[0]); terminal_name[node] = name

    # Small center disks provide keys; large disconnected components are links.
    keys: dict[str, str] = {}
    edge_components: list[tuple[str, np.ndarray]] = []
    for color, rgb in LINK_RGB.items():
        for comp in _components(_mask(arr, rgb), 40):
            if len(comp) >= 350:
                edge_components.append((color, comp))
        for node, (x, y) in nodes.items():
            if not node.startswith("j"):
                continue
            yy, xx = int(round(y)), int(round(x))
            patch = _mask(arr[max(0,yy-5):yy+6, max(0,xx-5):xx+6], rgb)
            if patch.size and patch.mean() > 0.55:
                if node in keys:
                    raise ValueError("abstain: ambiguous junction key")
                keys[node] = color
    if len(keys) != 3 or len(edge_components) != 7:
        raise ValueError(f"abstain: found {len(keys)} keys and {len(edge_components)} links")

    graph: dict[str, list[tuple[str, str]]] = {n: [] for n in nodes}
    node_items = list(nodes.items())
    for color, comp in edge_components:
        py = comp[:, 0].astype(float); px = comp[:, 1].astype(float)
        distances = []
        for name, (x, y) in node_items:
            d2 = np.min((px - x) ** 2 + (py - y) ** 2)
            distances.append((float(d2), name))
        distances.sort()
        (d1, a), (d2, b) = distances[:2]
        if d1 > 900 or d2 > 900 or a == b:
            raise ValueError("abstain: link endpoints unresolved")
        graph[a].append((b, color)); graph[b].append((a, color))

    prev, cur = None, "start"
    for _ in range(8):
        if cur in terminal_name:
            return terminal_name[cur]
        options = [(n, c) for n, c in graph[cur] if n != prev]
        if cur == "start":
            if len(options) != 1:
                raise ValueError("abstain: start degree is not one")
            nxt = options[0][0]
        else:
            if cur not in keys:
                raise ValueError("abstain: route reached an unkeyed nonterminal")
            matches = [n for n, c in options if c == keys[cur]]
            if len(matches) != 1:
                raise ValueError("abstain: key does not uniquely match a forward link")
            nxt = matches[0]
        prev, cur = cur, nxt
    raise ValueError("abstain: traversal did not terminate")
