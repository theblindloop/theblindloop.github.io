from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 520
BG = (250, 248, 242)
DARK = (48, 52, 58)
GRAY = (92, 98, 106)
GREEN = (34, 155, 84)
LINK_RGB = {"cyan": (0, 151, 190), "magenta": (202, 54, 143)}
TERMINAL_RGB = {
    "orange": (232, 126, 34),
    "purple": (126, 78, 181),
    "blue": (48, 103, 206),
    "red": (213, 62, 55),
}
ANSWERS = tuple(TERMINAL_RGB)


def _positions(mirror: bool, offset: list[int]) -> dict[str, list[int]]:
    raw = {
        "start": (62, 260), "j0": (184, 260), "t0": (288, 82),
        "j1": (322, 260), "t1": (418, 438), "j2": (462, 260),
        "t2": (638, 116), "t3": (638, 404),
    }
    dx, dy = offset
    return {k: [x + dx, (HEIGHT - y if mirror else y) + dy] for k, (x, y) in raw.items()}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1009 + 17)
    target = (seed // 4) % 4
    answer = ANSWERS[seed % 4]
    other = [c for c in ANSWERS if c != answer]
    rng.shuffle(other)
    terminals = other[:]
    terminals.insert(target, answer)
    colors = []
    keys = []
    desired = [0 if target == 0 else 1, 0 if target == 1 else 1, 0 if target == 2 else 1]
    for pick in desired:
        pair = ["cyan", "magenta"]
        rng.shuffle(pair)
        colors.append(pair)
        keys.append(pair[pick])
    scene = {
        "seed": seed,
        "nodes": _positions(rng.choice([False, True]), [rng.randint(-10, 10), rng.randint(-8, 8)]),
        "edges": [
            {"a": "start", "b": "j0", "color": rng.choice(["cyan", "magenta"])},
            {"a": "j0", "b": "t0", "color": colors[0][0]},
            {"a": "j0", "b": "j1", "color": colors[0][1]},
            {"a": "j1", "b": "t1", "color": colors[1][0]},
            {"a": "j1", "b": "j2", "color": colors[1][1]},
            {"a": "j2", "b": "t2", "color": colors[2][0]},
            {"a": "j2", "b": "t3", "color": colors[2][1]},
        ],
        "keys": {"j0": keys[0], "j1": keys[1], "j2": keys[2]},
        "terminal_colors": {f"t{i}": terminals[i] for i in range(4)},
        "palette": {"cyan": list(LINK_RGB["cyan"]), "magenta": list(LINK_RGB["magenta"])},
        "offset": [0, 0],
    }
    return scene


def _terminal_node(scene: dict) -> str:
    graph: dict[str, list[tuple[str, str]]] = {}
    for edge in scene["edges"]:
        graph.setdefault(edge["a"], []).append((edge["b"], edge["color"]))
        graph.setdefault(edge["b"], []).append((edge["a"], edge["color"]))
    prev, cur = None, "start"
    for _ in range(8):
        if cur in scene["terminal_colors"]:
            return cur
        options = [(n, c) for n, c in graph[cur] if n != prev]
        if cur == "start":
            nxt = options[0][0]
        else:
            matches = [n for n, c in options if c == scene["keys"][cur]]
            if len(matches) != 1:
                raise ValueError("junction key does not uniquely match a forward edge")
            nxt = matches[0]
        prev, cur = cur, nxt
    raise ValueError("route did not terminate")


def analytic_gold(scene: dict) -> str:
    return scene["terminal_colors"][_terminal_node(scene)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    pos = scene["nodes"]
    palette = {k: tuple(v) for k, v in scene["palette"].items()}
    for edge in scene["edges"]:
        a, b = pos[edge["a"]], pos[edge["b"]]
        draw.line([tuple(a), tuple(b)], fill=DARK, width=16)
        draw.line([tuple(a), tuple(b)], fill=palette[edge["color"]], width=9)
    # Draw terminals first, then junctions, so strokes stop visibly at markers.
    for node, name in scene["terminal_colors"].items():
        x, y = pos[node]
        draw.rounded_rectangle((x - 20, y - 20, x + 20, y + 20), radius=4,
                               fill=TERMINAL_RGB[name], outline=DARK, width=4)
    x, y = pos["start"]
    draw.ellipse((x - 19, y - 19, x + 19, y + 19), fill=BG, outline=GREEN, width=8)
    draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=GREEN)
    for node in ("j0", "j1", "j2"):
        x, y = pos[node]
        draw.ellipse((x - 20, y - 20, x + 20, y + 20), fill=GRAY, outline=DARK, width=3)
        draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=palette[scene["keys"][node]], outline=BG, width=2)
    return image


def margin(scene: dict) -> float:
    a, b = scene["palette"]["cyan"], scene["palette"]["magenta"]
    return round(sum((x - y) ** 2 for x, y in zip(a, b)) ** 0.5, 3)


def is_quarantined(scene: dict) -> bool:
    try:
        _terminal_node(scene)
    except (KeyError, ValueError):
        return True
    return margin(scene) < 100.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def robustness_transforms(scene: dict) -> list[tuple[str, dict]]:
    shifted = copy.deepcopy(scene)
    for point in shifted["nodes"].values():
        point[0] += 5
        point[1] -= 4
    swapped = copy.deepcopy(scene)
    swapped["palette"]["cyan"], swapped["palette"]["magenta"] = (
        swapped["palette"]["magenta"], swapped["palette"]["cyan"]
    )
    return [("translation", shifted), ("link_key_palette_swap", swapped)]
