"""Renderer for the bead-ring cyclic-order correspondence world.

A dark board is littered with decorative tile clusters. Somewhere on the board
sits one small cream card. The card carries three rings of six coloured beads:

* the KEY ring at the top of the card;
* two DIAL rings below it, one left and one right, each holding exactly the same
  six bead colours as the key ring;
* one wide coloured BAR under each dial ring.

One dial ring is the key ring rotated by a visible angle, so its clockwise bead
sequence is a rotation of the key's clockwise sequence. The other dial ring is
the key ring reflected, so its clockwise sequence is the key's read backwards.
Reflection preserves the multiset of bead colours and every unordered
neighbouring pair, so only the *chirality* of the cyclic order separates the two
dials. The answer is the colour of the bar under the rotated dial.

All decisive evidence lives inside one small cream card whose position is
randomised over the board; everything outside the card is decoration.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "beadring_cyclic_order_bind-0.1.0"

CANVAS_W = 640
CANVAS_H = 480

BEADS_PER_RING = 6
BEAD = 7
RING_R = 14
RING_SPAN = 2 * RING_R + BEAD  # 35
H_GAP = 18
V_GAP = 20
BAND_W = 24
BAND_H = 7
BAND_GAP = 6

# card-inner-relative anchors
LEFT_OX = 0
RIGHT_OX = RING_SPAN + H_GAP  # 53
CONTENT_W = RIGHT_OX + RING_SPAN  # 88
KEY_OX = (CONTENT_W - RING_SPAN) // 2  # 26
KEY_OY = 0
DIAL_OY = RING_SPAN + V_GAP  # 55
BAND_OY = DIAL_OY + RING_SPAN + BAND_GAP  # 96
CONTENT_H = BAND_OY + BAND_H  # 103

PANEL_PAD = 8
PANEL_W = CONTENT_W + 2 * PANEL_PAD  # 104
PANEL_H = CONTENT_H + 2 * PANEL_PAD  # 119

BOARD = (26, 32, 44)
CREAM = (240, 232, 208)

COLOR_NAMES = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (196, 52, 52),
    "blue": (44, 92, 196),
    "green": (36, 140, 80),
    "purple": (132, 60, 168),
}

# Every decorative colour gets exactly this many clutter tiles, so no colour can
# be picked out by decorative ink.
CLUTTER_TILES_PER_COLOR = 14
CLUTTER_CLUSTER_SIZES = (3, 4, 3, 4)
CLUTTER_TILE = 7
CLUTTER_PITCH = 10

# A dial whose rotation relative to the key is nearly zero would be solvable by
# plain superposition, so the visible turn is kept away from the identity.
MIN_TURN_DEG = 15
STEP_DEG = 360 // BEADS_PER_RING  # 60

QUARANTINE_MARGIN = 3.0


def _rotations(seq):
    n = len(seq)
    return [tuple(seq[(i + k) % n] for i in range(n)) for k in range(n)]


def _is_rotation(seq, reference) -> bool:
    return tuple(reference) in _rotations(seq)


def _reverse_gap(seq) -> int:
    """Positions by which the reflected ring fails to be a rotation of `seq`.

    Minimised over the six possible reflection alignments. Both sequences carry
    the same colour multiset, so this gap is always even: 2 is the tightest
    possible disagreement and 4 is a comfortable one.
    """
    seq = tuple(seq)
    reverse = tuple(reversed(seq))
    return min(
        sum(1 for a, b in zip(seq, rotated) if a != b) for rotated in _rotations(reverse)
    )


def _sample_key(rng: random.Random, want_near_boundary: bool):
    """A chiral six-bead cyclic colour order using three or four colours."""
    fallback = None
    for _ in range(4000):
        seq = tuple(COLOR_NAMES[rng.randrange(len(COLOR_NAMES))] for _ in range(BEADS_PER_RING))
        if len(set(seq)) < 3:
            continue
        gap = _reverse_gap(seq)
        if gap == 0:  # cyclically palindromic: the two dials would be identical
            continue
        if fallback is None:
            fallback = seq
        if (gap < QUARANTINE_MARGIN) == want_near_boundary:
            return seq
    if fallback is None:
        raise RuntimeError("no chiral bead order was found for this seed")
    return fallback


def _turn(rng: random.Random, base: int) -> int:
    """An offset whose turn away from `base` is visible modulo one bead step."""
    while True:
        offset = rng.randrange(360)
        delta = (offset - base) % STEP_DEG
        if MIN_TURN_DEG <= delta <= STEP_DEG - MIN_TURN_DEG:
            return offset


def _clutter(rng: random.Random, panel_box) -> list[dict]:
    """Non-overlapping decorative tile clusters, ink-balanced across colours."""
    px0, py0, px1, py1 = panel_box
    keep_out = (px0 - 12, py0 - 12, px1 + 12, py1 + 12)
    placed: list[tuple[int, int, int, int]] = []
    items: list[dict] = []
    for name in COLOR_NAMES:
        for size in CLUTTER_CLUSTER_SIZES:
            for _ in range(2000):
                cols = rng.randint(1, 3)
                rows = (size + cols - 1) // cols
                w = (cols - 1) * CLUTTER_PITCH + CLUTTER_TILE
                h = (rows - 1) * CLUTTER_PITCH + CLUTTER_TILE
                x = rng.randint(6, CANVAS_W - w - 6)
                y = rng.randint(6, CANVAS_H - h - 6)
                box = (x - 6, y - 6, x + w + 6, y + h + 6)
                if box[0] < keep_out[2] and keep_out[0] < box[2] and box[1] < keep_out[3] and keep_out[1] < box[3]:
                    continue
                if any(box[0] < o[2] and o[0] < box[2] and box[1] < o[3] and o[1] < box[3] for o in placed):
                    continue
                placed.append(box)
                offsets = [(i // cols, i % cols) for i in range(size)]
                items.append({"color": name, "x": x, "y": y, "cells": [list(o) for o in offsets]})
                break
            else:
                raise RuntimeError("could not place a decorative cluster")
    return items


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    # The answer colour cycles on seed % 4 and the near-boundary flag cycles on
    # (seed // 4) % 3, so answer identity is independent of the decision margin.
    want_near_boundary = (seed // len(COLOR_NAMES)) % 3 == 1

    key_seq = _sample_key(rng, want_near_boundary)
    key_offset = rng.randrange(360)
    match_side = "left" if rng.random() < 0.5 else "right"

    match_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    others = [name for name in COLOR_NAMES if name != match_color]
    other_color = others[rng.randrange(len(others))]
    band_colors = {
        match_side: match_color,
        ("right" if match_side == "left" else "left"): other_color,
    }

    offsets = {
        "key": key_offset,
        "left": _turn(rng, key_offset),
        "right": _turn(rng, key_offset),
    }

    panel_x = rng.randint(12, CANVAS_W - PANEL_W - 12)
    panel_y = rng.randint(12, CANVAS_H - PANEL_H - 12)
    clutter = _clutter(rng, (panel_x, panel_y, panel_x + PANEL_W, panel_y + PANEL_H))

    return {
        "panel": [panel_x, panel_y],
        "key_seq": list(key_seq),
        "match_side": match_side,
        "offsets": offsets,
        "band_colors": band_colors,
        "clutter": clutter,
    }


def dial_sequence(scene: dict, side: str) -> list[str]:
    """Clockwise bead colours drawn on one dial ring, starting at its first bead."""
    key = list(scene["key_seq"])
    if side == scene["match_side"]:
        return key
    return list(reversed(key))


def _ring_center(ox: int, oy: int) -> tuple[int, int]:
    return ox + RING_R + BEAD // 2, oy + RING_R + BEAD // 2


def _bead_boxes(ox: int, oy: int, offset_deg: int):
    cx, cy = _ring_center(ox, oy)
    boxes = []
    for index in range(BEADS_PER_RING):
        theta = math.radians(offset_deg + STEP_DEG * index)
        bx = int(round(cx + RING_R * math.cos(theta))) - BEAD // 2
        by = int(round(cy + RING_R * math.sin(theta))) - BEAD // 2
        boxes.append((bx, by))
    return boxes


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BOARD)
    draw = ImageDraw.Draw(image)

    for item in sorted(scene["clutter"], key=lambda d: (d["y"], d["x"], d["color"])):
        fill = PALETTE[item["color"]]
        for r, c in sorted(tuple(cell) for cell in item["cells"]):
            x = item["x"] + c * CLUTTER_PITCH
            y = item["y"] + r * CLUTTER_PITCH
            draw.rectangle([x, y, x + CLUTTER_TILE - 1, y + CLUTTER_TILE - 1], fill=fill)

    px, py = scene["panel"]
    draw.rectangle([px, py, px + PANEL_W - 1, py + PANEL_H - 1], fill=CREAM)
    ox, oy = px + PANEL_PAD, py + PANEL_PAD

    rings = (
        (ox + KEY_OX, oy + KEY_OY, scene["offsets"]["key"], list(scene["key_seq"])),
        (ox + LEFT_OX, oy + DIAL_OY, scene["offsets"]["left"], dial_sequence(scene, "left")),
        (ox + RIGHT_OX, oy + DIAL_OY, scene["offsets"]["right"], dial_sequence(scene, "right")),
    )
    for rx, ry, offset, seq in rings:
        for (bx, by), name in zip(_bead_boxes(rx, ry, offset), seq):
            draw.rectangle([bx, by, bx + BEAD - 1, by + BEAD - 1], fill=PALETTE[name])

    for side, base_x in (("left", LEFT_OX), ("right", RIGHT_OX)):
        bx = ox + base_x + (RING_SPAN - BAND_W) // 2
        by = oy + BAND_OY
        draw.rectangle(
            [bx, by, bx + BAND_W - 1, by + BAND_H - 1],
            fill=PALETTE[scene["band_colors"][side]],
        )

    return image


def analytic_gold(scene: dict) -> str:
    """Bar colour under the dial whose clockwise order rotates onto the key."""
    key = list(scene["key_seq"])
    matches = [
        side for side in ("left", "right") if _is_rotation(dial_sequence(scene, side), key)
    ]
    if len(matches) != 1:
        raise ValueError(f"cyclic-order assignment is not unique: {matches}")
    return scene["band_colors"][matches[0]]


def margin(scene: dict) -> float:
    """Runner-up gap: bead positions by which the reflected dial misses the key."""
    return float(_reverse_gap(scene["key_seq"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The clutter list and each cluster's cell list are sets: order is an alias."""
    reversed_cells = [
        {**item, "cells": list(reversed(item["cells"]))} for item in scene["clutter"]
    ]
    return [
        ("clutter_order", {**scene, "clutter": list(reversed(scene["clutter"]))}),
        ("clutter_cell_order", {**scene, "clutter": reversed_cells}),
    ]
